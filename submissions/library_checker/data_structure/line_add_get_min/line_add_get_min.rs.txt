pub fn solve() {
    crate::prepare!(fast);
    sc!(n, q);
    let mut tree = OfflineLiChaoTree::new();
    for (a, b) in sv!([(i32, i64); iter n]) {
        tree.add_line((a, b));
    }
    for _ in 0..q {
        sc!(query: Query);
        match query {
            Query::Add { a, b } => {
                tree.add_line((a, b));
            }
            Query::Get { x } => {
                tree.query_min(x);
            }
        }
    }
    pp!(@lf @it tree.execute().into_iter().map(Option::unwrap));
}
