1use competitive::prelude::*;
2use competitive::{
3    algebra::AdditiveOperation,
4    data_structure::{BinaryIndexedTree, FibHashMap},
5};
6
7#[verify::library_checker("static_range_count_distinct")]
8pub fn static_range_count_distinct(reader: impl Read, writer: impl Write) {
9    prepare_io!(reader, writer);
10    sc!(n, q, a: [u32; n], lr: [(usize, usize); q]);
11    let mut order: Vec<_> = (0..q).collect();
12    order.sort_unstable_by_key(|&i| lr[i].1);
13    let mut bit = BinaryIndexedTree::<AdditiveOperation<i64>>::new(n);
14    let mut last = FibHashMap::default();
15    let mut ans = vec![0; q];
16    let mut r = 0;
17    for i in order {
18        while r < lr[i].1 {
19            if let Some(prev) = last.insert(a[r], r) {
20                bit.update(prev, -1);
21            }
22            bit.update(r, 1);
23            r += 1;
24        }
25        ans[i] = bit.fold(lr[i].0, lr[i].1);
26    }
27    pp!(@lf @it ans);
28}
