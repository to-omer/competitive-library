8#[verify::library_checker(
9    "ordered_set",
10    include("RankSelectBitSet", "StaticSearch", "sort", "IteratorExt")

12pub fn ordered_set(reader: impl Read, writer: impl Write) {
13    prepare_io!(buffered; reader, writer);
14    sc!(n, q, a: [u32; n], queries: [(u8, u32); q]);
15    let mut updates: Vec<_> = queries
16        .iter()
17        .filter_map(|&(t, x)| (t == 0).then_some(x))
18        .collect();
19    let minimum = a
20        .first()
21        .copied()
22        .into_iter()
23        .chain(updates.iter().copied())
24        .min();
25    let maximum = a
26        .last()
27        .copied()
28        .into_iter()
29        .chain(updates.iter().copied())
30        .max();
31    let (values, words) = if let Some((minimum, maximum)) = minimum.zip(maximum)
32        && ((maximum - minimum) as usize) < 2 * (n + updates.len())
33    {
34        drop(updates);
35        let values: Vec<_> = (minimum..=maximum).collect();
36        let mut words = vec![0; values.len().div_ceil(64)];
37        for x in a {
38            let index = (x - minimum) as usize;
39            words[index / 64] |= 1 << (index % 64);
40        }
41        (values, words)
42    } else {
43        updates.radix_sort_by_key(|&x| x);
44        updates.dedup();
45        let mut values = Vec::with_capacity(n + updates.len());
46        let mut words = vec![0; (n + updates.len()).div_ceil(64)];
47        let merged = a
48            .into_iter()
49            .map(|x| (x, true))
50            .merge_by(updates.into_iter().map(|x| (x, false)), |x, y| x.0 <= y.0);
51        for (value, present) in merged {
52            if values.last() != Some(&value) {
53                values.push(value);
54            }
55            if present {
56                let k = values.len() - 1;
57                words[k / 64] |= 1 << (k % 64);
58            }
59        }
60        words.truncate(values.len().div_ceil(64));
61        (values, words)
62    };
63    let search = StaticSearch::from_sorted(&values);
64    let endpoints: Vec<_> = queries
65        .iter()
66        .filter_map(|&(t, x)| (t != 2).then_some(if t == 3 || t == 4 { x + 1 } else { x }))
67        .collect();
68    let mut positions = vec![0; endpoints.len()];
69    search.lower_bound_batch(&endpoints, &mut positions);
70    let mut set = RankSelectBitSet::from_words(&words, values.len());
71    let mut positions = positions.into_iter();
72    for (t, x) in queries {
73        let k = if t == 2 { 0 } else { positions.next().unwrap() };
74        match t {
75            0 => set.set(k, true),
76            1 => {
77                if values.get(k) == Some(&x) {
78                    set.set(k, false);
79                }
80            }
81            2 => {
82                pp!(set.select1(x as usize - 1).map_or(-1, |k| values[k] as i64));
83            }
84            3 => {
85                pp!(set.rank1(k));
86            }
87            4 => {
88                pp!(set.prev_one(k).map_or(-1, |k| values[k] as i64));
89            }
90            5 => {
91                pp!(set.next_one(k).map_or(-1, |k| values[k] as i64));
92            }
93            _ => unreachable!(),
94        }
95    }
96}

98#[verify::library_checker("ordered_set", include("SqrtTree"))]
99pub fn ordered_set_sqrt_set(reader: impl Read, writer: impl Write) {
100    prepare_io!(buffered; reader, writer);
101    sc!(n, q, a: [u32; iter n]);
102    let mut set: SqrtSet<_> = a.collect();
103    for _ in 0..q {
104        sc!(t: u8, x: u32);
105        match t {
106            0 => {
107                set.insert(x);
108            }
109            1 => {
110                set.remove(&x);
111            }
112            2 => pp!(set.get_at(x as usize - 1).map_or(-1, |&v| v as i64)),
113            3 => pp!(set.rank_le(&x)),
114            4 => pp!(set.range(..=x).next_back().map_or(-1, |&v| v as i64)),
115            5 => pp!(set.range(x..).next().map_or(-1, |&v| v as i64)),
116            _ => unreachable!(),
117        }
118    }
