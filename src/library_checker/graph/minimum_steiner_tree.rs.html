1use competitive::prelude::*;

7#[verify::library_checker("minimum_steiner_tree")]
8pub fn minimum_steiner_tree(reader: impl Read, writer: impl Write) {
9    prepare_io!(reader, writer);
10    sc!(n, m, (graph, weights): @UndirectedGraphScanner::<usize, u64>::new(n, m));
11    sc!(k, terminals: [usize; k]);
12    let tree = graph
13        .steiner_tree::<StandardSp<AdditiveOperation<_>>>()
14        .with_parent()
15        .solve(terminals[1..].iter().copied(), |eid| weights[eid]);
16    let edges = tree.edges_from_source(terminals[0]).unwrap();
17    pp!(tree.minimum_from_source(terminals[0]), edges.len(); @it edges);
18}
