65mod verification {

68    verus! {
69        fn payload_is_ignored<U>(a: i64, b: i64, left: U, right: U) {
70            let x = PartialIgnoredOrd(a, left);
71            let y = PartialIgnoredOrd(b, right);
72            let equal = x == y;
73            let compared = x.cmp(&y);
74            let partial = x.partial_cmp(&y);
75            proof {
76                #[verifier::proof_note("PartialIgnoredOrd equality and both ordering operations depend only on the key, for arbitrary payload types and values, without requiring the payload to implement any comparison trait")]
77                assert(equal == (a == b) && compared == a.cmp_spec(&b) && partial == Some(compared));
78            }
79        }
80    }
