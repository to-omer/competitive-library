4#[derive(Clone, Debug, Default)]
5pub struct AdjacencyListGraph {
6    pub vsize: usize,
7    pub esize: usize,
8    pub graph: Vec<Vec<Neighbor<usize, usize>>>,
9}
10impl AdjacencyListGraph {
11    pub fn new(vsize: usize) -> AdjacencyListGraph {
12        AdjacencyListGraph {
13            vsize,
14            esize: 0,
15            graph: vec![vec![]; vsize],
16        }
17    }
18    pub fn add_edge(&mut self, from: usize, to: usize) {
19        self.graph[from].push(Neighbor::new(to, self.esize));
20        self.esize += 1;
21    }
22    pub fn add_undirected_edge(&mut self, u: usize, v: usize) {
23        self.graph[u].push(Neighbor::new(v, self.esize));
24        self.graph[v].push(Neighbor::new(u, self.esize));
25        self.esize += 1;
26    }
27    pub fn vertices(&self) -> Range<usize> {
28        0..self.vsize
29    }
30}

120impl<U: IterScan<Output = usize>, T: IterScan> MarkedIterScan for AdjacencyListGraphScanner<U, T> {
121    type Output = (AdjacencyListGraph, Vec<<T as IterScan>::Output>);
122    fn mscan<'a, I: Iterator<Item = &'a str>>(self, iter: &mut I) -> Option<Self::Output> {
123        let mut graph = AdjacencyListGraph::new(self.vsize);
124        let mut rest = Vec::with_capacity(self.esize);
125        for _ in 0..self.esize {
126            let (from, to) = (U::scan(iter)?, U::scan(iter)?);
127            if self.directed {
128                graph.add_edge(from, to);
129            } else {
130                graph.add_undirected_edge(from, to);
131            }
132            rest.push(T::scan(iter)?);
133        }
134        Some((graph, rest))
135    }
