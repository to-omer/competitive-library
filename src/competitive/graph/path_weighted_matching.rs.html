12impl<T> PathWeightedMatching<T> {
13    pub fn from_edge_weights(weights: impl IntoIterator<Item = T>) -> Self
14    where
15        T: Copy + Zero + Sub<Output = T>,
16    {
17        let mut vertex = T::zero();
18        let vertices = std::iter::once(vertex).chain(weights.into_iter().map(|weight| {
19            vertex = weight - vertex;
20            vertex
21        }));
22        Self::from_vertex_weights(vertices)
23    }
24
25    /// Edge weights are the sums of adjacent vertex weights.
26    pub fn from_vertex_weights(weights: impl IntoIterator<Item = T>) -> Self {
27        Self {
28            weights: weights.into_iter().collect(),
29        }
30    }

32    /// Returns the maximum weight of a matching with exactly `k` edges in O(n) time.
33    pub fn maximum_weight(self, k: usize) -> T
34    where
35        T: Copy + Ord + Zero + Add<Output = T>,
36    {
37        assert!(k <= self.weights.len() / 2);
38        let mut gains = path_matching_candidates(self.weights, |x| x, |a, b| a + b);
39        if k != 0 && k < gains.len() {
40            gains.select_nth_unstable_by_key(k, |x| Reverse(*x));
41        }
42        gains.into_iter().take(k).fold(T::zero(), |a, b| a + b)
43    }
44
45    /// Returns the maximum weights for exactly 0, 1, ... edges, including all
46    /// feasible sizes, in O(n log n) time. The first entry is zero.
47    pub fn maximum_weights(self) -> Vec<T>
48    where
49        T: Copy + Ord + Zero + Add<Output = T>,
50    {
51        let mut gains = path_matching_candidates(self.weights, |x| x, |a, b| a + b);
52        gains.sort_unstable_by_key(|x| Reverse(*x));
53        gains.insert(0, T::zero());
54        for i in 1..gains.len() {
55            gains[i] = gains[i - 1] + gains[i];
56        }
57        gains
58    }
59
60    /// Returns the maximum weight and sorted edge indices for exactly `k` edges in O(n) time.
61    pub fn maximum_weight_matching(self, k: usize) -> (T, Vec<usize>)
62    where
63        T: Copy + Ord + Zero + Add<Output = T>,
64    {
65        let n = self.weights.len();
66        assert!(k <= n / 2);
67        let mut pairs = path_matching_candidates(
68            (0..n).collect(),
69            |i| self.weights[i],
70            |u, v| (self.weights[u] + self.weights[v], u, v),
71        );
72        if k != 0 && k < pairs.len() {
73            pairs.select_nth_unstable_by_key(k, |&(w, u, v)| (Reverse(w), u, v));
74        }
75        let mut matched = vec![false; n];
76        let mut weight = T::zero();
77        for (w, u, v) in pairs.into_iter().take(k) {
78            weight = weight + w;
79            matched[u] = true;
80            matched[v] = true;
81        }
82        let mut edges = Vec::with_capacity(k);
83        let mut i = 0;
84        while i < n {
85            if matched[i] {
86                edges.push(i);
87                i += 2;
88            } else {
89                i += 1;
90            }
91        }
92        (weight, edges)
93    }
94}

96/// Returns maximum matching weights for queries `(l, r, k)` in O((n + q) log n) time.
97/// Each query selects exactly `k` edges in the vertex interval `[l, r)`.
98/// Edge weights are sums of adjacent entries in `weights`; answers follow query order.
99pub fn path_weighted_matching_queries<T>(weights: &[T], queries: &[(usize, usize, usize)]) -> Vec<T>
100where
101    T: Copy + Ord + Zero + Add<Output = T> + Sub<Output = T>,
102{
103    let n = weights.len();
104    let mut prefix = Vec::with_capacity(n + 1);
105    prefix.push(T::zero());
106    for &weight in weights {
107        prefix.push(*prefix.last().unwrap() + weight);
108    }
109    let mut answers = vec![T::zero(); queries.len()];
110    let mut pending = false;
111    for (i, &(l, r, k)) in queries.iter().enumerate() {
112        assert!(l <= r && r <= n && k <= (r - l) / 2);
113        if 2 * k == r - l {
114            answers[i] = prefix[r] - prefix[l];
115        } else if k != 0 {
116            pending = true;
117        }
118    }
119    if !pending {
120        return answers;
121    }
122    let mut pairs = path_matching_candidates(
123        (0..n).collect(),
124        |i| weights[i],
125        |u, v| (weights[u] + weights[v], u, v),
126    );
127    pairs.sort_unstable_by_key(|&(w, u, v)| (Reverse(w), u, v));
128    let mut times = vec![pairs.len(); n];
129    for (t, &(_, u, v)) in pairs.iter().enumerate() {
130        times[u] = t;
131        times[v] = t;
132    }
133    let times = WaveletMatrix::new(times);
134    let mut heads = vec![!0; pairs.len()];
135    let mut next = vec![!0; queries.len()];
136    for (i, &(l, r, k)) in queries.iter().enumerate() {
137        if k != 0 && 2 * k != r - l {
138            let t = times.quantile(l..r, 2 * k - 1);
139            next[i] = heads[t];
140            heads[t] = i;
141        }
142    }
143    let minima: [_; 2] = std::array::from_fn(|parity| {
144        RangeMinimumQuery::new(
145            (parity..n)
146                .step_by(2)
147                .map(|i| (weights[i], Reverse(i)))
148                .collect(),
149        )
150    });
151    let mut state = SegmentTree::<RangeState<T>>::from_vec(
152        weights
153            .iter()
154            .enumerate()
155            .map(|(i, &w)| RangeState::unmatched(w, i))
156            .collect(),
157    );
158    let mut rollback = Vec::with_capacity(6);
159    for (t, &(_, u, v)) in pairs.iter().enumerate() {
160        let mut query = heads[t];
161        while query != !0 {
162            let (l, r, k) = queries[query];
163            let current = state.fold(l..r);
164            let used = r - l - current.free;
165            let left = (l - state.fold(..l).free) % 2;
166            let right = (left + used) % 2;
167            let mut answer = current.sum;
168            if left != 0 {
169                let first = current.first.unwrap().1;
170                let (_, Reverse(i)) = minima[l % 2].fold(l / 2, first.div_ceil(2));
171                rollback.push((i, state.get(i)));
172                state.set(i, RangeState::unmatched(weights[i], i));
173                answer = answer - weights[i];
174            }
175            if right != 0 {
176                let last = current.last.unwrap().1;
177                let (_, Reverse(i)) = minima[(r - 1) % 2].fold(last.div_ceil(2), r.div_ceil(2));
178                rollback.push((i, state.get(i)));
179                state.set(i, RangeState::unmatched(weights[i], i));
180                answer = answer - weights[i];
181            }
182            for _ in 0..(2 * k - used + left + right) / 2 {
183                let (gain, Reverse(a), Reverse(b)) = state.fold(l..r).best.unwrap();
184                answer = answer + gain;
185                for i in [a, b] {
186                    rollback.push((i, state.get(i)));
187                    state.set(i, RangeState::matched(weights[i]));
188                }
189            }
190            answers[query] = answer;
191            while let Some((i, value)) = rollback.pop() {
192                state.set(i, value);
193            }
194            query = next[query];
195        }
196        state.set(u, RangeState::matched(weights[u]));
197        state.set(v, RangeState::matched(weights[v]));
198    }
199    answers
200}

202fn path_matching_candidates<V, T, R>(
203    mut stack: Vec<V>,
204    weight: impl Fn(V) -> T,
205    pair: impl Fn(V, V) -> R,
206) -> Vec<R>
207where
208    V: Copy,
209    T: Ord,
210{
211    let mut pairs = Vec::with_capacity(stack.len() / 2 + 1);
212    let mut n = 0;
213    for i in 0..stack.len() {
214        stack[n] = stack[i];
215        n += 1;
216        while n >= 3 {
217            // Equal gains prefer the leftmost pair.
218            let local_maximum = weight(stack[n - 3]) >= weight(stack[n - 1])
219                && (n == 3 || weight(stack[n - 2]) > weight(stack[n - 4]));
220            if !local_maximum {
221                break;
222            }
223            pairs.push(pair(stack[n - 3], stack[n - 2]));
224            stack[n - 3] = stack[n - 1];
225            n -= 2;
226        }
227    }
228    while n >= 2 {
229        pairs.push(pair(stack[n - 2], stack[n - 1]));
230        n -= 2;
231    }
232    pairs
233}

235#[derive(Clone)]
236struct RangeState<T> {
237    free: usize,
238    sum: T,
239    first: Option<(T, usize)>,
240    last: Option<(T, usize)>,
241    best: Option<(T, Reverse<usize>, Reverse<usize>)>,
242}
243
244impl<T> RangeState<T> {
245    fn unmatched(weight: T, index: usize) -> Self
246    where
247        T: Copy + Zero,
248    {
249        Self {
250            free: 1,
251            sum: T::zero(),
252            first: Some((weight, index)),
253            last: Some((weight, index)),
254            best: None,
255        }
256    }
257
258    fn matched(weight: T) -> Self {
259        Self {
260            free: 0,
261            sum: weight,
262            first: None,
263            last: None,
264            best: None,
265        }
266    }
267}

269impl<T> Magma for RangeState<T>
270where
271    T: Copy + Ord + Add<Output = T>,
272{
273    type T = Self;
274    fn operate(x: &Self, y: &Self) -> Self {
275        let crossing = x
276            .last
277            .zip(y.first)
278            .map(|((a, i), (b, j))| (a + b, Reverse(i), Reverse(j)));
279        Self {
280            free: x.free + y.free,
281            sum: x.sum + y.sum,
282            first: x.first.or(y.first),
283            last: y.last.or(x.last),
284            best: x.best.max(y.best).max(crossing),
285        }
286    }
287}

308    fn maximum_weights_by_dp(weights: &[i128]) -> Vec<i128> {
309        let maximum_size = weights.len().div_ceil(2);
310        let mut previous = vec![None; maximum_size + 1];
311        let mut current = vec![None; maximum_size + 1];
312        previous[0] = Some(0);
313        current[0] = Some(0);
314        for (i, &weight) in weights.iter().enumerate() {
315            for k in (1..=i / 2 + 1).rev() {
316                let take = previous[k - 1].map(|total| total + weight);
317                previous[k] = current[k].max(take);
318            }
319            std::mem::swap(&mut previous, &mut current);
320        }
321        current.into_iter().map(Option::unwrap).collect()
322    }

324    fn check_matching(weights: &[i128], matching: PathWeightedMatching<i128>) {
325        let expected = maximum_weights_by_dp(weights);
326        assert_eq!(matching.clone().maximum_weights(), expected, "{weights:?}");
327        for (k, &weight) in expected.iter().enumerate() {
328            assert_eq!(
329                matching.clone().maximum_weight(k),
330                weight,
331                "{weights:?}, k={k}"
332            );
333        }
334    }

373        let low = i128::from(i64::MIN);
374        let high = i128::from(i64::MAX);
375        let mut rng = Xorshift::default();
376        let cases = exhaustive_sequences(-1i128..=1, 0..=10)

380            let weights: Vec<_> = vertices.windows(2).map(|pair| pair[0] + pair[1]).collect();
381            let expected = maximum_weights_by_dp(&weights);
382            for matching in [
383                PathWeightedMatching::from_vertex_weights(vertices),
384                PathWeightedMatching::from_edge_weights(weights.iter().copied()),

386                for (k, &expected) in expected.iter().enumerate() {

391                    assert!(edges.windows(2).all(|pair| pair[0] + 1 < pair[1]));
392                    let total: i128 = edges.iter().map(|&i| weights[i]).sum();
393                    assert_eq!(total, expected, "{weights:?}, k={k}, edges={edges:?}");
394                }

399    #[test]
400    fn test_path_weighted_matching_queries() {
401        let low = i128::from(i64::MIN);
402        let high = i128::from(i64::MAX);
403        let mut rng = Xorshift::default();
404        let cases = exhaustive_sequences(-1i128..=1, 0..=8)
405            .chain(exhaustive_sequences([low, 0, high], 0..=5))
406            .chain(sequence_cases(&mut rng, low..=high, 0..=128))
407            .chain([vec![0; 513], vec![low; 513], (0..512).chain([-1]).collect()]);
408        for vertices in cases {
409            let n = vertices.len();
410            let ranges: Vec<_> = if n <= 8 {
411                (0..=n).flat_map(|l| (l..=n).map(move |r| (l, r))).collect()
412            } else {
413                vec![
414                    (0, 0),
415                    (0, n),
416                    (1, n - 1),
417                    (0, n / 2),
418                    (n / 2, n),
419                    (n / 4, 3 * n / 4),
420                    (n, n),
421                ]
422            };
423            let mut queries = Vec::new();
424            let mut expected = Vec::new();
425            for (l, r) in ranges {
426                let weights: Vec<_> = vertices[l..r]
427                    .windows(2)
428                    .map(|pair| pair[0] + pair[1])
429                    .collect();
430                for (k, weight) in maximum_weights_by_dp(&weights).into_iter().enumerate() {
431                    queries.push((l, r, k));
432                    expected.push(weight);
433                }
434            }
435            queries.extend_from_within(..);
436            expected.extend_from_within(..);
437            queries.reverse();
438            expected.reverse();
439            assert_eq!(
440                path_weighted_matching_queries(&vertices, &queries),
441                expected,
442                "{vertices:?}"
443            );
444            assert!(path_weighted_matching_queries(&vertices, &[]).is_empty());
445        }
446    }
