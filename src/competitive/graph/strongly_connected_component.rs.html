3/// Strongly connected components in topological order.
4#[codesnip::entry("StronglyConnectedComponent")]
5#[derive(Debug, Clone)]
6pub struct StronglyConnectedComponent {
7    vertices: Vec<usize>,
8    // Vertices before starts[0] are singleton components removed by Kahn's algorithm.
9    starts: Vec<usize>,
10}
11
12#[codesnip::entry("StronglyConnectedComponent")]
13impl StronglyConnectedComponent {
14    /// Decomposes a directed graph on `0..n` in O(V + E) time and space.
15    pub fn from_edges(n: usize, edges: impl Iterator<Item = (usize, usize)>) -> Self {
16        // Each block stores the previous cursor followed by neighbors in insertion order.
17        const BLOCK: usize = 4;
18        let mut degree = vec![0usize; n];
19        let mut rows = vec![(n, n); n];
20        let mut overflow = Vec::new();
21        for (u, v) in edges {
22            degree[v] += 1;
23            let (first, rest) = rows[u];
24            let rest = if rest == n {
25                first
26            } else if rest > n && !(rest - n).is_multiple_of(BLOCK) {
27                overflow[rest - n] = first;
28                rest + 1
29            } else {
30                let i = overflow.len();
31                overflow.extend_from_slice(&[0; BLOCK]);
32                overflow[i] = rest;
33                overflow[i + 1] = first;
34                n + i + 2
35            };
36            rows[u] = (v, rest);
37        }
38        // A cursor below n is the last neighbor, n is the end, and larger
39        // values link to earlier edges of the same vertex.
40        let next_edge = |cursor: usize| {
41            if cursor <= n {
42                (cursor, n)
43            } else {
44                let i = cursor - n - 1;
45                let next = if i % BLOCK == 1 {
46                    overflow[i - 1]
47                } else {
48                    cursor - 1
49                };
50                (overflow[i], next)
51            }
52        };
53        let mut vertices = vec![0; n];
54        let mut tail = 0;
55        for (i, &d) in degree.iter().enumerate() {
56            vertices[tail] = i;
57            tail += usize::from(d == 0);
58        }
59        let mut head = 0;
60        while head < tail {
61            let u = vertices[head];
62            head += 1;
63            let (mut v, mut rest) = rows[u];
64            while v < n {
65                degree[v] -= 1;
66                // This edge keeps its target out of the queue until it is processed.
67                vertices[tail] = v;
68                tail += usize::from(degree[v] == 0);
69                (v, rest) = next_edge(rest);
70            }
71            rows[u].0 = usize::MAX;
72        }
73        let mut starts = if tail == n {
74            Vec::new()
75        } else {
76            Self::decompose(&mut vertices, tail, &mut rows, degree, next_edge)
77        };
78        starts.push(n);
79        Self { vertices, starts }
80    }
81
82    pub fn size(&self) -> usize {
83        self.starts[0] + (self.starts.len() - 1)
84    }
85
86    /// Returns the vertex slices in topological order. The order within each slice is unspecified.
87    pub fn components(&self) -> impl DoubleEndedIterator<Item = &[usize]> {
88        let singletons = self.vertices[..self.starts[0]]
89            .iter()
90            .map(std::slice::from_ref);
91        let groups = self.starts.windows(2).map(|r| &self.vertices[r[0]..r[1]]);
92        singletons.chain(groups)
93    }
94
95    /// Maps each vertex to its zero-based component index in topological order.
96    pub fn component_ids(&self) -> Vec<usize> {
97        let mut ids = vec![0; self.vertices.len()];
98        for (id, group) in self.components().enumerate() {
99            for &v in group {
100                ids[v] = id;
101            }
102        }
103        ids
104    }
105
106    /// Returns whether a component contains more than one vertex.
107    pub fn has_loop(&self) -> bool {
108        self.vertices.len() != self.size()
109    }
110
111    fn decompose(
112        vertices: &mut [usize],
113        tail: usize,
114        rows: &mut [(usize, usize)],
115        mut roots: Vec<usize>,
116        next_edge: impl Fn(usize) -> (usize, usize),
117    ) -> Vec<usize> {
118        let n = vertices.len();
119        let mut starts = Vec::new();
120        roots.clear();
121        let mut branches = Vec::new();
122        let mut open = tail;
123        let mut end = n;
124        for root in 0..n {
125            if rows[root].0 > n {
126                continue;
127            }
128            let mut v = root;
129            loop {
130                if v < n {
131                    let (first, rest) = rows[v];
132                    if first <= n {
133                        // The first edge is consumed on discovery. Its slot then holds the
134                        // complemented one-based stack position, or MAX for a closed SCC.
135                        // These marks exceed n since rows holds two words per vertex.
136                        vertices[open] = v;
137                        roots.push(open);
138                        open += 1;
139                        rows[v].0 = !open;
140                        if rest != n {
141                            branches.push((open, rest));
142                        }
143                        v = first;
144                        continue;
145                    }
146                    if first != usize::MAX {
147                        let rank = !first;
148                        while roots.last().is_some_and(|&i| i >= rank) {
149                            roots.pop();
150                        }
151                    }
152                }
153                // Roots deeper than the nearest ancestor with unexamined edges
154                // are complete. A vertex's last edge needs no return frame.
155                let parent = branches.last().map_or(0, |&(rank, _)| rank);
156                while roots.last().is_some_and(|&i| i >= parent) {
157                    let first = roots.pop().unwrap();
158                    let len = open - first;
159                    end -= len;
160                    vertices.copy_within(first..open, end);
161                    for &u in &vertices[end..end + len] {
162                        rows[u].0 = usize::MAX;
163                    }
164                    open = first;
165                    starts.push(end);
166                }
167                let Some((_, cursor)) = branches.last_mut() else {
168                    break;
169                };
170                let (to, rest) = next_edge(*cursor);
171                if rest == n {
172                    branches.pop();
173                } else {
174                    *cursor = rest;
175                }
176                v = to;
177            }
178        }
179        starts.reverse();
180        starts
181    }
182}

184#[codesnip::entry(when("StronglyConnectedComponent", "SparseGraph"))]
185impl StronglyConnectedComponent {
186    /// Builds the condensation graph from the original directed edges.
187    pub fn gen_cgraph(
188        &self,
189        edges: impl IntoIterator<Item = (usize, usize)>,
190    ) -> DirectedSparseGraph {
191        let ids = self.component_ids();
192        let mut used = std::collections::HashSet::new();
193        let edges = edges
194            .into_iter()
195            .filter_map(|(u, v)| {
196                let u = ids[u];
197                let v = ids[v];
198                (u != v && used.insert((u, v))).then_some((u, v))
199            })
200            .collect();
201        DirectedSparseGraph::from_edges(self.size(), edges)
202    }
203}

205#[cfg(test)]
206mod tests {
207    use super::*;
208    use crate::tools::{
209        Xorshift,
210        testutil::{integer_cases, sequence_cases},
211    };
212
213    #[test]
214    fn components() {
215        let check = |n: usize, edges: &[(usize, usize)]| {
216            let mut reachable: Vec<_> = (0..n).map(|v| 1u64 << v).collect();
217            for &(u, v) in edges {
218                reachable[u] |= 1 << v;
219            }
220            for k in 0..n {
221                let destinations = reachable[k];
222                for row in &mut reachable {
223                    if *row >> k & 1 != 0 {
224                        *row |= destinations;
225                    }
226                }
227            }
228            let scc = StronglyConnectedComponent::from_edges(n, edges.iter().copied());
229            let mut ids = vec![usize::MAX; n];
230            for (id, group) in scc.components().enumerate() {
231                assert!(!group.is_empty());
232                for &v in group {
233                    assert_eq!(std::mem::replace(&mut ids[v], id), usize::MAX);
234                }
235            }
236            assert!(!ids.contains(&usize::MAX));
237            assert_eq!(ids, scc.component_ids());
238            assert_eq!(scc.size(), scc.components().count());
239            assert_eq!(
240                scc.has_loop(),
241                scc.components().any(|group| group.len() > 1)
242            );
243            for u in 0..n {
244                for v in 0..n {
245                    if reachable[u] >> v & 1 != 0 {
246                        if reachable[v] >> u & 1 != 0 {
247                            assert_eq!(ids[u], ids[v]);
248                        } else {
249                            assert!(ids[u] < ids[v]);
250                        }
251                    } else {
252                        assert_ne!(ids[u], ids[v]);
253                    }
254                }
255            }
256            let mut expected: Vec<_> = edges
257                .iter()
258                .map(|&(u, v)| (ids[u], ids[v]))
259                .filter(|&(u, v)| u != v)
260                .collect();
261            expected.sort_unstable();
262            expected.dedup();
263            let mut graph = scc.gen_cgraph(edges.iter().copied());
264            assert_eq!(graph.vertices_size(), scc.size());
265            graph.edges.sort_unstable();
266            assert_eq!(graph.edges, expected);
267        };
268
269        for n in 0..=4 {
270            for mask in 0..1u32 << (n * n) {
271                let edges: Vec<_> = (0..n * n)
272                    .filter(|&i| mask >> i & 1 != 0)
273                    .map(|i| (i / n, i % n))
274                    .collect();
275                check(n, &edges);
276            }
277        }
278        let mut rng = Xorshift::default();
279        for n in integer_cases(1usize..=64) {
280            for pairs in sequence_cases(&mut rng, 0..=n * n - 1, 0..=n * n) {
281                let edges: Vec<_> = pairs.into_iter().map(|i| (i / n, i % n)).collect();
282                check(n, &edges);
283            }
284        }
285    }
