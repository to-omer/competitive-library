1use super::{Approx, ApproxOrd, Complex, Float, Zero};
2use std::ops::{Add, Mul, Sub};
3
4/// Signed area times two; positive for counterclockwise input.
5pub fn polygon_area2<T>(ps: &[Complex<T>]) -> T
6where
7    T: Copy + Zero + Add<Output = T> + Sub<Output = T> + Mul<Output = T>,
8{
9    ps.iter()
10        .zip(ps.iter().cycle().skip(1))
11        .fold(T::zero(), |sum, (&a, &b)| sum + a.cross(b))
12}
13
14/// `(area2, moment6)` contribution.
15pub fn polygon_edge_moments<T>(a: Complex<T>, b: Complex<T>) -> (T, Complex<T>)
16where
17    T: Copy + Add<Output = T> + Sub<Output = T> + Mul<Output = T>,
18{
19    let cross = a.cross(b);
20    (cross, (a + b) * cross)
21}
22
23/// `None` if the signed area is approximately zero.
24pub fn polygon_centroid<T>(ps: &[Complex<T>]) -> Option<Complex<T>>
25where
26    T: Float + ApproxOrd,
27{
28    let (area, moment) = polygon_moments(ps);
29    if Approx(area) == Approx(T::zero()) {
30        None
31    } else {
32        Some(moment / (area + area + area))
33    }
34}
35
36/// Signed `(area2, moment6)`; centroid is `moment6 / (3 * area2)`.
37pub fn polygon_moments<T>(ps: &[Complex<T>]) -> (T, Complex<T>)
38where
39    T: Copy + Zero + Add<Output = T> + Sub<Output = T> + Mul<Output = T>,
40{
41    ps.iter().zip(ps.iter().cycle().skip(1)).fold(
42        (T::zero(), Complex::zero()),
43        |(area, moment), (&a, &b)| {
44            let (cross, contribution) = polygon_edge_moments(a, b);
45            (area + cross, moment + contribution)
46        },
47    )
48}

50#[cfg(test)]
51mod tests {
52    use super::*;
53    use crate::tools::testutil::exhaustive_sequences;

55    type P = Complex<i128>;
56
57    #[test]
58    fn test_polygon_moments() {
59        let points: Vec<_> = (-1..=1)
60            .flat_map(|x| (-1..=1).map(move |y| P::new(x, y)))
61            .collect();
62        for mut ps in exhaustive_sequences(points.iter().copied(), 0..=5) {
63            let mut area = 0;
64            let mut moment = P::zero();
65            if let Some(&first) = ps.first() {
66                for triangle in ps[1..].windows(2) {
67                    let cross = (triangle[0] - first).cross(triangle[1] - first);
68                    area += cross;
69                    moment += (first + triangle[0] + triangle[1]) * cross;
70                }
71            }
72            assert_eq!(polygon_area2(&ps), area);
73            assert_eq!(polygon_moments(&ps), (area, moment));
74            ps.reverse();
75            assert_eq!(polygon_moments(&ps), (-area, -moment));
76            let translation = P::new(1000000000, -2000000000);
77            for p in &mut ps {
78                *p += translation;
79            }
80            assert_eq!(
81                polygon_moments(&ps),
82                (-area, -moment - translation * (3 * area))
83            );
84            if !ps.is_empty() {
85                ps.push(ps[0]);
86                ps.rotate_left(1);
87                assert_eq!(
88                    polygon_moments(&ps),
89                    (-area, -moment - translation * (3 * area))
90                );
91            }
92        }
93        let triangle = [P::new(1, 0), P::new(0, 1), P::zero()];
94        let scale = 1000000000000;
95        let scaled = triangle.map(|p| p * scale);
96        let (area, moment) = polygon_moments(&triangle);
97        assert_eq!(
98            polygon_moments(&scaled),
99            (area * scale * scale, moment * scale * scale * scale)
100        );
101    }

103    #[test]
104    fn test_polygon_centroid() {
105        let points: Vec<_> = (-1..=1)
106            .flat_map(|x| (-1..=1).map(move |y| P::new(x, y)))
107            .collect();
108        let empty: [Complex<f64>; 0] = [];
109        assert_eq!(polygon_centroid(&empty), None);
110        for &a in &points {
111            for &b in &points {
112                for &c in &points {
113                    let mut ps = [a, b, c].map(|p| p.map(|x| x as f64));
114                    if (b - a).cross(c - a) == 0 {
115                        assert_eq!(polygon_centroid(&ps), None);
116                    } else {
117                        let expected = (ps[0] + ps[1] + ps[2]) / 3.0;
118                        let got = polygon_centroid(&ps).unwrap();
119                        assert_eq!(got.map(Approx), expected.map(Approx));
120                        ps.reverse();
121                        assert_eq!(
122                            polygon_centroid(&ps).unwrap().map(Approx),
123                            expected.map(Approx)
124                        );
125                    }
126                }
127            }
128        }
129    }
