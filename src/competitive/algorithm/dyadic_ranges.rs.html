1/// Maximal aligned power-of-two intervals.
2pub fn dyadic_ranges(range: std::ops::Range<u64>) -> impl Iterator<Item = std::ops::Range<u64>> {
3    let mut left = range.start;
4    let right = range.end;
5    std::iter::from_fn(move || {
6        if left >= right {
7            return None;
8        }
9        let remaining = right - left;
10        let exponent = left.trailing_zeros().min(remaining.ilog2());
11        let end = left + (1u64 << exponent);
12        let result = left..end;
13        left = end;
14        Some(result)
15    })
16}

18#[cfg(test)]
19mod tests {
20    use super::*;
21    use crate::tools::Xorshift;
22
23    #[test]
24    fn test_dyadic_ranges() {
25        let mut rng = Xorshift::default();
26        let ranges = (0..=256)
27            .flat_map(|left| (0..=256).map(move |right| (left, right)))
28            .chain([
29                (0, u64::MAX),
30                (u64::MAX - 1000, u64::MAX),
31                (1 << 63, u64::MAX),
32                (u64::MAX, u64::MAX),
33                (u64::MAX, 0),
34            ])
35            .chain(rng.random_iter((0..=u64::MAX, 0..=u64::MAX)).take(1000));
36        for (left, right) in ranges {
37            if left > right {
38                assert!(dyadic_ranges(left..right).next().is_none());
39                continue;
40            }
41            let mut expected_start = left;
42            for range in dyadic_ranges(left..right) {
43                let length = range.end - range.start;
44                assert!(length.is_power_of_two());
45                assert_eq!(range.start % length, 0);
46                assert_eq!(range.start, expected_start);
47                assert!(range.end <= right);
48                let parent_length = 2 * length as u128;
49                let parent_start = range.start as u128 & !(parent_length - 1);
50                assert!(
51                    parent_start < left as u128 || parent_start + parent_length > right as u128
52                );
53                expected_start = range.end;
54            }
55            assert_eq!(expected_start, right);
56        }
57    }
