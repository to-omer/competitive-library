3pub fn static_range_count_distinct<T: Eq + std::hash::Hash>(
4    a: &[T],
5    queries: &[(usize, usize)],
6) -> Vec<u32> {
7    let end = queries.iter().map(|&(_, r)| r).max().unwrap_or(0);
8    let mut offsets = vec![0; end + 2];
9    for &(_, r) in queries {
10        offsets[r] += 1;
11    }
12    for r in 0..=end {
13        offsets[r + 1] += offsets[r];
14    }
15    let mut order = vec![(0, 0); queries.len()];
16    for (i, &(l, r)) in queries.iter().enumerate() {
17        offsets[r] -= 1;
18        order[offsets[r]] = (l as u32, i);
19    }
20    let mut bit = DaryPrefixSumTreeU32::new(end + 1);
21    let mut last = FibHashMap::default();
22    let mut ans = vec![0; queries.len()];
23    for r in 0..=end {
24        if r != 0 {
25            let previous = last.insert(&a[r - 1], r).unwrap_or(0);
26            bit.update(previous, 1);
27        }
28        for &(l, i) in &order[offsets[r]..offsets[r + 1]] {
29            ans[i] = bit.accumulate0(l as usize + 1) - l;
30        }
31    }
32    ans
33}

44    fn test_static_range_count_distinct() {
45        for values in exhaustive_sequences(0..3, 0..=7) {
46            let array: Vec<_> = values.into_iter().map(Key).collect();
47            for end in 0..=array.len() {
48                let mut queries: Vec<_> = (0..=end)
49                    .flat_map(|l| (l..=end).map(move |r| (l, r)))
50                    .collect();
51                queries.reverse();
52                let expected: Vec<_> = queries
53                    .iter()
54                    .map(|&(l, r)| {
55                        array[l..r]
56                            .iter()
57                            .collect::<std::collections::HashSet<_>>()
58                            .len() as u32
59                    })
60                    .collect();
61                assert_eq!(static_range_count_distinct(&array, &queries), expected);
62            }
