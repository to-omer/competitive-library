18impl CartesianTree {
19    #[cfg_attr(verus_keep_ghost, verifier::rlimit(40))]
20    #[cfg_attr(verus_keep_ghost, verus_spec(result =>
21        ensures
22            result.parents.len() == a.len(), result.children.len() == a.len(),
23            result.valid_parents(),
24            a.len() == 0 ==> result.root == !0usize,
25            a.len() > 0 ==> exists|ranges: Seq<(usize, usize)>| #[trigger] result.valid_ranges(ranges)

29    pub fn new<T>(a: &[T]) -> Self
30    where
31        T: PartialOrd,
32    {
33        let mut parents = vec![!0; a.len()];
34        let mut children = vec![[!0; 2]; a.len()];
35        let mut root = !0;
36        #[cfg(verus_keep_ghost)]
37        proof_decl! {
38            let ghost mut ranges = Seq::<(usize, usize)>::empty();
39            let ghost mut spine = Seq::<usize>::empty();
40        }
41        #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
42            invariant
43                parents.len() == a.len(), children.len() == a.len(),
44                Self::prefix_tree(parents@, children@, root, ranges, spine, iter.index() as nat),
45                Self::lawful_order(a@) ==> Self::range_minima(a@, ranges),
46                Self::lawful_order(a@) ==> Self::range_boundaries(a@, ranges),
47        ))]
48        for i in 0..a.len() {
49            #[cfg(verus_keep_ghost)]
50            proof_decl! { let ghost mut cut = spine.len(); }
51            let mut prev = !0;
52            let mut parent = i.wrapping_sub(1);
53            #[cfg(verus_keep_ghost)]
54            proof! {
55                let maximum = usize::MAX;
56                assert(maximum <= !0usize) by(bit_vector);
57                assert(!0usize == usize::MAX);
58                if i > 0 { assert(parent == i - 1); }
59            }
60            #[cfg_attr(verus_keep_ghost, verus_spec(
61                invariant
62                    i < a.len(), parents.len() == a.len(), children.len() == a.len(),
63                    Self::prefix_tree(parents@, children@, root, ranges, spine, i as nat),
64                    Self::lawful_order(a@) ==> Self::range_minima(a@, ranges),
65                    Self::lawful_order(a@) ==> Self::range_boundaries(a@, ranges),
66                    Self::lawful_order(a@) && prev != !0usize ==>
67                        !Self::not_greater(a@[prev as int], a@[i as int]),
68                    cut <= spine.len(),
69                    parent == if cut == 0 { !0usize } else { spine[cut as int - 1] },
70                    prev == if cut == spine.len() { !0usize } else { spine[cut as int] },
71                decreases cut,
72            ))]
73            while parent != !0 && a[i] < a[parent] {
74                #[cfg(verus_keep_ghost)]
75                proof! { assert(Self::spine_link(parents@, children@, spine, cut as int - 1)); }
76                prev = parent;
77                parent = parents[parent];
78                #[cfg(verus_keep_ghost)]
79                proof! { cut = (cut - 1) as nat; }
80            }
81            #[cfg(verus_keep_ghost)]
82            proof_decl! {
83                let ghost old_ranges = ranges;
84                let ghost old_spine = spine;
85                let ghost old_children = children@;
86                let ghost old_parents = parents@;
87                let ghost old_root = root;
88                let ghost retained = spine.take(cut as int);
89            }
90            parents[i] = parent;
91            if parent != !0 {
92                children[parent][1] = i;
93            } else {
94                root = i;
95            }
96            if prev != !0 {
97                parents[prev] = i;
98                children[i][0] = prev;
99            }
100            #[cfg(verus_keep_ghost)]
101            proof! {
102                broadcast use vstd::seq_lib::lemma_seq_contains_after_push;
103                Self::spine_prefix(old_spine, cut, i as nat);
104                let left = if parent == !0usize { 0usize } else { (parent + 1) as usize };
105                ranges = Seq::new(i as nat, |node: int| (
106                    old_ranges[node].0,
107                    if retained.contains(node as usize) { (i + 1) as usize } else { old_ranges[node].1 },
108                )).push((left, (i + 1) as usize));
109                spine = retained.push(i);
110                assert(Self::prefix_tree(old_parents, old_children, old_root, old_ranges, old_spine, i as nat));
111                assert forall|node: int| 0 <= node < children.len() implies
112                    (#[trigger] children@[node])[0] == if node == i { prev } else { old_children[node][0] }
113                    && children@[node][1] == if node == parent { i } else { old_children[node][1] }
114                    && (#[trigger] parents@[node]) == if node == i { parent }
115                        else if node == prev { i } else { old_parents[node] } by {}
116                if parent != !0usize {
117                    assert(cut > 0);
118                    assert(retained.contains(parent));
119                }
120                if prev != !0usize {
121                    assert(cut < old_spine.len());
122                    assert(!retained.contains(prev));
123                }
124                assert forall|node: int| 0 <= node < i + 1 implies
125                    #[trigger] Self::node_range_valid(children@, ranges, node) by {
126                    if node < i {
127                        assert(Self::node_range_valid(old_children, old_ranges, node));
128                        let left_child = old_children[node][0];
129                        if left_child != !0usize {
130                            assert(old_ranges[left_child as int].1 == node);
131                            assert(!retained.contains(left_child));
132                        }
133                        let right_child = old_children[node][1];
134                        if node != parent && right_child != !0usize {
135                            Self::preserved_right_path(
136                                old_parents, old_children, old_root, old_ranges, old_spine, cut, node,
137                            );
138                        }
139                    } else if prev != !0usize {
140                        assert(old_ranges[prev as int].1 == i);
141                        if parent != !0usize {
142                            assert(Self::spine_link(old_parents, old_children, old_spine, cut as int - 1));
143                            assert(Self::node_range_valid(old_children, old_ranges, parent as int));
144                        } else {
145                            assert(old_root == prev);
146                        }
147                        assert(old_ranges[prev as int].0 == left);
148                    } else {
149                        assert(left == i);
150                    }
151                }
152                if cut > 0 {
153                    assert(retained.contains(old_root));
154                }
155                assert(Self::range_structure(root, children@, ranges));
156                if prev != !0usize {
157                    assert(Self::node_range_valid(children@, ranges, i as int));
158                    assert(old_ranges[prev as int] == ranges[prev as int]);
159                    assert(old_ranges[prev as int] == (left, i));
160                }
161                if Self::lawful_order(a@) {
162                    assert forall|node: int| 0 <= node < i + 1 implies
163                        #[trigger] Self::node_minimum(a@, ranges, node) by {
164                        if node < i {
165                            assert(Self::node_minimum(a@, old_ranges, node));
166                            if retained.contains(node as usize) {
167                                assert(node <= parent < i);
168                                assert(Self::node_range_valid(old_children, old_ranges, node));
169                                assert(old_ranges[node].0 <= parent < old_ranges[node].1);
170                                assert(Self::not_greater(a@[node], a@[parent as int]));
171                                assert(Self::not_greater(a@[parent as int], a@[i as int]));
172                                assert(Self::not_greater(a@[node], a@[i as int]));
173                                Self::append_minimum(a@, node, old_ranges[node].0 as int, i as int);
174                            }
175                        } else if prev != !0usize {
176                            assert(Self::node_minimum(a@, old_ranges, prev as int));
177                            assert(old_ranges[prev as int] == (left, i));
178                            assert(!Self::not_greater(a@[prev as int], a@[i as int]));
179                            assert(Self::node_range_valid(old_children, old_ranges, prev as int));
180                            Self::append_minimum(a@, prev as int, left as int, i as int);
181                        } else {
182                            assert(left == i);
183                            assert(Self::not_greater(a@[i as int], a@[i as int]));
184                        }
185                    }
186                    assert(Self::range_minima(a@, ranges));
187                    assert forall|node: int| 0 <= node < i + 1 implies
188                        #[trigger] Self::node_boundaries(a@, ranges, node) by {
189                        if node < i {
190                            assert(Self::node_boundaries(a@, old_ranges, node));
191                            assert(Self::node_range_valid(old_children, old_ranges, node));
192                            if old_ranges[node].1 == i && !retained.contains(node as usize) {
193                                assert(old_spine.contains(node as usize));
194                                assert(prev <= node < i);
195                                assert(Self::node_range_valid(old_children, old_ranges, prev as int));
196                                assert(Self::node_minimum(a@, ranges, i as int));
197                                assert(left <= node < i);
198                                assert(Self::not_greater(a@[i as int], a@[node]));
199                                assert(!Self::not_greater(a@[node], a@[i as int]));
200                            }
201                        } else if left > 0 {
202                            assert(parent != !0usize && left == parent + 1);
203                            assert(Self::not_greater(a@[parent as int], a@[i as int]));
204                        }
205                    }
206                    assert(Self::range_boundaries(a@, ranges));
207                }
208                assert forall|node: int| 0 <= node < i + 1 implies
209                    #[trigger] Self::node_parent_valid(parents@, children@, root, (i + 1) as nat, node) by {
210                    if node < i {
211                        assert(Self::node_parent_valid(old_parents, old_children, old_root, i as nat, node));
212                        if parent != !0usize {
213                            assert(Self::spine_link(old_parents, old_children, old_spine, cut as int - 1));
214                        }
215                        if prev != !0usize {
216                            assert(Self::spine_link(old_parents, old_children, old_spine, cut as int));
217                        }
218                    }
219                }
220                assert forall|index: int| 0 <= index < spine.len() implies
221                    #[trigger] Self::spine_link(parents@, children@, spine, index) by {
222                    if index < cut {
223                        assert(Self::spine_link(old_parents, old_children, old_spine, index));
224                        assert(retained.contains(old_spine[index]));
225                        if index + 1 < cut {
226                            assert(Self::spine_ordered(old_spine, index, cut as int - 1));
227                            assert(old_spine[index] != parent);
228                        } else {
229                            assert(old_spine[index] == parent);
230                        }
231                    } else {
232                        assert(spine[index] == i);
233                    }
234                }
235                assert forall|node: int| #![trigger parents@[node]] #![trigger children@[node]]
236                    i + 1 <= node < parents.len() implies parents@[node] == !0usize
237                        && children@[node][0] == !0usize && children@[node][1] == !0usize by {
238                    assert(old_parents[node] == !0usize);
239                    assert(old_children[node][0] == !0usize && old_children[node][1] == !0usize);
240                }
241                assert forall|node: int| 0 <= node < i + 1 implies
242                    (#[trigger] ranges[node]).1 == i + 1 <==> spine.contains(node as usize) by {
243                    if node < i {
244                        assert(Self::node_range_valid(old_children, old_ranges, node));
245                    }
246                }
247                assert forall|left: int, right: int| 0 <= left < right < spine.len() implies
248                    #[trigger] Self::spine_ordered(spine, left, right) by {
249                    if right < cut {
250                        assert(Self::spine_ordered(old_spine, left, right));
251                        assert(old_spine[left] < old_spine[right]);
252                    } else {
253                        assert(retained.contains(spine[left]));
254                    }
255                }
256                assert(ranges.len() == i + 1 && spine.len() == cut + 1 <= i + 1);
257                assert(spine[0] == root && spine.last() == i);
258                assert forall|index: int| 0 <= index < spine.len() implies
259                    #[trigger] spine[index] < i + 1 by {
260                    if index < cut {
261                        assert(retained.contains(spine[index]));
262                    }
263                }
264                assert(Self::prefix_tree(parents@, children@, root, ranges, spine, (i + 1) as nat)) by {}
265            }
266        }
267        #[cfg(verus_keep_ghost)]
268        proof! {
269            let result = Self { root, parents, children };
270            #[verifier::proof_note("For every input length and any PartialOrd comparison outcomes, CartesianTree::new creates reciprocal parent/child links; the root alone has the absent-parent marker and every child is on the correct side of its parent")]
271            assert(result.valid_parents());
272            if a.len() > 0 {
273                #[verifier::proof_note("For every nonempty input length and any PartialOrd comparison outcomes, CartesianTree::new constructs valid subtree intervals: the root spans the full input and each node's children split its interval around its own index")]
274                assert(result.valid_ranges(ranges));
275                if Self::lawful_order(a@) {
276                    #[verifier::proof_note("For every nonempty input whose stable comparison model defines a total preorder on its values, every constructed Cartesian subtree is rooted at the leftmost minimum of its interval")]
277                    assert(Self::range_minima(a@, ranges));
278                    #[verifier::proof_note("Under the same total-preorder assumption, every subtree interval is maximal for its leftmost minimum: a preceding boundary value is no greater than the root, and a following boundary value is strictly smaller than the root")]
279                    assert(Self::range_boundaries(a@, ranges));
280                }
281            }
282        }
283        Self {
284            root,
285            parents,
286            children,
287        }
288    }
289    #[cfg_attr(verus_keep_ghost, verus_spec(
290        requires exists|ranges: Seq<(usize, usize)>| #![auto]
291            self.valid_ranges(ranges) && Self::callbacks_defined(ranges, f),
292    ))]
293    pub fn with_ranges(&self, mut f: impl FnMut(usize, Range<usize>)) {
294        #[cfg(verus_keep_ghost)]
295        proof_decl! {
296            let ghost ranges = choose|ranges: Seq<(usize, usize)>| #![auto]
297                self.valid_ranges(ranges) && Self::callbacks_defined(ranges, f);
298            let ghost mut visited = Seq::<usize>::empty();
299        }
300        let mut stack = vec![(self.root, 0, self.parents.len())];
301        #[cfg(verus_keep_ghost)]
302        proof_decl! { let ghost mut before = stack@; }
303        #[cfg_attr(verus_keep_ghost, verus_spec(
304            invariant
305                self.valid_ranges(ranges), Self::callbacks_defined(ranges, f),
306                self.walk_partition(ranges, stack@, visited), before == stack@,
307                visited.len() <= self.parents.len(),
308            ensures stack.len() == 0,
309            decreases self.parents.len() - visited.len(),
310        ))]
311        while let Some((v, l, r)) = stack.pop() {
312            #[cfg(verus_keep_ghost)]
313            proof_decl! { let ghost seen = visited; }
314            #[cfg(verus_keep_ghost)]
315            proof! {
316                assert(Self::node_range_valid(self.children@, ranges, v as int));
317            }
318            f(v, l..r);
319            #[cfg(verus_keep_ghost)]
320            proof! {
321                assert(!visited.contains(v));
322                visited = visited.push(v);
323            }
324            if self.children[v][1] != !0 {
325                #[cfg(verus_keep_ghost)]
326                proof! {
327                    assert(Self::node_range_valid(self.children@, ranges, self.children@[v as int][1] as int));
328                }
329                stack.push((self.children[v][1], v + 1, r));
330            }
331            if self.children[v][0] != !0 {
332                #[cfg(verus_keep_ghost)]
333                proof! { assert(Self::node_range_valid(self.children@, ranges, self.children@[v as int][0] as int)); }
334                stack.push((self.children[v][0], l, v));
335            }
336            #[cfg(verus_keep_ghost)]
337            proof! {
338                broadcast use vstd::seq_lib::lemma_seq_contains_after_push;
339                assert forall|i: int| 0 <= i < stack.len() implies
340                    if i < before.len() - 1 {
341                        #[trigger] stack@[i] == before[i]
342                    } else {
343                        l <= (#[trigger] stack@[i]).1 < stack@[i].2 <= r
344                            && (stack@[i].2 <= v || v < stack@[i].1)
345                    } by {}
346                assert forall|node: usize| #[trigger] visited.contains(node) implies
347                    node < self.parents.len() && !Self::pending(stack@, node as int) by {
348                    if Self::pending(stack@, node as int) {
349                        let i = choose|i: int| 0 <= i < stack.len()
350                            && (#[trigger] stack@[i]).1 <= node < stack@[i].2;
351                        if seen.contains(node) {
352                            if i < before.len() - 1 {
353                                assert(before[i].1 <= node < before[i].2);
354                            } else {
355                                assert(before[before.len() - 1].1 <= node < before[before.len() - 1].2);
356                            }
357                            assert(Self::pending(before, node as int));
358                        } else if i < before.len() - 1 {
359                            assert(before[before.len() - 1].2 <= before[i].1);
360                        }
361                    }
362                }
363                assert forall|node: usize| node < self.parents.len() implies
364                    visited.contains(node) || #[trigger] Self::pending(stack@, node as int) by {
365                    if !visited.contains(node) {
366                        assert(Self::pending(before, node as int));
367                        let i = choose|i: int| 0 <= i < before.len()
368                            && (#[trigger] before[i]).1 <= node < before[i].2;
369                        if i + 1 < before.len() {
370                            assert(stack@[i] == before[i]);
371                        } else if node < v {
372                            assert(stack@[stack.len() - 1].1 <= node < stack@[stack.len() - 1].2);
373                        } else {
374                            let right = before.len() - 1;
375                            assert(stack@[right].1 <= node < stack@[right].2);
376                        }
377                    }
378                }
379                assert(visited.to_set().subset_of(Set::<usize>::range(0, self.parents.len())));
380                visited.unique_seq_to_set();
381                vstd::set_lib::lemma_len_subset(
382                    visited.to_set(), Set::<usize>::range(0, self.parents.len()),
383                );
384            }
385            #[cfg(verus_keep_ghost)]
386            proof! { before = stack@; }
387        }
388        #[cfg(verus_keep_ghost)]
389        proof! {
390            assert forall|node: usize| node < self.parents.len() implies
391                #[trigger] visited.contains(node) by {
392                assert(!Self::pending(stack@, node as int));
393            }
394            #[verifier::proof_note("On normal return, with_ranges has invoked its callback exactly once per node with that node's subtree interval, for every nonempty tree whose children split those intervals and whose callback precondition holds for every reported interval")]
395            assert(visited.no_duplicates()
396                && forall|node: usize| node < self.parents.len() <==> #[trigger] visited.contains(node));
397        }
398    }

403    impl CartesianTree {
404        pub open spec fn valid_ranges(&self, ranges: Seq<(usize, usize)>) -> bool {
405            &&& self.parents.len() == ranges.len() && self.children.len() == ranges.len()
406            &&& Self::range_structure(self.root, self.children@, ranges)
407        }
408
409        pub open spec fn range_structure(
410            root: usize, children: Seq<[usize; 2]>, ranges: Seq<(usize, usize)>,
411        ) -> bool {
412            let n = ranges.len();
413            &&& 0 < n <= usize::MAX && root < n && n <= children.len()
414            &&& ranges[root as int] == (0usize, n as usize)
415            &&& forall|node: int| 0 <= node < n ==>
416                #[trigger] Self::node_range_valid(children, ranges, node)
417        }
418
419        pub open spec fn node_range_valid(
420            children: Seq<[usize; 2]>, ranges: Seq<(usize, usize)>, node: int,
421        ) -> bool {
422            let n = ranges.len();
423            let (l, r) = ranges[node];
424            &&& l <= node < r <= n
425            &&& if children[node][0] == !0usize {
426                l == node
427            } else {
428                children[node][0] < n
429                    && ranges[children[node][0] as int] == (l, node as usize)
430            }
431            &&& if children[node][1] == !0usize {
432                r == node + 1
433            } else {
434                children[node][1] < n
435                    && ranges[children[node][1] as int] == ((node + 1) as usize, r)
436            }
437        }
438
439        pub open spec fn callbacks_defined<F>(ranges: Seq<(usize, usize)>, callback: F) -> bool
440            where F: FnMut(usize, Range<usize>),
441        {
442            forall|node: usize| node < ranges.len() ==>
443                #[trigger] call_requires(callback, (node, Range {
444                    start: ranges[node as int].0, end: ranges[node as int].1,
445                }))
446        }
447
448        pub proof fn unique_ranges(&self, first: Seq<(usize, usize)>, second: Seq<(usize, usize)>)
449            requires self.valid_ranges(first), self.valid_ranges(second),
450            ensures first == second,
451        {
452            assert forall|node: int| 0 <= node < first.len() implies
453                #[trigger] first[node] == second[node] by {
454                self.subtree_ranges_unique(first, second, self.root as int, node);
455            }
456            #[verifier::proof_note("For every nonempty Cartesian tree with valid subtree intervals, its root and child links determine exactly one interval for every node; this uniqueness needs no assumptions about the input ordering")]
457            assert(first =~= second);
458        }
459
460        pub open spec fn valid_parents(&self) -> bool {
461            let n = self.parents.len();
462            &&& self.children.len() == n
463            &&& if n == 0 { self.root == !0usize } else { self.root < n }
464            &&& forall|node: int| 0 <= node < n ==>
465                #[trigger] Self::node_parent_valid(self.parents@, self.children@, self.root, n as nat, node)
466        }
467
468        pub open spec fn node_parent_valid(
469            parents: Seq<usize>, children: Seq<[usize; 2]>, root: usize, n: nat, node: int,
470        ) -> bool {
471            let parent = parents[node];
472            let left = children[node][0];
473            let right = children[node][1];
474            &&& if node == root { parent == !0usize } else {
475                parent < n && (node < parent && children[parent as int][0] == node
476                    || parent < node && children[parent as int][1] == node)
477            }
478            &&& left != !0usize ==> left < node && parents[left as int] == node
479            &&& right != !0usize ==> node < right < n && parents[right as int] == node
480        }
481
482        pub open spec fn not_greater<T>(left: T, right: T) -> bool
483            where T: PartialOrd,
484        {
485            !right.is_lt(&left)
486        }
487
488        pub open spec fn lawful_order<T>(values: Seq<T>) -> bool
489            where T: PartialOrd,
490        {
491            &&& T::obeys_partial_cmp_spec()
492            &&& forall|i: int| 0 <= i < values.len() ==>
493                #[trigger] Self::not_greater(values[i], values[i])
494            &&& forall|i: int, j: int| 0 <= i < values.len() && 0 <= j < values.len() ==>
495                #[trigger] Self::not_greater(values[i], values[j])
496                    || #[trigger] Self::not_greater(values[j], values[i])
497            &&& forall|i: int, j: int, k: int|
498                0 <= i < values.len() && 0 <= j < values.len() && 0 <= k < values.len()
499                    && #[trigger] Self::not_greater(values[i], values[j])
500                    && #[trigger] Self::not_greater(values[j], values[k]) ==>
501                        Self::not_greater(values[i], values[k])
502        }
503
504        pub open spec fn range_minima<T>(values: Seq<T>, ranges: Seq<(usize, usize)>) -> bool
505            where T: PartialOrd,
506        {
507            ranges.len() <= values.len() && forall|node: int| 0 <= node < ranges.len() ==>
508                #[trigger] Self::node_minimum(values, ranges, node)
509        }
510
511        pub open spec fn node_minimum<T>(
512            values: Seq<T>, ranges: Seq<(usize, usize)>, node: int,
513        ) -> bool
514            where T: PartialOrd,
515        {
516            Self::interval_minimum(values, node, ranges[node].0 as int, ranges[node].1 as int)
517        }
518
519        pub open spec fn range_boundaries<T>(values: Seq<T>, ranges: Seq<(usize, usize)>) -> bool
520            where T: PartialOrd,
521        {
522            ranges.len() <= values.len() && forall|node: int| 0 <= node < ranges.len() ==>
523                #[trigger] Self::node_boundaries(values, ranges, node)
524        }
525
526        pub open spec fn node_boundaries<T>(
527            values: Seq<T>, ranges: Seq<(usize, usize)>, node: int,
528        ) -> bool
529            where T: PartialOrd,
530        {
531            let (left, right) = ranges[node];
532            &&& left > 0 ==> Self::not_greater(values[left as int - 1], values[node])
533            &&& right < ranges.len() ==> !Self::not_greater(values[node], values[right as int])
534        }
535
536        pub open spec fn interval_minimum<T>(values: Seq<T>, node: int, left: int, right: int) -> bool
537            where T: PartialOrd,
538        {
539            forall|index: int| left <= index < right ==>
540                #[trigger] Self::not_greater(values[node], values[index])
541                    && (index < node ==> !Self::not_greater(values[index], values[node]))
542        }

544        proof fn subtree_ranges_unique(
545            &self, first: Seq<(usize, usize)>, second: Seq<(usize, usize)>, root: int, node: int,
546        )
547            requires self.valid_ranges(first), self.valid_ranges(second),
548                0 <= root < first.len(), first[root] == second[root],
549                first[root].0 <= node < first[root].1,
550            ensures first[node] == second[node],
551            decreases first[root].1 - first[root].0,
552        {
553            assert(Self::node_range_valid(self.children@, first, root));
554            assert(Self::node_range_valid(self.children@, second, root));
555            if node < root {
556                let child = self.children@[root][0] as int;
557                self.subtree_ranges_unique(first, second, child, node);
558            } else if root < node {
559                let child = self.children@[root][1] as int;
560                self.subtree_ranges_unique(first, second, child, node);
561            }
562        }
563
564        proof fn append_minimum<T>(values: Seq<T>, node: int, left: int, right: int)
565            where T: PartialOrd,
566            requires
567                Self::lawful_order(values), 0 <= left <= node < right < values.len(),
568                Self::interval_minimum(values, node, left, right),
569            ensures
570                Self::not_greater(values[node], values[right]) ==>
571                    Self::interval_minimum(values, node, left, right + 1),
572                !Self::not_greater(values[node], values[right]) ==>
573                    Self::interval_minimum(values, right, left, right + 1),
574        {
575            if Self::not_greater(values[node], values[right]) {
576                assert forall|index: int| left <= index < right + 1 implies
577                    #[trigger] Self::not_greater(values[node], values[index])
578                        && (index < node ==> !Self::not_greater(values[index], values[node])) by {}
579            } else {
580                assert(Self::not_greater(values[right], values[node]));
581                assert forall|index: int| left <= index < right + 1 implies
582                    #[trigger] Self::not_greater(values[right], values[index])
583                        && (index < right ==> !Self::not_greater(values[index], values[right])) by {
584                    if index < right {
585                        assert(Self::not_greater(values[node], values[index]));
586                        assert(Self::not_greater(values[right], values[index]));
587                        assert(!Self::not_greater(values[index], values[right]));
588                    } else {
589                        assert(Self::not_greater(values[right], values[right]));
590                    }
591                }
592            }
593        }

595        spec fn pending(stack: Seq<(usize, usize, usize)>, node: int) -> bool {
596            exists|i: int| 0 <= i < stack.len()
597                && (#[trigger] stack[i]).1 <= node < stack[i].2
598        }
599
600        spec fn walk_partition(
601            &self, ranges: Seq<(usize, usize)>, stack: Seq<(usize, usize, usize)>,
602            visited: Seq<usize>,
603        ) -> bool {
604            &&& visited.no_duplicates()
605            &&& forall|node: usize| #[trigger] visited.contains(node) ==>
606                node < self.parents.len() && !Self::pending(stack, node as int)
607            &&& forall|node: usize| node < self.parents.len() ==>
608                visited.contains(node) || #[trigger] Self::pending(stack, node as int)
609            &&& forall|i: int| 0 <= i < stack.len() ==> {
610                let (node, l, r) = #[trigger] stack[i];
611                node < ranges.len() && ranges[node as int] == (l, r)
612            }
613            &&& forall|i: int, j: int| 0 <= i < j < stack.len() ==>
614                (#[trigger] stack[j]).2 <= (#[trigger] stack[i]).1
615        }
616
617        spec fn prefix_tree(
618            parents: Seq<usize>, children: Seq<[usize; 2]>, root: usize,
619            ranges: Seq<(usize, usize)>, spine: Seq<usize>, processed: nat,
620        ) -> bool {
621            &&& parents.len() == children.len() && processed <= parents.len()
622            &&& ranges.len() == processed && spine.len() <= processed
623            &&& forall|node: int| #![trigger parents[node]] #![trigger children[node]]
624                processed <= node < parents.len() ==> parents[node] == !0usize
625                    && children[node][0] == !0usize && children[node][1] == !0usize
626            &&& if processed == 0 {
627                root == !0usize && spine.len() == 0
628            } else {
629                &&& Self::range_structure(root, children, ranges)
630                &&& forall|node: int| 0 <= node < processed ==>
631                    #[trigger] Self::node_parent_valid(parents, children, root, processed, node)
632                &&& spine.len() > 0 && spine[0] == root && spine.last() == processed - 1
633                &&& forall|index: int| 0 <= index < spine.len() ==> #[trigger] spine[index] < processed
634                &&& forall|index: int| 0 <= index < spine.len() ==>
635                    #[trigger] Self::spine_link(parents, children, spine, index)
636                &&& forall|left: int, right: int| 0 <= left < right < spine.len() ==>
637                    #[trigger] Self::spine_ordered(spine, left, right)
638                &&& forall|node: int| 0 <= node < processed ==>
639                    ((#[trigger] ranges[node]).1 == processed <==> spine.contains(node as usize))
640            }
641        }
642
643        spec fn spine_link(
644            parents: Seq<usize>, children: Seq<[usize; 2]>, spine: Seq<usize>, index: int,
645        ) -> bool {
646            let node = spine[index];
647            &&& parents[node as int] == if index == 0 { !0usize } else { spine[index - 1] }
648            &&& children[node as int][1] == if index + 1 == spine.len() {
649                !0usize
650            } else { spine[index + 1] }
651        }
652
653        spec fn spine_ordered(spine: Seq<usize>, left: int, right: int) -> bool {
654            spine[left] < spine[right]
655        }
656
657        proof fn spine_prefix(spine: Seq<usize>, cut: nat, processed: nat)
658            requires
659                cut <= spine.len(),
660                forall|index: int| 0 <= index < spine.len() ==> #[trigger] spine[index] < processed,
661                forall|left: int, right: int| 0 <= left < right < spine.len() ==>
662                    #[trigger] Self::spine_ordered(spine, left, right),
663            ensures
664                forall|index: int| 0 <= index < spine.len() ==>
665                    (#[trigger] spine.take(cut as int).contains(spine[index])) == (index < cut),
666                forall|node: usize| #[trigger] spine.take(cut as int).contains(node) ==>
667                    spine.contains(node) && node < processed && cut > 0 && node <= spine[cut as int - 1],
668                forall|node: usize| #[trigger] spine.contains(node)
669                    && !spine.take(cut as int).contains(node) ==>
670                    cut < spine.len() && spine[cut as int] <= node < processed,
671        {
672            let retained = spine.take(cut as int);
673            assert forall|index: int| 0 <= index < spine.len() implies
674                (#[trigger] retained.contains(spine[index])) == (index < cut) by {
675                if index < cut {
676                    assert(retained[index] == spine[index]);
677                } else if retained.contains(spine[index]) {
678                    let k = choose|k: int| 0 <= k < retained.len() && retained[k] == spine[index];
679                    assert(spine[k] == spine[index]);
680                    assert(Self::spine_ordered(spine, k, index));
681                }
682            }
683            assert forall|node: usize| #[trigger] retained.contains(node) implies
684                spine.contains(node) && node < processed && cut > 0 && node <= spine[cut as int - 1] by {
685                let index = choose|index: int| 0 <= index < retained.len() && retained[index] == node;
686                assert(spine[index] == node);
687                if index + 1 < cut {
688                    assert(Self::spine_ordered(spine, index, cut as int - 1));
689                }
690            }
691            assert forall|node: usize| #[trigger] spine.contains(node) && !retained.contains(node) implies
692                cut < spine.len() && spine[cut as int] <= node < processed by {
693                let index = choose|index: int| 0 <= index < spine.len() && spine[index] == node;
694                assert(index >= cut);
695                if index > cut {
696                    assert(Self::spine_ordered(spine, cut as int, index));
697                }
698            }
699        }
700
701        proof fn preserved_right_path(
702            parents: Seq<usize>, children: Seq<[usize; 2]>, root: usize,
703            ranges: Seq<(usize, usize)>, spine: Seq<usize>, cut: nat, node: int,
704        )
705            requires
706                Self::prefix_tree(parents, children, root, ranges, spine, ranges.len()),
707                cut <= spine.len(), 0 <= node < ranges.len(),
708                node != if cut == 0 { !0usize } else { spine[cut as int - 1] },
709                children[node][1] != !0usize,
710            ensures
711                spine.take(cut as int).contains(node as usize)
712                    == spine.take(cut as int).contains(children[node][1]),
713        {
714            let retained = spine.take(cut as int);
715            let right = children[node][1];
716            Self::spine_prefix(spine, cut, ranges.len());
717            assert(Self::node_range_valid(children, ranges, node));
718            if spine.contains(node as usize) {
719                let index = choose|index: int| 0 <= index < spine.len() && spine[index] == node;
720                assert(Self::spine_link(parents, children, spine, index));
721                assert(index + 1 < spine.len());
722                assert(spine[index + 1] == right);
723                assert(index + 1 != cut);
724            } else {
725                assert(!spine.contains(right));
726            }
727        }
728    }

828    use crate::{

834
835    #[test]
836    fn test_cartesian_tree() {
837        const Q: usize = 1000;
838        const N: usize = 100;
839        const A: i64 = 100;
840        let mut rng = Xorshift::default();
841        let arrays = exhaustive_sequences(-1..=1, 0..=6).chain((0..Q).map(|_| {
842            let n = rng.random(1..=N);
843            rng.random_iter(0..A).take(n).collect()
844        }));
845        for a in arrays {
846            let n = a.len();
847            let mut seg = SegmentTree::<MinOperation<_>>::from_vec(
848                a.iter().enumerate().map(|(i, &a)| (a, i)).collect(),
849            );
850            let mut parents = vec![!0; n];
851            let mut root = !0;
852            let mut children = vec![[!0; 2]; n];
853            let mut ranges = vec![];
854            crecurse!(
855                unsafe fn dfs(l: usize, r: usize, p: usize, ci: usize) {
856                    if l >= r {
857                        return;
858                    }
859                    let m = seg.fold(l..r).1;
860                    ranges.push((m, l..r));
861                    if p == !0 {
862                        root = m;
863                    } else {
864                        parents[m] = p;
865                        children[p][ci] = m;
866                    }
867                    seg.set(m, (i64::MAX, usize::MAX));
868                    dfs!(l, m, m, 0);
869                    dfs!(m + 1, r, m, 1);
870                }
871            )(0, n, !0, !0);
872
873            let ct = CartesianTree::new(&a);
874            assert_eq!(ct.root, root);
875            assert_eq!(ct.parents, parents);
876            assert_eq!(ct.children, children);
877            let mut ct_ranges = vec![];
878            if !a.is_empty() {
879                ct.with_ranges(|v, range| {
880                    ct_ranges.push((v, range));
881                });
882            }
883            assert_eq!(ct_ranges, ranges);
884        }
885    }
