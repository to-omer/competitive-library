4/// Enumerates a finite monoid defined by word equations using the HLT strategy.
5#[derive(Debug, Clone)]
6pub struct ToddCoxeter {
7    relations: Vec<(Vec<usize>, Vec<usize>)>,
8    table: Vec<Vec<usize>>,
9    uf: UnionFind,
10}
11
12impl ToddCoxeter {
13    /// Generators are numbered `0..alphabet_size`; the empty word is the identity.
14    pub fn new(alphabet_size: usize) -> Self {
15        Self {
16            relations: Vec::new(),
17            table: vec![vec![usize::MAX; alphabet_size]],
18            uf: UnionFind::new(1),
19        }
20    }
21
22    /// Adds an equation valid in every left and right context, not a directed rewrite.
23    pub fn add_relation<I, J>(&mut self, lhs: I, rhs: J)
24    where
25        I: IntoIterator<Item = usize>,
26        J: IntoIterator<Item = usize>,
27    {
28        let lhs: Vec<_> = lhs.into_iter().collect();
29        let rhs: Vec<_> = rhs.into_iter().collect();
30        assert!(lhs.iter().chain(&rhs).all(|&c| c < self.table[0].len()));
31        self.relations.push((lhs, rhs));
32    }
33
34    /// Returns the right generator transitions `table[state][generator]`.
35    /// State 0 is the identity; states follow breadth-first order with ascending generators.
36    ///
37    /// The presented monoid must be finite.
38    ///
39    /// With P created states, K generators, and L the sum of `1 + lhs.len() + rhs.len()`
40    /// over the equations, time is O(P(1 + K + L) alpha(P)) and space O(P(1 + K) + L).
41    /// P can greatly exceed the final number of states and depends on equation order.
42    ///
43    /// ```
44    /// use competitive::algorithm::ToddCoxeter;
45    /// let mut tc = ToddCoxeter::new(2);
46    /// tc.add_relation([0, 0], []);
47    /// tc.add_relation([1, 1, 1], []);
48    /// tc.add_relation([0, 1, 0, 1], []);
49    /// let table = tc.solve();
50    /// assert_eq!(table.len(), 6);
51    /// assert_eq!([1, 0, 1].into_iter().fold(0, |q, c| table[q][c]), table[0][0]);
52    /// ```
53    pub fn solve(mut self) -> Vec<Vec<usize>> {
54        let relations = take(&mut self.relations);
55        let mut stack = Vec::new();
56        let alphabet_size = self.table[0].len();
57        let mut state = 0;
58        while state < self.table.len() {
59            if self.uf.find_root(state) == state {
60                for (lhs, rhs) in &relations {
61                    let (lc, lp) = lhs
62                        .split_last()
63                        .map_or((None, &[][..]), |(&c, p)| (Some(c), p));
64                    let (rc, rp) = rhs
65                        .split_last()
66                        .map_or((None, &[][..]), |(&c, p)| (Some(c), p));
67                    let left = self.follow(state, lp);
68                    let right = self.follow(state, rp);
69                    // Both prefixes must exist before reading either last edge (e.g. a = aa).
70                    let x = lc.map_or(left, |c| self.table[left][c]);
71                    let y = rc.map_or(right, |c| self.table[right][c]);
72                    if x == usize::MAX {
73                        let y = if y == usize::MAX {
74                            self.define(right, rc.unwrap())
75                        } else {
76                            y
77                        };
78                        self.table[left][lc.unwrap()] = y;
79                    } else if y == usize::MAX {
80                        self.table[right][rc.unwrap()] = x;
81                    } else {
82                        self.merge(x, y, &mut stack);
83                    }
84                }
85                for c in 0..alphabet_size {
86                    self.define(state, c);
87                }
88            }
89            state += 1;
90        }
91
92        let identity = self.uf.find_root(0);
93        let mut order = vec![identity];
94        let mut index = vec![usize::MAX; self.table.len()];
95        index[identity] = 0;
96        let mut table = Vec::new();
97        let mut i = 0;
98        while i < order.len() {
99            let mut row = Vec::with_capacity(alphabet_size);
100            for &target in &self.table[order[i]] {
101                let target = self.uf.find_root(target);
102                if index[target] == usize::MAX {
103                    index[target] = order.len();
104                    order.push(target);
105                }
106                row.push(index[target]);
107            }
108            table.push(row);
109            i += 1;
110        }
111        table
112    }
113
114    fn define(&mut self, state: usize, c: usize) -> usize {
115        let state = self.uf.find_root(state);
116        if self.table[state][c] == usize::MAX {
117            assert!(self.table.len() < i32::MAX as usize);
118            self.table[state][c] = self.table.len();
119            self.table.push(vec![usize::MAX; self.table[0].len()]);
120            self.uf.push();
121        }
122        self.uf.find_root(self.table[state][c])
123    }
124
125    fn follow(&mut self, state: usize, word: &[usize]) -> usize {
126        let mut state = self.uf.find_root(state);
127        for &c in word {
128            state = self.define(state, c);
129        }
130        state
131    }
132
133    fn merge(&mut self, x: usize, y: usize, stack: &mut Vec<(usize, usize)>) {
134        stack.push((x, y));
135        while let Some((x, y)) = stack.pop() {
136            let mut x = self.uf.find_root(x);
137            let mut y = self.uf.find_root(y);
138            if !self.uf.unite(x, y) {
139                continue;
140            }
141            if self.uf.find_root(x) == y {
142                swap(&mut x, &mut y);
143            }
144            for c in 0..self.table[0].len() {
145                let a = self.table[x][c];
146                let b = self.table[y][c];
147                if a == usize::MAX {
148                    self.table[x][c] = b;
149                } else if b != usize::MAX {
150                    stack.push((a, b));
151                }
152            }
153        }
154    }
155}

167    fn transformation_table(
168        degree: usize,
169        generators: &[Vec<usize>],
170    ) -> (Vec<Vec<usize>>, Vec<Vec<usize>>) {
171        let identity: Vec<_> = (0..degree).collect();
172        let mut index = HashMap::from([(identity.clone(), 0)]);
173        let mut elements = vec![identity];
174        let mut words = vec![vec![]];
175        let mut table = Vec::new();
176        let mut state = 0;
177        while state < elements.len() {
178            let mut row = Vec::new();
179            for (c, generator) in generators.iter().enumerate() {
180                let next: Vec<_> = elements[state].iter().map(|&x| generator[x]).collect();
181                let target = *index.entry(next.clone()).or_insert_with(|| {
182                    let target = elements.len();
183                    elements.push(next);
184                    let mut word = words[state].clone();
185                    word.push(c);
186                    words.push(word);
187                    target
188                });
189                row.push(target);
190            }
191            table.push(row);
192            state += 1;
193        }
194        (table, words)
195    }

198    fn test_todd_coxeter_cyclic_monoid() {
199        for period_end in 1..=128 {
200            for cycle_start in 0..period_end {
201                let mut tc = ToddCoxeter::new(1);
202                let mut lhs = vec![0; cycle_start];
203                let mut rhs = vec![0; period_end];
204                if (cycle_start + period_end) % 2 == 0 {
205                    swap(&mut lhs, &mut rhs);
206                }
207                tc.add_relation(lhs, rhs);
208                let table = tc.solve();
209                let expected: Vec<_> = (0..period_end)
210                    .map(|i| {
211                        let target = if i + 1 == period_end {
212                            cycle_start
213                        } else {
214                            i + 1
215                        };
216                        vec![target]
217                    })
218                    .collect();
219                assert_eq!(table, expected);
220            }
221        }
222    }

224    #[test]
225    fn test_todd_coxeter_transformation_monoid() {
226        let mut rng = Xorshift::default();
227        let maps: Vec<_> = exhaustive_sequences(0..3, 3..=3).collect();
228        let small = exhaustive_sequences(0..maps.len(), 0..=2)
229            .map(|ids| (3, ids.into_iter().map(|i| maps[i].clone()).collect()));
230        let full = (2..=4).map(|n| {
231            let cycle = (0..n).map(|i| (i + 1) % n).collect();
232            let mut transposition: Vec<_> = (0..n).collect();
233            transposition.swap(0, 1);
234            let mut projection: Vec<_> = (0..n).collect();
235            projection[1] = 0;
236            (n, vec![cycle, transposition, projection])
237        });
238        for (degree, generators) in small.chain(full) {
239            let (expected, words) = transformation_table(degree, &generators);
240            let mut relations = Vec::new();
241            for (state, row) in expected.iter().enumerate() {
242                for (c, &target) in row.iter().enumerate() {
243                    let mut word = words[state].clone();
244                    word.push(c);
245                    relations.push((word, words[target].clone()));
246                }
247            }
248            // Redundant equations exercise overlapping prefixes and different scan orders.
249            for word in exhaustive_sequences(0..generators.len(), 0..=3) {
250                let target = word.iter().fold(0, |q, &c| expected[q][c]);
251                relations.push((word, words[target].clone()));
252            }
253            rng.shuffle(&mut relations);
254            let mut tc = ToddCoxeter::new(generators.len());
255            for (mut lhs, mut rhs) in relations {
256                if rng.gen_bool(0.5) {
257                    swap(&mut lhs, &mut rhs);
258                }
259                tc.add_relation(lhs, rhs);
260            }
261            assert_eq!(tc.solve(), expected);
262        }
263    }

266    fn test_todd_coxeter_symmetric_group() {
267        for n in 1..=6 {
268            let generators: Vec<_> = (0..n - 1)
269                .map(|i| {
270                    let mut permutation: Vec<_> = (0..n).collect();
271                    permutation.swap(i, i + 1);
272                    permutation
273                })
274                .collect();
275            let (expected, _) = transformation_table(n, &generators);
276            let mut tc = ToddCoxeter::new(generators.len());
277            for i in 0..generators.len() {
278                tc.add_relation([i, i], []);
279                for j in i + 1..generators.len() {
280                    if j == i + 1 {
281                        tc.add_relation([i, j, i], [j, i, j]);
282                    } else {
283                        tc.add_relation([i, j], [j, i]);
284                    }
285                }
286            }
287            assert_eq!(tc.solve(), expected);
288        }
289    }

300    #[test]
301    fn test_todd_coxeter_counting() {
302        let generators = [
303            vec![vec![0], vec![0]],
304            vec![vec![1, 0, 2], vec![1, 2, 0]],
305            vec![vec![1, 0, 3, 2], vec![1, 2, 0, 3]],
306            vec![vec![1, 0, 2, 3], vec![0, 2, 3, 1]],
307            vec![vec![1, 0, 3, 2, 4], vec![0, 2, 4, 3, 1]],
308        ];
309        for (i, generators) in generators.iter().enumerate() {
310            let (expected, _) = transformation_table(generators[0].len(), generators);
311            let mut tc = ToddCoxeter::new(2);
312            tc.add_relation([0, 0], []);
313            tc.add_relation([1, 1, 1], []);
314            tc.add_relation((0..=i).flat_map(|_| [0, 1]), []);
315            let table = tc.solve();
316            assert_eq!(table, expected);
317
318            let target = [0, 1, 0].into_iter().fold(0, |q, c| table[q][c]);
319            let mut dp = vec![M::new(0); table.len()];
320            dp[0] = M::new(1);
321            let mut terms = Vec::new();
322            for n in 0..1000 {
323                if n < 2 * table.len() {
324                    terms.push(dp[target]);
325                }
326                let mut next = vec![M::new(0); table.len()];
327                for (row, &count) in table.iter().zip(&dp) {
328                    for &q in row {
329                        next[q] += count;
330                    }
331                }
332                dp = next;
333            }
334            assert_eq!(Fps998244353::kth_term(terms, 1000), dp[target]);
335            if table.len() == 1 {
336                assert_eq!(dp[0], M::new(2).pow(1000));
337            }
338        }
339    }
