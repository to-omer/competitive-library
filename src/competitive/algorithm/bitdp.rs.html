4pub trait BitDpExt:
5    Sized
6    + Copy
7    + Default
8    + PartialEq
9    + Eq
10    + PartialOrd
11    + Ord
12    + Not<Output = Self>
13    + BitAnd<Output = Self>
14    + BitOr<Output = Self>
15    + BitXor<Output = Self>
16    + Shl<usize, Output = Self>
17    + Shr<usize, Output = Self>
18    + Add<Output = Self>
19    + Sub<Output = Self>
20    + Div<Output = Self>
21    + Zero
22    + One
23{
24    fn contains(self, x: usize) -> bool {
25        self & (Self::one() << x) != Self::zero()
26    }
27    fn insert(self, x: usize) -> Self {
28        self | (Self::one() << x)
29    }
30    fn remove(self, x: usize) -> Self {
31        self & !(Self::one() << x)
32    }
33    fn is_subset(self, elements: Self) -> bool {
34        self & elements == elements
35    }
36    fn is_superset(self, elements: Self) -> bool {
37        elements.is_subset(self)
38    }
39    fn subsets(self) -> Subsets<Self> {
40        Subsets {
41            mask: self,
42            cur: Some(self),
43        }
44    }
45    fn combinations(n: usize, k: usize) -> Combinations<Self> {
46        Combinations {
47            mask: Self::one() << n,
48            cur: Some((Self::one() << k) - Self::one()),
49        }
50    }
51}

60#[derive(Debug, Clone)]
61pub struct Subsets<T> {
62    mask: T,
63    cur: Option<T>,
64}
65
66impl<T> Iterator for Subsets<T>
67where
68    T: BitDpExt,
69{
70    type Item = T;
71    fn next(&mut self) -> Option<Self::Item> {
72        if let Some(cur) = self.cur {
73            self.cur = if cur.is_zero() {
74                None
75            } else {
76                Some((cur - T::one()) & self.mask)
77            };
78            Some(cur)
79        } else {
80            None
81        }
82    }
83}

85#[derive(Debug, Clone)]
86pub struct Combinations<T> {
87    mask: T,
88    cur: Option<T>,
89}
90
91impl<T> Iterator for Combinations<T>
92where
93    T: BitDpExt,
94{
95    type Item = T;
96    fn next(&mut self) -> Option<Self::Item> {
97        if let Some(cur) = self.cur {
98            if cur < self.mask {
99                self.cur = if cur == T::zero() {
100                    None
101                } else {
102                    let x = cur & (!cur + T::one());
103                    let y = cur + x;
104                    Some(((cur & !y) / x / (T::one() + T::one())) | y)
105                };
106                Some(cur)
107            } else {
108                None
109            }
110        } else {
111            None
112        }
113    }
114}

117mod verification {
118    use super::*;

120    #[kani::proof]
121    fn proof_subset_step() {
122        let mask: u64 = kani::any();
123        let current: u64 = kani::any();
124        let candidate: u64 = kani::any();
125        kani::assume(current & mask == current);
126        kani::assert(
127            mask.subsets().next() == Some(mask),
128            "For every u64 mask, subset enumeration starts with the entire mask",
129        );
130        // Every submask is a reachable iterator state; verify a transition without enumerating 2^64 states.
131        let mut iter = Subsets {
132            mask,
133            cur: Some(current),
134        };
135        kani::assert(
136            iter.next() == Some(current),
137            "For every u64 mask and current submask, the iterator yields that submask",
138        );
139        match iter.next() {
140            Some(next) => {
141                kani::assert(
142                    next < current && next & mask == next,
143                    "Each u64 subset step strictly decreases while remaining a submask",
144                );
145                kani::assert(
146                    !(candidate < current && candidate & mask == candidate) || candidate <= next,
147                    "Each u64 subset step chooses the greatest smaller submask, so it skips none",
148                );
149            }
150            None => kani::assert(
151                current == 0 && iter.next().is_none(),
152                "u64 subset enumeration ends exactly after zero and stays exhausted",
153            ),
154        }
155        kani::cover!(mask == u64::MAX && current == 1 << 63);
156        kani::cover!(mask == 0);
157    }
158
159    #[kani::proof]
160    fn proof_combination_start() {
161        let n: usize = kani::any();
162        let k: usize = kani::any();
163        let candidate: u64 = kani::any();
164        kani::assume(k <= n && n < u64::BITS as usize);
165        let first = u64::combinations(n, k).next().unwrap();
166        kani::assert(
167            first < 1u64 << n && first.count_ones() as usize == k,
168            "For every 0 <= k <= n < 64, u64 combination enumeration starts with a valid k-bit selection among n bits",
169        );
170        kani::assert(
171            candidate.count_ones() as usize != k || first <= candidate,
172            "The first u64 combination is the smallest integer with the requested number of selected bits",
173        );
174        kani::cover!(n == 63 && k == 63);
175        kani::cover!(n == 0 && k == 0);
176    }
177
178    #[kani::proof]
179    fn proof_combination_step() {
180        let n: usize = kani::any();
181        let current: u64 = kani::any();
182        let candidate: u64 = kani::any();
183        kani::assume(n < u64::BITS as usize);
184        let mask = 1u64 << n;
185        kani::assume(current < mask);
186        // A valid state of any fixed-popcount enumeration on at most sixty-three bits.
187        let mut iter = Combinations {
188            mask,
189            cur: Some(current),
190        };
191        kani::assert(
192            iter.next() == Some(current),
193            "For every u64 combination state on 0 through 63 bits, the iterator yields the current combination",
194        );
195        match iter.next() {
196            Some(next) => {
197                kani::assert(
198                    current < next && next < mask && next.count_ones() == current.count_ones(),
199                    "Every next u64 combination on at most 63 bits increases and preserves the number of selected bits",
200                );
201                kani::assert(
202                    !(current < candidate
203                        && candidate < mask
204                        && candidate.count_ones() == current.count_ones())
205                        || next <= candidate,
206                    "The next u64 combination is the smallest larger value with the same popcount, so it skips none",
207                );
208            }
209            None => kani::assert(
210                !(current < candidate
211                    && candidate < mask
212                    && candidate.count_ones() == current.count_ones())
213                    && iter.next().is_none(),
214                "u64 combination enumeration ends exactly when no larger combination remains and stays exhausted",
215            ),
216        }
217        kani::cover!(n == 0);
218        kani::cover!(n == 63 && current == 0b0101011);
219        kani::cover!(n == 63 && current == (1u64 << 63) - 1);
220        kani::cover!(n == 63 && current == 1u64 << 62);
221    }
222}

224#[cfg(test)]
225mod tests {
226    use super::*;
227    use crate::tools::Xorshift;
228    use std::collections::BTreeSet;
229
230    #[test]
231    fn test_bit_operations() {
232        let mut rng = Xorshift::default();
233        let pairs: Vec<_> = (0..256usize)
234            .flat_map(|a| (0..256).map(move |b| (a, b)))
235            .chain(
236                rng.random_iter((0..1usize << 12, 0..1usize << 12))
237                    .take(1000),
238            )
239            .collect();
240        for (a, b) in pairs {
241            let set: BTreeSet<_> = (0..12).filter(|&i| a >> i & 1 != 0).collect();
242            let other: BTreeSet<_> = (0..12).filter(|&i| b >> i & 1 != 0).collect();
243            for i in 0..12 {
244                assert_eq!(a.contains(i), set.contains(&i));
245                let mut inserted = set.clone();
246                inserted.insert(i);
247                assert_eq!(a.insert(i), inserted.iter().map(|i| 1 << i).sum());
248                let mut removed = set.clone();
249                removed.remove(&i);
250                assert_eq!(a.remove(i), removed.iter().map(|i| 1 << i).sum());
251            }
252            assert_eq!(a.is_subset(b), other.is_subset(&set));
253            assert_eq!(a.is_superset(b), other.is_superset(&set));
254        }
255        for a in 0..1usize << 12 {
256            let mut subsets: Vec<_> = a.subsets().collect();
257            subsets.sort();
258            assert_eq!(subsets, (0..=a).filter(|x| x & a == *x).collect::<Vec<_>>());
259        }
260        for n in 0..=12 {
261            for k in 0..=n {
262                let expected: Vec<_> = (0..1usize << n)
263                    .filter(|x| x.count_ones() as usize == k)
264                    .collect();
265                assert_eq!(usize::combinations(n, k).collect::<Vec<_>>(), expected);
266            }
267        }
268    }
