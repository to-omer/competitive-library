205    pub unsafe fn index_to_key_unchecked(&self, index: usize) -> RangeInclusive<usize> {
206        #[cfg(verus_keep_ghost)]
207        proof! { self.index_bounds(); self.index_value_properties(index as int); }
208        if index < self.sqrt {
209            #[cfg(verus_keep_ghost)]
210            proof! {
211                let value = index + 1;
212                floor_membership(self.num, value as usize);
213                assert forall|key: int| 1 <= key <= self.num implies
214                    ((self.num as int / (value + 1) < key <= self.num as int / value)
215                        == (#[trigger] (self.num as int / key) == self.value_at(index as int))) by {
216                    quotient_interval(self.num as int, value, key);
217                }
218            }
219            self.num / (index + 2) + 1..=self.num / (index + 1)
220        } else {
221            let key = self.len() - index;
222            #[cfg(verus_keep_ghost)]
223            proof! {
224                assert forall|k: int| 1 <= k <= self.num implies
225                    ((k == key) == (#[trigger] (self.num as int / k) == self.value_at(index as int))) by {
226                    if self.num as int / k == self.value_at(index as int) {
227                        quotient_at_least(self.num as int, k, self.sqrt + 1);
228                        small_floor_quotient(self.num as int, self.sqrt as int, k);
229                        floor_reciprocal(self.num as int, k);
230                    }
231                }
232            }
233            key..=key
234        }

289    pub unsafe fn value_to_key_unchecked(&self, value: usize) -> RangeInclusive<usize> {
290        #[cfg(verus_keep_ghost)]
291        proof! { floor_membership(self.num, value); }
292        if value == usize::MAX {
293            #[cfg(verus_keep_ghost)]
294            proof! {
295                assert forall|k: int| 1 <= k <= self.num implies
296                    ((k == 1) == (#[trigger] (self.num as int / k) == value)) by {
297                    if k > 1 { vstd::arithmetic::div_mod::lemma_div_decreases(self.num as int, k); }
298                }
299                #[verifier::proof_note("The maximum usize quotient occurs only for numerator usize::MAX and key 1")]
300                assert(self.num == usize::MAX && forall|k: int| 1 <= k <= self.num ==>
301                    ((k == 1) == (#[trigger] (self.num as int / k) == value)));
302            }
303            return 1..=1;
304        }
305        let start = (self.num / (value + 1)) + 1;
306        let end = self.num / value;
307        #[cfg(verus_keep_ghost)]
308        proof! {
309            vstd::arithmetic::div_mod::lemma_div_nonincreasing(self.num as int, value as int);
310            assert forall|k: int| 1 <= k <= self.num implies
311                ((start <= k <= end) == (#[trigger] (self.num as int / k) == value)) by {
312                quotient_interval(self.num as int, value as int, k);
313            }
314            #[verifier::proof_note("For every present floor quotient below usize::MAX, value_to_key_unchecked returns a nonempty interval containing exactly all keys producing that quotient")]
315            assert(1 <= start <= end <= self.num && forall|k: int| 1 <= k <= self.num ==>
316                ((start <= k <= end) == (#[trigger] (self.num as int / k) == value)));
317        }
318        start..=end
319    }

504#[cfg(verus_keep_ghost)]
505verus! {
506    impl FloorQuotientIndex {
507        pub closed spec fn index_invariant(&self) -> bool {
508            self.num > 0 && self.sqrt > 0
509            && self.sqrt as int * self.sqrt as int <= self.num
510            && self.num < (self.sqrt + 1) * (self.sqrt + 1)
511            && self.pivot == self.num as int / self.sqrt as int
512        }
513
514        pub closed spec fn index_len(&self) -> int {
515            2 * self.sqrt - if self.pivot == self.sqrt { 1int } else { 0int }
516        }
517
518        pub closed spec fn value_at(&self, index: int) -> int
519            recommends self.index_invariant(), 0 <= index < self.index_len(),
520        {
521            if index < self.sqrt { index + 1 } else { self.num as int / (self.index_len() - index) }
522        }
523
524        proof fn index_bounds(&self)
525            requires self.index_invariant(),
526            ensures
527                2 * self.sqrt <= usize::MAX,
528                0 < self.index_len() <= usize::MAX,
529                self.sqrt <= self.index_len() <= 2 * self.sqrt,
530                self.index_len() - self.sqrt == self.num as int / (self.sqrt + 1),
531        {
532            floor_split(self.num as int, self.sqrt as int);
533            assert(2 * self.sqrt <= usize::MAX) by(nonlinear_arith)
534                requires self.sqrt > 0, self.sqrt as int * self.sqrt as int <= self.num,
535                    self.num <= usize::MAX, usize::MAX >= 2;
536        }
537
538        proof fn index_value_properties(&self, index: int)
539            requires self.index_invariant(), 0 <= index < self.index_len(),
540            ensures
541                1 <= self.value_at(index) <= self.num,
542                floor_quotient(self.num as int, self.value_at(index)),
543                index >= self.sqrt ==> self.sqrt < self.value_at(index)
544                    && self.num as int / self.value_at(index) == self.index_len() - index,
545        {
546            self.index_bounds();
547            let n = self.num as int;
548            let s = self.sqrt as int;
549            if index < s {
550                small_floor_quotient(n, s, index + 1);
551                floor_reciprocal(n, index + 1);
552            } else {
553                let key = self.index_len() - index;
554                small_floor_quotient(n, s, key);
555                floor_reciprocal(n, key);
556                quotient_at_least(n, key, s + 1);
557                assert(1 <= key <= n && n / key == self.value_at(index));
558            }
559        }
560
561        proof fn index_order(&self, i: int, j: int)
562            requires self.index_invariant(), 0 <= i < j < self.index_len(),
563            ensures self.value_at(i) < self.value_at(j),
564        {
565            self.index_value_properties(i);
566            self.index_value_properties(j);
567            if i >= self.sqrt {
568                vstd::arithmetic::div_mod::lemma_div_is_ordered_by_denominator(
569                    self.num as int, self.index_len() - j, self.index_len() - i);
570            }
571        }
572    }

593        pub fn collect_quotients(num: usize, sqrt: usize)
594            requires sqrt > 0, sqrt as int * sqrt as int <= num < (sqrt + 1) * (sqrt + 1),
595        {
596            assert(num > 0) by(nonlinear_arith)
597                requires sqrt > 0, sqrt as int * sqrt as int <= num;
598            let index = FloorQuotientIndex { num, sqrt, pivot: num / sqrt };
599            let len = index.len();
600            let values: Vec<usize> = index.values().collect();
601            let key_values: Vec<(RangeInclusive<usize>, usize)> = index.key_values().collect();
602            assert(values.len() == len && key_values.len() == len);
603            assert forall|i: int, j: int| 0 <= i < j < values.len() implies
604                #[trigger] values@[i] < #[trigger] values@[j] by {
605                index.index_order(i, j);
606            }
607            proof {
608                index.index_bounds();
609                assert forall|value: usize| floor_quotient(num as int, value as int) implies
610                    #[trigger] values@.contains(value) by {
611                    floor_reciprocal(num as int, value as int);
612                    let position = if value <= sqrt { value - 1 } else { len - num as int / value as int };
613                    if value > sqrt {
614                        lemma_div_is_ordered_by_denominator(num as int, sqrt + 1, value as int);
615                    }
616                    assert(0 <= position < len && index.value_at(position) == value);
617                    assert(values@[position] == value);
618                }
619                assert forall|value: usize| #[trigger] values@.contains(value) implies
620                    floor_quotient(num as int, value as int) by {
621                    let position = choose|i: int| 0 <= i < values.len() && #[trigger] values@[i] == value;
622                    index.index_value_properties(position);
623                }
624            }
625            #[verifier::proof_note("Collecting values for any valid floor quotient index returns exactly all distinct floor(n/k) values for 1 <= k <= n, in strictly increasing order")]
626            assert((forall|value: usize| #[trigger] values@.contains(value)
627                == floor_quotient(num as int, value as int))
628                && forall|i: int, j: int| 0 <= i < j < values.len() ==>
629                    #[trigger] values@[i] < #[trigger] values@[j]);
630            #[verifier::proof_note("Collecting key_values returns one nonempty exact key interval per quotient, in the same order and with the same length as values")]
631            assert(key_values.len() == values.len()
632                && forall|i: int| 0 <= i < key_values.len() ==> {
633                    let entry = #[trigger] key_values@[i];
634                    entry.1 == values@[i]
635                        && 1 <= entry.0@.start <= entry.0@.end <= num
636                        && forall|key: int| 1 <= key <= num ==>
637                            ((entry.0@.start <= key <= entry.0@.end)
638                                == (#[trigger] (num as int / key) == entry.1))
639                });
640        }

642        pub fn index_round_trip(num: usize, sqrt: usize, position: usize, key: usize)
643            requires sqrt > 0, sqrt as int * sqrt as int <= num < (sqrt + 1) * (sqrt + 1),
644                1 <= key <= num,
645        {
646            let index = FloorQuotientIndex { num, sqrt, pivot: num / sqrt };
647            let len = index.len();
648            if position < len {
649                let value = index.index_to_value(position);
650                let reverse = index.value_to_index(value);
651                proof {
652                    if reverse < position { index.index_order(reverse as int, position as int); }
653                    if position < reverse { index.index_order(position as int, reverse as int); }
654                }
655                let keys = index.index_to_key(position);
656                let value_keys = index.value_to_key(value);
657                assert(keys@.start == value_keys@.start && keys@.end == value_keys@.end) by {
658                    assert(num as int / keys@.start as int == value);
659                    assert(num as int / keys@.end as int == value);
660                    assert(num as int / value_keys@.start as int == value);
661                    assert(num as int / value_keys@.end as int == value);
662                }
663                #[verifier::proof_note("For every positive usize numerator, its integer square root and every valid index, index-to-value and value-to-index round trip exactly, and index-to-key returns the same complete interval as value-to-key")]
664                assert(reverse == position && keys@.start == value_keys@.start
665                    && keys@.end == value_keys@.end);
666            }
667            let position = index.key_to_index(key);
668            let unchecked_position = unsafe { index.key_to_index_unchecked(key) };
669            let value = index.index_to_value(position);
670            let expected = index.key_to_value(key);
671            proof {
672                if position < unchecked_position { index.index_order(position as int, unchecked_position as int); }
673                if unchecked_position < position { index.index_order(unchecked_position as int, position as int); }
674            }
675            #[verifier::proof_note("For every valid key, both checked and unchecked key-to-index produce the same index, and mapping that index to a value yields exactly the key's floor quotient")]
676            assert(position == unchecked_position && value == expected);
677            assert forall|i: int, j: int| 0 <= i < j < len implies
678                #[trigger] index.value_at(i) < #[trigger] index.value_at(j) by {
679                index.index_order(i, j);
680            }
681            #[verifier::proof_note("The indexed quotient values are strictly increasing, so no two valid indices represent the same floor quotient")]
682            assert(forall|i: int, j: int| 0 <= i < j < len ==>
683                #[trigger] index.value_at(i) < #[trigger] index.value_at(j));
684        }

687            requires index@ > 0, 1 <= key <= index@,

689            let present_key = index.contains_key(key);
690            let value = index.key_to_value(key);
691            let present_value = index.contains_value(value);
692            let keys = index.value_to_key(value);

694            assert(present_key && present_value && keys@.start <= key <= keys@.end
695                && forall|k: int| 1 <= k <= index@ ==>
696                    ((keys@.start <= k <= keys@.end) == (#[trigger] (index@ as int / k) == value)));
697        }

700            exists|k: int| 1 <= k <= n && #[trigger] (n / k) == value

703        pub proof fn small_floor_quotient(n: int, s: int, value: int)
704            requires 0 < s, s * s <= n, 1 <= value <= s,
705            ensures floor_quotient(n, value),
706        {
707            assert(s * value <= s * s) by(nonlinear_arith)
708                requires 0 < s, 1 <= value <= s;
709            lemma_div_is_ordered(s * value, n, value);
710            lemma_div_by_multiple(s, value);
711            lemma_div_nonincreasing(n, value);
712            lemma_fundamental_div_mod(n, value);
713            lemma_mod_bound(n, value);
714            let key = n / value;
715            lemma_fundamental_div_mod_converse(n, key, value, n % value);
716            assert(1 <= key <= n && n / key == value);

719        pub proof fn quotient_at_least(n: int, key: int, value: int)
720            requires n >= 0, key > 0, value > 0,
721            ensures (value <= n / key) == (key <= n / value),
722        {
723            lemma_fundamental_div_mod(n, key);
724            lemma_mod_bound(n, key);
725            lemma_fundamental_div_mod(n, value);
726            lemma_mod_bound(n, value);
727            assert((value <= n / key) == (key * value <= n)) by(nonlinear_arith)
728                requires key > 0, n == key * (n / key) + n % key, 0 <= n % key < key;
729            assert((key <= n / value) == (key * value <= n)) by(nonlinear_arith)
730                requires value > 0, n == value * (n / value) + n % value, 0 <= n % value < value;
731        }

734            requires n > 0, floor_quotient(n, value),

745        pub proof fn floor_split(n: int, s: int)
746            requires s > 0, s * s <= n < (s + 1) * (s + 1),
747            ensures s <= n / s,
748                n / (s + 1) == s - if n / s == s { 1int } else { 0int },
749        {
750            lemma_fundamental_div_mod(n, s);
751            lemma_mod_bound(n, s);
752            lemma_fundamental_div_mod(n, s + 1);
753            lemma_mod_bound(n, s + 1);
754            assert(s <= n / s) by(nonlinear_arith)
755                requires s > 0, s * s <= n, n == s * (n / s) + n % s, 0 <= n % s < s;
756            assert(s - 1 <= n / (s + 1) <= s) by(nonlinear_arith)
757                requires s > 0, s * s <= n < (s + 1) * (s + 1),
758                    n == (s + 1) * (n / (s + 1)) + n % (s + 1), 0 <= n % (s + 1) < s + 1;
759            if n / s == s {
760                assert(n / (s + 1) == s - 1) by(nonlinear_arith)
761                    requires s > 0, n / s == s, s - 1 <= n / (s + 1) <= s,
762                        n == s * (n / s) + n % s, 0 <= n % s < s,
763                        n == (s + 1) * (n / (s + 1)) + n % (s + 1), 0 <= n % (s + 1);
764            } else {
765                assert(n / (s + 1) == s) by(nonlinear_arith)
766                    requires s > 0, s < n / s, s - 1 <= n / (s + 1) <= s,
767                        n == s * (n / s) + n % s, 0 <= n % s,
768                        n == (s + 1) * (n / (s + 1)) + n % (s + 1), n % (s + 1) < s + 1;
769            }
770        }

772        pub proof fn quotient_interval(n: int, v: int, k: int)
773            requires n > 0, v > 0, k > 0,
774            ensures (n / k == v) == (n / (v + 1) < k <= n / v),
775        {
776            lemma_fundamental_div_mod(n, k);
777            lemma_mod_bound(n, k);
778            lemma_fundamental_div_mod(n, v);
779            lemma_mod_bound(n, v);
780            lemma_fundamental_div_mod(n, v + 1);
781            lemma_mod_bound(n, v + 1);
782            #[verifier::proof_note("For positive integers n, v and k, n/k equals v exactly when n/(v+1) < k <= n/v")]
783            assert((n / k == v) == (n / (v + 1) < k <= n / v)) by(nonlinear_arith)
784                requires n > 0, v > 0, k > 0,
785                    n == k * (n / k) + n % k, 0 <= n % k < k,
786                    n == v * (n / v) + n % v, 0 <= n % v < v,
787                    n == (v + 1) * (n / (v + 1)) + n % (v + 1), 0 <= n % (v + 1) < v + 1;
788        }

790        pub proof fn floor_membership(n: usize, value: usize)
791            requires n > 0,
792            ensures floor_quotient(n as int, value as int) ==
793                (value > 0 && value <= n && (value == usize::MAX
794                    || (n as int) / (value as int) > (n as int) / (value + 1))),
795        {
796            let v = value as int;
797            if value == 0 || value > n {
798                assert forall|k: int| 1 <= k <= n implies #[trigger] (n as int / k) != v by {
799                    lemma_div_non_zero(n as int, k);
800                    lemma_div_nonincreasing(n as int, k);
801                }
802            } else if value == usize::MAX {
803                assert(n == usize::MAX);
804                assert(n as int / 1 == v);
805            } else {
806                let k = n as int / v;
807                if (n as int) / v > (n as int) / (v + 1) {
808                    lemma_div_non_zero(n as int, v);
809                    lemma_div_nonincreasing(n as int, v);
810                    quotient_interval(n as int, v, k);
811                    assert(n as int / k == v);
812                } else {
813                    assert forall|k: int| 1 <= k <= n implies #[trigger] (n as int / k) != v by {
814                        quotient_interval(n as int, v, k);
815                    }
816                }
817            }
818            #[verifier::proof_note("A positive floor quotient is present exactly when its key interval contains an integer, with zero, out-of-range values and usize::MAX handled explicitly")]
819            assert(floor_quotient(n as int, value as int) ==
820                (value > 0 && value <= n && (value == usize::MAX
821                    || (n as int) / (value as int) > (n as int) / (value + 1))));
822        }

827    #[kani::unwind(6)]
828    fn proof_quotient_constructors_u64_max() {
829        let distance: u64 = kani::any();
830        let root = u32::MAX as usize;
831        kani::assume(distance <= 2 * root as u64);
832        let num = usize::MAX - distance as usize;
833        let floor = FloorQuotientIndex::new(num);
834        let ceil = CeilQuotientIndex::new(num);
835        let floor_pivot = if distance == 0 {
836            root + 2
837        } else if distance <= root as u64 {
838            root + 1
839        } else {
840            root
841        };
842        let ceil_pivot = if distance < root as u64 {
843            root + 2
844        } else if distance < 2 * root as u64 {
845            root + 1
846        } else {
847            root
848        };
849        kani::assert(
850            floor.num == num && floor.sqrt == root && floor.pivot == floor_pivot,
851            "On 64-bit targets, for every numerator from (2^32-1)^2 through usize::MAX, the actual floor constructor stores square root 2^32-1 and its exact floor quotient; all three pivot values and both square-interval endpoints are covered",
852        );
853        kani::assert(
854            ceil.num == num && ceil.sqrt == root && ceil.pivot == ceil_pivot,
855            "On 64-bit targets, for every numerator from (2^32-1)^2 through usize::MAX, the actual ceiling constructor stores square root 2^32-1 and its exact ceiling quotient; all three pivot values and both square-interval endpoints are covered",
856        );
857        kani::cover!(distance == 0);
858        kani::cover!(distance == 1);
859        kani::cover!(distance == root as u64);
860        kani::cover!(distance == root as u64 + 1);
861        kani::cover!(distance == 2 * root as u64);
862    }
