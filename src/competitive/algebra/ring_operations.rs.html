6mod gf2_63 {
7    use super::*;
8    pub enum Gf2_63 {}
9    impl Gf2_63 {
10        pub const MOD: u64 = 1 << 63;
11    }
12    impl Magma for Gf2_63 {
13        type T = u64;
14        fn operate(x: &Self::T, y: &Self::T) -> Self::T {
15            #[cfg(target_arch = "x86_64")]
16            {
17                use core::arch::x86_64::{_mm_clmulepi64_si128, _mm_extract_epi64, _mm_set_epi64x};
18                unsafe {
19                    let a = _mm_set_epi64x(0, *x as i64);
20                    let b = _mm_set_epi64x(0, *y as i64);
21                    let c = _mm_clmulepi64_si128(a, b, 0);
22                    let lo = _mm_extract_epi64(c, 0) as u64;
23                    let hi = _mm_extract_epi64(c, 1) as u64;
24                    let hi = (hi << 1) | (lo >> 63);
25                    let lo = lo & !(!(0u64) << 63);
26                    lo ^ hi ^ (hi << 1)
27                }
28            }
29            #[cfg(not(target_arch = "x86_64"))]
30            {
31                let x = *x as u128;
32                let mut bit = 0u128;
33                for i in 0u32..63 {
34                    if y >> i & 1 != 0 {
35                        bit ^= x << i;
36                    }
37                }
38                let hi = (bit >> 64) as u64;
39                let lo = (bit & ((1 << 64) - 1)) as u64;
40                let hi = (hi << 1) | (lo >> 63);
41                let lo = lo & !(!(0u64) << 63);
42                lo ^ hi ^ (hi << 1)
43            }
44        }
45    }
46    impl Unital for Gf2_63 {
47        fn unit() -> Self::T {
48            1
49        }
50    }
51    impl Associative for Gf2_63 {}
52    impl Commutative for Gf2_63 {}
53    impl SemiRing for Gf2_63 {
54        type T = u64;
55        type Additive = BitXorOperation<u64>;
56        type Multiplicative = Self;
57    }
58}

68    pub enum Mersenne61Add {}
69    #[cfg_attr(verus_keep_ghost, verus_verify)]
70    impl Magma for Mersenne61Add {
71        type T = u64;
72        #[cfg_attr(verus_keep_ghost, verus_spec(result =>
73            requires (*x as int) + (*y as int) < 2 * (Mersenne61::MOD as int),
74            ensures
75                result < Mersenne61::MOD,
76                result == ((*x as int) + (*y as int)) % (Mersenne61::MOD as int),
77        ))]
78        fn operate(x: &Self::T, y: &Self::T) -> Self::T {
79            let mut z = x + y;
80            if z >= Mersenne61::MOD {
81                z -= Mersenne61::MOD
82            }
83            #[cfg(verus_keep_ghost)]
84            proof! {
85                lemma_fundamental_div_mod_converse(
86                    (*x as int) + (*y as int), (Mersenne61::MOD as int),
87                    if (*x as int) + (*y as int) >= Mersenne61::MOD { 1int } else { 0int },
88                    z as int,
89                );
90                #[verifier::proof_note("For all u64 operands with sum below twice 2^61 - 1, addition returns the canonical residue modulo 2^61 - 1 without overflow")]
91                assert(z < Mersenne61::MOD && z == ((*x as int) + (*y as int)) % (Mersenne61::MOD as int));
92            }
93            z
94        }
95    }
96    impl Unital for Mersenne61Add {
97        fn unit() -> Self::T {
98            0
99        }
100    }
101    impl Associative for Mersenne61Add {}
102    impl Commutative for Mersenne61Add {}
103    #[cfg_attr(verus_keep_ghost, verus_verify)]
104    impl Invertible for Mersenne61Add {
105        #[cfg_attr(verus_keep_ghost, verus_spec(result =>
106            requires *x < Mersenne61::MOD,
107            ensures result < Mersenne61::MOD, ((*x as int) + result as int) % (Mersenne61::MOD as int) == 0,
108        ))]
109        fn inverse(x: &Self::T) -> Self::T {
110            #[cfg(verus_keep_ghost)]
111            proof! {
112                let inverse = if *x == 0 { 0int } else { (Mersenne61::MOD as int) - (*x as int) };
113                lemma_small_mod(0, Mersenne61::MOD as nat);
114                lemma_mod_self_0((Mersenne61::MOD as int));
115                #[verifier::proof_note("For every residue below 2^61 - 1, additive inversion returns a canonical residue whose sum with the input is zero modulo 2^61 - 1")]
116                assert(0 <= inverse < Mersenne61::MOD && ((*x as int) + inverse) % (Mersenne61::MOD as int) == 0);
117            }
118            if *x == 0 { 0 } else { Mersenne61::MOD - x }
119        }
120    }

122    pub enum Mersenne61 {}
123    #[cfg_attr(verus_keep_ghost, verus_verify)]
124    impl Mersenne61 {
125        pub const MOD: u64 = 0x1fff_ffff_ffff_ffff; // 2^61 - 1
126    }
127    #[cfg_attr(verus_keep_ghost, verus_verify)]
128    impl Magma for Mersenne61 {
129        type T = u64;
130        #[cfg_attr(verus_keep_ghost, verus_spec(result =>
131            requires *x < Self::MOD, *y < Self::MOD,
132            ensures result < Self::MOD, result == ((*x as int) * (*y as int)) % (Self::MOD as int),
133        ))]
134        fn operate(x: &Self::T, y: &Self::T) -> Self::T {
135            #[cfg(verus_keep_ghost)]
136            proof! {
137                assert(Self::MOD == (1u64 << 61) - 1) by(bit_vector);
138                assert(0 <= (*x as int) * (*y as int) < (Self::MOD as int) * (Self::MOD as int))
139                    by(nonlinear_arith) requires *x < Self::MOD, *y < Self::MOD, *x >= 0, *y >= 0;
140                assert((*x as int) * (*y as int) <= u128::MAX) by(nonlinear_arith)
141                    requires *x < Self::MOD, *y < Self::MOD, Self::MOD == 0x1fff_ffff_ffff_ffff;
142            }
143            let z = *x as u128 * *y as u128;
144            #[cfg(verus_keep_ghost)]
145            proof! {
146                let radix = ((Self::MOD as int) + 1) as u128;
147                assert(z >> 61 < (Self::MOD as u128)) by(bit_vector)
148                    requires z < (Self::MOD as u128) * (Self::MOD as u128);
149                assert(z >> 61 == z / radix) by(bit_vector)
150                    requires radix == ((Self::MOD as int) + 1) as u128;
151                let hi = (z >> 61) as u64;
152                let lo = (#[verifier::truncate] (z as u64)) & Self::MOD;
153                assert(lo as u128 == z % radix) by(bit_vector)
154                    requires lo == (#[verifier::truncate] (z as u64)) & Self::MOD,
155                        radix == ((Self::MOD as int) + 1) as u128;
156                lemma_fundamental_div_mod(z as int, radix as int);
157                assert(hi < Self::MOD && lo <= Self::MOD);
158                assert(z == hi as int * radix as int + lo) by(nonlinear_arith)
159                    requires z == radix as int * (z / radix) + z % radix,
160                        hi == z / radix, lo == z % radix;
161                lemma_mod_self_0((Self::MOD as int));
162                lemma_small_mod(0, Self::MOD as nat);
163                vstd::arithmetic::mul::lemma_mul_basics(hi as int);
164                lemma_mul_mod_noop_right(hi as int, (Self::MOD as int), (Self::MOD as int));
165                lemma_add_mod_noop(hi as int * (Self::MOD as int), hi as int + lo as int, (Self::MOD as int));
166                assert(z == hi as int * (Self::MOD as int) + (hi as int + lo as int))
167                    by(nonlinear_arith) requires z == hi as int * radix as int + lo,
168                        radix == (Self::MOD as int) + 1;
169                #[verifier::proof_note("For all residues below 2^61 - 1, the u128 product fits and folding its high 61-bit part into its low part preserves the residue; final addition yields the canonical product modulo 2^61 - 1")]
170                assert(z as int % (Self::MOD as int) == (hi as int + lo as int) % (Self::MOD as int));
171            }
172            Mersenne61Add::operate(&((z >> 61) as _), &(z as u64 & Self::MOD))
173        }
174    }
175    impl Unital for Mersenne61 {
176        fn unit() -> Self::T {
177            1
178        }
179    }

183        type T = u64;
184        type Additive = Mersenne61Add;
