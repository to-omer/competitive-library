8/// Prepare `sc!`, `sv!`, `pp!`, and `dg!` with fast I/O for the supplied reader and writer.
9/// Every read must satisfy [`FastInput`]'s token requirements.
10#[macro_export]
11macro_rules! prepare_io {
12    (@inner ($dol:tt) $reader:expr, $writer:expr) => {
13        #[allow(unused_imports)]
14        use $crate::tools::{FastInput, FastIterPrint, FastOutput, FastPrint, IterPrint, ScanSource as _, read_all_unchecked};
15        let mut __in_buf = read_all_unchecked($reader);
16        __in_buf.push_str("                 ");
17        #[allow(unused_mut, unused_variables)]
18        let mut __scanner = unsafe { FastInput::from_slice(__in_buf.as_bytes()) };
19        #[allow(unused_mut, unused_variables)]
20        let mut __out = FastOutput::new($writer);
21        #[allow(unused_macros)]
22        macro_rules! sc { ($dol($dol t:tt)*) => { $crate::scan!(__scanner, $dol($dol t)*) } }
23        #[allow(unused_macros)]
24        macro_rules! sv { ($dol($dol t:tt)*) => { $crate::scan_value!(__scanner, $dol($dol t)*) } }
25        #[allow(unused_macros)]
26        macro_rules! pp { ($dol($dol t:tt)*) => { $crate::iter_print!(fast; __out, $dol($dol t)*) } }
27        $crate::prepare!(@debug ($));
28    };
29    ($reader:expr, $writer:expr $(,)?) => {
30        $crate::prepare_io!(@inner ($) $reader, $writer);
31    };
32}
