11#[cfg_attr(verus_keep_ghost, verus_verify)]
12impl Zarray {
13    // Called operations must terminate. The comparison bound counts Eq calls,
14    // excluding their internal work and allocation; loop variants prove loop termination.
15    #[cfg_attr(verus_keep_ghost, verus_spec(result =>
16        requires
17            #![verifier::proof_note("Zarray::new is called with nonempty input")]
18            0 < s.len(),
19            #![verifier::proof_note("Zarray::new is called with a lawful, stable Eq matching its specification")]
20            lawful_eq::<T>(),
21        ensures
22            result@.len() == s.len(),
23            forall|i: int| 0 <= i < s.len() ==> z_value(s@, i, result@[i] as int),
24    ))]
25    pub fn new<T>(s: &[T]) -> Self
26    where
27        T: Eq,
28    {
29        let n = s.len();
30        let mut z = vec![0; n];
31        z[0] = n;
32        let (mut i, mut j) = (1, 0);
33        #[cfg(verus_keep_ghost)]
34        proof_decl! {
35            let ghost mut successes = 0int;
36            let ghost mut failures = 0int;
37        }
38        #[cfg_attr(verus_keep_ghost, verus_spec(
39            invariant
40                n == s.len(), 0 < n,
41                lawful_eq::<T>(),
42                z.len() == n,
43                1 <= i <= n,
44                matches(s@, i as int, j as int),
45                forall|p: int| 0 <= p < i ==> z_value(s@, p, z@[p] as int),
46                0 <= successes <= i + j - 1,
47                0 <= failures <= i - 1,
48            decreases n - i,
49        ))]
50        while i < n {
51            #[cfg_attr(verus_keep_ghost, verus_spec(
52                invariant
53                    n == s.len(), 0 < i < n,
54                    lawful_eq::<T>(),
55                    matches(s@, i as int, j as int),
56                    0 <= successes <= i + j - 1,
57                decreases n - i - j,
58            ))]
59            while i + j < n && s[j] == s[i + j] {
60                #[cfg(verus_keep_ghost)]
61                proof! { successes = successes + 1; }
62                j += 1;
63            }
64            #[cfg(verus_keep_ghost)]
65            proof! { if i + j < n { failures = failures + 1; } }
66            z[i] = j;
67            if j == 0 {
68                i += 1;
69                continue;
70            }
71            let mut k = 1;
72            #[cfg_attr(verus_keep_ghost, verus_spec(
73                invariant
74                    n == s.len(), 0 < i < n,
75                    lawful_eq::<T>(),
76                    z.len() == n,
77                    1 <= k <= j,
78                    z_value(s@, i as int, j as int),
79                    forall|p: int| 0 <= p < i + k ==> z_value(s@, p, z@[p] as int),
80                decreases j - k,
81            ))]
82            while i + k < n && k + z[k] < j {
83                #[cfg(verus_keep_ghost)]
84                proof! { copy_match(s@, i as int, j as int, k as int, z@[k as int] as int); }
85                z[i + k] = z[k];
86                k += 1;
87            }
88            #[cfg(verus_keep_ghost)]
89            proof! {
90                if i + k < n {
91                    carry_match(s@, i as int, j as int, k as int, z@[k as int] as int);
92                }
93            }
94            i += k;
95            j -= k;
96        }
97        #[cfg(verus_keep_ghost)]
98        proof! {
99            #[verifier::proof_note("The Z-array has the same length as the input")]
100            assert(z.len() == s.len());
101            #[verifier::proof_note("For any nonempty input s with lawful Eq, z[i] is the longest common prefix length of s and s[i..]")]
102            assert(forall|p: int| 0 <= p < s.len() ==> z_value(s@, p, z@[p] as int));
103            #[verifier::proof_note("At most 2n - 2 Eq calls for any nonempty input of length n")]
104            assert(successes + failures <= 2 * n - 2);
105        }
106        Self { z }
107    }
108    #[cfg_attr(verus_keep_ghost, verus_spec(result =>
109        requires
110            pat.len() > 0,
111            pat.len() + s.len() < usize::MAX,
112            lawful_eq::<T>(),
113        ensures
114            forall|q: int| 0 <= q < result.len() ==> occurs(s@, pat@, result@[q] as int),
115            forall|p: int| 0 <= p < s.len()
116                ==> (occurs(s@, pat@, p) == result@.contains(p as usize)),
117            forall|q: int, r: int| 0 <= q < r < result.len() ==> result@[q] < result@[r],
118    ))]
119    pub fn search<T: Eq>(s: &[T], pat: &[T], sep: T) -> Vec<usize> {
120        let mut res: Vec<usize> = vec![];
121        let mut t: Vec<&T> = Vec::with_capacity(pat.len() + 1 + s.len());
122        #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
123            invariant
124                t.len() == iter.index(),
125                forall|j: int| 0 <= j < iter.index() ==> *t@[j] == pat@[j],
126        ))]
127        for x in pat {
128            t.push(x);
129        }
130        t.push(&sep);
131        #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
132            invariant
133                t.len() == pat.len() + 1 + iter.index(),
134                forall|j: int| 0 <= j < pat.len() ==> *t@[j] == pat@[j],
135                forall|j: int| 0 <= j < iter.index() ==> *t@[pat.len() + 1 + j] == s@[j],
136        ))]
137        for x in s {
138            t.push(x);
139        }
140        #[cfg(verus_keep_ghost)]
141        proof! { assert(lawful_eq::<&T>()); }
142        let zarray = Self::new(&t);
143        #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
144            invariant
145                pat.len() > 0, pat.len() + s.len() < usize::MAX,
146                lawful_eq::<T>(), lawful_eq::<&T>(),
147                t.len() == pat.len() + 1 + s.len(),
148                forall|j: int| 0 <= j < pat.len() ==> *t@[j] == pat@[j],
149                forall|j: int| 0 <= j < s.len() ==> *t@[pat.len() + 1 + j] == s@[j],
150                zarray@.len() == t.len(),
151                forall|j: int| 0 <= j < t.len() ==> z_value(t@, j, zarray@[j] as int),
152                res.len() <= iter.index(),
153                forall|q: int| 0 <= q < res.len()
154                    ==> res@[q] < iter.index() && occurs(s@, pat@, res@[q] as int),
155                forall|p: int| 0 <= p < iter.index()
156                    ==> (occurs(s@, pat@, p) == res@.contains(p as usize)),
157                forall|q: int, r: int| 0 <= q < r < res.len() ==> res@[q] < res@[r],
158        ))]
159        for (i, &z) in (pat.len() + 1..t.len()).zip(&zarray.z[pat.len() + 1..]) {
160            #[cfg(verus_keep_ghost)]
161            proof_decl! { let ghost before = res@; }
162            #[cfg(verus_keep_ghost)]
163            proof! {
164                let pos = i as int - pat.len() - 1;
165                let z = zarray@[i as int] as int;
166                assert(z_value(t@, i as int, z));
167                if z >= pat.len() {
168                    assert forall|j: int| 0 <= j < pat.len() implies
169                        #[trigger] pat@[j].eq_spec(&s@[pos + j]) by {
170                        assert(<&T as PartialEqSpec>::eq_spec(&t@[j], &t@[i + j]));
171                    }
172                } else if occurs(s@, pat@, pos) {
173                    assert(pat@[z].eq_spec(&s@[pos + z]));
174                    assert(<&T as PartialEqSpec>::eq_spec(&t@[z], &t@[i + z]));
175                }
176                assert((z >= pat.len()) == occurs(s@, pat@, pos));
177            }
178            if z >= pat.len() {
179                res.push(i - pat.len() - 1);
180            }
181            #[cfg(verus_keep_ghost)]
182            proof! {
183                broadcast use vstd::seq_lib::lemma_seq_contains_after_push;
184                let pos = i as int - pat.len() - 1;
185                assert forall|p: int| 0 <= p <= pos implies
186                    (occurs(s@, pat@, p) == res@.contains(p as usize)) by {
187                    if p < pos {
188                        assert(occurs(s@, pat@, p) == before.contains(p as usize));
189                    } else if before.contains(p as usize) {
190                        let q = before.lemma_contains_to_index(p as usize);
191                        assert(before[q] < pos);
192                    }
193                }
194                assert forall|q: int| 0 <= q < res.len() implies
195                    res@[q] <= pos && occurs(s@, pat@, res@[q] as int) by {
196                    if q < before.len() { assert(res@[q] == before[q]); }
197                }
198                assert forall|q: int, r: int| 0 <= q < r < res.len() implies
199                    res@[q] < res@[r] by {
200                    if r < before.len() { assert(before[q] < before[r]); }
201                }
202            }
203        }
204        #[cfg(verus_keep_ghost)]
205        proof! {
206            #[verifier::proof_note("For arbitrary text and nonempty pattern lengths whose combined allocation fits usize, lawful Eq, and any separator, Z search returns exactly all matching positions in increasing order, including overlaps and separator collisions")]
207            assert((forall|q: int| 0 <= q < res.len() ==> occurs(s@, pat@, res@[q] as int))
208                && (forall|p: int| 0 <= p < s.len() ==> (occurs(s@, pat@, p) == res@.contains(p as usize)))
209                && (forall|q: int, r: int| 0 <= q < r < res.len() ==> res@[q] < res@[r]));
210        }
211        res
212    }

230pub open spec fn lawful_eq<T>() -> bool
231where
232    T: Eq,
233{
234    &&& T::obeys_eq_spec()
235    &&& forall|x: T| #[trigger] x.eq_spec(&x)
236    &&& forall|x: T, y: T| #[trigger] x.eq_spec(&y) == y.eq_spec(&x)
237    &&& forall|x: T, y: T, z: T|
238        #[trigger] x.eq_spec(&y) && #[trigger] y.eq_spec(&z) ==> x.eq_spec(&z)
239}
240
241pub open spec fn matches<T>(s: Seq<T>, i: int, n: int) -> bool
242where
243    T: Eq,
244{
245    &&& 0 <= i <= s.len()
246    &&& 0 <= n <= s.len() - i
247    &&& forall|k: int| 0 <= k < n ==> #[trigger] s[k].eq_spec(&s[i + k])
248}
249
250pub open spec fn z_value<T>(s: Seq<T>, i: int, n: int) -> bool
251where
252    T: Eq,
253{
254    matches(s, i, n) && (i + n == s.len() || !s[n].eq_spec(&s[i + n]))
255}
256
257pub open spec fn occurs<T>(s: Seq<T>, pat: Seq<T>, pos: int) -> bool
258where
259    T: Eq,
260{
261    &&& 0 <= pos && pos + pat.len() <= s.len()
262    &&& forall|j: int| 0 <= j < pat.len() ==> #[trigger] pat[j].eq_spec(&s[pos + j])
263}
264
265proof fn copy_match<T>(s: Seq<T>, i: int, n: int, k: int, m: int)
266    where T: Eq,
267    requires lawful_eq::<T>(), z_value(s, i, n), z_value(s, k, m), 0 < k, k + m < n,
268    ensures z_value(s, i + k, m),
269{
270    assert forall|t: int| 0 <= t < m implies
271        #[verifier::proof_note("The m matching prefix elements transfer to the shifted position i + k")]
272        #[trigger] s[t].eq_spec(&s[i + k + t]) by {
273        assert(s[t].eq_spec(&s[k + t]));
274        assert(s[k + t].eq_spec(&s[i + (k + t)]));
275    }
276    #[verifier::proof_note("The known Z-box also matches at the copied prefix's mismatch boundary")]
277    assert(s[k + m].eq_spec(&s[i + (k + m)]));
278}
279
280proof fn carry_match<T>(s: Seq<T>, i: int, n: int, k: int, m: int)
281    where T: Eq,
282    requires lawful_eq::<T>(), matches(s, i, n), matches(s, k, m), 0 < k <= n, n <= k + m,
283    ensures matches(s, i + k, n - k),
284{
285    assert forall|t: int| 0 <= t < n - k implies
286        #[verifier::proof_note("Shifting an overlapping prefix match by k preserves n - k matching elements")]
287        #[trigger] s[t].eq_spec(&s[i + k + t]) by {
288        assert(s[t].eq_spec(&s[k + t]));
289        assert(s[k + t].eq_spec(&s[i + (k + t)]));
290    }
291}

309    }

323    pub fn byte_search(text: &[u8], pattern: &[u8], separator: u8)
324        requires pattern.len() > 0, pattern.len() + text.len() < usize::MAX,
325    {
326        let positions = Zarray::search(text, pattern, separator);
327        #[verifier::proof_note("For byte texts and nonempty patterns fitting the combined buffer, Z search's contract is usable for any separator byte")]
328        assert(forall|p: int| 0 <= p < text.len()
329            ==> (occurs(text@, pattern@, p) == positions@.contains(p as usize)));
330    }

333        requires s.len() > 0,
334    {

336        assert(lawful_eq::<Classified>());

353mod tests {
354    use super::*;
355    use crate::tools::{
356        Xorshift,
357        testutil::{exhaustive_sequences, sequence_cases},
358    };
359    use std::cell::Cell;
360
361    #[test]
362    fn test_zarray() {
363        struct Classified<'a> {
364            class: u8,
365            _payload: usize,
366            comparisons: &'a Cell<usize>,
367        }
368        impl PartialEq for Classified<'_> {
369            fn eq(&self, other: &Self) -> bool {
370                self.comparisons.set(self.comparisons.get() + 1);
371                self.class == other.class
372            }
373        }
374        impl Eq for Classified<'_> {}
375
376        let mut rng = Xorshift::default();
377        for s in
378            exhaustive_sequences(0..3, 1..=9).chain(sequence_cases(&mut rng, 0..=u8::MAX, 1..=1024))
379        {
380            let comparisons = Cell::new(0);
381            let classified: Vec<_> = s
382                .iter()
383                .enumerate()
384                .map(|(i, &class)| Classified {
385                    class,
386                    _payload: i,
387                    comparisons: &comparisons,
388                })
389                .collect();
390            let z = Zarray::new(&classified);
391            assert!(comparisons.get() <= 2 * s.len() - 2);
392            assert_eq!(z.z.len(), s.len());
393            for i in 0..s.len() {
394                let expected = s.iter().zip(&s[i..]).take_while(|(a, b)| a == b).count();
395                assert_eq!(z[i], expected, "s={s:?}, i={i}");
396            }
397        }
398    }
