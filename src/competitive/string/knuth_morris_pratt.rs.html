9pub struct KnuthMorrisPratt<T: Eq> {
10    pattern: Vec<T>,
11    table: Vec<usize>,
12}
13#[cfg_attr(verus_keep_ghost, verus_verify)]
14impl<T: Eq> KnuthMorrisPratt<T> {
15    #[cfg_attr(verus_keep_ghost, verus_spec(result =>
16        requires
17            pattern.len() < usize::MAX,
18            lawful_eq::<T>(),
19        ensures
20            result@ == pattern@,
21            result.valid(),
22    ))]
23    pub fn new(pattern: Vec<T>) -> Self {
24        let mut table = vec![0; pattern.len() + 1];
25        // Each iteration compares at most twice outside fallback steps.
26        #[cfg(verus_keep_ghost)]
27        proof_decl! { let ghost mut fallbacks = 0int; }
28        #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
29            invariant
30                pattern.len() < usize::MAX,
31                lawful_eq::<T>(),
32                table.len() == pattern.len() + 1,
33                0 <= fallbacks,
34                fallbacks + table@[iter.index()] <= iter.index(),
35                forall|q: int| 0 <= q < pattern.len() && q < iter.index() + 1
36                    ==> longest_border(pattern@, q + 1, table@[q] as int),
37        ))]
38        for i in 1..pattern.len() {
39            let mut j = table[i - 1];
40            #[cfg(verus_keep_ghost)]
41            proof! {
42                assert forall|k: int| j < k < i implies
43                    !#[trigger] prefix_suffix(pattern@, pattern@, i + 1, k + 1) by {
44                    extend_border(pattern@, pattern@, i as int, k);
45                }
46            }
47            #[cfg_attr(verus_keep_ghost, verus_spec(
48                invariant
49                    0 < i < pattern.len() < usize::MAX,
50                    lawful_eq::<T>(),
51                    table.len() == pattern.len() + 1,
52                    forall|q: int| 0 <= q < i
53                        ==> longest_border(pattern@, q + 1, table@[q] as int),
54                    j < i,
55                    0 <= fallbacks, fallbacks + j <= i - 1,
56                    prefix_suffix(pattern@, pattern@, i as int, j as int),
57                    forall|k: int| j < k < i
58                        ==> !#[trigger] prefix_suffix(pattern@, pattern@, i + 1, k + 1),
59                decreases j,
60            ))]
61            while j > 0 && pattern[i] != pattern[j] {
62                #[cfg(verus_keep_ghost)]
63                proof! {
64                    let next = table@[j - 1] as int;
65                    nested_borders(pattern@, pattern@, i as int, j as int, next);
66                    assert forall|k: int| next < k < i implies
67                        !#[trigger] prefix_suffix(pattern@, pattern@, i + 1, k + 1) by {
68                        extend_border(pattern@, pattern@, i as int, k);
69                        if k < j && prefix_suffix(pattern@, pattern@, i as int, k) {
70                            nested_borders(pattern@, pattern@, i as int, j as int, k);
71                        }
72                    }
73                    fallbacks = fallbacks + 1;
74                }
75                j = table[j - 1];
76            }
77            #[cfg(verus_keep_ghost)]
78            proof! {
79                extend_border(pattern@, pattern@, i as int, j as int);
80                assert forall|k: int| j < k < i implies
81                    !#[trigger] prefix_suffix(pattern@, pattern@, i + 1, k + 1) by {}
82            }
83            table[i] = j + (pattern[i] == pattern[j]) as usize;
84            #[cfg(verus_keep_ghost)]
85            proof! {
86                assert forall|k: int| table@[i as int] < k < i + 1 implies
87                    !#[trigger] prefix_suffix(pattern@, pattern@, i + 1, k) by {
88                    extend_border(pattern@, pattern@, i as int, k - 1);
89                }
90                assert(longest_border(pattern@, i + 1, table@[i as int] as int));
91            }
92        }
93        #[cfg(verus_keep_ghost)]
94        proof! {
95            #[verifier::proof_note("For any pattern with lawful Eq and length below usize::MAX, each KMP table entry is the longest proper prefix matching the corresponding prefix's suffix")]
96            assert(forall|i: int| 0 <= i < pattern.len()
97                ==> #[trigger] longest_border(pattern@, i + 1, table@[i] as int));
98            #[verifier::proof_note("KMP preprocessing uses at most 3n Eq calls for a pattern of length n, excluding comparison internals and allocation")]
99            assert(fallbacks + 2 * pattern.len() <= 3 * pattern.len());
100        }
101        Self { pattern, table }
102    }
103    #[cfg_attr(verus_keep_ghost, verifier::external)]
104    pub fn search_all(&self, s: &[T]) -> Vec<usize> {
105        let mut res = vec![];
106        let mut j = 0;
107        for (i, s) in s.iter().enumerate() {
108            while j > 0 && s != &self.pattern[j] {
109                j = self.table[j - 1];
110            }
111            if s == &self.pattern[j] {
112                j += 1;
113            }
114            if j == self.pattern.len() {
115                res.push(i - (j - 1));
116                j = self.table[j - 1];
117            }
118        }
119        res
120    }
121}

131impl<T: Eq> KnuthMorrisPratt<T> {
132    pub closed spec fn valid(&self) -> bool {
133        self.table.len() == self.pattern.len() + 1
134        && forall|i: int| 0 <= i < self.pattern.len()
135            ==> longest_border(self.pattern@, i + 1, self.table@[i] as int)
136    }
137}

139pub open spec fn prefix_suffix<T>(p: Seq<T>, s: Seq<T>, n: int, k: int) -> bool
140where
141    T: Eq,
142{
143    &&& 0 <= k <= p.len()
144    &&& k <= n <= s.len()
145    &&& forall|t: int| 0 <= t < k ==> #[trigger] p[t].eq_spec(&s[n - k + t])
146}
147
148pub open spec fn longest_border<T>(p: Seq<T>, n: int, k: int) -> bool
149where
150    T: Eq,
151{
152    &&& 0 <= k < n
153    &&& prefix_suffix(p, p, n, k)
154    &&& forall|j: int| k < j < n ==> !prefix_suffix(p, p, n, j)
155}

158proof fn extend_border<T>(p: Seq<T>, s: Seq<T>, n: int, k: int)
159    where T: Eq,
160    requires 0 <= k <= n < s.len(), k < p.len(),
161    ensures prefix_suffix(p, s, n + 1, k + 1)
162        == (prefix_suffix(p, s, n, k) && p[k].eq_spec(&s[n])),
163{
164    if prefix_suffix(p, s, n + 1, k + 1) {
165        assert(p[k].eq_spec(&s[n + 1 - (k + 1) + k]));
166        assert forall|t: int| 0 <= t < k implies
167            #[trigger] p[t].eq_spec(&s[n - k + t]) by {
168            assert(p[t].eq_spec(&s[n + 1 - (k + 1) + t]));
169        }
170    }
171    if prefix_suffix(p, s, n, k) && p[k].eq_spec(&s[n]) {
172        assert forall|t: int| 0 <= t < k + 1 implies
173            #[trigger] p[t].eq_spec(&s[n - k + t]) by {}
174    }
175    #[verifier::proof_note("A prefix match extends by one exactly when the next pattern and text elements compare equal")]
176    assert(prefix_suffix(p, s, n + 1, k + 1)
177        == (prefix_suffix(p, s, n, k) && p[k].eq_spec(&s[n])));
178}
179
180proof fn nested_borders<T>(p: Seq<T>, s: Seq<T>, n: int, j: int, k: int)
181    where T: Eq,
182    requires lawful_eq::<T>(), prefix_suffix(p, s, n, j), 0 <= k < j,
183    ensures prefix_suffix(p, s, n, k) == prefix_suffix(p, p, j, k),
184{
185    assert forall|t: int| 0 <= t < k implies
186        #[trigger] p[t].eq_spec(&s[n - k + t]) == p[t].eq_spec(&p[j - k + t]) by {
187        assert(p[j - k + t].eq_spec(&s[n - j + (j - k + t)]));
188        assert(p[j - k + t].eq_spec(&s[n - k + t]));
189    }
190    if prefix_suffix(p, s, n, k) {
191        assert forall|t: int| 0 <= t < k implies
192            #[trigger] p[t].eq_spec(&p[j - k + t]) by {
193            assert(p[t].eq_spec(&s[n - k + t]));
194        }
195    } else if prefix_suffix(p, p, j, k) {
196        assert forall|t: int| 0 <= t < k implies
197            #[trigger] p[t].eq_spec(&s[n - k + t]) by {
198            assert(p[t].eq_spec(&p[j - k + t]));
199        }
200    }
201    #[verifier::proof_note("Inside a matched suffix of length j, a shorter length k matches exactly when it is a border of the pattern prefix of length j")]
202    assert(prefix_suffix(p, s, n, k) == prefix_suffix(p, p, j, k));
203}
