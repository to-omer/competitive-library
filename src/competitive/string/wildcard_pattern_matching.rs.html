5/// Matches bytes with `?` as a wildcard on either side.
6/// Requires round-to-nearest binary64 arithmetic with gradual underflow and no fast-math.
7pub fn wildcard_pattern_matching(p: &[u8], s: &[u8]) -> Vec<bool> {
8    assert!(!p.is_empty());
9    assert!(p.len() <= s.len());
10    matching_with_limit(p, s, FFT_LIMIT)
11}
12
13// The score is sum(mask_s * mask_p * (s - p)^2), an integer >= 1 on a mismatch.
14// For n,m <= 2^22, 65536 * gamma_3295 * sqrt(n*m) + 2^-900 < 0.100556.
15// This includes fixed-root generation, partial transforms, leaf products and the
16// final addition of exact square sums. Each tile owns its outputs; scores from
17// different tiles or pattern pieces are never added in floating point.
18const FFT_LIMIT: usize = 1 << 22;
19
20fn matching_with_limit(p: &[u8], s: &[u8], limit: usize) -> Vec<bool> {
21    if s.len() <= limit {
22        return bounded(p, s, |value| value.abs() < 0.5);
23    }
24    let mut result = vec![true; s.len() - p.len() + 1];
25    for offset in (0..p.len()).step_by(limit / 2) {
26        let part = &p[offset..offset + (p.len() - offset).min(limit / 2)];
27        let step = limit - part.len() + 1;
28        for start in (0..result.len()).step_by(step) {
29            let count = step.min(result.len() - start);
30            let text = &s[start + offset..start + offset + count + part.len() - 1];
31            let values = bounded(part, text, |value| value.abs() < 0.5);
32            for (answer, value) in result[start..start + count].iter_mut().zip(values) {
33                *answer &= value;
34            }
35        }
36    }
37    result
38}

40fn bounded<T>(p: &[u8], s: &[u8], convert: impl FnMut(f64) -> T) -> Vec<T> {
41    #[cfg(target_arch = "x86_64")]
42    if is_x86_feature_detected!("avx2") && is_x86_feature_detected!("fma") {
43        match simd_backend() {
44            SimdBackend::Avx512 => return unsafe { avx512::run(p, s, convert) },
45            SimdBackend::Avx2 => return unsafe { avx2::run(p, s, convert) },
46            SimdBackend::Scalar => {}
47        }
48    }
49    scalar::run(p, s, convert)
50}

52fn capacity<T>(len: usize) -> Vec<T> {
53    let mut result = Vec::with_capacity(len);
54    advise_huge_pages(&mut result);
55    result

58fn square(ch: u8) -> u64 {
59    let value = i64::from(ch) - 128;
60    if ch == b'?' {
61        0
62    } else {
63        (value * value) as u64
64    }
65}

67// Reuse pattern transforms only when the estimated transform work decreases.
68fn transform_length(p: usize, s: usize, lanes: usize, terms: usize) -> usize {
69    let full = s.next_power_of_two().max(2 * lanes);
70    let mut length = full;
71    let mut best = (2 * terms + 1) * full * full.ilog2() as usize;
72    let mut option = (2 * p - 1).next_power_of_two().max(2 * lanes);
73    while option < full {
74        let blocks = (s - p + 1).div_ceil(option - p + 1);
75        let cost = (terms + (terms + 1) * blocks) * option * option.ilog2() as usize;
76        if cost < best {
77            length = option;
78            best = cost;
79        }
80        option *= 2;
81    }
82    length
83}

85macro_rules! matching_kernel {
86    ([$($qualifier:tt)*] $(#[$attr:meta])*; $block:ident, $lanes:literal) => {
87        $(#[$attr])*
88        pub $($qualifier)* fn run<T>(p: &[u8], s: &[u8], convert: impl FnMut(f64) -> T) -> Vec<T> {
89            match (!p.contains(&b'?'), !s.contains(&b'?')) {
90                (true, true) => general::<_, 1, true, true>(p, s, convert),
91                (true, false) => general::<_, 2, true, false>(p, s, convert),
92                (false, true) => general::<_, 2, false, true>(p, s, convert),
93                (false, false) => general::<_, 3, false, false>(p, s, convert),
94            }
95        }
96
97        $(#[$attr])*
98        $($qualifier)* fn general<T, const TERMS: usize, const PCLEAR: bool, const SCLEAR: bool>(
99            p: &[u8],
100            s: &[u8],
101            mut convert: impl FnMut(f64) -> T,
102        ) -> Vec<T> {
103            let length = transform_length(p.len(), s.len(), $lanes, TERMS);
104            let n = length / 2;
105            let step = length - p.len() + 1;
106            let mut text: [Vec<$block>; TERMS] = std::array::from_fn(|_| capacity(n / $lanes));
107            let mut pattern: [Vec<$block>; TERMS] = std::array::from_fn(|_| capacity(n / $lanes));
108            pack::<TERMS, true, PCLEAR>(p, n, &mut pattern);
109            for part in &mut pattern {
110                fft_soa(part);
111            }
112            let mut window = if PCLEAR {
113                s[..p.len()].iter().map(|&ch| square(ch)).sum::<u64>()
114            } else {
115                0
116            };
117            let pattern_square = if SCLEAR {
118                p.iter().map(|&ch| square(ch)).sum::<u64>()
119            } else {
120                0
121            };
122            let mut result = Vec::with_capacity(s.len() - p.len() + 1);
123            for start in (0..s.len() - p.len() + 1).step_by(step) {
124                let count = step.min(s.len() - p.len() + 1 - start);
125                let bytes = &s[start..start + count + p.len() - 1];
126                pack::<TERMS, false, PCLEAR>(bytes, n, &mut text);
127                for part in &mut text {
128                    fft_soa(part);
129                }
130                with_roots(n / 2, |cache| {
131                    for block in 0..text[0].len() {
132                        let mut br = [splat(0.); TERMS];
133                        let mut bi = br;
134                        for term in 0..TERMS {
135                            (br[term], bi[term]) = load(&pattern[term][block]);
136                        }
137                        let mut rr = splat(0.);
138                        let mut ri = rr;
139                        let w = eval_twiddle(cache, 1, text[0].len(), block);
140                        let wr = first(w.re, 1.);
141                        let wi = first(w.im, 0.);
142                        for lane in 0..$lanes {
143                            for term in 0..TERMS {
144                                let ar = splat(text[term][block].re[lane]);
145                                let ai = splat(text[term][block].im[lane]);
146                                multiply_accumulate(&mut rr, &mut ri, ar, ai, br[term], bi[term]);
147                            }
148                            if lane != $lanes - 1 {
149                                for term in 0..TERMS {
150                                    br[term] = rotate(br[term]);
151                                    bi[term] = rotate(bi[term]);
152                                    (br[term], bi[term]) = complex_mul(br[term], bi[term], wr, wi);
153                                }
154                            }
155                        }
156                        store(&mut text[0][block], rr, ri);
157                    }
158                });
159                ifft_soa(&mut text[0]);
160                for i in p.len() - 1..p.len() - 1 + count {
161                    let mut value = if i < n {
162                        text[0][i / $lanes].re[i % $lanes]
163                    } else {
164                        text[0][(i - n) / $lanes].im[i % $lanes]
165                    };
166                    if PCLEAR || SCLEAR {
167                        value += (window + pattern_square) as f64;
168                    }
169                    if PCLEAR {
170                        let position = start + i - (p.len() - 1);
171                        if position + 1 < s.len() - p.len() + 1 {
172                            window -= square(s[position]);
173                            window += square(s[position + p.len()]);
174                        }
175                    }
176                    result.push(convert(value));
177                }
178            }
179            result
180        }
181
182        $(#[$attr])*
183        #[inline]
184        $($qualifier)* fn pack<const TERMS: usize, const PATTERN: bool, const PCLEAR: bool>(
185            bytes: &[u8],
186            n: usize,
187            parts: &mut [Vec<$block>; TERMS],
188        ) {
189            for part in parts.iter_mut() {
190                part.clear();
191            }
192            for block in 0..n / $lanes {
193                let (xr, mr) = values(bytes, block * $lanes, PATTERN);
194                let (xi, mi) = values(bytes, block * $lanes + n, PATTERN);
195                for (term, part) in parts.iter_mut().enumerate() {
196                    let (re, im) = match (PATTERN, term + usize::from(PCLEAR)) {
197                        (true, 0) | (false, 2) => (mr, mi),
198                        (true, 1) => (mul(xr, mr), mul(xi, mi)),
199                        (false, 1) => (mul(mul(splat(-2.), xr), mr), mul(mul(splat(-2.), xi), mi)),
200                        _ => (mul(mul(xr, xr), mr), mul(mul(xi, xi), mi)),
201                    };
202                    let mut value = $block::default();
203                    store(&mut value, re, im);
204                    part.push(value);
205                }
206            }
207        }
208    };
209}

212#[inline]
213fn byte_block<const LANES: usize>(bytes: &[u8], start: usize, reverse: bool) -> [u8; LANES] {
214    let count = bytes.len().saturating_sub(start).min(LANES);
215    let mut block = [b'?'; LANES];
216    if count == LANES {
217        let at = if reverse {
218            bytes.len() - start - LANES
219        } else {
220            start
221        };
222        block.copy_from_slice(&bytes[at..at + LANES]);
223        if reverse {
224            block.reverse();
225        }
226    } else {
227        for (lane, value) in block.iter_mut().take(count).enumerate() {
228            *value = bytes[if reverse {
229                bytes.len() - start - lane - 1
230            } else {
231                start + lane
232            }];
233        }
234    }
235    block
236}

238mod scalar {
239    use super::partial_fft::{eval_twiddle, scalar::*, with_roots};
240    use super::{capacity, square, transform_length};
241    fn values(bytes: &[u8], start: usize, reverse: bool) -> ([f64; 4], [f64; 4]) {
242        let count = bytes.len().saturating_sub(start).min(4);
243        let mut x = [0.; 4];
244        let mut mask = [0.; 4];
245        for lane in 0..count {
246            let ch = bytes[if reverse {
247                bytes.len() - start - lane - 1
248            } else {
249                start + lane
250            }];
251            x[lane] = f64::from(ch) - 128.;
252            mask[lane] = f64::from(ch != b'?');
253        }
254        (x, mask)
255    }
256    matching_kernel!([]; Complex4, 4);
257}

260mod avx2 {
261    // The dispatcher checks the target features before calling this module.
262    #![allow(unsafe_op_in_unsafe_fn)]
263    use super::partial_fft::{avx2::*, eval_twiddle, with_roots};
264    use super::{byte_block, capacity, square, transform_length};
265    #[target_feature(enable = "avx2,fma")]
266    #[inline]
267    unsafe fn values(
268        bytes: &[u8],
269        start: usize,
270        reverse: bool,
271    ) -> (std::arch::x86_64::__m256d, std::arch::x86_64::__m256d) {
272        use std::arch::x86_64::*;
273        let raw = u32::from_le_bytes(byte_block(bytes, start, reverse));
274        let raw = _mm_cvtsi32_si128(raw as i32);
275        let ch = _mm256_cvtepi32_pd(_mm_cvtepu8_epi32(raw));
276        let x = _mm256_sub_pd(ch, _mm256_set1_pd(128.));
277        let mask = _mm256_andnot_pd(
278            _mm256_cmp_pd::<_CMP_EQ_OQ>(ch, _mm256_set1_pd(63.)),
279            _mm256_set1_pd(1.),
280        );
281        (x, mask)
282    }
283
284    matching_kernel!([unsafe] #[target_feature(enable = "avx2,fma")]; Complex4, 4);
285}

436    #[test]
437    fn test_wildcard_pattern_matching_scores() {
438        let mut rng = Xorshift::default();
439        let boundaries = integer_cases(1usize..=129)
440            .map(|m| (0..m).map(|i| [0, 1, 127, 128, 254, 255][i % 6]).collect());
441        let structured = sequence_cases(&mut rng, 0..=u8::MAX, 1..=129);
442        for mut input in boundaries.chain(structured) {
443            for ch in &mut input {
444                if *ch == b'?' {
445                    *ch = 0;
446                }
447            }
448            let m = input.len();
449            for n in [m, 2 * m + 3, 8 * m + 17, 1025] {
450                for mask in 0..4 {
451                    let mut p = input.clone();
452                    let mut s: Vec<_> = input.iter().copied().cycle().take(n).collect();
453                    s[n / 2] ^= 1;
454                    if s[n / 2] == b'?' {
455                        s[n / 2] ^= 2;
456                    }
457                    if mask & 1 != 0 {
458                        for ch in p.iter_mut().step_by(5) {
459                            *ch = b'?';
460                        }
461                    }
462                    if mask & 2 != 0 {
463                        for ch in s.iter_mut().step_by(7) {
464                            *ch = b'?';
465                        }
466                    }
467                    let expected: Vec<_> = s
468                        .windows(m)
469                        .map(|window| {
470                            p.iter()
471                                .zip(window)
472                                .map(|(&a, &b)| {
473                                    if a == b'?' || b == b'?' {
474                                        0
475                                    } else {
476                                        let difference = i64::from(a) - i64::from(b);
477                                        difference * difference
478                                    }
479                                })
480                                .sum::<i64>() as f64
481                        })
482                        .collect();
483                    let check = |actual: Vec<f64>| {
484                        assert_eq!(actual.len(), expected.len());
485                        for (actual, expected) in actual.into_iter().zip(&expected) {
486                            assert!(
487                                (actual - expected).abs() < 0.5,
488                                "m={m}, n={n}, mask={mask}, actual={actual}, expected={expected}"
489                            );
490                        }
491                    };
492                    check(scalar::run(&p, &s, |x| x));
493                    #[cfg(target_arch = "x86_64")]
494                    unsafe {
495                        if is_x86_feature_detected!("avx2") && is_x86_feature_detected!("fma") {
496                            check(super::avx2::run(&p, &s, |x| x));
497                            if is_x86_feature_detected!("avx512f") {
498                                check(super::avx512::run(&p, &s, |x| x));
499                            }
500                        }
501                    }
502                }
503            }
504        }
505    }
