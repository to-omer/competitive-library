61{
62    #[inline]

68#[codesnip::entry(when("MIntBase", "coding"))]
69impl<M> SerdeByteStr for MInt<M>
70where
71    M: MIntBase<Inner: SerdeByteStr>,
72{
73    fn serialize(&self, buf: &mut Vec<u8>) {
74        self.inner().serialize(buf)
75    }
76
77    fn deserialize<I>(iter: &mut I) -> Self
78    where
79        I: Iterator<Item = u8>,
80    {
81        Self::new_unchecked(M::Inner::deserialize(iter))
82    }
83}
