8struct MontgomeryReduction64 {
9    modulus: u64,
10    inverse: u64,
11}
12
13#[cfg_attr(verus_keep_ghost, verus_verify)]
14impl MontgomeryReduction64 {
15    #[cfg_attr(verus_keep_ghost, verus_spec(result =>
16        requires modulus % 2 == 1,
17        ensures result.modulus == modulus,
18            (modulus as int * result.inverse as int) % radix() == 1,
19    ))]
20    fn new(modulus: u64) -> Self {
21        let mut inverse = modulus;
22        #[cfg(verus_keep_ghost)]
23        proof! {
24            reveal(vstd::arithmetic::power::pow);
25            vstd::arithmetic::div_mod::lemma_mul_mod_noop(modulus as int, modulus as int, 2);
26            vstd::arithmetic::div_mod::lemma_sub_mod_noop(modulus as int * modulus as int, 1, 2);
27            assert(pow2(pow2(0)) == 2) by(compute);
28            assert((modulus as int * inverse as int - 1) % 2 == 0);
29        }
30        #[cfg_attr(verus_keep_ghost, verus_spec(iter => invariant
31            modulus % 2 == 1,
32            (modulus as int * inverse as int - 1) % (pow2(pow2(iter.index() as nat)) as int) == 0,
33        ))]
34        for _ in 0..6 {
35            #[cfg(verus_keep_ghost)]
36            proof_decl! { let ghost previous = inverse; }
37            inverse = inverse.wrapping_mul(2u64.wrapping_sub(modulus.wrapping_mul(inverse)));
38            #[cfg(verus_keep_ghost)]
39            proof! {
40                inverse_precision(iter.index() as nat);
41                inverse_refinement(modulus, previous, pow2(pow2(iter.index() as nat)) as int);
42            }
43        }
44        #[cfg(verus_keep_ghost)]
45        proof! {
46            reveal(vstd::arithmetic::power::pow);
47            assert(pow2(pow2(6)) == radix()) by(compute);
48            vstd::arithmetic::div_mod::lemma_mod_adds(modulus as int * inverse as int - 1, 1, radix());
49            #[verifier::proof_note("For every odd u64 modulus, six Newton iterations with wrapping arithmetic compute its exact multiplicative inverse modulo 2^64")]
50            assert((modulus as int * inverse as int) % radix() == 1);
51        }
52        Self { modulus, inverse }
53    }
54    #[cfg_attr(verus_keep_ghost, verifier::external)]
55    fn sub(&self, lhs: u64, rhs: u64) -> u64 {
56        let (value, borrow) = lhs.overflowing_sub(rhs);
57        value.wrapping_add((borrow as u64).wrapping_neg() & self.modulus)
58    }
59
60    #[cfg_attr(verus_keep_ghost, verifier::external)]
61    fn mul(&self, lhs: u64, rhs: u64) -> u64 {
62        let product = lhs as u128 * rhs as u128;
63        let (value, borrow) = ((product >> 64) as u64).overflowing_sub(
64            (((product as u64).wrapping_mul(self.inverse) as u128 * self.modulus as u128) >> 64)
65                as u64,
66        );
67        value.wrapping_add((borrow as u64).wrapping_neg() & self.modulus)
68    }
69}

71fn find_factor(n: u64) -> Option<u64> {
72    let br = BarrettReduction::<u128>::new(n as u128);
73    if miller_rabin_with_br(n, &br) {
74        return None;
75    }
76    let mr = MontgomeryReduction64::new(n);
77    let mut rng = Xorshift::default();
78    let (mut y0, mut c) = (0, n - 1);
79    loop {
80        let (mut x, mut y, mut ys, mut g, mut q, mut r, mut k) = (0, y0, 0, 1, 1, 1, 0);
81        while g == 1 && r <= 1 << 20 {
82            x = y;
83            while k < r && g == 1 {
84                ys = y;
85                for _ in 0..1024.min(r - k) {
86                    y = mr.sub(mr.mul(y, y), c);
87                    q = mr.mul(q, mr.sub(x, y));
88                }
89                g = gcd(q, n);
90                k += 1024;
91            }
92            k = r;
93            r <<= 1;
94        }
95        if g == n {
96            g = 1;
97            y = ys;
98            while g == 1 {
99                y = mr.sub(mr.mul(y, y), c);
100                g = gcd(mr.sub(x, y), n);
101            }
102        }
103        if g != 1 && g != n {
104            return Some(g);
105        }
106        y0 = ((rng.rand64() as u128 * (n - 2) as u128) >> 64) as u64 + 2;
107        c = ((rng.rand64() as u128 * (n - 1) as u128) >> 64) as u64 + 1;
108    }
109}

111pub fn prime_factors_flatten(mut n: u64) -> Vec<u64> {
112    if n == 0 {
113        return vec![];
114    }
115    let k = n.trailing_zeros();
116    let mut res = vec![2; k as usize];
117    n >>= k;
118    while n.is_multiple_of(3) {
119        res.push(3);
120        n /= 3;
121    }
122    if n != 1 {
123        let mut c = vec![n];
124        while let Some(n) = c.pop() {
125            if let Some(m) = find_factor(n) {
126                c.push(m);
127                c.push(n / m);
128            } else {
129                res.push(n);
130            }
131        }
132    }
133    res.sort_unstable();
134    res
135}

137pub fn prime_factors(n: u64) -> Vec<(u64, u32)> {
138    let mut res = Vec::new();
139    for a in prime_factors_flatten(n) {
140        if let Some((p, len)) = res.last_mut()
141            && p == &a
142        {
143            *len += 1;
144            continue;
145        }
146        res.push((a, 1));
147    }
148    res
149}

151pub fn divisors(n: u64) -> Vec<u64> {
152    let mut d = vec![1u64];
153    for (p, c) in prime_factors(n) {
154        let k = d.len();
155        let mut acc = 1;
156        for _ in 0..c {
157            acc *= p;
158            for i in 0..k {
159                d.push(d[i] * acc);
160            }
161        }
162    }
163    d.sort_unstable();
164    d
165}

174    #[cfg(kani)]
175    #[kani::proof]
176    fn proof_modular_subtraction() {
177        let modulus: u64 = kani::any();
178        let lhs: u64 = kani::any();
179        let rhs: u64 = kani::any();
180        kani::assume(modulus > 0 && lhs < modulus && rhs < modulus);
181        let reduction = super::MontgomeryReduction64 {
182            modulus,
183            inverse: 0,
184        };
185        let expected = if lhs >= rhs {
186            lhs - rhs
187        } else {
188            (modulus as u128 + lhs as u128 - rhs as u128) as u64
189        };
190        kani::assert(
191            reduction.sub(lhs, rhs) == expected,
192            "For every positive u64 modulus and pair of reduced residues, Montgomery subtraction returns their canonical modular difference, including borrowing at u64::MAX",
193        );
194        kani::cover!(modulus == u64::MAX && lhs == 0 && rhs == u64::MAX - 1);
195        kani::cover!(modulus == 1);
196        kani::cover!(lhs == rhs);
197    }

200    verus! {
201        pub open spec fn radix() -> int { 0x1_0000_0000_0000_0000 }

203        pub open spec fn inverse_step(m: u64, x: u64) -> u64 {
204            u64_specs::wrapping_mul(x, u64_specs::wrapping_sub(2, u64_specs::wrapping_mul(m, x)))
205        }

207        pub proof fn inverse_refinement(m: u64, x: u64, p: int)
208            requires p > 1, radix() % (p * p) == 0, (m as int * x as int - 1) % p == 0,
209            ensures (m as int * inverse_step(m, x) as int - 1) % (p * p) == 0,
210        {
211            let big = radix();
212            let a = u64_specs::wrapping_mul(m, x);
213            let b = u64_specs::wrapping_sub(2, a);
214            let next = inverse_step(m, x);
215            let qa = m as int * x as int / big;
216            let qx = x as int * b as int / big;
217            let c = if a > 2 { 1int } else { 0int };
218            let t = (m as int * x as int - 1) / p;
219            let co = big / (p * p);
220            lemma_mod_bound(m as int * x as int, big);
221            lemma_mod_bound(x as int * b as int, big);
222            lemma_fundamental_div_mod(m as int * x as int, big);
223            lemma_fundamental_div_mod(x as int * b as int, big);
224            lemma_fundamental_div_mod(m as int * x as int - 1, p);
225            assert(p * p > 0) by(nonlinear_arith) requires p > 1;
226            lemma_fundamental_div_mod(big, p * p);
227            assert(m as int * x as int == qa * big + a);
228            assert(b as int == 2 - a + big * c);
229            assert(x as int * b as int == qx * big + next);
230            let correction = x as int * (qa + c) - qx;
231            assert(next as int == x as int * (2 - m as int * x as int) + big * correction) by(nonlinear_arith)
232                requires
233                    m as int * x as int == qa * big + a,
234                    b as int == 2 - a + big * c,
235                    x as int * b as int == qx * big + next,
236                    correction == x as int * (qa + c) - qx;
237            let error = m as int * x as int - 1;
238            assert(m as int * next as int - 1 == -error * error + big * m * correction) by(nonlinear_arith)
239                requires next as int == x as int * (2 - m as int * x as int) + big * correction,
240                    error == m as int * x as int - 1;
241            let witness = co * m * correction - t * t;
242            assert(m as int * next as int - 1 == (p * p) * witness) by(nonlinear_arith)
243                requires
244                    m as int * next as int - 1 == -error * error + big * m * correction,
245                    error == p * t,
246                    big == co * (p * p),
247                    witness == co * m * correction - t * t;
248            lemma_mod_multiples_basic(witness, p * p);
249            #[verifier::proof_note("A wrapping Newton step squares the inverse-error divisor whenever its square divides 2^64")]
250            assert((m as int * next as int - 1) % (p * p) == 0);
251        }

253        pub proof fn inverse_precision(i: nat)
254            requires i < 6,
255            ensures
256                pow2(pow2(i + 1)) == pow2(pow2(i)) * pow2(pow2(i)),
257                pow2(pow2(i)) > 1,
258                radix() % ((pow2(pow2(i)) * pow2(pow2(i))) as int) == 0,
259        {
260            reveal(vstd::arithmetic::power::pow);
261            lemma_pow2_unfold(i + 1);
262            lemma_pow2_adds(pow2(i), pow2(i));
263            assert(pow2(5) == 32 && pow2(64) == radix()) by(compute);
264            if i < 5 { lemma_pow2_strictly_increases(i, 5); }
265            lemma_pow2_pos(i);
266            lemma_pow2_strictly_increases(0, pow2(i));
267            assert(pow2(0) == 1) by(compute);
268            let e = 2 * pow2(i);
269            lemma_pow2_adds(e, (64 - e) as nat);
270            lemma_pow2_pos(e);
271            assert(pow2(e) == pow2(pow2(i)) * pow2(pow2(i)));
272            assert(radix() == pow2(e) * pow2((64 - e) as nat));
273            vstd::arithmetic::mul::lemma_mul_is_commutative(pow2(e) as int, pow2((64 - e) as nat) as int);
274            lemma_mod_multiples_basic(pow2((64 - e) as nat) as int, pow2(e) as int);
275            #[verifier::proof_note("The six Newton steps double inverse precision from 1 bit to 64 bits without requiring a modulus larger than 2^64")]
276            assert(pow2(pow2(i + 1)) == pow2(pow2(i)) * pow2(pow2(i))
277                && radix() % (pow2(pow2(i + 1)) as int) == 0);
278        }
279    }

287    pub fn naive_divisors(n: u64) -> Vec<u64> {
288        let mut res = vec![];
289        for i in 1..(n as f32).sqrt() as u64 + 1 {
290            if n.is_multiple_of(i) {
291                res.push(i);
292                if i * i != n {
293                    res.push(n / i);
294                }
295            }
296        }
297        res.sort_unstable();
298        res
299    }
