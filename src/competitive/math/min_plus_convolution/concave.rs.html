6pub(super) fn is_concave<T>(values: &[T]) -> bool
7where
8    T: Signed,
9{
10    let n = values.len().saturating_sub(2);
11    (0..n).step_by(128).all(|start| {
12        let end = n.min(start + 128);
13        values[start..end + 2]
14            .windows(3)
15            .fold(true, |valid, w| valid & (w[1] - w[0] >= w[2] - w[1]))
16    })
17}

19/// Computes convolution when one finite input is concave using offline envelopes.
20///
21/// The running time is `O((n + m) log(n + m))`. The arbitrary input may
22/// contain `T::maximum()`.
23///
24/// # Panics
25///
26/// Panics unless at least one input is finite and concave.
27pub fn min_plus_convolution_concave_envelope<T>(a: &[T], b: &[T]) -> Vec<T>
28where
29    T: Signed,
30{
31    let len = output_len(a.len(), b.len());
32    if len == 0 {
33        return Vec::new();
34    }
35    let (arbitrary, concave) = if !b.iter().any(T::is_maximum) && is_concave(b) {
36        (a, b)
37    } else if !a.iter().any(T::is_maximum) && is_concave(a) {
38        (b, a)
39    } else {
40        panic!("at least one min-plus convolution input must be finite and concave")
41    };
42    concave_envelope(arbitrary, concave)
43}

45pub(super) fn concave_envelope<T>(arbitrary: &[T], concave: &[T]) -> Vec<T>
46where
47    T: Signed,
48{
49    if concave.len() == 1 {
50        return arbitrary
51            .iter()
52            .map(|&value| {
53                if value.is_maximum() {
54                    T::maximum()
55                } else {
56                    value + concave[0]
57                }
58            })
59            .collect();
60    }
61    if arbitrary.len() == 1 {
62        return if arbitrary[0].is_maximum() {
63            vec![T::maximum(); concave.len()]
64        } else {
65            concave.iter().map(|&value| arbitrary[0] + value).collect()
66        };
67    }
68
69    let n = concave.len();
70    if n <= CONCAVE_NAIVE_CUTOFF || arbitrary.len() == 2 {
71        return min_plus_convolution_naive(arbitrary, concave);
72    }
73    let mut result = vec![T::maximum(); output_len(arbitrary.len(), n)];
74    let mut stack = Vec::new();
75    let mut ranks = None;
76    for (block, values) in arbitrary.chunks(n).enumerate() {
77        let output = &mut result[block * n..][..n + values.len() - 1];
78        sweep::<false, _>(concave, values, &mut output[..n], &mut stack, &mut ranks);
79        sweep::<true, _>(concave, values, &mut output[n..], &mut stack, &mut ranks);
80    }
81    result
82}

84/// Computes convolution of two concave inputs from antidiagonal endpoints.
85///
86/// The running time is `O(n + m)`.
87///
88/// # Panics
89///
90/// Panics unless both inputs are finite and concave.
91pub fn min_plus_convolution_concave_both<T>(a: &[T], b: &[T]) -> Vec<T>
92where
93    T: Signed,
94{
95    let len = output_len(a.len(), b.len());
96    if len == 0 {
97        return Vec::new();
98    }
99    assert_finite(a);
100    assert_finite(b);
101    assert!(
102        is_concave(a) && is_concave(b),
103        "both inputs must be concave"
104    );
105    concave_both(a, b)
106}

108pub(super) fn concave_both<T>(a: &[T], b: &[T]) -> Vec<T>
109where
110    T: Signed,
111{
112    let len = output_len(a.len(), b.len());
113    let mut result = Vec::with_capacity(len);
114    for output in 0..len {
115        let first = output.saturating_sub(b.len() - 1);
116        let last = output.min(a.len() - 1);
117        result.push((a[first] + b[output - first]).min(a[last] + b[output - last]));
118    }
119    result
120}

122/// At each row, the newest shift wins a prefix of the remaining rows.
123/// Reversing both coordinates gives the same invariant past the end of a block.
124fn sweep<const REVERSE: bool, T>(
125    concave: &[T],
126    values: &[T],
127    output: &mut [T],
128    stack: &mut Vec<(usize, T, usize)>,
129    ranks: &mut Option<SlopeRanks<T>>,
130) where
131    T: Signed,
132{
133    let len = output.len();
134    let value = |row, column, bias| {
135        let offset: usize = row - column;
136        let index = if REVERSE {
137            concave.len() - 1 - offset
138        } else {
139            offset
140        };
141        concave[index] + bias
142    };
143    let count = len.min(values.len());
144    let bias_at = |row: usize| values[if REVERSE { values.len() - 1 - row } else { row }];
145    let Some(first) = (0..count).find(|&row| !bias_at(row).is_maximum()) else {
146        return;
147    };
148    let dest = |row| if REVERSE { len - 1 - row } else { row };
149    let joining = if REVERSE {
150        concave[concave.len() - 1]
151    } else {
152        concave[0]
153    };
154    let bias = bias_at(first);
155    output[dest(first)] = output[dest(first)].min(joining + bias);
156    stack.clear();
157    let mut top = (first, bias, len);
158    let mut wide = 0;
159    let mut crossing =
160        |(row, bias): (usize, T), (col, old): (usize, T), mut lo: usize, mut hi: usize| {
161            let d = row - col;
162            if hi - lo > 32 && d <= 64 {
163                if ranks.is_none() {
164                    wide += 1;
165                    if wide == 64 {
166                        *ranks = Some(SlopeRanks::new(concave));
167                    }
168                }
169                if let Some(ranks) = ranks {
170                    let (low, high) = ranks.crossing::<REVERSE>(bias, old, d, concave.len() - 1);
171                    lo = lo.max(row + low.saturating_sub(d));
172                    hi = hi.min(row.saturating_add(high));
173                }
174            }
175            if hi - lo >= 8 {
176                let offset = |column| {
177                    if REVERSE {
178                        concave.len() - 8 - (lo - column)
179                    } else {
180                        lo - column
181                    }
182                };
183                let a = &concave[offset(row)..][..8];
184                let b = &concave[offset(col)..][..8];
185                let wins: u8 = a
186                    .iter()
187                    .zip(b)
188                    .map(|(&a, &b)| u8::from(a + bias <= b + old))
189                    .sum();
190                if wins < 8 {
191                    return lo + usize::from(wins);
192                }
193                lo += 7;
194            }
195            let mut step = 1;
196            while hi - lo > step {
197                let probe = lo + step;
198                if value(probe, row, bias) > value(probe, col, old) {
199                    hi = probe;
200                    break;
201                }
202                lo = probe;
203                step = step.saturating_mul(2);
204            }
205            while hi - lo > 1 {
206                let mid = lo + (hi - lo) / 2;
207                if value(mid, row, bias) <= value(mid, col, old) {
208                    lo = mid;
209                } else {
210                    hi = mid;
211                }
212            }
213            hi
214        };
215    for row in first + 1..count {
216        if top.2 == row {
217            top = stack.pop().unwrap();
218        }
219        let current = value(row, top.0, top.1);
220        let bias = bias_at(row);
221        let best = if !bias.is_maximum() && joining + bias <= current {
222            let mut lo = row;
223            let mut end = len;
224            loop {
225                let (col, old, upper) = top;
226                if value(upper - 1, row, bias) > value(upper - 1, col, old) {
227                    stack.push(top);
228                    end = crossing((row, bias), (col, old), lo, upper - 1);
229                    break;
230                }
231                lo = upper - 1;
232                let Some(previous) = stack.pop() else { break };
233                top = previous;
234            }
235            top = (row, bias, end);
236            joining + bias
237        } else {
238            current
239        };
240        output[dest(row)] = output[dest(row)].min(best);
241    }
242
243    let mut row = values.len().min(len);
244    for (col, bias, end) in std::iter::once(top).chain(stack.drain(..).rev()) {
245        if row < end {
246            // Only the forward sweep can have rows left after the last insertion.
247            for (dest, &v) in output[row..end].iter_mut().zip(&concave[row - col..]) {
248                *dest = (*dest).min(v + bias);
249            }
250            row = end;
251        }
252    }
253}

255struct SlopeRanks<T>
256where
257    T: Signed,
258{
259    minimum: T,
260    maximum: T,
261    width: T::Unsigned,
262    counts: Vec<usize>,
263    distances: [T::Unsigned; 64],
264}
265
266impl<T> SlopeRanks<T>
267where
268    T: Signed,
269{
270    fn new(a: &[T]) -> Self {
271        let minimum = a[a.len() - 1] - a[a.len() - 2];
272        let maximum = a[1] - a[0];
273        let span = maximum.abs_diff(minimum);
274        let shift = if span.is_zero() {
275            0
276        } else {
277            span.ilog2().saturating_sub(12)
278        };
279        let mut distance = T::zero();
280        let mut ranks = Self {
281            minimum,
282            maximum,
283            width: (T::one() + T::one()).unsigned().pow(shift),
284            counts: Vec::new(),
285            distances: std::array::from_fn(|_| {
286                distance += T::one();
287                distance.unsigned()
288            }),
289        };
290        ranks.counts.resize(ranks.bucket(maximum) + 2, 0);
291        for w in a.windows(2) {
292            let index = ranks.bucket(w[1] - w[0]);
293            ranks.counts[index] += 1;
294        }
295        for i in (1..ranks.counts.len()).rev() {
296            ranks.counts[i - 1] += ranks.counts[i];
297        }
298        ranks
299    }
300
301    fn bucket(&self, slope: T) -> usize {
302        let delta = slope.abs_diff(self.minimum);
303        let mut key = if self.width == T::one().unsigned() {
304            delta
305        } else {
306            delta / self.width
307        };
308        let two = (T::one() + T::one()).unsigned();
309        // The key fits 13 bits, including for integer types wider than usize.
310        (0..13).fold(0, |index, bit| {
311            let digit = usize::from(!key.is_multiple_of(two));
312            key /= two;
313            index | digit << bit
314        })
315    }
316
317    fn crossing<const REVERSE: bool>(
318        &self,
319        new: T,
320        old: T,
321        d: usize,
322        length: usize,
323    ) -> (usize, usize) {
324        let gap = new.abs_diff(old);
325        let magnitude = if new < old {
326            gap / self.distances[d - 1]
327        } else {
328            gap.div_ceil(self.distances[d - 1])
329        };
330        if magnitude > T::maximum().unsigned() {
331            return (0, length);
332        }
333        let magnitude = magnitude.signed();
334        let slope = if (new < old) != REVERSE {
335            -magnitude
336        } else {
337            magnitude
338        };
339        if slope < self.minimum || slope > self.maximum {
340            return (0, length);
341        }
342        let i = self.bucket(slope);
343        // The sum of d consecutive slopes lies between d times the endpoint slopes.
344        // A reversed sweep counts slopes <= -ceil(gap/d) in the original kernel.
345        if self.width == T::one().unsigned() {
346            let rank = self.counts[i + usize::from(REVERSE)];
347            let rank = if REVERSE { length - rank } else { rank };
348            (rank, rank)
349        } else if REVERSE {
350            (length - self.counts[i], length - self.counts[i + 1])
351        } else {
352            (self.counts[i + 1], self.counts[i])
353        }
354    }
355}

362    #[test]
363    fn test_concave_algorithms_randomly() {
364        let mut rng = Xorshift::default();
365        for _ in 0..1_000 {
366            let arbitrary_len = rng.random(0..=4 * CONCAVE_NAIVE_CUTOFF);
367            let concave_len = rng.random(0..=2 * CONCAVE_NAIVE_CUTOFF);
368            let arbitrary: Vec<_> = (0..arbitrary_len)
369                .map(|_| {
370                    if rng.random(0_u64..8) == 0 {
371                        i64::MAX
372                    } else {
373                        rng.random(-50_i64..=50)
374                    }
375                })
376                .collect();
377            let mut slopes: Vec<_> = rng
378                .random_iter(-20_i64..=20)
379                .take(concave_len.saturating_sub(1))
380                .collect();
381            slopes.sort_unstable_by(|a, b| b.cmp(a));
382            let mut concave = Vec::with_capacity(concave_len);
383            if concave_len != 0 {
384                concave.push(rng.random(-50_i64..=50));
385            }
386            for slope in slopes {
387                concave.push(concave[concave.len() - 1] + slope);
388            }
389            let other_len: usize = rng.random(0..=32);
390            let mut slopes: Vec<_> = rng
391                .random_iter(-20_i64..=20)
392                .take(other_len.saturating_sub(1))
393                .collect();
394            slopes.sort_unstable_by(|a, b| b.cmp(a));
395            let mut other = Vec::with_capacity(other_len);
396            if other_len != 0 {
397                other.push(rng.random(-50_i64..=50));
398            }
399            for slope in slopes {
400                other.push(other[other.len() - 1] + slope);
401            }
402            assert_eq!(
403                min_plus_convolution_concave_envelope(&arbitrary, &concave),
404                min_plus_convolution_naive(&arbitrary, &concave)
405            );
406            assert_eq!(
407                min_plus_convolution_concave_both(&concave, &other),
408                min_plus_convolution_naive(&concave, &other)
409            );
410        }
411        let constant = [1_000_000_000i32; 33];
412        assert_eq!(
413            min_plus_convolution_concave_envelope(&constant, &constant),
414            [2_000_000_000; 65]
415        );
416        for (n, m) in [
417            (1, 4097),
418            (2, 4097),
419            (4097, 1),
420            (4097, 2),
421            (31, 1025),
422            (32, 1025),
423            (33, 1025),
424            (63, 511),
425            (64, 511),
426            (65, 511),
427            (1024, 2051),
428        ] {
429            for scale in [1i64, 65537] {
430                let kernel: Vec<_> = (0..n).map(|i| i * (2 * n - i) * scale).collect();
431                let mut value = 0;
432                let mut arbitrary: Vec<_> = (0..m)
433                    .map(|_| {
434                        value += rng.random(0..=2 * n * scale);
435                        value
436                    })
437                    .collect();
438                arbitrary[m / 3] = i64::MAX;
439                for _ in 0..2 {
440                    assert_eq!(
441                        min_plus_convolution_concave_envelope(&arbitrary, &kernel),
442                        min_plus_convolution_naive(&arbitrary, &kernel)
443                    );
444                    arbitrary.reverse();
445                }
446                if n == 1024 {
447                    // The gap isolates the final two shifts after the slope table has been built.
448                    arbitrary.truncate(n as usize);
449                    arbitrary.resize(2 * n as usize, i64::MAX);
450                    arbitrary.extend([0, (n + 1) * scale + 1]);
451                    assert_eq!(
452                        min_plus_convolution_concave_envelope(&arbitrary, &kernel),
453                        min_plus_convolution_naive(&arbitrary, &kernel)
454                    );
455                }
456            }
457        }
458    }

461    fn test_concave_envelope_exhaustively() {
462        let values = [-2_i64, 0, 3, i64::MAX];
463        let mut inputs = vec![Vec::new()];
464        for _ in 0..3 {
465            let prefixes = inputs.clone();
466            for prefix in prefixes {
467                for &value in &values {
468                    let mut input = prefix.clone();
469                    input.push(value);
470                    inputs.push(input);
471                }
472            }
473        }
474        inputs.sort();
475        inputs.dedup();
476        let concave: Vec<_> = inputs
477            .iter()
478            .filter(|input| !input.contains(&i64::MAX) && is_concave(input))
479            .collect();
480        for arbitrary in &inputs {
481            for &structured in &concave {
482                assert_eq!(
483                    min_plus_convolution_concave_envelope(arbitrary, structured),
484                    min_plus_convolution_naive(arbitrary, structured)
485                );
486            }
