10impl<R> SubsetConvolve<R>
11where
12    R: Ring<T: PartialEq, Additive: Invertible>,

14    /// Sums products of block weights over the partitions of each subset.
15    /// The input has power-of-two length and zero constant term over a commutative ring.
16    pub fn exp(weights: &[R::T]) -> Vec<R::T> {
17        assert!(weights.len().is_power_of_two() && weights[0] == R::zero());
18        let mut result = vec![R::one()];
19        while result.len() < weights.len() {
20            let half = result.len();
21            let upper =
22                Self::convolve_impl(result.as_slice().into(), (&weights[half..2 * half]).into());
23            result.extend(upper);
24        }
25        result
26    }

28    fn ranked(t: Vec<R::T>, len: usize) -> (Vec<R::T>, usize) {
29        let width = len.trailing_zeros() as usize + 1;
30        let mut ranked = Vec::with_capacity(len * width);
31        advise_huge_pages(&mut ranked);
32        ranked.resize(len * width, R::zero());
33        for (i, value) in t.into_iter().enumerate() {
34            ranked[i * width + i.count_ones() as usize] = value;
35        }
36        (ranked, width)
37    }

39    fn diagonal(ranked: Vec<R::T>, width: usize) -> Vec<R::T> {
40        ranked
41            .chunks_exact(width)
42            .enumerate()
43            .map(|(i, row)| row[i.count_ones() as usize].clone())
44            .collect()
45    }

47    #[inline]
48    fn multiply_row(
49        x: &[R::T],
50        y: &[R::T],
51        right: &mut [R::T],
52        output: &mut [R::T],
53        rank: usize,
54    ) -> usize {
55        for (right, y) in right[..=rank].iter_mut().zip(y[..=rank].iter().rev()) {
56            right.clone_from(y);
57        }
58        let end = (rank * 2).min(x.len() - 1);
59        for (degree, output) in output.iter_mut().enumerate().take(end + 1).skip(rank) {
60            let first = degree - rank;
61            *output = R::dot_product(&x[first..=rank], &right[..=rank - first]);
62        }
63        end
64    }

66    fn convolve_impl(a: Cow<'_, [R::T]>, b: Cow<'_, [R::T]>) -> Vec<R::T> {
67        const BATCH: usize = 128;
68        assert_eq!(a.len(), b.len());
69        let len = a.len();
70        if len >= 8 * BATCH
71            && let Some(inverse_two) = R::inverse_two()
72        {
73            assert!(len.is_power_of_two());
74            crate::avx_helper!(@versions avx512, avx2; <R>(
75                a: &[R::T], b: &[R::T], inverse_two: R::T,
76            ) -> Vec<R::T> where R: Ring<T: PartialEq, Additive: Invertible> {
77                SubsetConvolve::<R>::ranked_xor::<BATCH>(a, b, inverse_two)
78            });
79            // SAFETY: the dispatcher checks the features used by each kernel.
80            return crate::avx_helper!(@dispatch simd_backend, SimdBackend;
81                unsafe { avx512::<R>(&a, &b, inverse_two) },
82                unsafe { avx2::<R>(&a, &b, inverse_two) },
83                Self::ranked_xor::<BATCH>(&a, &b, inverse_two)
84            );
85        }
86        let same = a == b;
87        let (mut x, width) = Self::ranked(a.into_owned(), len);
88        let (mut y, _) = if same {
89            (x.clone(), width)
90        } else {
91            Self::ranked(b.into_owned(), len)
92        };
93        let mut right = vec![R::zero(); width];
94        let mut output = vec![R::zero(); width];
95        for i in 0..len {
96            for bit in (0..(i | len).trailing_zeros() as usize).rev() {
97                let half = width << bit;
98                let start = i * width;
99                let (lower, upper) = x[start..start + half * 2].split_at_mut(half);
100                for (upper, lower) in upper.iter_mut().zip(lower) {
101                    R::add_assign(upper, lower);
102                }
103                let (lower, upper) = y[start..start + half * 2].split_at_mut(half);
104                for (upper, lower) in upper.iter_mut().zip(lower) {
105                    R::add_assign(upper, lower);
106                }
107            }
108
109            let rank = i.count_ones() as usize;
110            let start = i * width;
111            let x_row = &x[start..start + width];
112            let y_row = &y[start..start + width];
113            output.fill(R::zero());
114            Self::multiply_row(x_row, y_row, &mut right, &mut output, rank);
115            x[start..start + width].clone_from_slice(&output);
116
117            for bit in 0..i.trailing_ones() as usize {
118                let end = (i + 1) * width;
119                let half = width << bit;
120                let (lower, upper) = x[end - half * 2..end].split_at_mut(half);
121                for (upper, lower) in upper.iter_mut().zip(lower) {
122                    R::sub_assign(upper, lower);
123                }
124            }
125        }
126        Self::diagonal(x, width)
127    }

129    #[inline(always)]
130    fn ranked_xor<const W: usize>(a: &[R::T], b: &[R::T], inverse_two: R::T) -> Vec<R::T> {
131        let n = a.len().ilog2() as usize;
132        let width = n - 1;
133        let ranks = std::array::from_fn(|i| i.count_ones() as usize);
134        let zero = std::array::from_fn(|_| R::zero());
135        let mut x = vec![zero.clone(); a.len() / (8 * W) * width];
136        let mut y = x.clone();
137        let mut local = vec![zero.clone(); W.ilog2() as usize + 4];
138        let mut result = vec![R::zero(); a.len()];
139
140        // |A| + |B| - |A xor B| = 2 |A & B|. The rank sum also fixes the parity
141        // of A xor B, so its lowest bit can be omitted from the Walsh transform.
142        for frequency in 0..4 {
143            local.fill(zero.clone());
144            Self::project_ranks::<W, false>(a, &mut x, &mut local, frequency, width, &ranks);
145            Self::project_ranks::<W, true>(b, &mut y, &mut local, frequency, width, &ranks);
146            for (x, y) in x.chunks_exact_mut(width).zip(y.chunks_exact(width)) {
147                for k in (0..width).rev() {
148                    x[k] = R::dot_product_batch(&x[..=k], &y[width - 1 - k..]);
149                }
150            }
151            BitwisexorConvolve::<R::Additive>::hadamard_transform_rows(
152                x.as_flattened_mut(),
153                width * W,
154            );
155            for (middle, (block, output)) in x
156                .chunks_exact(width)
157                .zip(result.chunks_exact_mut(8 * W))
158                .enumerate()
159            {
160                let base = middle.count_ones() as usize;
161                // Omitting both constant terms leaves product ranks 2 through n.
162                for (k, row) in local.iter_mut().enumerate() {
163                    if base + k < 2 {
164                        row.fill(R::zero());
165                    } else {
166                        row.clone_from(&block[base + k - 2]);
167                        BitwisexorConvolve::<R::Additive>::hadamard_transform(row);
168                    }
169                }
170                for (j, (out, &rank)) in output
171                    .as_chunks_mut::<8>()
172                    .0
173                    .iter_mut()
174                    .zip(&ranks)
175                    .enumerate()
176                {
177                    let low_ranks = [0, 1, 1, 2, 1, 2, 2, 3];
178                    for ((i, value), low_rank) in out.iter_mut().enumerate().zip(low_ranks) {
179                        let term = &local[rank + low_rank][j];
180                        if frequency == 0 {
181                            value.clone_from(term);
182                        } else if [false, true, true, false][frequency & (i >> 1)] {
183                            R::sub_assign(value, term);
184                        } else {
185                            R::add_assign(value, term);
186                        }
187                    }
188                }
189            }
190        }
191        let mut scale = R::one();
192        for _ in 1..n {
193            R::mul_assign(&mut scale, &inverse_two);
194        }
195        result[0] = R::mul(&a[0], &b[0]);
196        for i in 1..a.len() {
197            result[i] = R::dot_product(
198                &[result[i].clone(), a[0].clone(), a[i].clone()],
199                &[scale.clone(), b[i].clone(), b[0].clone()],
200            );
201        }
202        result
203    }

205    #[inline(always)]
206    fn project_ranks<const W: usize, const REVERSE: bool>(
207        input: &[R::T],
208        transformed: &mut [[R::T; W]],
209        local: &mut [[R::T; W]],
210        frequency: usize,
211        width: usize,
212        ranks: &[usize; W],
213    ) {
214        let first = |x: &R::T, y: &R::T| {
215            if frequency & 1 == 0 {
216                R::add(x, y)
217            } else {
218                R::sub(x, y)
219            }
220        };
221        let second = |x: &R::T, y: &R::T| {
222            if frequency & 2 == 0 {
223                R::add(x, y)
224            } else {
225                R::sub(x, y)
226            }
227        };
228        let zero = std::array::from_fn(|_| R::zero());
229        for (middle, (src, block)) in input
230            .chunks_exact(8 * W)
231            .zip(transformed.chunks_exact_mut(width))
232            .enumerate()
233        {
234            for (j, (s, &k)) in src.as_chunks::<8>().0.iter().zip(ranks).enumerate() {
235                local[k][j].clone_from(&s[0]);
236                local[k + 1][j] = second(&first(&s[1], &s[2]), &s[4]);
237                let low = if frequency & 1 == 0 {
238                    s[3].clone()
239                } else {
240                    R::neg(&s[3])
241                };
242                local[k + 2][j] = second(&low, &first(&s[5], &s[6]));
243                local[k + 3][j] = if frequency == 1 || frequency == 2 {
244                    R::neg(&s[7])
245                } else {
246                    s[7].clone()
247                };
248            }
249            let base = middle.count_ones() as usize;
250            let end = (base + local.len() - 1).min(width);
251            block.fill(zero.clone());
252            for rank in base.max(1)..=end {
253                let source = &local[rank - base];
254                let index = if REVERSE { width - rank } else { rank - 1 };
255                block[index].clone_from(source);
256                BitwisexorConvolve::<R::Additive>::hadamard_transform(&mut block[index]);
257            }
258        }
259        BitwisexorConvolve::<R::Additive>::hadamard_transform_rows(
260            transformed.as_flattened_mut(),
261            width * W,
262        );
263    }

266impl<R> ConvolveSteps for SubsetConvolve<R>
267where
268    R: Ring<T: PartialEq, Additive: Invertible>,
269{
270    type T = Vec<R::T>;
271    type F = (Vec<R::T>, usize);
272
273    fn length(t: &Self::T) -> usize {
274        t.len()
275    }
276
277    fn transform(t: Self::T, len: usize) -> Self::F {
278        let (mut f, width) = Self::ranked(t, len);
279        let k = width - 1;
280        for bit in 0..k {
281            let half = 1 << bit;
282            for base in (0..len).step_by(half * 2) {
283                for lower in base..base + half {
284                    let upper = lower + half;
285                    let ranks = lower.count_ones() as usize + 1;
286                    let (lower_rows, upper_rows) = f.split_at_mut(upper * width);
287                    let lower_row = &lower_rows[lower * width..lower * width + ranks];
288                    let upper_row = &mut upper_rows[..ranks];
289                    for (upper, lower) in upper_row.iter_mut().zip(lower_row) {
290                        R::add_assign(upper, lower);
291                    }
292                }
293            }
294        }
295        (f, width)
296    }
297
298    fn inverse_transform((mut f, width): Self::F, len: usize) -> Self::T {
299        let k = width - 1;
300        for bit in 0..k {
301            let half = 1 << bit;
302            for base in (0..len).step_by(half * 2) {
303                for lower in base..base + half {
304                    let upper = lower + half;
305                    let rank = lower.count_ones() as usize;
306                    let (lower_rows, upper_rows) = f.split_at_mut(upper * width);
307                    let lower_row = &lower_rows[lower * width + rank..lower * width + width];
308                    let upper_row = &mut upper_rows[rank..width];
309                    for (upper, lower) in upper_row.iter_mut().zip(lower_row) {
310                        R::sub_assign(upper, lower);
311                    }
312                }
313            }
314        }
315        Self::diagonal(f, width)
316    }
317
318    fn multiply(f: &mut Self::F, g: &Self::F) {
319        let (f, width) = f;
320        let (g, _) = g;
321        let mut right = vec![R::zero(); *width];
322        let mut output = vec![R::zero(); *width];
323        for (i, f) in f.chunks_exact_mut(*width).enumerate() {
324            let rank = i.count_ones() as usize;
325            let g = &g[i * *width..(i + 1) * *width];
326            let end = Self::multiply_row(f, g, &mut right, &mut output, rank);
327            f[rank..=end].clone_from_slice(&output[rank..=end]);
328        }
329    }
330
331    fn convolve(a: Self::T, b: Self::T) -> Self::T {
332        Self::convolve_impl(a.into(), b.into())
333    }
334}

348    fn test_subset_convolve() {
349        let mut rng = Xorshift::default();
350
351        for k in 0..12 {
352            let n = 1 << k;
353            rand!(rng, f: [-A..A; n], g: [-A..A; n]);
354            let mut h = vec![0i64; n];
355            for i in 0..n {
356                for j in 0..n {
357                    if i & j == 0 {
358                        h[i | j] += f[i] * g[j];
359                    }
360                }
361            }
362            let mut transformed = SubsetConvolve::<AddMulOperation<i64>>::transform(f.clone(), n);
363            let other = SubsetConvolve::<AddMulOperation<i64>>::transform(g.clone(), n);
364            SubsetConvolve::<AddMulOperation<i64>>::multiply(&mut transformed, &other);
365            let j = SubsetConvolve::<AddMulOperation<i64>>::inverse_transform(transformed, n);
366            let i = SubsetConvolve::<AddMulOperation<i64>>::convolve(f, g);
367            assert_eq!(h, i);
368            assert_eq!(h, j);
369        }
370    }

373    fn set_partitions() {
374        use crate::num::{One, Zero, mint_basic::MInt998244353 as M};
375
376        let mut rng = Xorshift::default();
377        for k in 0..=12 {
378            let len = 1 << k;
379            for weights in sequence_cases(&mut rng, -2i64..=2, len..=len).take(16) {
380                let mut weights: Vec<_> = weights.into_iter().map(M::from).collect();
381                weights[0] = M::zero();
382                let mut expected = vec![M::zero(); len];
383                expected[0] = M::one();
384                for mask in 1..len {
385                    let rest = mask & (mask - 1);
386                    let mut subset = rest;
387                    loop {
388                        let term = expected[subset] * weights[mask ^ subset];
389                        expected[mask] += term;
390                        if subset == 0 {
391                            break;
392                        }
393                        subset = (subset - 1) & rest;
394                    }
395                }
396                assert_eq!(
397                    SubsetConvolve::<AddMulOperation<_>>::exp(&weights),
398                    expected
399                );
400            }
401        }

404    #[test]
405    fn modular_subset_products() {
406        use crate::num::{MInt, MIntConvert, MIntDotProduct, mint_basic::DynModuloU32, montgomery};
407
408        fn check<M>(rng: &mut Xorshift)
409        where
410            M: MIntDotProduct<Inner = u32> + MIntConvert<u32>,
411        {
412            let modulus = M::get_mod();
413            for bits in 0..=11 {
414                let n = 1 << bits;
415                let mut coefficient = || MInt::<M>::new(rng.random(..modulus));
416                let a: Vec<_> = (0..n).map(|_| coefficient()).collect();
417                let b: Vec<_> = (0..n).map(|_| coefficient()).collect();
418                let expected: Vec<_> = (0..n)
419                    .map(|mask| {
420                        let mut sub = mask;
421                        let mut sum = 0u128;
422                        loop {
423                            sum += u128::from(a[sub].inner()) * u128::from(b[mask ^ sub].inner());
424                            if sub == 0 {
425                                break;
426                            }
427                            sub = (sub - 1) & mask;
428                        }
429                        MInt::<M>::new((sum % u128::from(modulus)) as u32)
430                    })
431                    .collect();
432                assert_eq!(
433                    SubsetConvolve::<AddMulOperation<_>>::convolve(a, b),
434                    expected
435                );
436            }
437        }
438        let mut rng = Xorshift::default();
439        for modulus in [1, 2, 9, 1 << 31, u32::MAX] {
440            DynModuloU32::set_mod(modulus);
441            check::<DynModuloU32>(&mut rng);
442        }
443        DynModuloU32::set_mod(1_000_000_007);
444        check::<montgomery::Modulo998244353>(&mut rng);
