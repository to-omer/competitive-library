2use std::marker::PhantomData;

4pub struct SubsetConvolve<M> {
5    _marker: PhantomData<fn() -> M>,
6}
7
8impl<R> SubsetConvolve<R>
9where
10    R: Ring<T: PartialEq, Additive: Invertible>,
11{
12    fn ranked(t: Vec<R::T>, len: usize) -> (Vec<R::T>, usize) {
13        let width = len.trailing_zeros() as usize + 1;
14        let mut ranked = vec![R::zero(); len * width];
15        for (i, value) in t.into_iter().enumerate() {
16            ranked[i * width + i.count_ones() as usize] = value;
17        }
18        (ranked, width)
19    }
20
21    fn diagonal(ranked: Vec<R::T>, width: usize) -> Vec<R::T> {
22        ranked
23            .chunks_exact(width)
24            .enumerate()
25            .map(|(i, row)| row[i.count_ones() as usize].clone())
26            .collect()
27    }
28
29    #[inline]
30    fn multiply_row(
31        x: &[R::T],
32        y: &[R::T],
33        right: &mut [R::T],
34        output: &mut [R::T],
35        rank: usize,
36    ) -> usize {
37        for (right, y) in right[..=rank].iter_mut().zip(y[..=rank].iter().rev()) {
38            right.clone_from(y);
39        }
40        let end = (rank * 2).min(x.len() - 1);
41        for (degree, output) in output.iter_mut().enumerate().take(end + 1).skip(rank) {
42            let first = degree - rank;
43            *output = R::dot_product(&x[first..=rank], &right[..=rank - first]);
44        }
45        end
46    }
47}

49impl<R> ConvolveSteps for SubsetConvolve<R>
50where
51    R: Ring<T: PartialEq, Additive: Invertible>,
52{
53    type T = Vec<R::T>;
54    type F = (Vec<R::T>, usize);
55
56    fn length(t: &Self::T) -> usize {
57        t.len()
58    }
59
60    fn transform(t: Self::T, len: usize) -> Self::F {
61        let (mut f, width) = Self::ranked(t, len);
62        let k = width - 1;
63        for bit in 0..k {
64            let half = 1 << bit;
65            for base in (0..len).step_by(half * 2) {
66                for lower in base..base + half {
67                    let upper = lower + half;
68                    let ranks = lower.count_ones() as usize + 1;
69                    let (lower_rows, upper_rows) = f.split_at_mut(upper * width);
70                    let lower_row = &lower_rows[lower * width..lower * width + ranks];
71                    let upper_row = &mut upper_rows[..ranks];
72                    for (upper, lower) in upper_row.iter_mut().zip(lower_row) {
73                        R::add_assign(upper, lower);
74                    }
75                }
76            }
77        }
78        (f, width)
79    }
80
81    fn inverse_transform((mut f, width): Self::F, len: usize) -> Self::T {
82        let k = width - 1;
83        for bit in 0..k {
84            let half = 1 << bit;
85            for base in (0..len).step_by(half * 2) {
86                for lower in base..base + half {
87                    let upper = lower + half;
88                    let rank = lower.count_ones() as usize;
89                    let (lower_rows, upper_rows) = f.split_at_mut(upper * width);
90                    let lower_row = &lower_rows[lower * width + rank..lower * width + width];
91                    let upper_row = &mut upper_rows[rank..width];
92                    for (upper, lower) in upper_row.iter_mut().zip(lower_row) {
93                        R::sub_assign(upper, lower);
94                    }
95                }
96            }
97        }
98        Self::diagonal(f, width)
99    }
100
101    fn multiply(f: &mut Self::F, g: &Self::F) {
102        let (f, width) = f;
103        let (g, _) = g;
104        let mut right = vec![R::zero(); *width];
105        let mut output = vec![R::zero(); *width];
106        for (i, f) in f.chunks_exact_mut(*width).enumerate() {
107            let rank = i.count_ones() as usize;
108            let g = &g[i * *width..(i + 1) * *width];
109            let end = Self::multiply_row(f, g, &mut right, &mut output, rank);
110            f[rank..=end].clone_from_slice(&output[rank..=end]);
111        }
112    }
113
114    fn convolve(a: Self::T, b: Self::T) -> Self::T {
115        assert_eq!(a.len(), b.len());
116        let len = a.len();
117        let same = a == b;
118        let (mut x, width) = Self::ranked(a, len);
119        let (mut y, _) = if same {
120            (x.clone(), width)
121        } else {
122            Self::ranked(b, len)
123        };
124        let mut right = vec![R::zero(); width];
125        let mut output = vec![R::zero(); width];
126        for i in 0..len {
127            for bit in (0..(i | len).trailing_zeros() as usize).rev() {
128                let half = width << bit;
129                let start = i * width;
130                let (lower, upper) = x[start..start + half * 2].split_at_mut(half);
131                for (upper, lower) in upper.iter_mut().zip(lower) {
132                    R::add_assign(upper, lower);
133                }
134                let (lower, upper) = y[start..start + half * 2].split_at_mut(half);
135                for (upper, lower) in upper.iter_mut().zip(lower) {
136                    R::add_assign(upper, lower);
137                }
138            }
139
140            let rank = i.count_ones() as usize;
141            let start = i * width;
142            let x_row = &x[start..start + width];
143            let y_row = &y[start..start + width];
144            output.fill(R::zero());
145            Self::multiply_row(x_row, y_row, &mut right, &mut output, rank);
146            x[start..start + width].clone_from_slice(&output);
147
148            for bit in 0..i.trailing_ones() as usize {
149                let end = (i + 1) * width;
150                let half = width << bit;
151                let (lower, upper) = x[end - half * 2..end].split_at_mut(half);
152                for (upper, lower) in upper.iter_mut().zip(lower) {
153                    R::sub_assign(upper, lower);
154                }
155            }
156        }
157        Self::diagonal(x, width)
158    }
159}

161#[cfg(test)]
162mod tests {
163    use super::*;
164    use crate::{algebra::AddMulOperation, rand, tools::Xorshift};
165
166    const A: i64 = 100_000;
167
168    #[test]
169    fn test_subset_convolve() {
170        let mut rng = Xorshift::default();
171
172        for k in 0..12 {
173            let n = 1 << k;
174            rand!(rng, f: [-A..A; n], g: [-A..A; n]);
175            let mut h = vec![0i64; n];
176            for i in 0..n {
177                for j in 0..n {
178                    if i & j == 0 {
179                        h[i | j] += f[i] * g[j];
180                    }
181                }
182            }
183            let mut transformed = SubsetConvolve::<AddMulOperation<i64>>::transform(f.clone(), n);
184            let other = SubsetConvolve::<AddMulOperation<i64>>::transform(g.clone(), n);
185            SubsetConvolve::<AddMulOperation<i64>>::multiply(&mut transformed, &other);
186            let j = SubsetConvolve::<AddMulOperation<i64>>::inverse_transform(transformed, n);
187            let i = SubsetConvolve::<AddMulOperation<i64>>::convolve(f, g);
188            assert_eq!(h, i);
189            assert_eq!(h, j);
190        }
191    }
