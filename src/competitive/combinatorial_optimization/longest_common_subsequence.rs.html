5pub fn longest_common_subsequence_length<T>(x: &[T], y: &[T]) -> usize
6where
7    T: Eq + Hash,
8{
9    let (x, y) = if x.len() <= y.len() { (x, y) } else { (y, x) };
10    if x.is_empty() {
11        return 0;
12    }
13    if x.len() == 1 {
14        return usize::from(y.contains(&x[0]));
15    }
16    // SAFETY: simd_lanes checks the CPU features required by the selected lane count.
17    let (mut state, x, y) = unsafe { LcsState::new(x, y, simd_lanes(x.len())) };
18    state.lengths(&x, &y, false, |_| {})
19}

21/// Returns one longest common subsequence as increasing pairs of indices into `x` and `y`.
22pub fn longest_common_subsequence<T>(x: &[T], y: &[T]) -> Vec<(usize, usize)>
23where
24    T: Eq + Hash,
25{
26    let (x, y, flipped) = if x.len() <= y.len() {
27        (x, y, false)
28    } else {
29        (y, x, true)
30    };
31    let mut indices = Vec::with_capacity(x.len());
32    if x.is_empty() {
33        return indices;
34    }
35    if x.len() == 1 {
36        if let Some(j) = y.iter().position(|value| value == &x[0]) {
37            indices.push(if flipped { (j, 0) } else { (0, j) });
38        }
39        return indices;
40    }
41    // SAFETY: simd_lanes checks the CPU features required by the selected lane count.
42    let (mut state, x, y) = unsafe { LcsState::new(x, y, simd_lanes(x.len())) };
43    state.reconstruct(&x, &y, 0, 0, flipped, &mut indices);
44    indices
45}
46
47fn simd_lanes(m: usize) -> usize {
48    #[cfg(target_arch = "x86_64")]
49    if m >= 256 && is_x86_feature_detected!("avx2") {
50        return if avx512_enabled() && is_x86_feature_detected!("avx512f") {
51            8
52        } else {
53            4
54        };
55    }
56    let _ = m;
57    1
58}

60struct LcsState {
61    masks: Vec<u64>,
62    lanes: usize,
63}
64
65impl LcsState {
66    // lanes is 1, 4, or 8, with AVX2 available for 4/8 and AVX-512F for 8.
67    unsafe fn new<T>(x: &[T], y: &[T], lanes: usize) -> (Self, Vec<usize>, Vec<usize>)
68    where
69        T: Eq + Hash,
70    {
71        let mut ids = HashMap::new();
72        let mut pattern: Vec<_> = x
73            .iter()
74            .map(|value| {
75                let id = ids.len();
76                *ids.entry(value).or_insert(id)
77            })
78            .collect();
79        let absent = ids.len();
80        let mut text: Vec<_> = y
81            .iter()
82            .map(|value| ids.get(value).copied().unwrap_or(absent))
83            .collect();
84        drop(ids);
85        let mut offsets = vec![usize::MAX; absent + 1];
86        for &id in &text {
87            offsets[id] = 0;
88        }
89        let mut mask_len = 0;
90        for offset in &mut offsets[..absent] {
91            if *offset == 0 {
92                *offset = mask_len;
93                mask_len += lanes;
94            }
95        }
96        offsets[absent] = mask_len;
97        // Values exclusive to either input remain distinct when recursion swaps the inputs.
98        for offset in &mut offsets[..absent] {
99            if *offset == usize::MAX {
100                *offset = mask_len + lanes;
101            }
102        }
103        for id in pattern.iter_mut().chain(&mut text) {
104            *id = offsets[*id];
105        }
106        let state = Self {
107            masks: vec![0; mask_len + 3 * lanes],
108            lanes,
109        };
110        (state, pattern, text)
111    }
112
113    fn lengths<F>(&mut self, x: &[usize], y: &[usize], reverse: bool, mut row: F) -> usize
114    where
115        F: FnMut(usize),
116    {
117        let reversed: Vec<_>;
118        let pattern = if reverse {
119            reversed = x.iter().rev().copied().collect();
120            &reversed
121        } else {
122            x
123        };
124        let n = y.len();
125        // Seven zero-mask columns on each side cover the staggered SIMD lanes.
126        let absent = self.masks.len() - self.lanes;
127        let mut text = vec![absent; n + 14];
128        if reverse {
129            for (id, &value) in text[7..n + 7].iter_mut().zip(y.iter().rev()) {
130                *id = value;
131            }
132        } else {
133            text[7..n + 7].copy_from_slice(y);
134        }
135        let masks = &mut self.masks;
136        let mut horizontal = vec![0; n + 14];
137        let start = 0;
138        #[cfg(target_arch = "x86_64")]
139        let start = {
140            let mut start = start;
141            if self.lanes == 8 {
142                start = pattern.len() / 512 * 512;
143                // SAFETY: construction requires AVX-512F. Mask rows have eight words,
144                // and text/horizontal have seven padding entries on each side.
145                unsafe {
146                    advance_avx512(&pattern[..start], &text, &mut horizontal, masks);
147                }
148            }
149            if self.lanes >= 4 {
150                let end = pattern.len() / 256 * 256;
151                // SAFETY: construction requires AVX2; mask rows have at least four words,
152                // and the same padding and offset bounds apply.
153                unsafe {
154                    advance_avx2(&pattern[start..end], &text, &mut horizontal, masks);
155                }
156                start = end;
157            }
158            start
159        };
160        for block in pattern[start..].chunks(64) {
161            for (i, &id) in block.iter().enumerate() {
162                masks[id] |= 1 << i;
163            }
164            // Zero bits encode increases of the LCS length along the pattern.
165            let mut state = u64::MAX;
166            for (&id, h) in text[7..n + 7].iter().zip(&mut horizontal[7..n + 7]) {
167                let matched = state & masks[id];
168                let (sum, carry) = state.overflowing_add(matched);
169                let (sum, carry_in) = sum.overflowing_add(u64::from(*h));
170                *h = u8::from(carry | carry_in);
171                state = sum | (state ^ matched);
172            }
173            for &id in block {
174                masks[id] = 0;
175            }
176        }
177        let mut length = 0;
178        for &h in &horizontal[7..n + 7] {
179            length += usize::from(h);
180            row(length);
181        }
182        length
183    }
184
185    fn reconstruct(
186        &mut self,
187        x: &[usize],
188        y: &[usize],
189        x_offset: usize,
190        y_offset: usize,
191        flipped: bool,
192        indices: &mut Vec<(usize, usize)>,
193    ) {
194        if x.len() > y.len() {
195            return self.reconstruct(y, x, y_offset, x_offset, !flipped, indices);
196        }
197        if x.len() <= 1 {
198            if let Some(value) = x.first()
199                && let Some(position) = y.iter().position(|other| other == value)
200            {
201                let pair = (x_offset, y_offset + position);
202                indices.push(if flipped { (pair.1, pair.0) } else { pair });
203            }
204            return;
205        }
206        if x.len() <= 64 {
207            for (i, &id) in x.iter().enumerate() {
208                self.masks[id] |= 1 << i;
209            }
210            let mut state = u64::MAX;
211            // Each row stores the LCS length differences along x.
212            let mut rows = Vec::with_capacity(y.len());
213            for &id in y {
214                let matched = state & self.masks[id];
215                state = state.wrapping_add(matched) | (state ^ matched);
216                rows.push(!state);
217            }
218            for &id in x {
219                self.masks[id] = 0;
220            }
221            let start = indices.len();
222            let (mut i, mut j) = (x.len(), y.len());
223            while i != 0 && j != 0 {
224                if x[i - 1] == y[j - 1] {
225                    let pair = (x_offset + i - 1, y_offset + j - 1);
226                    indices.push(if flipped { (pair.1, pair.0) } else { pair });
227                    i -= 1;
228                    j -= 1;
229                } else if rows[j - 1] >> (i - 1) & 1 == 0 {
230                    i -= 1;
231                } else {
232                    j -= 1;
233                }
234            }
235            indices[start..].reverse();
236            return;
237        }
238        let middle = x.len() / 2;
239        let mut forward = Vec::with_capacity(y.len() + 1);
240        forward.push(0);
241        self.lengths(&x[..middle], y, false, |score| forward.push(score));
242        let mut split = y.len();
243        let mut length = forward[split];
244        let mut index = y.len();
245        self.lengths(&x[middle..], y, true, |score| {
246            index -= 1;
247            let candidate = forward[index] + score;
248            if candidate > length {
249                length = candidate;
250                split = index;
251            }
252        });
253        drop(forward);
254        if length == 0 {
255            return;
256        }
257        self.reconstruct(
258            &x[..middle],
259            &y[..split],
260            x_offset,
261            y_offset,
262            flipped,
263            indices,
264        );
265        self.reconstruct(
266            &x[middle..],
267            &y[split..],
268            x_offset + middle,
269            y_offset + split,
270            flipped,
271            indices,
272        );
273    }
274}

278unsafe fn advance_avx2(
279    pattern: &[usize],
280    text: &[usize],
281    horizontal: &mut [u8],
282    masks: &mut [u64],
283) {
284    use std::arch::x86_64::*;
285    let n = text.len() - 14;
286    let offsets = _mm256_set_epi64x(3, 2, 1, 0);
287    for group in pattern.as_chunks::<256>().0 {
288        for (i, &id) in group.iter().enumerate() {
289            masks[id + i / 64] |= 1 << (i % 64);
290        }
291        let mut state = _mm256_set1_epi64x(-1);
292        let mut carry = _mm256_setzero_si256();
293        for t in 0..n + 3 {
294            let incoming = _mm256_blend_epi32::<3>(
295                _mm256_permute4x64_epi64::<0x93>(carry),
296                _mm256_set1_epi64x(i64::from(horizontal[t + 7])),
297            );
298            // SAFETY: t + 4 through t + 7 are inside the padded text. Each
299            // gathered index is offset + lane within a complete mask row.
300            let eq = unsafe {
301                let ids = _mm256_loadu_si256(text.as_ptr().add(t + 4).cast());
302                let indices = _mm256_add_epi64(_mm256_permute4x64_epi64::<0x1b>(ids), offsets);
303                _mm256_i64gather_epi64::<8>(masks.as_ptr().cast(), indices)
304            };
305            let matched = _mm256_and_si256(state, eq);
306            let sum = _mm256_add_epi64(_mm256_add_epi64(state, matched), incoming);
307            carry =
308                _mm256_srli_epi64::<63>(_mm256_or_si256(matched, _mm256_andnot_si256(sum, state)));
309            horizontal[t + 4] = _mm256_extract_epi64::<3>(carry) as u8;
310            state = _mm256_or_si256(sum, _mm256_xor_si256(state, matched));
311        }
312        for (i, &id) in group.iter().enumerate() {
313            masks[id + i / 64] = 0;
314        }
315    }
316}

318#[cfg(target_arch = "x86_64")]
319#[target_feature(enable = "avx512f")]
320unsafe fn advance_avx512(
321    pattern: &[usize],
322    text: &[usize],
323    horizontal: &mut [u8],
324    masks: &mut [u64],
325) {
326    use std::arch::x86_64::*;
327    let n = text.len() - 14;
328    let offsets = _mm512_set_epi64(7, 6, 5, 4, 3, 2, 1, 0);
329    let reverse = _mm512_set_epi64(0, 1, 2, 3, 4, 5, 6, 7);
330    let rotate = _mm512_set_epi64(6, 5, 4, 3, 2, 1, 0, 7);
331    for group in pattern.as_chunks::<512>().0 {
332        for (i, &id) in group.iter().enumerate() {
333            masks[id + i / 64] |= 1 << (i % 64);
334        }
335        let mut state = _mm512_set1_epi64(-1);
336        let mut carry = _mm512_setzero_si512();
337        for t in 0..n + 7 {
338            let incoming = _mm512_mask_mov_epi64(
339                _mm512_permutexvar_epi64(rotate, carry),
340                1,
341                _mm512_set1_epi64(i64::from(horizontal[t + 7])),
342            );
343            // SAFETY: t through t + 7 are inside the padded text. Each gathered
344            // index is offset + lane within a complete mask row.
345            let eq = unsafe {
346                let ids = _mm512_loadu_si512(text.as_ptr().add(t).cast());
347                let indices = _mm512_add_epi64(_mm512_permutexvar_epi64(reverse, ids), offsets);
348                _mm512_i64gather_epi64::<8>(indices, masks.as_ptr().cast())
349            };
350            let matched = _mm512_and_si512(state, eq);
351            let sum = _mm512_add_epi64(_mm512_add_epi64(state, matched), incoming);
352            carry =
353                _mm512_srli_epi64::<63>(_mm512_or_si512(matched, _mm512_andnot_si512(sum, state)));
354            let out = _mm512_extracti32x4_epi32::<3>(carry);
355            horizontal[t] = _mm_cvtsi128_si64(_mm_srli_si128::<8>(out)) as u8;
356            state = _mm512_or_si512(sum, _mm512_xor_si512(state, matched));
357        }
358        for (i, &id) in group.iter().enumerate() {
359            masks[id + i / 64] = 0;
360        }
361    }

369    fn naive_lengths<T>(x: &[T], y: &[T]) -> Vec<usize>
370    where
371        T: Eq,
372    {
373        let mut dp = vec![0; x.len() + 1];
374        let mut row = vec![0];
375        for b in y {
376            let mut diagonal = 0;
377            for (i, a) in x.iter().enumerate() {
378                let old = dp[i + 1];
379                dp[i + 1] = if a == b { diagonal + 1 } else { old.max(dp[i]) };
380                diagonal = old;
381            }
382            row.push(dp[x.len()]);
383        }
384        row
385    }

387    fn check<T>(x: &[T], y: &[T])
388    where
389        T: Eq + Hash,
390    {
391        let (x, y) = if x.len() <= y.len() { (x, y) } else { (y, x) };
392        let expected = naive_lengths(x, y);
393        assert_eq!(longest_common_subsequence_length(x, y), expected[y.len()]);
394        assert_eq!(longest_common_subsequence_length(y, x), expected[y.len()]);
395        for lanes in [1, 4, 8] {
396            #[cfg(not(target_arch = "x86_64"))]
397            if lanes != 1 {
398                continue;
399            }
400            #[cfg(target_arch = "x86_64")]
401            if (lanes >= 4 && !is_x86_feature_detected!("avx2"))
402                || (lanes == 8 && !is_x86_feature_detected!("avx512f"))
403            {
404                continue;
405            }
406            let mut row = vec![0];
407            // SAFETY: the CPU features required by each lane count were checked.
408            let (mut state, x, y) = unsafe { LcsState::new(x, y, lanes) };
409            let actual = state.lengths(&x, &y, false, |score| row.push(score));
410            assert_eq!(actual, expected[y.len()]);
411            assert_eq!(row, expected);
412        }
413    }

415    fn check_reconstruction<T>(x: &[T], y: &[T])
416    where
417        T: Eq + Hash,
418    {
419        let expected = naive_lengths(x, y)[y.len()];
420        for (x, y) in [(x, y), (y, x)] {
421            let indices = longest_common_subsequence(x, y);
422            assert_eq!(indices.len(), expected);
423            assert!(indices.iter().all(|&(i, j)| x[i] == y[j]));
424            assert!(
425                indices
426                    .windows(2)
427                    .all(|pair| pair[0].0 < pair[1].0 && pair[0].1 < pair[1].1)
428            );
429        }
430    }
