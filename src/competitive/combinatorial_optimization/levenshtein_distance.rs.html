5/// Returns the edit distance with unit insertion, deletion, and substitution costs.
6/// Uses O(m + n + n ceil(m / 64)) expected time and O(m + n) space, where m <= n.
7pub fn levenshtein_distance<T>(x: &[T], y: &[T]) -> usize
8where
9    T: Eq + Hash,
10{
11    let (x, y) = if x.len() <= y.len() { (x, y) } else { (y, x) };
12    if x.is_empty() {
13        return y.len();
14    }
15    levenshtein_distance_impl(x.iter(), y.iter(), |_| {})
16}

18#[derive(Debug, Clone, Copy, PartialEq, Eq)]
19pub enum LevenshteinEdit {
20    Match,
21    Substitute,
22    Insert,
23    Delete,
24}
25
26/// Returns the unit-cost distance and one optimal edit sequence from `x` to `y`.
27/// Edits run left to right: `Insert` consumes one element of `y`, `Delete` of `x`,
28/// and `Match`/`Substitute` of each.
29pub fn levenshtein_distance_with_edits<T>(x: &[T], y: &[T]) -> (usize, Vec<LevenshteinEdit>)
30where
31    T: Eq + Hash,
32{
33    let mut edits = Vec::with_capacity(x.len().max(y.len()));
34    let distance = levenshtein_edits(x, y, false, &mut edits);
35    (distance, edits)
36}

38fn levenshtein_edits<T>(x: &[T], y: &[T], flipped: bool, edits: &mut Vec<LevenshteinEdit>) -> usize
39where
40    T: Eq + Hash,
41{
42    if x.len() > y.len() {
43        return levenshtein_edits(y, x, !flipped, edits);
44    }
45    if x.len() <= 1 {
46        let insert = if flipped {
47            LevenshteinEdit::Delete
48        } else {
49            LevenshteinEdit::Insert
50        };
51        if x.is_empty() {
52            edits.extend(std::iter::repeat_n(insert, y.len()));
53            return y.len();
54        }
55        let position = y.iter().position(|value| value == &x[0]);
56        let index = position.unwrap_or(0);
57        edits.extend(std::iter::repeat_n(insert, index));
58        edits.push(if position.is_some() {
59            LevenshteinEdit::Match
60        } else {
61            LevenshteinEdit::Substitute
62        });
63        edits.extend(std::iter::repeat_n(insert, y.len() - index - 1));
64        return y.len() - usize::from(position.is_some());
65    }
66    let middle = x.len() / 2;
67    let mut forward = Vec::with_capacity(y.len() + 1);
68    forward.push(middle);
69    levenshtein_distance_impl(x[..middle].iter(), y.iter(), |score| forward.push(score));
70    let mut split = y.len();
71    let mut distance = forward[split] + x.len() - middle;
72    let mut index = y.len();
73    // Reversed prefix scores are suffix scores in the original coordinates.
74    levenshtein_distance_impl(x[middle..].iter().rev(), y.iter().rev(), |score| {
75        index -= 1;
76        let candidate = forward[index] + score;
77        if candidate < distance {
78            distance = candidate;
79            split = index;
80        }
81    });
82    drop(forward);
83    levenshtein_edits(&x[..middle], &y[..split], flipped, edits);
84    levenshtein_edits(&x[middle..], &y[split..], flipped, edits);
85    distance
86}

88fn levenshtein_distance_impl<I, F>(x: I, y: I, row: F) -> usize
89where
90    I: ExactSizeIterator,
91    I::Item: Eq + Hash,
92    F: FnMut(usize),
93{
94    #[cfg(target_arch = "x86_64")]
95    if x.len() >= 256 && is_x86_feature_detected!("avx2") {
96        let avx512 = avx512_enabled() && is_x86_feature_detected!("avx512f");
97        // SAFETY: AVX2 and, when requested, AVX-512F support were checked.
98        return unsafe { levenshtein_distance_simd(x, y, avx512, row) };
99    }
100    levenshtein_distance_scalar(x, y, row)
101}

103fn levenshtein_distance_scalar<I, F>(x: I, y: I, mut row: F) -> usize
104where
105    I: ExactSizeIterator,
106    I::Item: Eq + Hash,
107    F: FnMut(usize),
108{
109    let m = x.len();
110    let mut masks: HashMap<I::Item, Vec<(usize, u64)>> = HashMap::new();
111    for (i, value) in x.enumerate() {
112        let entries = masks.entry(value).or_default();
113        let block = i / 64;
114        if let Some(last) = entries.last_mut().filter(|last| last.0 == block) {
115            last.1 |= 1 << (i % 64);
116        } else {
117            entries.push((block, 1 << (i % 64)));
118        }
119    }
120    let mut state = vec![(u64::MAX, 0); m.div_ceil(64)];
121    let top = 1 << ((m - 1) % 64);
122    let mut distance = m;
123    for value in y {
124        let mut entries = masks.get(&value).map(Vec::as_slice).unwrap_or(&[]);
125        let (mut carry, mut hp, mut hm) = (false, 1, 0);
126        let mut last = (0, 0);
127        for (block, (pv, mv)) in state.iter_mut().enumerate() {
128            let eq = match entries.first() {
129                Some(&(index, mask)) if index == block => {
130                    entries = &entries[1..];
131                    mask
132                }
133                _ => 0,
134            };
135            let xv = eq | *mv;
136            let (sum, overflow) = (eq & *pv).overflowing_add(*pv);
137            let (sum, overflow_carry) = sum.overflowing_add(u64::from(carry));
138            carry = overflow | overflow_carry;
139            let xh = (sum ^ *pv) | eq;
140            let ph = *mv | !(xh | *pv);
141            let mh = *pv & xh;
142            let ph_shifted = (ph << 1) | hp;
143            let mh_shifted = (mh << 1) | hm;
144            hp = ph >> 63;
145            hm = mh >> 63;
146            *pv = mh_shifted | !(xv | ph_shifted);
147            *mv = ph_shifted & xv;
148            last = (ph, mh);
149        }
150        distance += usize::from(last.0 & top != 0);
151        distance -= usize::from(last.1 & top != 0);
152        row(distance);
153    }
154    distance
155}

157#[cfg(target_arch = "x86_64")]
158unsafe fn levenshtein_distance_simd<I, F>(x: I, y: I, avx512: bool, mut row: F) -> usize
159where
160    I: ExactSizeIterator,
161    I::Item: Eq + Hash,
162    F: FnMut(usize),
163{
164    // Adjacent lane masks reuse cache lines across the staggered steps.
165    let lanes = if avx512 { 8 } else { 4 };
166    let m = x.len();
167    let mut ids = HashMap::new();
168    let mut pattern: Vec<_> = x
169        .map(|value| {
170            let id = ids.len();
171            *ids.entry(value).or_insert(id)
172        })
173        .collect();
174    let absent = ids.len();
175    let n = y.len();
176    // Seven zero-mask columns on each side cover the staggered SIMD lanes.
177    let mut text = vec![absent; n + 14];
178    for (id, value) in text[7..n + 7].iter_mut().zip(y) {
179        *id = ids.get(&value).copied().unwrap_or(absent);
180    }
181    drop(ids);
182    // Only values shared by both sequences need distinct mask rows.
183    let mut offsets = vec![usize::MAX; absent + 1];
184    for &id in &text[7..n + 7] {
185        offsets[id] = 0;
186    }
187    let mut mask_len = 0;
188    for offset in &mut offsets[..absent] {
189        if *offset == 0 {
190            *offset = mask_len;
191            mask_len += lanes;
192        }
193    }
194    offsets[absent] = mask_len;
195    // Pattern-only values must not share the zero mask used by text padding.
196    for offset in &mut offsets[..absent] {
197        if *offset == usize::MAX {
198            *offset = mask_len + lanes;
199        }
200    }
201    for id in pattern.iter_mut().chain(&mut text) {
202        *id = offsets[*id];
203    }
204    drop(offsets);
205    let mut masks = vec![0; mask_len + 2 * lanes];
206    let mut horizontal = vec![0; n + 14];
207    horizontal[7..n + 7].fill(1);
208    let mut start = 0;
209    if avx512 {
210        start = pattern.len() / 512 * 512;
211        // SAFETY: the caller checked AVX-512F. Mask offsets identify complete
212        // eight-word rows, and text/horizontal have seven padding entries on each side.
213        unsafe {
214            advance_avx512(&pattern[..start], &text, &mut horizontal, &mut masks);
215        }
216    }
217    let end = pattern.len() / 256 * 256;
218    // SAFETY: the caller checked AVX2. The same padding and offset bounds apply;
219    // each mask row has at least four words, and pattern chunks are complete.
220    unsafe {
221        advance_avx2(&pattern[start..end], &text, &mut horizontal, &mut masks);
222    }
223    for block in pattern[end..].chunks(64) {
224        for (i, &id) in block.iter().enumerate() {
225            masks[id] |= 1 << i;
226        }
227        let shift = block.len() - 1;
228        let (mut pv, mut mv) = (u64::MAX, 0);
229        for (&id, h) in text[7..n + 7].iter().zip(&mut horizontal[7..n + 7]) {
230            let neg = u64::from(*h < 0);
231            let pos = u64::from(*h > 0);
232            let eq = masks[id];
233            let xv = eq | mv;
234            let eq = eq | neg;
235            let xh = ((eq & pv).wrapping_add(pv) ^ pv) | eq;
236            let ph = mv | !(xh | pv);
237            let mh = pv & xh;
238            *h = ((ph >> shift) & 1) as i8 - ((mh >> shift) & 1) as i8;
239            let ph = (ph << 1) | pos;
240            let mh = (mh << 1) | neg;
241            pv = mh | !(xv | ph);
242            mv = ph & xv;
243        }
244        for &id in block {
245            masks[id] = 0;
246        }
247    }
248    let mut delta = 0isize;
249    for &h in &horizontal[7..n + 7] {
250        delta += h as isize;
251        row((m as isize + delta) as usize);
252    }
253    (m as isize + delta) as usize
254}

256#[cfg(target_arch = "x86_64")]
257#[target_feature(enable = "avx2")]
258unsafe fn advance_avx2(
259    pattern: &[usize],
260    text: &[usize],
261    horizontal: &mut [i8],
262    masks: &mut [u64],
263) {
264    use std::arch::x86_64::*;
265    let n = text.len() - 14;
266    let offsets = _mm256_set_epi64x(3, 2, 1, 0);
267    let all = _mm256_set1_epi64x(-1);
268    for group in pattern.as_chunks::<256>().0 {
269        for (i, &id) in group.iter().enumerate() {
270            masks[id + i / 64] |= 1 << (i % 64);
271        }
272        let mut pv = all;
273        let mut mv = _mm256_setzero_si256();
274        let mut hp = _mm256_setzero_si256();
275        let mut hm = _mm256_setzero_si256();
276        for t in 0..n + 3 {
277            let incoming = horizontal[t + 7];
278            let pos = _mm256_blend_epi32::<3>(
279                _mm256_permute4x64_epi64::<0x93>(hp),
280                _mm256_set1_epi64x(i64::from(incoming > 0)),
281            );
282            let neg = _mm256_blend_epi32::<3>(
283                _mm256_permute4x64_epi64::<0x93>(hm),
284                _mm256_set1_epi64x(i64::from(incoming < 0)),
285            );
286            // SAFETY: t + 4 through t + 7 are inside the padded text. Each
287            // gathered index is offset + lane within a complete mask row.
288            let eq = unsafe {
289                let ids = _mm256_loadu_si256(text.as_ptr().add(t + 4).cast());
290                let indices = _mm256_add_epi64(_mm256_permute4x64_epi64::<0x1b>(ids), offsets);
291                _mm256_i64gather_epi64::<8>(masks.as_ptr().cast(), indices)
292            };
293            let xv = _mm256_or_si256(eq, mv);
294            let eq = _mm256_or_si256(eq, neg);
295            let xh = _mm256_or_si256(
296                _mm256_xor_si256(_mm256_add_epi64(_mm256_and_si256(eq, pv), pv), pv),
297                eq,
298            );
299            let ph = _mm256_or_si256(mv, _mm256_xor_si256(_mm256_or_si256(xh, pv), all));
300            let mh = _mm256_and_si256(pv, xh);
301            hp = _mm256_srli_epi64::<63>(ph);
302            hm = _mm256_srli_epi64::<63>(mh);
303            horizontal[t + 4] = _mm256_extract_epi64::<3>(_mm256_sub_epi64(hp, hm)) as i8;
304            let ph = _mm256_or_si256(_mm256_slli_epi64::<1>(ph), pos);
305            let mh = _mm256_or_si256(_mm256_slli_epi64::<1>(mh), neg);
306            pv = _mm256_or_si256(mh, _mm256_xor_si256(_mm256_or_si256(xv, ph), all));
307            mv = _mm256_and_si256(ph, xv);
308        }
309        for (i, &id) in group.iter().enumerate() {
310            masks[id + i / 64] = 0;
311        }
312    }
313}

315#[cfg(target_arch = "x86_64")]
316#[target_feature(enable = "avx512f")]
317unsafe fn advance_avx512(
318    pattern: &[usize],
319    text: &[usize],
320    horizontal: &mut [i8],
321    masks: &mut [u64],
322) {
323    use std::arch::x86_64::*;
324    let n = text.len() - 14;
325    let offsets = _mm512_set_epi64(7, 6, 5, 4, 3, 2, 1, 0);
326    let reverse = _mm512_set_epi64(0, 1, 2, 3, 4, 5, 6, 7);
327    let rotate = _mm512_set_epi64(6, 5, 4, 3, 2, 1, 0, 7);
328    let all = _mm512_set1_epi64(-1);
329    for group in pattern.as_chunks::<512>().0 {
330        for (i, &id) in group.iter().enumerate() {
331            masks[id + i / 64] |= 1 << (i % 64);
332        }
333        let mut pv = all;
334        let mut mv = _mm512_setzero_si512();
335        let mut hp = _mm512_setzero_si512();
336        let mut hm = _mm512_setzero_si512();
337        for t in 0..n + 7 {
338            let incoming = horizontal[t + 7];
339            let pos = _mm512_mask_mov_epi64(
340                _mm512_permutexvar_epi64(rotate, hp),
341                1,
342                _mm512_set1_epi64(i64::from(incoming > 0)),
343            );
344            let neg = _mm512_mask_mov_epi64(
345                _mm512_permutexvar_epi64(rotate, hm),
346                1,
347                _mm512_set1_epi64(i64::from(incoming < 0)),
348            );
349            // SAFETY: t through t + 7 are inside the padded text. Each gathered
350            // index is offset + lane within a complete mask row.
351            let eq = unsafe {
352                let ids = _mm512_loadu_si512(text.as_ptr().add(t).cast());
353                let indices = _mm512_add_epi64(_mm512_permutexvar_epi64(reverse, ids), offsets);
354                _mm512_i64gather_epi64::<8>(indices, masks.as_ptr().cast())
355            };
356            let xv = _mm512_or_si512(eq, mv);
357            let eq = _mm512_or_si512(eq, neg);
358            let xh = _mm512_or_si512(
359                _mm512_xor_si512(_mm512_add_epi64(_mm512_and_si512(eq, pv), pv), pv),
360                eq,
361            );
362            let ph = _mm512_or_si512(mv, _mm512_xor_si512(_mm512_or_si512(xh, pv), all));
363            let mh = _mm512_and_si512(pv, xh);
364            hp = _mm512_srli_epi64::<63>(ph);
365            hm = _mm512_srli_epi64::<63>(mh);
366            let out = _mm512_extracti32x4_epi32::<3>(_mm512_sub_epi64(hp, hm));
367            horizontal[t] = _mm_cvtsi128_si64(_mm_srli_si128::<8>(out)) as i8;
368            let ph = _mm512_or_si512(_mm512_slli_epi64::<1>(ph), pos);
369            let mh = _mm512_or_si512(_mm512_slli_epi64::<1>(mh), neg);
370            pv = _mm512_or_si512(mh, _mm512_xor_si512(_mm512_or_si512(xv, ph), all));
371            mv = _mm512_and_si512(ph, xv);
372        }
373        for (i, &id) in group.iter().enumerate() {
374            masks[id + i / 64] = 0;
375        }
376    }
377}

384    fn naive_distances<T>(x: &[T], y: &[T]) -> Vec<usize>
385    where
386        T: Eq,
387    {
388        let mut dp: Vec<_> = (0..=x.len()).collect();
389        let mut scores = vec![x.len()];
390        for (j, b) in y.iter().enumerate() {
391            let mut diagonal = dp[0];
392            dp[0] = j + 1;
393            for (i, a) in x.iter().enumerate() {
394                let old = dp[i + 1];
395                dp[i + 1] = (old.min(dp[i]) + 1).min(diagonal + usize::from(a != b));
396                diagonal = old;
397            }
398            scores.push(dp[x.len()]);
399        }
400        scores
401    }

403    fn check<T>(x: &[T], y: &[T])
404    where
405        T: Eq + Hash,
406    {
407        let (x, y) = if x.len() <= y.len() { (x, y) } else { (y, x) };
408        let scores = naive_distances(x, y);
409        let expected = scores[y.len()];
410        assert_eq!(levenshtein_distance(x, y), expected);
411        assert_eq!(levenshtein_distance(y, x), expected);
412        if x.is_empty() {
413            return;
414        }
415        let mut row = vec![x.len()];
416        assert_eq!(
417            levenshtein_distance_scalar(x.iter(), y.iter(), |score| row.push(score)),
418            expected
419        );
420        assert_eq!(row, scores);
421        #[cfg(target_arch = "x86_64")]
422        if is_x86_feature_detected!("avx2") {
423            for avx512 in [false, true] {
424                if avx512 && !is_x86_feature_detected!("avx512f") {
425                    continue;
426                }
427                row.truncate(1);
428                // SAFETY: AVX2 and, when requested, AVX-512F support were checked.
429                let actual = unsafe {
430                    levenshtein_distance_simd(x.iter(), y.iter(), avx512, |score| row.push(score))
431                };
432                assert_eq!(actual, expected);
433                assert_eq!(row, scores);
434            }
435        }
436    }

438    fn check_edits<T>(x: &[T], y: &[T])
439    where
440        T: Eq + Hash,
441    {
442        let expected = naive_distances(x, y)[y.len()];
443        for (x, y) in [(x, y), (y, x)] {
444            let (distance, edits) = levenshtein_distance_with_edits(x, y);
445            assert_eq!(distance, expected);
446            let (mut i, mut j, mut cost) = (0, 0, 0);
447            let mut restored = Vec::new();
448            for edit in edits {
449                cost += usize::from(edit != LevenshteinEdit::Match);
450                match edit {
451                    LevenshteinEdit::Match => {
452                        assert!(x[i] == y[j]);
453                        restored.push(&x[i]);
454                        i += 1;
455                        j += 1;
456                    }
457                    LevenshteinEdit::Substitute => {
458                        assert!(x[i] != y[j]);
459                        restored.push(&y[j]);
460                        i += 1;
461                        j += 1;
462                    }
463                    LevenshteinEdit::Insert => {
464                        restored.push(&y[j]);
465                        j += 1;
466                    }
467                    LevenshteinEdit::Delete => {
468                        assert!(i < x.len());
469                        i += 1;
470                    }
471                }
472            }
473            assert_eq!((i, j, cost), (x.len(), y.len(), distance));
474            assert!(restored.into_iter().eq(y));
475        }
476    }
