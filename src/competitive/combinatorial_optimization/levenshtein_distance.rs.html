15pub fn levenshtein_distance<T: PartialEq>(x: &[T], y: &[T]) -> usize {
16    let n = x.len();
17    let m = y.len();
18    let mut dp = vec![vec![0; m + 1]; n + 1];
19    #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
20        invariant
21            n == x.len(), m == y.len(), table_shape(dp@, n as nat, m as nat),
22            forall|p: int| 0 <= p < iter.index() ==> #[trigger] dp@[p]@[0] == p,
23    ))]
24    // Verus does not model Enumerate; explicit indices keep the table invariant available.
25    #[allow(clippy::needless_range_loop)]
26    for i in 0..=n {
27        dp[i][0] = i;
28    }
29    #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
30        invariant
31            n == x.len(), m == y.len(), table_shape(dp@, n as nat, m as nat),
32            forall|p: int| 0 <= p <= n ==> #[trigger] dp@[p]@[0] == p,
33            forall|q: int| 0 <= q < iter.index() ==> #[trigger] dp@[0]@[q] == q,
34    ))]
35    #[allow(clippy::needless_range_loop)] // Same table invariant as the row initialization.
36    for j in 0..=m {
37        dp[0][j] = j;
38    }
39    #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
40        invariant
41            n == x.len(), m == y.len(), n + m < usize::MAX, T::obeys_eq_spec(),
42            table_shape(dp@, n as nat, m as nat),
43            forall|p: int| 0 <= p <= n ==> #[trigger] dp@[p]@[0] == p,
44            forall|p: int, q: int| 0 <= p <= iter.index() && 0 <= q <= m
45                ==> #[trigger] dp@[p]@[q] == distance(x@, y@, p as nat, q as nat),
46    ))]
47    for (i, x_value) in (0..n).zip(x) {
48        #[cfg_attr(verus_keep_ghost, verus_spec(iter =>
49            invariant
50                n == x.len(), m == y.len(), n + m < usize::MAX, T::obeys_eq_spec(),
51                i < n, *x_value == x@[i as int], table_shape(dp@, n as nat, m as nat),
52                forall|p: int| 0 <= p <= n ==> #[trigger] dp@[p]@[0] == p,
53                forall|p: int, q: int| 0 <= p <= i && 0 <= q <= m
54                    ==> #[trigger] dp@[p]@[q] == distance(x@, y@, p as nat, q as nat),
55                forall|q: int| 0 <= q <= iter.index()
56                    ==> #[trigger] dp@[i + 1]@[q] == distance(x@, y@, (i + 1) as nat, q as nat),
57        ))]
58        for (j, y_value) in (0..m).zip(y) {
59            #[cfg(verus_keep_ghost)]
60            proof! {
61                distance_bound(x@, y@, i as nat, (j + 1) as nat);
62                distance_bound(x@, y@, (i + 1) as nat, j as nat);
63                distance_bound(x@, y@, i as nat, j as nat);
64            }
65            dp[i + 1][j + 1] =
66                (dp[i][j + 1].min(dp[i + 1][j]) + 1).min(dp[i][j] + (x_value != y_value) as usize);
67        }
68    }
69    #[cfg(verus_keep_ghost)]
70    proof! {
71        optimal_alignment(x@, y@, n as nat, m as nat);
72        #[verifier::proof_note("For arbitrary input lengths whose sum is below usize::MAX and stable PartialEq, the result is attainable by insertions, deletions and substitutions of unit cost, with matching pairs free")]
73        assert(exists|script: Seq<Edit>| #[trigger] script_cost(x@, y@, n as nat, m as nat, script) == Some(dp@[n as int]@[m as int] as nat));
74        #[verifier::proof_note("Every alignment consuming both input sequences has cost at least the returned distance, so the returned edit cost is minimal, including empty inputs")]
75        assert(forall|script: Seq<Edit>| #[trigger] script_cost(x@, y@, n as nat, m as nat, script) is Some
76            ==> dp@[n as int]@[m as int] <= script_cost(x@, y@, n as nat, m as nat, script).unwrap());
77    }
78    dp[n][m]
79}

82mod verification {
83    use super::*;
84    use vstd::math::{max, min};

86    verus! {
87        pub enum Edit { Delete, Insert, Align }
88
89        // Recursive specifications have termination checks; the lemmas below connect
90        // the recurrence to the minimum cost of an edit script.
91        pub open spec fn script_cost<T: PartialEq>(
92            x: Seq<T>, y: Seq<T>, i: nat, j: nat, script: Seq<Edit>,
93        ) -> Option<nat>
94            recommends i <= x.len(), j <= y.len(),
95            decreases script.len(),
96        {
97            if script.len() == 0 {
98                if i == 0 && j == 0 { Some(0nat) } else { None }
99            } else {
100                let previous = match script.last() {
101                    Edit::Delete => if i > 0 {
102                        script_cost(x, y, (i - 1) as nat, j, script.drop_last())
103                    } else { None },
104                    Edit::Insert => if j > 0 {
105                        script_cost(x, y, i, (j - 1) as nat, script.drop_last())
106                    } else { None },
107                    Edit::Align => if i > 0 && j > 0 {
108                        script_cost(x, y, (i - 1) as nat, (j - 1) as nat, script.drop_last())
109                    } else { None },
110                };
111                match previous {
112                    Some(cost) => Some(cost + match script.last() {
113                        Edit::Align => if x[i - 1].eq_spec(&y[j - 1]) { 0nat } else { 1nat },
114                        _ => 1nat,
115                    }),
116                    None => None,
117                }
118            }
119        }
120
121        pub open spec fn distance<T: PartialEq>(x: Seq<T>, y: Seq<T>, i: nat, j: nat) -> nat
122            recommends i <= x.len(), j <= y.len(),
123            decreases i + j,
124        {
125            if i == 0 {
126                j
127            } else if j == 0 {
128                i
129            } else {
130                let deletion = distance(x, y, (i - 1) as nat, j) + 1;
131                let insertion = distance(x, y, i, (j - 1) as nat) + 1;
132                let alignment = distance(x, y, (i - 1) as nat, (j - 1) as nat)
133                    + if x[i - 1].eq_spec(&y[j - 1]) { 0nat } else { 1nat };
134                min(min(deletion as int, insertion as int), alignment as int) as nat
135            }
136        }
137
138        pub open spec fn table_shape(dp: Seq<Vec<usize>>, n: nat, m: nat) -> bool {
139            dp.len() == n + 1
140                && forall|i: int| 0 <= i <= n ==> #[trigger] dp[i]@.len() == m + 1
141        }
142
143        pub proof fn distance_bound<T: PartialEq>(x: Seq<T>, y: Seq<T>, i: nat, j: nat)
144            requires i <= x.len(), j <= y.len(),
145            ensures distance(x, y, i, j) <= max(i as int, j as int),
146            decreases i + j,
147        {
148            if i > 0 && j > 0 {
149                distance_bound(x, y, (i - 1) as nat, (j - 1) as nat);
150            }
151            #[verifier::proof_note("The unit-cost edit distance of any two prefixes is at most the larger prefix length")]
152            assert(distance(x, y, i, j) <= max(i as int, j as int));
153        }
154
155        pub proof fn optimal_alignment<T: PartialEq>(x: Seq<T>, y: Seq<T>, i: nat, j: nat)
156            requires i <= x.len(), j <= y.len(),
157            ensures
158                exists|script: Seq<Edit>| #[trigger] script_cost(x, y, i, j, script) == Some(distance(x, y, i, j)),
159                forall|script: Seq<Edit>| #[trigger] script_cost(x, y, i, j, script) is Some
160                    ==> script_cost(x, y, i, j, script).unwrap() >= distance(x, y, i, j),
161            decreases i + j,
162        {
163            if i == 0 && j == 0 {
164                assert(script_cost(x, y, i, j, Seq::empty()) == Some(0nat));
165            } else {
166                let step = if i == 0 {
167                    Edit::Insert
168                } else if j == 0 {
169                    Edit::Delete
170                } else {
171                    optimal_alignment(x, y, (i - 1) as nat, j);
172                    optimal_alignment(x, y, i, (j - 1) as nat);
173                    optimal_alignment(x, y, (i - 1) as nat, (j - 1) as nat);
174                    let deletion = distance(x, y, (i - 1) as nat, j) + 1;
175                    let insertion = distance(x, y, i, (j - 1) as nat) + 1;
176                    let alignment = distance(x, y, (i - 1) as nat, (j - 1) as nat)
177                        + if x[i - 1].eq_spec(&y[j - 1]) { 0nat } else { 1nat };
178                    if deletion <= insertion && deletion <= alignment {
179                        Edit::Delete
180                    } else if insertion <= alignment {
181                        Edit::Insert
182                    } else {
183                        Edit::Align
184                    }
185                };
186                if i == 0 { optimal_alignment(x, y, i, (j - 1) as nat); }
187                if j == 0 { optimal_alignment(x, y, (i - 1) as nat, j); }
188                let (previous_i, previous_j) = match step {
189                    Edit::Delete => ((i - 1) as nat, j),
190                    Edit::Insert => (i, (j - 1) as nat),
191                    Edit::Align => ((i - 1) as nat, (j - 1) as nat),
192                };
193                let previous = choose|script: Seq<Edit>|
194                    script_cost(x, y, previous_i, previous_j, script)
195                        == Some(distance(x, y, previous_i, previous_j));
196                let result = previous.push(step);
197                assert(result.drop_last() == previous);
198                assert(script_cost(x, y, i, j, result) == Some(distance(x, y, i, j)));
199            }
200            assert forall|script: Seq<Edit>| #[trigger] script_cost(x, y, i, j, script) is Some
201                implies script_cost(x, y, i, j, script).unwrap() >= distance(x, y, i, j) by {
202                if script.len() > 0 {
203                    match script.last() {
204                        Edit::Delete => {
205                            assert(script_cost(x, y, (i - 1) as nat, j, script.drop_last()) is Some);
206                        },
207                        Edit::Insert => {
208                            assert(script_cost(x, y, i, (j - 1) as nat, script.drop_last()) is Some);
209                        },
210                        Edit::Align => {
211                            assert(script_cost(x, y, (i - 1) as nat, (j - 1) as nat, script.drop_last()) is Some);
212                        },
213                    }
214                }
215            }
216            #[verifier::proof_note("The edit-distance recurrence is attainable by an alignment and is a lower bound for every valid alignment")]
217            assert((exists|script: Seq<Edit>| #[trigger] script_cost(x, y, i, j, script) == Some(distance(x, y, i, j)))
218                && (forall|script: Seq<Edit>| #[trigger] script_cost(x, y, i, j, script) is Some
219                    ==> script_cost(x, y, i, j, script).unwrap() >= distance(x, y, i, j)));
220        }
221    }
