5struct RootedTree {
6    parents: Vec<usize>,
7    vs: Vec<usize>,
8}
9
10impl RootedTree {
11    fn len(&self) -> usize {
12        self.vs.len()
13    }
14
15    fn split_centroid(self) -> CentroidSplit {
16        let n = self.len();
17        assert!(n > 2);
18        let parents = &self.parents;
19        let vs = &self.vs;
20        let mut size = vec![1; n];
21        let mut c = usize::MAX;
22        for i in (0..n).rev() {
23            if size[i] >= n.div_ceil(2) {
24                c = i;
25                break;
26            }
27            size[parents[i]] += size[i];
28        }
29        let mut side = vec![u8::MAX; n];
30        let mut order = vec![usize::MAX; n];
31        order[c] = 0;
32        let mut count = 1usize;
33        let mut taken = 0usize;
34        for u in 1..n {
35            if parents[u] == c && taken + size[u] <= (n - 1) / 2 {
36                taken += size[u];
37                side[u] = 0;
38                order[u] = count;
39                count += 1;
40            }
41        }
42        for u in 1..n {
43            if side[parents[u]] == 0 {
44                side[u] = 0;
45                order[u] = count;
46                count += 1;
47            }
48        }
49        let lsize = count - 1;
50        {
51            let mut u = parents[c];
52            while u != usize::MAX {
53                side[u] = 1;
54                order[u] = count;
55                count += 1;
56                u = parents[u];
57            }
58        }
59        for u in 0..n {
60            if u != c && side[u] == u8::MAX {
61                side[u] = 1;
62                order[u] = count;
63                count += 1;
64            }
65        }
66        assert_eq!(count, n);
67        let mut whole_parents = vec![usize::MAX; n];
68        let mut whole_vs = vec![usize::MAX; n];
69        for u in 0..n {
70            whole_vs[order[u]] = vs[u];
71        }
72        for u in 1..n {
73            let mut x = order[u];
74            let mut y = order[parents[u]];
75            if x > y {
76                swap(&mut x, &mut y);
77            }
78            whole_parents[y] = x;
79        }
80        let left = RootedTree {
81            parents: whole_parents[..=lsize].to_vec(),
82            vs: whole_vs[..=lsize].to_vec(),
83        };
84        let right = RootedTree {
85            parents: std::iter::once(usize::MAX)
86                .chain(
87                    whole_parents[lsize + 1..]
88                        .iter()
89                        .map(|&p| if p == 0 { 0 } else { p - lsize }),
90                )
91                .collect(),
92            vs: std::iter::once(whole_vs[0])
93                .chain(whole_vs[lsize + 1..].iter().copied())
94                .collect(),
95        };
96        CentroidSplit {
97            whole: RootedTree {
98                parents: whole_parents,
99                vs: whole_vs,
100            },
101            left,
102            right,
103            lsize,
104        }
105    }
106
107    fn centroid_decomposition(self, f: &mut impl FnMut(&[usize], &[usize], usize, usize)) {
108        if self.len() <= 2 {
109            return;
110        }
111        let split = self.split_centroid();
112        f(
113            &split.whole.parents,
114            &split.whole.vs,
115            split.lsize,
116            split.rsize(),
117        );
118        split.left.centroid_decomposition(f);
119        split.right.centroid_decomposition(f);
120    }
121}

123impl From<&UndirectedSparseGraph> for RootedTree {
124    fn from(graph: &UndirectedSparseGraph) -> Self {
125        let n = graph.vertices_size();
126        let mut vs = Vec::with_capacity(n);
127        let mut parent = vec![usize::MAX; n];
128        vs.push(0usize);
129        for i in 0..n {
130            let u = vs[i];
131            for a in graph.neighbors(u) {
132                if a.to != parent[u] {
133                    vs.push(a.to);
134                    parent[a.to] = u;
135                }
136            }
137        }
138        let mut new_idx = vec![0; n];
139        for (i, &v) in vs.iter().enumerate() {
140            new_idx[v] = i;
141        }
142        let mut parents = vec![usize::MAX; n];
143        for v in 1..n {
144            parents[new_idx[v]] = new_idx[parent[v]];
145        }
146        Self { parents, vs }
147    }
148}

171pub struct ContourQueryRange {

180impl ContourQueryRange {
181    pub fn len(&self) -> usize {
182        self.comp_range.last().copied().unwrap_or_default()
183    }
184
185    pub fn is_empty(&self) -> bool {
186        self.len() == 0
187    }
188
189    pub fn component_sizes(&self) -> impl ExactSizeIterator<Item = usize> + '_ {
190        self.comp_range.windows(2).map(|range| range[1] - range[0])
191    }

193    pub fn build_point_add<M: AbelianGroup>(&self, values: &[M::T]) -> ContourQueryPointAdd<'_, M> {
194        assert_eq!(values.len() + 1, self.info_indptr.len());
195        let mut components: Vec<_> = self.component_sizes().map(|n| vec![M::unit(); n]).collect();
196        for (v, value) in values.iter().enumerate() {
197            self.for_each_index(v, |c, i| M::operate_assign(&mut components[c][i], value));
198        }
199        ContourQueryPointAdd {
200            geometry: self,
201            values: values.to_vec(),
202            components: components
203                .into_iter()
204                .map(|values| {
205                    if values.len() <= 32 {
206                        ContourComponent::Values(values)
207                    } else {
208                        ContourComponent::Bit(BinaryIndexedTree::from_slice(&values))
209                    }
210                })
211                .collect(),
212        }
213    }
214
215    /// Calls `f(component, index)` for each position representing `v`.
216    pub fn for_each_index(&self, v: usize, mut f: impl FnMut(usize, usize)) {
217        for info in &self.infos[self.info_indptr[v]..self.info_indptr[v + 1]] {
218            f(info.comp as usize, info.dep as usize);
219        }
220        let (comp, index) = self.local_info[v];
221        if comp != usize::MAX {
222            f(
223                self.comp_range.len() - 1 - self.local_offsets.len() + comp,
224                index,
225            );
226        }
227    }
228
229    /// Calls `f(component, start, end)` for disjoint ranges at distances in `l..r` from `v`.
230    /// The ranges exclude `v` itself, even when `l == 0`.
231    pub fn for_each_contour_range(
232        &self,
233        v: usize,
234        l: usize,
235        r: usize,
236        mut f: impl FnMut(usize, usize, usize),
237    ) {
238        for info in &self.infos[self.info_indptr[v]..self.info_indptr[v + 1]] {
239            let comp = (info.comp ^ 1) as usize;
240            let start = self.comp_range[comp];
241            let len = self.comp_range[comp + 1] - start;
242            let lo = l.saturating_sub(info.dep as usize).min(len);
243            let hi = r.saturating_sub(info.dep as usize).min(len);
244            if lo < hi {
245                f(comp, lo, hi);
246            }
247        }
248        let (local, index) = self.local_info[v];
249        if local != usize::MAX {
250            let comp = self.comp_range.len() - 1 - self.local_offsets.len() + local;
251            let len = self.comp_range[comp + 1] - self.comp_range[comp];
252            let lo = l.max(1).min(len);
253            let hi = r.min(len);
254            if lo < hi {
255                let offset = self.local_offsets[local] + index * (len + 1);
256                let mut mask = self.local_masks[offset + hi] ^ self.local_masks[offset + lo];
257                while mask != 0 {
258                    let start = mask.trailing_zeros();
259                    let end = start + (mask >> start).trailing_ones();
260                    f(comp, start as usize, end as usize);
261                    mask &= mask.wrapping_add(1 << start);
262                }
263            }
264        }
265    }
266}

268enum ContourComponent<M: AbelianGroup> {
269    Values(Vec<M::T>),
270    Bit(BinaryIndexedTree<M>),
271}
272
273pub struct ContourQueryPointAdd<'a, M: AbelianGroup> {
274    geometry: &'a ContourQueryRange,
275    values: Vec<M::T>,
276    components: Vec<ContourComponent<M>>,
277}
278
279impl<M: AbelianGroup> ContourQueryPointAdd<'_, M> {
280    pub fn update(&mut self, v: usize, value: M::T) {
281        M::operate_assign(&mut self.values[v], &value);
282        self.geometry
283            .for_each_index(v, |c, i| match &mut self.components[c] {
284                ContourComponent::Values(values) => M::operate_assign(&mut values[i], &value),
285                ContourComponent::Bit(bit) => bit.update(i, value.clone()),
286            });
287    }
288
289    pub fn fold(&self, v: usize, l: usize, r: usize) -> M::T {
290        let mut result = if l == 0 && r != 0 {
291            self.values[v].clone()
292        } else {
293            M::unit()
294        };
295        self.geometry.for_each_contour_range(
296            v,
297            l,
298            r,
299            #[inline(always)]
300            |c, start, end| match &self.components[c] {
301                ContourComponent::Values(values) => {
302                    for value in &values[start..end] {
303                        M::operate_assign(&mut result, value);
304                    }
305                }
306                ContourComponent::Bit(bit) => {
307                    M::operate_assign(&mut result, &bit.fold_abelian(start, end))
308                }
309            },
310        );
311        result
312    }
313}

315impl UndirectedSparseGraph {
316    /// 1/3 centroid decomposition
317    ///
318    /// - f: (parents: &[usize], vs: &[usize], lsize: usize, rsize: usize)
319    /// - 0: root, 1..=lsize: left subtree, lsize+1..=lsize+rsize: right subtree
320    pub fn centroid_decomposition(&self, mut f: impl FnMut(&[usize], &[usize], usize, usize)) {
321        if self.vertices_size() <= 1 {
322            return;
323        }
324        RootedTree::from(self).centroid_decomposition(&mut f);
325    }

327    pub fn contour_query_range(&self) -> ContourQueryRange {
328        let n = self.vertices_size();
329        assert!(n <= u32::MAX as usize / 2);
330        if n <= 1 {
331            return ContourQueryRange {
332                comp_range: vec![0],
333                info_indptr: vec![0; n + 1],
334                infos: vec![],
335                local_info: vec![(usize::MAX, 0); n],
336                local_offsets: vec![],
337                local_masks: vec![],
338            };
339        }
340        let (vertices, start, neighbors) = {
341            let (vertices, parents) = self.tree_order(0);
342            let mut indices = vec![0; n];
343            for (i, &v) in vertices.iter().enumerate() {
344                indices[v] = i;
345            }
346            let parents: Vec<_> = vertices
347                .iter()
348                .skip(1)
349                .map(|&v| indices[parents[v]] as u32)
350                .collect();
351            let mut start = vec![0usize; n + 1];
352            for (i, &p) in parents.iter().enumerate() {
353                start[i + 1] += 1;
354                start[p as usize] += 1;
355            }
356            for i in 1..=n {
357                start[i] += start[i - 1];
358            }
359            let mut neighbors = vec![0u32; 2 * (n - 1)];
360            for (i, &p) in parents.iter().enumerate() {
361                start[i + 1] -= 1;
362                neighbors[start[i + 1]] = p;
363                start[p as usize] -= 1;
364                neighbors[start[p as usize]] = i as u32 + 1;
365            }
366            (vertices, start, neighbors)
367        };
368        let mut comp_range = vec![0usize];
369        let mut vertex_info = Vec::with_capacity(n * (n.ilog2() as usize + 1));
370        let mut info_indptr = vec![0usize; n + 1];
371        let mut local_info = vec![(usize::MAX, 0); n];
372        let mut local_offsets = Vec::new();
373        let mut local_masks = Vec::new();
374        let mut distances = Vec::new();
375        let mut local_sizes = Vec::new();
376        let mut removed = vec![false; n];
377        let mut parents = vec![usize::MAX; n];
378        let mut sizes = vec![0usize; n];
379        let mut tasks = vec![0];
380        let mut order = Vec::with_capacity(n);
381        let mut entries = Vec::with_capacity(n);
382        let mut boundaries = Vec::new();
383        let mut groups = Vec::new();
384        while let Some(root) = tasks.pop() {
385            order.clear();
386            order.push(root);
387            parents[root] = usize::MAX;
388            let mut i = 0;
389            while i < order.len() {
390                let v = order[i];
391                sizes[v] = 1;
392                for &to in &neighbors[start[v]..start[v + 1]] {
393                    let to = to as usize;
394                    if !removed[to] && to != parents[v] {
395                        parents[to] = v;
396                        order.push(to);
397                    }
398                }
399                i += 1;
400            }
401            if order.len() <= 32 {
402                let len = order.len();
403                if len > 1 {
404                    let comp = local_offsets.len();
405                    let offset = local_masks.len();
406                    local_offsets.push(offset);
407                    local_sizes.push(len);
408                    local_masks.resize(offset + len * (len + 1), 0u32);
409                    distances.clear();
410                    distances.resize(len * len, 0u8);
411                    for (i, &v) in order.iter().enumerate() {
412                        local_info[vertices[v]] = (comp, i);
413                        sizes[v] = i;
414                        local_masks[offset + i * (len + 1) + 1] = 1 << i;
415                    }
416                    for (i, &v) in order.iter().enumerate().skip(1) {
417                        let parent = sizes[parents[v]];
418                        for j in 0..i {
419                            let distance = distances[parent * len + j] + 1;
420                            distances[i * len + j] = distance;
421                            distances[j * len + i] = distance;
422                            local_masks[offset + i * (len + 1) + distance as usize + 1] |= 1 << j;
423                            local_masks[offset + j * (len + 1) + distance as usize + 1] |= 1 << i;
424                        }
425                    }
426                    for row in local_masks[offset..].chunks_exact_mut(len + 1) {
427                        for d in 1..=len {
428                            row[d] |= row[d - 1];
429                        }
430                    }
431                }
432                continue;
433            }
434            let mut centroid = root;
435            for &v in order.iter().rev() {
436                if sizes[v] >= order.len().div_ceil(2) {
437                    centroid = v;
438                    break;
439                }
440                sizes[parents[v]] += sizes[v];
441            }
442            removed[centroid] = true;
443            entries.clear();
444            entries.push((centroid, 0));
445            boundaries.clear();
446            boundaries.extend([0, 1]);
447            for &v in &neighbors[start[centroid]..start[centroid + 1]] {
448                let v = v as usize;
449                if removed[v] {
450                    continue;
451                }
452                tasks.push(v);
453                parents[v] = centroid;
454                let mut i = entries.len();
455                entries.push((v, 1));
456                while i < entries.len() {
457                    let (v, distance) = entries[i];
458                    for &to in &neighbors[start[v]..start[v + 1]] {
459                        let to = to as usize;
460                        if !removed[to] && to != parents[v] {
461                            parents[to] = v;
462                            entries.push((to, distance + 1));
463                        }
464                    }
465                    i += 1;
466                }
467                boundaries.push(entries.len());
468            }
469            groups.push((0, boundaries.len() - 1));
470            while let Some((first, last)) = groups.pop() {
471                if last - first < 2 {
472                    continue;
473                }
474                let weight = boundaries[last] - boundaries[first];
475                let target = boundaries[first] + weight.div_ceil(2);
476                let mut middle =
477                    first + 1 + boundaries[first + 1..last].partition_point(|&p| p < target);
478                middle = middle.min(last - 1);
479                if middle > first + 1 {
480                    let x = boundaries[middle] - boundaries[first];
481                    let y = boundaries[middle - 1] - boundaries[first];
482                    if y.max(weight - y) < x.max(weight - x) {
483                        middle -= 1;
484                    }
485                }
486                for (l, r) in [(first, middle), (middle, last)] {
487                    let comp = comp_range.len() - 1;
488                    let mut max_distance = 0;
489                    for &(v, dep) in &entries[boundaries[l]..boundaries[r]] {
490                        vertex_info.push((
491                            vertices[v] as u32,
492                            ContourInfo {
493                                comp: comp as u32,
494                                dep: dep as u32,
495                            },
496                        ));
497                        info_indptr[vertices[v] + 1] += 1;
498                        max_distance = max_distance.max(dep);
499                    }
500                    comp_range.push(comp_range[comp] + max_distance + 1);
501                }
502                groups.extend([(middle, last), (first, middle)]);
503            }
504        }
505        for len in local_sizes {
506            comp_range.push(comp_range.last().unwrap() + len);
507        }
508        for v in 1..=n {
509            info_indptr[v] += info_indptr[v - 1];
510        }
511        let mut infos = vec![ContourInfo { comp: 0, dep: 0 }; vertex_info.len()];
512        let mut positions = info_indptr.clone();
513        for (v, info) in vertex_info {
514            let v = v as usize;
515            infos[positions[v]] = info;
516            positions[v] += 1;
517        }
518        ContourQueryRange {
519            comp_range,
520            info_indptr,
521            infos,
522            local_info,
523            local_offsets,
524            local_masks,
525        }
526    }

541    #[test]
542    fn test_contour_query_range() {
543        let mut rng = Xorshift::default();
544        let mut graphs = vec![UndirectedSparseGraph::from_edges(0, vec![])];
545        for n in 1..=5 {
546            for parents in exhaustive_sequences(0..n, n - 1..=n - 1) {
547                if parents.iter().enumerate().all(|(i, &p)| p <= i) {
548                    graphs.push(UndirectedSparseGraph::from_edges(
549                        n,
550                        parents
551                            .into_iter()
552                            .enumerate()
553                            .map(|(i, p)| (p, i + 1))
554                            .collect(),
555                    ));
556                }
557            }
558        }
559        for n in 1..=80 {
560            graphs.extend([rng.random(PathTree(n)), rng.random(StarTree(n))]);
561        }
562        for n in sample_usize(&mut rng, 5, 1..=257, 10) {
563            graphs.extend([rng.random(PathTree(n)), rng.random(StarTree(n))]);
564        }
565        graphs.extend((0..200).map(|_| rng.random(MixedTree(1usize..80))));
566        for graph in graphs {
567            let n = graph.vertices_size();
568            let query = graph.contour_query_range();
569            let mut values: Vec<_> = (0..n).map(|v| v as i64 + 1).collect();
570            let mut point_add = query.build_point_add::<AdditiveOperation<_>>(&values);
571            let mut data: Vec<_> = query.component_sizes().map(|n| vec![0i64; n]).collect();
572            for (v, &value) in values.iter().enumerate() {
573                query.for_each_index(v, |c, i| data[c][i] += value);
574            }
575            assert_eq!(query.len(), data.iter().map(Vec::len).sum());
576            assert_eq!(query.is_empty(), n <= 1);
577            let updates: Vec<_> = if n <= 5 {
578                (0..n)
579                    .flat_map(|u| (-1..=1).map(move |delta| (u, delta)))
580                    .collect()
581            } else {
582                (0..200)
583                    .map(|_| (rng.random(0..n), rng.random(-100..=100i64)))
584                    .collect()
585            };
586            for (u, delta) in updates {
587                values[u] += delta;
588                point_add.update(u, delta);
589                query.for_each_index(u, |c, i| data[c][i] += delta);
590                let ranges: Vec<_> = if n <= 5 {
591                    (0..n)
592                        .flat_map(|v| {
593                            (0..=n).flat_map(move |l| (l..=n + 1).map(move |r| (v, l, r)))
594                        })
595                        .collect()
596                } else {
597                    let v = rng.random(0..n);
598                    let l = rng.random(0..=n);
599                    vec![(v, l, rng.random(l..=n + 1))]
600                };
601                for (v, l, r) in ranges {
602                    let distances = graph.tree_depth(v);
603                    let expected: i64 = (0..n)
604                        .filter(|&u| {
605                            u != v && l <= distances[u] as usize && (distances[u] as usize) < r
606                        })
607                        .map(|u| values[u])
608                        .sum();
609                    let mut actual = 0;
610                    query.for_each_contour_range(v, l, r, |c, start, end| {
611                        actual += data[c][start..end].iter().sum::<i64>()
612                    });
613                    assert_eq!(actual, expected);
614                    assert_eq!(
615                        point_add.fold(v, l, r),
616                        expected + if l == 0 && r != 0 { values[v] } else { 0 }
617                    );
618                }
619            }
620        }
621    }
