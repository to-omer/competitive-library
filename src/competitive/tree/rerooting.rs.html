1//! dynamic programming on all-rooted trees
2
3use crate::algebra::{AbelianGroup, Monoid};
4use crate::graph::{Graph, Neighbor, UndirectedSparseGraph};
5
6#[codesnip::entry("ReRooting", include("algebra", "tree_order"))]
7/// dynamic programming on all-rooted trees
8///
9/// Neighbors are merged in adjacency order before applying `rooting`.
10#[derive(Clone, Debug)]
11pub struct ReRooting<'a, M: Monoid, F: Fn(&M::T, usize, Option<usize>) -> M::T> {
12    graph: &'a UndirectedSparseGraph,
13    /// dp\[v\]: result of v-rooted tree
14    pub dp: Vec<M::T>,
15    /// ep\[e\]: result of e-subtree; e >= m denotes the reverse direction.
16    pub ep: Vec<M::T>,
17    /// rooting(data, vid, (Optional)eid): add root node(vid), result subtree is edge(eid)
18    rooting: F,
19}
20#[codesnip::entry("ReRooting")]
21impl<'a, M, F> ReRooting<'a, M, F>
22where
23    M: Monoid,
24    F: Fn(&M::T, usize, Option<usize>) -> M::T,
25{
26    pub fn new(graph: &'a UndirectedSparseGraph, rooting: F) -> Self {
27        Self::build(graph, rooting, None::<fn(&M::T, &M::T) -> M::T>)
28    }
29
30    pub fn new_with_inverse(graph: &'a UndirectedSparseGraph, rooting: F) -> Self
31    where
32        M: AbelianGroup,
33    {
34        Self::build(graph, rooting, Some(M::rinv_operate))
35    }
36
37    fn build<I>(graph: &'a UndirectedSparseGraph, rooting: F, inverse: Option<I>) -> Self
38    where
39        I: Fn(&M::T, &M::T) -> M::T,
40    {
41        let dp = vec![M::unit(); graph.vertices_size()];
42        let ep = vec![M::unit(); graph.vertices_size() * 2];
43        let mut self_ = Self {
44            graph,
45            dp,
46            ep,
47            rooting,
48        };
49        self_.rerooting(inverse);
50        self_
51    }
52    #[inline]
53    fn eidx(&self, u: usize, a: Neighbor<usize, usize>) -> usize {
54        a.label + self.graph.edges_size() * (u > a.to) as usize
55    }
56    #[inline]
57    fn reidx(&self, u: usize, a: Neighbor<usize, usize>) -> usize {
58        a.label + self.graph.edges_size() * (u < a.to) as usize
59    }
60    #[inline]
61    fn merge(&self, x: &M::T, y: &M::T) -> M::T {
62        M::operate(x, y)
63    }
64    #[inline]
65    fn add_subroot(&self, x: &M::T, vid: usize, eid: usize) -> M::T {
66        (self.rooting)(x, vid, Some(eid))
67    }
68    #[inline]
69    fn add_root(&self, x: &M::T, vid: usize) -> M::T {
70        (self.rooting)(x, vid, None)
71    }
72    fn rerooting<I: Fn(&M::T, &M::T) -> M::T>(&mut self, inverse: Option<I>) {
73        let (order, parents) = self.graph.tree_order(0);
74        for &u in order.iter().skip(1).rev() {
75            let mut sum = M::unit();
76            let mut parent = None;
77            for a in self.graph.neighbors(u) {
78                if a.to == parents[u] {
79                    parent = Some(a);
80                } else {
81                    sum = self.merge(&sum, &self.ep[self.eidx(u, a)]);
82                }
83            }
84            let a = parent.unwrap();
85            let i = self.reidx(u, a);
86            self.ep[i] = self.add_subroot(&sum, u, a.label);
87            if inverse.is_some() {
88                self.dp[u] = sum;
89            }
90        }
91        if let Some(inverse) = inverse {
92            self.dp[0] = self.graph.neighbors(0).fold(M::unit(), |sum, a| {
93                self.merge(&sum, &self.ep[self.eidx(0, a)])
94            });
95            for u in order {
96                let sum = std::mem::replace(&mut self.dp[u], M::unit());
97                self.dp[u] = self.add_root(&sum, u);
98                for a in self.graph.neighbors(u) {
99                    if a.to != parents[u] {
100                        let value = inverse(&sum, &self.ep[self.eidx(u, a)]);
101                        let i = self.reidx(u, a);
102                        let value = self.add_subroot(&value, u, a.label);
103                        self.dp[a.to] = self.merge(&self.dp[a.to], &value);
104                        self.ep[i] = value;
105                    }
106                }
107            }
108            return;
109        }
110        let mut prefix = Vec::new();
111        for u in order {
112            prefix.clear();
113            prefix.push(M::unit());
114            for a in self.graph.neighbors(u) {
115                prefix.push(self.merge(prefix.last().unwrap(), &self.ep[self.eidx(u, a)]));
116            }
117            self.dp[u] = self.add_root(prefix.last().unwrap(), u);
118            let mut suffix = M::unit();
119            for (k, a) in self.graph.neighbors(u).enumerate().rev() {
120                if a.to != parents[u] {
121                    let i = self.reidx(u, a);
122                    self.ep[i] = self.add_subroot(&self.merge(&prefix[k], &suffix), u, a.label);
123                }
124                suffix = self.merge(&self.ep[self.eidx(u, a)], &suffix);
125            }
126        }
127    }
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133    use crate::{
134        algebra::{AdditiveOperation, ConcatenateOperation},
135        tools::{Xorshift, testutil::exhaustive_sequences},
136        tree::{MixedTree, PathTree, StarTree},
137    };
138
139    #[test]
140    fn test_rerooting() {
141        let mut rng = Xorshift::default();
142        let mut graphs = Vec::new();
143        for n in 1..=5 {
144            for parents in exhaustive_sequences(0..n, n - 1..=n - 1) {
145                if parents.iter().enumerate().all(|(i, &p)| p <= i) {
146                    graphs.push(UndirectedSparseGraph::from_edges(
147                        n,
148                        parents
149                            .into_iter()
150                            .enumerate()
151                            .map(|(i, p)| (p, i + 1))
152                            .collect(),
153                    ));
154                }
155            }
156        }
157        for n in 1..=32 {
158            graphs.extend([
159                rng.random(PathTree(n)),
160                rng.random(StarTree(n)),
161                rng.random(MixedTree(n)),
162            ]);
163        }
164        for graph in graphs {
165            let n = graph.vertices_size();
166            let dp = ReRooting::<ConcatenateOperation<_>, _>::new(&graph, |xs, v, edge| {
167                let mut xs = xs.clone();
168                xs.push((v, edge));
169                xs
170            });
171            let sums = ReRooting::<AdditiveOperation<i64>, _>::new_with_inverse(
172                &graph,
173                |&sum, v, edge| sum + v as i64 + edge.map_or(0, |e| e as i64),
174            );
175            for root in 0..n {
176                for parent in std::iter::once(None).chain(graph.neighbors(root).map(Some)) {
177                    let mut expected = Vec::new();
178                    let mut stack = vec![(
179                        root,
180                        parent.map_or(n, |a| a.to),
181                        parent.map(|a| a.label),
182                        false,
183                    )];
184                    while let Some((u, p, edge, visited)) = stack.pop() {
185                        if visited {
186                            expected.push((u, edge));
187                        } else {
188                            stack.push((u, p, edge, true));
189                            stack.extend(
190                                graph
191                                    .neighbors(u)
192                                    .rev()
193                                    .filter(|a| a.to != p)
194                                    .map(|a| (a.to, u, Some(a.label), false)),
195                            );
196                        }
197                    }
198                    let actual = if let Some(parent) = parent {
199                        &dp.ep[dp.reidx(root, parent)]
200                    } else {
201                        &dp.dp[root]
202                    };
203                    assert_eq!(*actual, expected);
204                    let sum = if let Some(parent) = parent {
205                        sums.ep[sums.reidx(root, parent)]
206                    } else {
207                        sums.dp[root]
208                    };
209                    assert_eq!(
210                        sum,
211                        expected
212                            .iter()
213                            .map(|&(v, e)| v as i64 + e.map_or(0, |e| e as i64))
214                            .sum()
215                    );
216                }
217            }
218        }
219    }
