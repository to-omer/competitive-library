25/// A sequence with amortized O(sqrt(N)) insertion/removal and O(sqrt(N)) indexing.
26/// An individual insertion or removal can take O(N) when rebuilding.
27#[derive(Clone)]
28pub struct SqrtVec<T> {
29    inner: Buckets<T, ()>,
30}

32impl<T> SqrtVec<T> {
33    pub fn new() -> Self {
34        Self {
35            inner: Buckets::new(),
36        }
37    }
38    pub fn len(&self) -> usize {
39        self.inner.len
40    }
41    pub fn is_empty(&self) -> bool {
42        self.len() == 0
43    }
44    pub fn clear(&mut self) {
45        *self = Self::new();
46    }
47
48    pub fn get(&self, index: usize) -> Option<&T> {
49        if index >= self.len() {
50            return None;
51        }
52        self.inner.at(self.inner.locate(index))
53    }
54
55    pub fn get_mut(&mut self, index: usize) -> Option<&mut T> {
56        if index >= self.len() {
57            return None;
58        }
59        let (i, j) = self.inner.locate(index);
60        Some(&mut self.inner.buckets[i].values[j])
61    }
62
63    pub fn first(&self) -> Option<&T> {
64        self.inner.buckets.first()?.values.first()
65    }
66    pub fn last(&self) -> Option<&T> {
67        self.inner.buckets.last()?.values.last()
68    }
69
70    pub fn insert(&mut self, index: usize, value: T) {
71        assert!(index <= self.len());
72        self.inner.insert(self.inner.locate(index), value);
73    }
74
75    pub fn remove(&mut self, index: usize) -> T {
76        assert!(index < self.len());
77        self.inner.remove(self.inner.locate(index))
78    }
79
80    pub fn push(&mut self, value: T) {
81        self.inner.insert((self.inner.buckets.len(), 0), value);
82    }
83
84    pub fn pop(&mut self) -> Option<T> {
85        let i = self.inner.buckets.len().checked_sub(1)?;
86        Some(
87            self.inner
88                .remove((i, self.inner.buckets[i].values.len() - 1)),
89        )
90    }
91
92    pub fn iter(&self) -> SqrtIter<'_, T> {
93        SqrtIter {
94            buckets: &self.inner.buckets,
95            front: (0, 0),
96            back: (self.inner.buckets.len(), 0),
97        }
98    }
99
100    pub fn iter_mut(&mut self) -> impl DoubleEndedIterator<Item = &mut T> {
101        self.inner
102            .buckets
103            .iter_mut()
104            .flat_map(|b| b.values.iter_mut())
105    }
106}

177pub struct SqrtIter<'a, T> {

179    front: (usize, usize),

181}

191impl<'a, T> Iterator for SqrtIter<'a, T> {
192    type Item = &'a T;
193    fn next(&mut self) -> Option<Self::Item> {
194        if self.front >= self.back {
195            return None;
196        }
197        let (i, j) = self.front;
198        let values = &self.buckets[i].values;
199        self.front = if j + 1 == values.len() {
200            (i + 1, 0)
201        } else {
202            (i, j + 1)
203        };
204        Some(&values[j])
205    }
206}
207impl<T> DoubleEndedIterator for SqrtIter<'_, T> {
208    fn next_back(&mut self) -> Option<Self::Item> {
209        if self.front >= self.back {
210            return None;
211        }
212        if self.back.1 == 0 {
213            self.back.0 -= 1;
214            self.back.1 = self.buckets[self.back.0].values.len();
215        }
216        self.back.1 -= 1;
217        Some(&self.buckets[self.back.0].values[self.back.1])
218    }
219}

222#[derive(Clone, Debug)]
223pub struct SqrtIntoIter<T> {
224    buckets: std::vec::IntoIter<Bucket<T, ()>>,
225    front: std::vec::IntoIter<T>,
226    back: std::vec::IntoIter<T>,
227    remaining: usize,
228}
229impl<T> Iterator for SqrtIntoIter<T> {
230    type Item = T;
231    fn next(&mut self) -> Option<T> {
232        let value = loop {
233            if let Some(value) = self.front.next() {
234                break Some(value);
235            }
236            if let Some(bucket) = self.buckets.next() {
237                self.front = bucket.values.into_iter();
238            } else {
239                break self.back.next();
240            }
241        };
242        self.remaining -= usize::from(value.is_some());
243        value
244    }
245    fn size_hint(&self) -> (usize, Option<usize>) {
246        (self.remaining, Some(self.remaining))
247    }
248}
249impl<T> DoubleEndedIterator for SqrtIntoIter<T> {
250    fn next_back(&mut self) -> Option<T> {
251        let value = loop {
252            if let Some(value) = self.back.next_back() {
253                break Some(value);
254            }
255            if let Some(bucket) = self.buckets.next_back() {
256                self.back = bucket.values.into_iter();
257            } else {
258                break self.front.next_back();
259            }
260        };
261        self.remaining -= usize::from(value.is_some());
262        value
263    }
264}

412    where
413        T: Ord,
414    {
415        let p = self.inner.inner.partition_point(|v| v < &value);
416        if self.inner.inner.at(p) == Some(&value) {

418        }

420        true
421    }

475/// An ordered map with O(log N) key lookup and amortized O(sqrt(N)) updates.
476/// Rank and indexing take O(sqrt(N)).
477#[derive(Clone, Debug, PartialEq, Eq)]
478pub struct SqrtMap<K, V> {
479    inner: SqrtVec<(K, V)>,
480}

482impl<K, V> SqrtMap<K, V> {
483    pub fn new() -> Self {
484        Self {
485            inner: SqrtVec::new(),
486        }
487    }
488    pub fn len(&self) -> usize {
489        self.inner.len()
490    }
491    pub fn is_empty(&self) -> bool {
492        self.inner.is_empty()
493    }
494    pub fn clear(&mut self) {
495        self.inner.clear();
496    }
497    pub fn iter(&self) -> impl DoubleEndedIterator<Item = (&K, &V)> {
498        self.inner.iter().map(|(k, v)| (k, v))
499    }
500    pub fn iter_mut(&mut self) -> impl DoubleEndedIterator<Item = (&K, &mut V)> {
501        self.inner.iter_mut().map(|(k, v)| (&*k, v))
502    }
503    pub fn get_key_value_at(&self, index: usize) -> Option<(&K, &V)> {
504        self.inner.get(index).map(|(k, v)| (k, v))
505    }
506    pub fn remove_at(&mut self, index: usize) -> Option<(K, V)> {
507        (index < self.len()).then(|| self.inner.remove(index))
508    }
509    pub fn first_key_value(&self) -> Option<(&K, &V)> {
510        self.inner.first().map(|(k, v)| (k, v))
511    }
512    pub fn last_key_value(&self) -> Option<(&K, &V)> {
513        self.inner.last().map(|(k, v)| (k, v))
514    }
515
516    pub fn get<Q>(&self, key: &Q) -> Option<&V>
517    where
518        K: Borrow<Q>,
519        Q: Ord + ?Sized,
520    {
521        self.get_key_value(key).map(|(_, v)| v)
522    }
523
524    pub fn get_key_value<Q>(&self, key: &Q) -> Option<(&K, &V)>
525    where
526        K: Borrow<Q>,
527        Q: Ord + ?Sized,
528    {
529        let p = self.inner.inner.partition_point(|(k, _)| k.borrow() < key);
530        self.inner
531            .inner
532            .at(p)
533            .filter(|(k, _)| k.borrow() == key)
534            .map(|(k, v)| (k, v))
535    }
536
537    pub fn get_mut<Q>(&mut self, key: &Q) -> Option<&mut V>
538    where
539        K: Borrow<Q>,
540        Q: Ord + ?Sized,
541    {
542        let (i, j) = self.inner.inner.partition_point(|(k, _)| k.borrow() < key);
543        let (k, v) = self.inner.inner.buckets.get_mut(i)?.values.get_mut(j)?;
544        (<K as Borrow<Q>>::borrow(k) == key).then_some(v)
545    }
546
547    pub fn contains_key<Q>(&self, key: &Q) -> bool
548    where
549        K: Borrow<Q>,
550        Q: Ord + ?Sized,
551    {
552        self.get(key).is_some()
553    }
554
555    pub fn insert(&mut self, key: K, value: V) -> Option<V>
556    where
557        K: Ord,
558    {
559        let (i, j) = self.inner.inner.partition_point(|(k, _)| k < &key);
560        if self.inner.inner.at((i, j)).is_some_and(|(k, _)| k == &key) {
561            return Some(mem::replace(
562                &mut self.inner.inner.buckets[i].values[j].1,
563                value,
564            ));
565        }
566        self.inner.inner.insert((i, j), (key, value));
567        None
568    }

571    where
572        K: Borrow<Q>,
573        Q: Ord + ?Sized,
574    {
575        self.remove_entry(key).map(|(_, v)| v)
576    }
577
578    pub fn remove_entry<Q>(&mut self, key: &Q) -> Option<(K, V)>
579    where
580        K: Borrow<Q>,
581        Q: Ord + ?Sized,
582    {
583        let p = self.inner.inner.partition_point(|(k, _)| k.borrow() < key);
584        self.inner.inner.at(p).filter(|(k, _)| k.borrow() == key)?;
585        Some(self.inner.inner.remove(p))
586    }
587
588    /// Counts keys strictly less than `key`.
589    pub fn rank<Q>(&self, key: &Q) -> usize
590    where
591        K: Borrow<Q>,
592        Q: Ord + ?Sized,
593    {
594        self.inner
595            .inner
596            .position(self.inner.inner.partition_point(|(k, _)| k.borrow() < key))
597    }
598
599    /// Counts keys less than or equal to `key`.
600    pub fn rank_le<Q>(&self, key: &Q) -> usize
601    where
602        K: Borrow<Q>,
603        Q: Ord + ?Sized,
604    {
605        self.inner
606            .inner
607            .position(self.inner.inner.partition_point(|(k, _)| k.borrow() <= key))
608    }
609
610    pub fn range<Q, R>(&self, range: R) -> impl DoubleEndedIterator<Item = (&K, &V)>
611    where
612        K: Borrow<Q>,
613        Q: Ord + ?Sized,
614        R: RangeBounds<Q>,
615    {
616        self.inner
617            .inner
618            .range(range, |(k, _)| k.borrow())
619            .map(|(k, v)| (k, v))
620    }

678    #[test]
679    fn sets() {
680        let mut rng = Xorshift::default();
681        let cases: Vec<_> = sequence_cases(&mut rng, -256i64..=256, 0..=256).collect();
682        for mut values in cases {
683            let mut set: SqrtSet<_> = values.iter().copied().collect();
684            let mut expected: BTreeSet<_> = values.iter().copied().collect();
685            let mut multi: SqrtMultiset<_> = values.iter().copied().collect();
686            values.sort_unstable();
687            for step in 0..512 {
688                let x = if step == 0 {
689                    i64::MIN
690                } else if step == 1 {
691                    i64::MAX
692                } else {
693                    rng.random(-256..=256)
694                };
695                match rng.random(0..4) {
696                    0 | 1 => {
697                        assert_eq!(set.insert(x), expected.insert(x));
698                        multi.insert(x);
699                        values.push(x);
700                        values.sort_unstable();
701                    }
702                    2 => {
703                        assert_eq!(set.remove(&x), expected.remove(&x));
704                        let pos = values.iter().position(|v| *v == x);
705                        assert_eq!(multi.remove(&x), pos.is_some());
706                        if let Some(i) = pos {
707                            values.remove(i);
708                        }
709                    }
710                    _ => {
711                        let i = rng.random(0..=set.len());
712                        let x = expected.iter().nth(i).copied();
713                        assert_eq!(set.remove_at(i), x);
714                        if let Some(x) = x {
715                            expected.remove(&x);
716                        }
717                        let i = rng.random(0..=values.len());
718                        assert_eq!(
719                            multi.remove_at(i),
720                            (i < values.len()).then(|| values.remove(i))
721                        );

805        for values in cases {
806            let mut map: SqrtMap<_, _> = values.iter().copied().zip(0i64..).collect();
807            let mut folded: SqrtFoldMap<_, ConcatenateOperation<i64>> = values
808                .iter()
809                .copied()
810                .zip(0i64..)
811                .map(|(k, v)| (k, vec![v]))
812                .collect();
813            let mut expected: BTreeMap<_, _> = values.into_iter().zip(0i64..).collect();
814            for step in 0..512 {
815                let key = rng.random(-256..=256);
816                match rng.random(0..5) {
817                    0 | 1 => {
818                        let old = expected.insert(key, step);
819                        assert_eq!(map.insert(key, step), old);
820                        assert_eq!(folded.insert(key, vec![step]), old.map(|v| vec![v]));
821                    }
822                    2 => {
823                        let old = expected.remove(&key);
824                        assert_eq!(map.remove(&key), old);
825                        assert_eq!(folded.remove(&key), old.map(|v| vec![v]));
826                    }
827                    3 => {
828                        let i = rng.random(0..=map.len());
829                        let entry = expected.iter().nth(i).map(|(&k, &v)| (k, v));

832                        if let Some((k, _)) = entry {
833                            expected.remove(&k);
834                        }
835                    }
836                    _ => {

838                            folded.modify(&key, |v| vec![v[0] + 1]),

890    #[test]
891    fn sequence() {
892        let mut rng = Xorshift::default();
893        let cases: Vec<_> = sequence_cases(&mut rng, -256i64..=256, 0..=256).collect();
894        for mut values in cases {
895            let mut seq: SqrtVec<_> = values.iter().copied().collect();
896            let mut folded: SqrtFoldVec<ConcatenateOperation<i64>> =
897                values.iter().map(|&v| vec![v]).collect();
898            for _ in 0..512 {
899                let i = rng.random(0..=values.len());
900                let x = rng.random(-256..=256);
901                match rng.random(0..5) {
902                    0 => {
903                        seq.insert(i, x);
904                        folded.insert(i, vec![x]);
905                        values.insert(i, x);
906                    }
907                    1 if i < values.len() => {
908                        let old = values.remove(i);
909                        assert_eq!(seq.remove(i), old);
910                        assert_eq!(folded.remove(i), [old]);
911                    }
912                    2 => {
913                        seq.push(x);
914                        folded.push(vec![x]);
915                        values.push(x);
916                    }
917                    3 => {
918                        let old = values.pop();
919                        assert_eq!(seq.pop(), old);
920                        assert_eq!(folded.pop(), old.map(|v| vec![v]));
921                    }
922                    _ => {
923                        assert_eq!(seq.get(i), values.get(i));
924                        assert_eq!(folded.get(i).map(|v| &v[0]), values.get(i));
925                        if let Some(v) = seq.get_mut(i) {
926                            *v = x;
927                        }
928                        if let Some(v) = values.get_mut(i) {
929                            *v = x;
930                            folded.modify(i, |_| vec![x]);
931                        }
932                    }
933                }
934                assert_eq!(seq.len(), values.len());
935                assert!(seq.iter().eq(values.iter()));
936                let l = i.min(values.len());
937                let r = rng.random(l..=values.len());
938                assert_eq!(folded.fold(l..r), values[l..r]);
939                check_buckets(&seq.inner);
940            }
941            assert!(folded.iter().map(|v| &v[0]).eq(values.iter()));
942            for x in seq.iter_mut() {
943                *x += 1;
944            }
945            for x in &mut values {
946                *x += 1;
947            }
948            assert_eq!(seq, values.iter().copied().collect());
949            let mut iter = seq.into_iter();
950            let mut expected = VecDeque::from(values);
951            while !expected.is_empty() {
952                assert_eq!(iter.len(), expected.len());
953                if rng.random(0..2) == 0 {
954                    assert_eq!(iter.next(), expected.pop_front());
955                } else {
956                    assert_eq!(iter.next_back(), expected.pop_back());
957                }
958            }
959            assert_eq!(iter.next(), None);
960            assert_eq!(iter.next_back(), None);
961        }
962        // Retain sparse buckets while transferring elements to the tail at constant length.
963        let mut seq: SqrtVec<_> = (0..256).collect();
964        let mut expected: Vec<_> = (0..256).collect();
965        for x in 0..4096 {
966            let mut index = 0;
967            for bucket in &seq.inner.buckets {
968                if bucket.values.len() > 1 {
969                    break;
970                }
971                index += bucket.values.len();
972            }
973            assert_eq!(seq.remove(index), expected.remove(index));
974            seq.push(x);
975            expected.push(x);
976            assert!(seq.iter().eq(expected.iter()));
977            check_buckets(&seq.inner);
978        }
979        while let Some(x) = expected.pop() {
980            assert_eq!(seq.pop(), Some(x));
981            check_buckets(&seq.inner);
982        }
983        struct Value;
984        let mut seq = SqrtVec::new();
985        seq.push(Value);
986        seq.insert(0, Value);
987        seq.remove(1);
988        assert!(seq.get_mut(0).is_some());
989        assert_eq!(seq.iter().clone().count(), 1);
990    }
