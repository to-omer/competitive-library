13struct FoldState<M, P, O = ()>
14where
15    M: MapMonoid,
16{
17    agg: MapAgg<M>,
18    order: O,
19    _marker: PhantomData<fn() -> P>,
20}

22impl<T, M, P, O> BucketState<T> for FoldState<M, P, O>
23where
24    M: MapMonoid,
25    P: BucketValue<T, Value = M::Key>,
26    O: BucketOrder,
27{
28    fn new(values: &[T]) -> Self {
29        let agg = P::fold::<M>(M::agg_unit(), values, false);
30        Self {
31            agg,
32            order: O::default(),
33            _marker: PhantomData,
34        }
35    }
36    fn push(&mut self, values: &mut [T]) {
37        self.order.push(values);
38    }
39    fn is_reversed(&self) -> bool {
40        self.order.is_reversed()
41    }
42}

44impl<T, M, P> ReversibleBucketState<T> for FoldState<M, P, bool>
45where
46    M: ReversibleMapMonoid,
47    P: BucketValue<T, Value = M::Key>,
48{
49    fn toggle(&mut self) {
50        M::toggle(&mut self.agg);
51        self.order ^= true;
52    }
53}

55impl<T, M, P, O> Buckets<T, FoldState<M, P, O>>
56where
57    M: MapMonoid,
58    P: BucketValue<T, Value = M::Key>,
59    O: BucketOrder,
60{
61    fn fold(&self, range: Range<usize>) -> MapAgg<M> {
62        assert!(range.start <= range.end);
63        let mut result = M::agg_unit();
64        let (start, index) = self.locate(range.start);
65        let mut offset = range.start - index;
66        for Bucket { values, state } in &self.buckets[start..] {
67            let end = offset + values.len();
68            if range.end <= offset {
69                break;
70            }
71            if range.start <= offset && end <= range.end {
72                result = M::agg_operate(&result, &state.agg);
73            } else if range.start < end {
74                let l = range.start.saturating_sub(offset);
75                let r = range.end.min(end) - offset;
76                if state.order.is_reversed() {
77                    result =
78                        P::fold::<M>(result, &values[values.len() - r..values.len() - l], true);
79                } else {
80                    result = P::fold::<M>(result, &values[l..r], false);
81                }
82            }
83            offset = end;
84        }
85        result
86    }
87}

89/// A sequence with ordered range aggregation and no lazy updates.
90/// With O(1) monoid operations, folds and amortized updates take O(sqrt(N)).
91pub struct SqrtFoldVec<M>
92where
93    M: Monoid,
94{
95    inner: Buckets<M::T, FoldState<EmptyActLazy<M>, SequenceValue>>,
96}

98impl<M> SqrtFoldVec<M>
99where
100    M: Monoid,
101{
102    pub fn new() -> Self {
103        Self {
104            inner: Buckets::new(),
105        }
106    }
107    pub fn len(&self) -> usize {
108        self.inner.len
109    }
110    pub fn is_empty(&self) -> bool {
111        self.len() == 0
112    }
113    pub fn clear(&mut self) {
114        *self = Self::new();
115    }
116    pub fn get(&self, index: usize) -> Option<&M::T> {
117        if index >= self.len() {
118            return None;
119        }
120        self.inner.at(self.inner.locate(index))
121    }
122    pub fn modify(&mut self, index: usize, f: impl FnOnce(&M::T) -> M::T) {
123        assert!(index < self.len());
124        let (i, j) = self.inner.locate(index);
125        let Bucket { values, state } = &mut self.inner.buckets[i];
126        values[j] = f(&values[j]);
127        *state = FoldState::new(values);
128    }
129    pub fn insert(&mut self, index: usize, value: M::T) {
130        assert!(index <= self.len());
131        self.inner.insert(self.inner.locate(index), value);
132    }
133    pub fn remove(&mut self, index: usize) -> M::T {
134        assert!(index < self.len());
135        self.inner.remove(self.inner.locate(index))
136    }
137    pub fn push(&mut self, value: M::T) {
138        self.inner.insert((self.inner.buckets.len(), 0), value);
139    }
140    pub fn pop(&mut self) -> Option<M::T> {
141        let i = self.inner.buckets.len().checked_sub(1)?;
142        Some(
143            self.inner
144                .remove((i, self.inner.buckets[i].values.len() - 1)),
145        )
146    }
147    pub fn iter(&self) -> impl DoubleEndedIterator<Item = &M::T> {
148        self.inner.buckets.iter().flat_map(|b| b.values.iter())
149    }
150    pub fn fold<R>(&self, range: R) -> M::T
151    where
152        R: RangeBounds<usize>,
153    {
154        let range = range
155            .to_range_bounded(0, self.len())
156            .expect("invalid range");
157        self.inner.fold(range)
158    }
159}

320/// An ordered map with value aggregation in key order and no lazy updates.
321/// Key lookup takes O(log N); folds and amortized updates take O(sqrt(N)) for O(1) monoid operations.
322pub struct SqrtFoldMap<K, M>
323where
324    M: Monoid,
325{
326    inner: Buckets<(K, M::T), FoldState<EmptyActLazy<M>, MapValue>>,
327}
328
329impl<K, M> SqrtFoldMap<K, M>
330where
331    M: Monoid,
332{
333    pub fn new() -> Self {
334        Self {
335            inner: Buckets::new(),
336        }
337    }
338    pub fn len(&self) -> usize {
339        self.inner.len
340    }
341    pub fn is_empty(&self) -> bool {
342        self.len() == 0
343    }
344    pub fn clear(&mut self) {
345        *self = Self::new();
346    }
347    pub fn get<Q>(&self, key: &Q) -> Option<&M::T>
348    where
349        K: Borrow<Q>,
350        Q: Ord + ?Sized,
351    {
352        self.get_key_value(key).map(|(_, v)| v)
353    }
354    pub fn get_key_value<Q>(&self, key: &Q) -> Option<(&K, &M::T)>
355    where
356        K: Borrow<Q>,
357        Q: Ord + ?Sized,
358    {
359        let p = self.inner.partition_point(|(k, _)| k.borrow() < key);
360        self.inner
361            .at(p)
362            .filter(|(k, _)| k.borrow() == key)
363            .map(|(k, v)| (k, v))
364    }
365    pub fn contains_key<Q>(&self, key: &Q) -> bool
366    where
367        K: Borrow<Q>,
368        Q: Ord + ?Sized,
369    {
370        self.get(key).is_some()
371    }
372    pub fn get_key_value_at(&self, index: usize) -> Option<(&K, &M::T)> {
373        if index >= self.len() {
374            return None;
375        }
376        self.inner.at(self.inner.locate(index)).map(|(k, v)| (k, v))
377    }
378    pub fn insert(&mut self, key: K, value: M::T) -> Option<M::T>
379    where
380        K: Ord,
381    {
382        let (i, j) = self.inner.partition_point(|(k, _)| k < &key);
383        if self.inner.at((i, j)).is_some_and(|(k, _)| k == &key) {
384            let Bucket { values, state } = &mut self.inner.buckets[i];
385            let old = mem::replace(&mut values[j].1, value);
386            *state = FoldState::new(values);
387            return Some(old);
388        }
389        self.inner.insert((i, j), (key, value));
390        None
391    }
392    /// Returns whether the key was present.
393    pub fn modify<Q>(&mut self, key: &Q, f: impl FnOnce(&M::T) -> M::T) -> bool
394    where
395        K: Borrow<Q>,
396        Q: Ord + ?Sized,
397    {
398        let (i, j) = self.inner.partition_point(|(k, _)| k.borrow() < key);
399        if !self
400            .inner
401            .at((i, j))
402            .is_some_and(|(k, _)| k.borrow() == key)
403        {
404            return false;
405        }
406        let Bucket { values, state } = &mut self.inner.buckets[i];
407        values[j].1 = f(&values[j].1);
408        *state = FoldState::new(values);
409        true
410    }
411    pub fn remove<Q>(&mut self, key: &Q) -> Option<M::T>
412    where
413        K: Borrow<Q>,
414        Q: Ord + ?Sized,
415    {
416        self.remove_entry(key).map(|(_, v)| v)
417    }
418    pub fn remove_entry<Q>(&mut self, key: &Q) -> Option<(K, M::T)>
419    where
420        K: Borrow<Q>,
421        Q: Ord + ?Sized,
422    {
423        let p = self.inner.partition_point(|(k, _)| k.borrow() < key);
424        self.inner.at(p).filter(|(k, _)| k.borrow() == key)?;
425        Some(self.inner.remove(p))
426    }
427    pub fn remove_at(&mut self, index: usize) -> Option<(K, M::T)> {
428        (index < self.len()).then(|| self.inner.remove(self.inner.locate(index)))
429    }
430    /// Counts keys strictly less than `key`.
431    pub fn rank<Q>(&self, key: &Q) -> usize
432    where
433        K: Borrow<Q>,
434        Q: Ord + ?Sized,
435    {
436        self.inner
437            .position(self.inner.partition_point(|(k, _)| k.borrow() < key))
438    }
439    /// Counts keys less than or equal to `key`.
440    pub fn rank_le<Q>(&self, key: &Q) -> usize
441    where
442        K: Borrow<Q>,
443        Q: Ord + ?Sized,
444    {
445        self.inner
446            .position(self.inner.partition_point(|(k, _)| k.borrow() <= key))
447    }
448    pub fn iter(&self) -> impl DoubleEndedIterator<Item = (&K, &M::T)> {
449        self.inner
450            .buckets
451            .iter()
452            .flat_map(|b| b.values.iter().map(|(k, v)| (k, v)))
453    }
454    pub fn fold<Q, R>(&self, range: R) -> M::T
455    where
456        K: Borrow<Q>,
457        Q: Ord + ?Sized,
458        R: RangeBounds<Q>,
459    {
460        let (start, end) = self.inner.bounds(range, |(k, _)| k.borrow());
461        self.inner
462            .fold(self.inner.position(start)..self.inner.position(end))
463    }
464}
