3/// A fixed-length mutable bitset with a prefix-count index over its words.
4///
5/// Updates, rank, and select1 take logarithmic time. The default select0 uses
6/// binary search over rank. The bit length must fit in u32.
7#[derive(Clone, Debug)]
8pub struct RankSelectBitSet {
9    bits: BitSet,
10    counts: DaryPrefixSumTreeU32,
11}
12
13impl RankSelectBitSet {
14    pub fn new(len: usize) -> Self {
15        assert!(u32::try_from(len).is_ok());
16        Self {
17            bits: BitSet::new(len),
18            counts: DaryPrefixSumTreeU32::new(len.div_ceil(64)),
19        }
20    }
21
22    /// Builds from low-bit-first words, ignoring unused high bits of the last word.
23    pub fn from_words(words: &[u64], len: usize) -> Self {
24        assert!(u32::try_from(len).is_ok());
25        assert_eq!(words.len(), len.div_ceil(64));
26        let mut bits = BitSet::new(len);
27        bits.words_mut()[..words.len()].copy_from_slice(words);
28        if !len.is_multiple_of(64) {
29            bits.words_mut()[len / 64] &= (1 << (len % 64)) - 1;
30        }
31        let counts: Vec<_> = bits.words()[..words.len()]
32            .iter()
33            .map(|word| word.count_ones())
34            .collect();
35        Self {
36            bits,
37            counts: DaryPrefixSumTreeU32::from_slice(&counts),
38        }
39    }
40
41    pub fn len(&self) -> usize {
42        self.bits.len()
43    }
44
45    pub fn is_empty(&self) -> bool {
46        self.bits.is_empty()
47    }
48
49    pub fn set(&mut self, index: usize, value: bool) {
50        assert!(index < self.len());
51        let word = index / 64;
52        let bits = &mut self.bits.words_mut()[word];
53        let mask = 1u64 << (index % 64);
54        if (*bits & mask != 0) != value {
55            *bits ^= mask;
56            self.counts.set(word, bits.count_ones());
57        }
58    }
59
60    pub fn count_ones(&self) -> u64 {
61        self.counts.fold_all() as u64
62    }
63
64    pub fn count_zeros(&self) -> u64 {
65        self.len() as u64 - self.count_ones()
66    }
67
68    /// Returns the first set bit in `start..self.len()`.
69    pub fn next_one(&self, start: usize) -> Option<usize> {
70        assert!(start <= self.len());
71        if start == self.len() {
72            return None;
73        }
74        let word = start / 64;
75        let bits = self.bits.words()[word] & (u64::MAX << (start % 64));
76        if bits != 0 {
77            return Some(word * 64 + bits.trailing_zeros() as usize);
78        }
79        let count = self.counts.accumulate0(word + 1);
80        let word = self.counts.partition_point_acc(count);
81        (word < self.counts.len())
82            .then(|| word * 64 + self.bits.words()[word].trailing_zeros() as usize)
83    }
84
85    /// Returns the last set bit in `0..end`.
86    pub fn prev_one(&self, end: usize) -> Option<usize> {
87        assert!(end <= self.len());
88        if end == 0 {
89            return None;
90        }
91        let word = (end - 1) / 64;
92        let bits = self.bits.words()[word] & (u64::MAX >> (63 - (end - 1) % 64));
93        if bits != 0 {
94            return Some(word * 64 + 63 - bits.leading_zeros() as usize);
95        }
96        let count = self.counts.accumulate0(word);
97        if count == 0 {
98            return None;
99        }
100        let word = self.counts.partition_point_acc(count - 1);
101        Some(word * 64 + 63 - self.bits.words()[word].leading_zeros() as usize)
102    }
103}
104
105impl RankSelectDictionaries for RankSelectBitSet {
106    fn bit_length(&self) -> usize {
107        self.len()
108    }
109
110    fn access(&self, index: usize) -> bool {
111        assert!(index < self.len());
112        self.bits.get(index)
113    }
114
115    fn rank1(&self, end: usize) -> usize {
116        assert!(end <= self.len());
117        let word = end / 64;
118        let mut count = self.counts.accumulate0(word) as usize;
119        if !end.is_multiple_of(64) {
120            count += self.bits.words()[word].rank1(end % 64);
121        }
122        count
123    }
124
125    fn select1(&self, rank: usize) -> Option<usize> {
126        if rank >= self.counts.fold_all() as usize {
127            return None;
128        }
129        let (word, prefix) = self.counts.partition_point_acc_with_sum(rank as u32);
130        let local = rank - prefix as usize;
131        Some(word * 64 + BitVector::select_word(self.bits.words()[word], local))
132    }
133}

135#[cfg(test)]
136mod tests {
137    use super::*;
138    use crate::tools::{Xorshift, testutil::sequence_cases};
139
140    #[test]
141    fn test_rank_select_bitset() {
142        let mut rng = Xorshift::default();
143        let cases: Vec<_> = sequence_cases(&mut rng, 0u8..=1, 0..=4097).collect();
144        for mut expected in cases {
145            let len = expected.len();
146            let mut words = vec![u64::MAX; len.div_ceil(64)];
147            for (index, &bit) in expected.iter().enumerate() {
148                if bit == 0 {
149                    words[index / 64] &= !(1 << (index % 64));
150                }
151            }
152            let mut actual = RankSelectBitSet::from_words(&words, len);
153            for step in 0..160 {
154                if len != 0 && step != 0 {
155                    let index = match step % 4 {
156                        0 => 0,
157                        1 => len - 1,
158                        _ => rng.random(0..len),
159                    };
160                    let value = rng.random(0u8..=1);
161                    actual.set(index, value != 0);
162                    actual.set(index, value != 0);
163                    expected[index] = value;
164                    assert_eq!(actual.access(index), value != 0);
165                }
166                assert_eq!(actual.bit_length(), len);
167                assert_eq!(actual.is_empty(), len == 0);
168                let ones: Vec<_> = (0..len).filter(|&i| expected[i] != 0).collect();
169                let zeros: Vec<_> = (0..len).filter(|&i| expected[i] == 0).collect();
170                assert_eq!(actual.count_ones(), ones.len() as u64);
171                assert_eq!(actual.count_zeros(), zeros.len() as u64);
172                for end in [0, len, rng.random(0..=len)] {
173                    let rank = ones.partition_point(|&i| i < end);
174                    assert_eq!(actual.rank1(end), rank);
175                    assert_eq!(actual.rank0(end), zeros.partition_point(|&i| i < end));
176                    assert_eq!(actual.next_one(end), ones.get(rank).copied());
177                    assert_eq!(actual.prev_one(end), rank.checked_sub(1).map(|i| ones[i]));
178                }
179                for rank in [0, ones.len() / 2, ones.len(), usize::MAX] {
180                    assert_eq!(actual.select1(rank), ones.get(rank).copied());
181                }
182                for rank in [0, zeros.len() / 2, zeros.len(), usize::MAX] {
183                    assert_eq!(actual.select0(rank), zeros.get(rank).copied());
184                }
185            }
186            let empty = RankSelectBitSet::new(len);
187            assert_eq!(empty.rank1(len), 0);
188            assert_eq!(empty.select1(0), None);
189        }
190    }
