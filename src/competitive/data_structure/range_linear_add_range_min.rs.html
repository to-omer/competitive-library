1use super::RangeBoundsExt;
2use std::ops::{Range, RangeBounds};
3
4const BLOCK_SHIFT: u32 = 5;
5const FANOUT: usize = 8;
6const ROOT_CAPACITY: usize = 128;
7const SCAN_LIMIT: u8 = 64;
8
9type Line = (i128, i128);
10
11/// Range minimum queries with additions of an affine function of the index.
12///
13/// For `i64` values, construction and q operations take O(n + q log² n) time and O(n) space.
14#[derive(Debug, Clone)]
15pub struct RangeLinearAddRangeMin {
16    values: Vec<i64>,
17    levels: Vec<Level>,
18}
19
20#[derive(Debug, Clone)]
21struct Node {
22    // Let L interpolate `left` and `right` over the full block width.
23    // All values satisfy a[i] >= L(i), and at least one satisfies a[i] <= L(i) + spread.
24    left: i128,
25    right: i128,
26    spread: i128,
27    lazy: Line,
28}
29
30impl Node {
31    fn new(minimum: i128) -> Self {
32        Self {
33            left: minimum,
34            right: minimum,
35            spread: 0,
36            lazy: (0, 0),
37        }
38    }
39}
40
41#[derive(Debug, Clone, Default)]
42struct Envelope {
43    visits: u8,
44    // In index order, each child beats its predecessor below its cut.
45    switches: Vec<(i128, usize)>,
46}
47
48#[derive(Debug, Clone)]
49struct Level {
50    shift: u32,
51    nodes: Vec<Node>,
52    envelopes: Vec<Envelope>,
53}
54
55impl RangeLinearAddRangeMin {
56    pub fn from_keys(keys: impl ExactSizeIterator<Item = i64>) -> Self {
57        let values: Vec<_> = keys.collect();
58        let blocks = values
59            .chunks(1 << BLOCK_SHIFT)
60            .map(|v| Node::new(i128::from(*v.iter().min().unwrap())))
61            .collect();
62        let mut level = Level {
63            shift: BLOCK_SHIFT,
64            nodes: blocks,
65            envelopes: Vec::new(),
66        };
67        let mut levels = Vec::new();
68        while level.nodes.len() > ROOT_CAPACITY {
69            let nodes: Vec<_> = level
70                .nodes
71                .chunks(FANOUT)
72                .map(|v| Node::new(v.iter().map(|n| n.left).min().unwrap()))
73                .collect();
74            let next = Level {
75                shift: level.shift + FANOUT.ilog2(),
76                envelopes: vec![Envelope::default(); nodes.len()],
77                nodes,
78            };
79            levels.push(level);
80            level = next;
81        }
82        levels.push(level);
83        Self { values, levels }
84    }
85
86    /// Adds `slope * i + intercept` to every element at index `i` in the range.
87    /// Each resulting element value must fit in `i64`.
88    pub fn update<R>(&mut self, range: R, slope: i64, intercept: i64)
89    where
90        R: RangeBounds<usize>,
91    {
92        let range = range
93            .to_range_bounded(0, self.values.len())
94            .expect("invalid range");
95        if !range.is_empty() {
96            self.update_at(
97                self.levels.len() - 1,
98                range,
99                (0, 0),
100                (slope.into(), intercept.into()),
101            );
102        }
103    }
104
105    /// Returns the minimum in the range, or `i64::MAX` for an empty range.
106    pub fn fold<R>(&mut self, range: R) -> i64
107    where
108        R: RangeBounds<usize>,
109    {
110        let range = range
111            .to_range_bounded(0, self.values.len())
112            .expect("invalid range");
113        if range.is_empty() {
114            return i64::MAX;
115        }
116        self.fold_at(self.levels.len() - 1, range, (0, 0), i64::MAX as i128) as i64
117    }
118
119    fn split(&self, range: Range<usize>, shift: u32) -> ([Range<usize>; 2], Range<usize>) {
120        let mask = (1 << shift) - 1;
121        let n = self.values.len();
122        let l = (range.start + mask) >> shift;
123        let r = (range.end + if range.end == n { mask } else { 0 }) >> shift;
124        if l > r {
125            return ([range, 0..0], l..l);
126        }
127        (
128            [
129                range.start..(l << shift).min(n),
130                (r << shift).min(n)..range.end,
131            ],
132            l..r,
133        )
134    }

136    fn update_at(
137        &mut self,
138        height: usize,
139        range: Range<usize>,
140        ancestors: Line,
141        (slope, intercept): Line,
142    ) {
143        let shift = self.levels[height].shift;
144        let width = 1 << shift;
145        let (edges, full) = self.split(range, shift);
146        for edge in edges {
147            if edge.is_empty() {
148                continue;
149            }
150            let k = edge.start >> shift;
151            let start = k << shift;
152            let tag = self.levels[height].nodes[k].lazy;
153            let (b, c) = (ancestors.0 + tag.0, ancestors.1 + tag.1);
154            let (lower, upper, tag) = if height == 0 {
155                let end = (start + width).min(self.values.len());
156                let mut minimum = i64::MAX;
157                for (part, b, c) in [
158                    (start..edge.start, b, c),
159                    (edge.clone(), b + slope, c + intercept),
160                    (edge.end..end, b, c),
161                ] {
162                    for (i, x) in part.clone().zip(&mut self.values[part]) {
163                        *x = (*x as i128 + b * i as i128 + c) as i64;
164                        minimum = minimum.min(*x);
165                    }
166                }
167                // The raw values now include the ancestors' pending additions.
168                (
169                    minimum as i128,
170                    minimum as i128,
171                    (-ancestors.0, -ancestors.1),
172                )
173            } else {
174                self.update_at(height - 1, edge, (b, c), (slope, intercept));
175                let first = k * FANOUT;
176                let nodes = &self.levels[height - 1].nodes;
177                let (mut lower, mut upper) = (i128::MAX, i128::MAX);
178                for child in &nodes[first..(first + FANOUT).min(nodes.len())] {
179                    lower = lower.min(child.left.min(child.right));
180                    upper = upper.min(child.left.max(child.right) + child.spread);
181                }
182                let envelope = &mut self.levels[height].envelopes[k];
183                envelope.visits = 0;
184                envelope.switches.clear();
185                (lower, upper, tag)
186            };
187            let node = &mut self.levels[height].nodes[k];
188            node.lazy = tag;
189            node.left = lower + tag.0 * start as i128 + tag.1;
190            node.right = node.left + (tag.0 << shift);
191            node.spread = upper - lower;
192        }
193        let step = slope << shift;
194        let mut delta = slope * (full.start << shift) as i128 + intercept;
195        for node in &mut self.levels[height].nodes[full] {
196            node.left += delta;
197            delta += step;
198            node.right += delta;
199            node.lazy.0 += slope;
200            node.lazy.1 += intercept;
201        }
202    }

204    fn fold_at(
205        &mut self,
206        height: usize,
207        range: Range<usize>,
208        ancestors: Line,
209        mut best: i128,
210    ) -> i128 {
211        let shift = self.levels[height].shift;
212        let width = 1 << shift;
213        let (edges, full) = self.split(range, shift);
214        let step = ancestors.0 << shift;
215        let initial = ancestors.0 * (full.start << shift) as i128 + ancestors.1;
216        if height != 0 {
217            let mut delta = initial;
218            for node in &self.levels[height].nodes[full.clone()] {
219                let left = node.left + delta;
220                delta += step;
221                best = best.min(left.max(node.right + delta) + node.spread);
222            }
223        }
224        for edge in edges {
225            if edge.is_empty() {
226                continue;
227            }
228            let k = edge.start >> shift;
229            let tag = self.levels[height].nodes[k].lazy;
230            let line = (ancestors.0 + tag.0, ancestors.1 + tag.1);
231            best = if height == 0 {
232                best.min(self.minimum(edge, line) as i128)
233            } else {
234                self.fold_at(height - 1, edge, line, best)
235            };
236        }
237        let mut delta = initial;
238        for k in full {
239            let node = &self.levels[height].nodes[k];
240            let left = node.left + delta;
241            delta += step;
242            let right = node.right + delta;
243            if left.min(right) >= best {
244                continue;
245            }
246            if left == right && node.spread == 0 {
247                best = left;
248                continue;
249            }
250            let start = k << shift;
251            let end = (start + width).min(self.values.len());
252            let line = (ancestors.0 + node.lazy.0, ancestors.1 + node.lazy.1);
253            let upper = left.max(right) + node.spread;
254            let minimum = if height == 0 {
255                i128::from(self.minimum(start..end, line))
256            } else if self.levels[height].envelopes[k].visits < SCAN_LIMIT {
257                self.levels[height].envelopes[k].visits += 1;
258                // Use this node's own upper bound so the cached minimum is exact.
259                self.fold_at(height - 1, start..end, line, upper)
260            } else {
261                self.ensure_envelope(height, k, line.0);
262                let (block, line) = self.winning_block(height, k, ancestors);
263                i128::from(self.minimum(block, line))
264            };
265            best = best.min(minimum);
266            let node = &mut self.levels[height].nodes[k];
267            node.right = minimum - delta;
268            node.left = node.right + step;
269            node.spread = 0;
270        }
271        best
272    }

274    fn minimum(&self, range: Range<usize>, (b, c): Line) -> i64 {
275        range
276            .clone()
277            .zip(&self.values[range])
278            .map(|(i, &v)| (v as i128 + b * i as i128 + c) as i64)
279            .min()
280            .unwrap()
281    }
282
283    fn winning_block(
284        &self,
285        mut height: usize,
286        mut k: usize,
287        (mut slope, mut intercept): Line,
288    ) -> (Range<usize>, Line) {
289        while height > 0 {
290            let level = &self.levels[height];
291            let (a, b) = level.nodes[k].lazy;
292            slope += a;
293            intercept += b;
294            let switches = &level.envelopes[k].switches;
295            let i = switches[1..].partition_point(|&(cut, _)| slope < cut);
296            k = switches[i].1;
297            height -= 1;
298        }
299        let start = k << BLOCK_SHIFT;
300        let end = (start + (1 << BLOCK_SHIFT)).min(self.values.len());
301        let (a, b) = self.levels[0].nodes[k].lazy;
302        (start..end, (slope + a, intercept + b))
303    }
304
305    fn support(&self, height: usize, k: usize, slope: i128) -> (usize, i128) {
306        let (range, (a, b)) = self.winning_block(height, k, (slope, 0));
307        // Trial slopes in a crossing search can give values outside i64.
308        range
309            .clone()
310            .zip(&self.values[range])
311            .map(|(x, &y)| (x, i128::from(y) + a * x as i128 + b))
312            .min_by_key(|&(_, y)| y)
313            .unwrap()
314    }
315
316    // First integer slope with left_min <= right_min. Larger right indices
317    // make right_min - left_min increase by at least one per slope step.
318    fn crossing(&self, height: usize, left: usize, right: usize, mut slope: i128) -> i128 {
319        let (mut lo, mut hi) = (i128::MIN, i128::MAX);
320        let mut secant = true;
321        loop {
322            let (x, y) = self.support(height, left, slope);
323            let (z, w) = self.support(height, right, slope);
324            let gap = w - y;
325            if gap == 0 {
326                return slope;
327            }
328            if gap > 0 {
329                hi = slope;
330                lo = lo.max(slope - gap - 1);
331            } else {
332                lo = slope;
333                hi = hi.min(slope - gap);
334            }
335            if hi - lo == 1 {
336                return hi;
337            }
338            slope = if secant {
339                let dx = (z - x) as i128;
340                let q = gap.div_euclid(dx);
341                (slope - q).clamp(lo + 1, hi - 1)
342            } else {
343                lo + (hi - lo) / 2
344            };
345            secant = !secant;
346        }
347    }
348
349    fn ensure_envelope(&mut self, height: usize, k: usize, slope: i128) {
350        if !self.levels[height].envelopes[k].switches.is_empty() {
351            return;
352        }
353        let mut table = std::mem::take(&mut self.levels[height].envelopes[k].switches);
354        let first = k * FANOUT;
355        for child in first..(first + FANOUT).min(self.levels[height - 1].nodes.len()) {
356            if height > 1 {
357                self.ensure_envelope(
358                    height - 1,
359                    child,
360                    slope + self.levels[height - 1].nodes[child].lazy.0,
361                );
362            }
363            let cut = loop {
364                let Some(&(prev, old)) = table.last() else {
365                    break i128::MAX;
366                };
367                let cut = self.crossing(height - 1, old, child, slope);
368                if cut < prev {
369                    break cut;
370                }
371                // The previous child has no winning integer slope.
372                table.pop();
373            };
374            table.push((cut, child));
375        }
376        let envelope = &mut self.levels[height].envelopes[k];
377        envelope.visits = SCAN_LIMIT;
378        envelope.switches = table;
379    }

387    #[test]
388    fn test_range_linear_add_range_min() {
389        let mut rng = Xorshift::default();
390        let mut input_rng = Xorshift::default();
391        let inputs = sequence_cases(&mut input_rng, -1000..=1000, 0..=1025)
392            .chain([(-256i64..=256).map(|x| x * x).collect()]);
393        for mut values in inputs {
394            let n = values.len();
395            let mut data = RangeLinearAddRangeMin::from_keys(values.iter().copied());
396            assert_eq!(
397                data.fold(..),
398                values.iter().copied().min().unwrap_or(i64::MAX)
399            );
400            for step in 0..=2 * n {
401                let (l, r) = rng.random(WithEmptySegment(n));
402                let range = if step % 4 == 0 { 0..n } else { l..r };
403                let b = rng.random(-99i64..=99);
404                let c = rng.random(-1000i64..=1000);
405                data.update(range.clone(), b, c);
406                for (i, x) in range.clone().zip(&mut values[range.clone()]) {
407                    *x += b * i as i64 + c;
408                }
409                let (l, r) = rng.random(WithEmptySegment(n));
410                for query in [range, l..r, 0..n] {
411                    let expected = values[query.clone()]
412                        .iter()
413                        .copied()
414                        .min()
415                        .unwrap_or(i64::MAX);
416                    assert_eq!(data.fold(query), expected);
417                }
418            }
419        }
420    }

446    #[test]
447    fn test_group_bounds_at_i64_limits() {
448        use std::ops::Bound::{Excluded, Included};
449
450        let len = ROOT_CAPACITY * (1 << BLOCK_SHIFT) + 1;
451        for (limit, slope) in [(i64::MAX, 99), (i64::MIN, -99)] {
452            let mut values: Vec<_> = (0..len).map(|i| limit - slope * i as i64).collect();
453            let mut data = RangeLinearAddRangeMin::from_keys(values.iter().copied());
454            data.update(.., slope, 0);
455            values.fill(limit);
456            assert_eq!(data.fold(..), limit);
457            data.update((Excluded(4), Included(23)), -slope, 0);
458            for (i, value) in (5..=23).zip(&mut values[5..=23]) {
459                *value -= slope * i;
460            }
461            for l in 0..32 {
462                for r in l..32 {
463                    assert_eq!(data.fold(l..=r), *values[l..=r].iter().min().unwrap());
464                }
465            }
466            assert_eq!(data.fold(32..), limit);
467            assert_eq!(data.fold(len..), i64::MAX);
468        }

471    #[test]
472    fn test_envelope_crossings_and_invalidation() {
473        let mut rng = Xorshift::default();
474        let block = 1 << BLOCK_SHIFT;
475        let limit = usize::from(SCAN_LIMIT);
476        for span in [block * FANOUT, block * FANOUT * FANOUT] {
477            let len = span / FANOUT * ROOT_CAPACITY + rng.random(span / 2..span);
478            let parabola = (0..len)
479                .map(|i| {
480                    let x = (i % span) as i64 - (span / 2) as i64;
481                    x * x / (span / 64) as i64 + rng.random(-16i64..=16)
482                })
483                .collect();
484            let extremes = (0..len)
485                .map(|i| {
486                    let sign = if (i / block).is_multiple_of(2) { -1 } else { 1 };
487                    sign * (i64::MAX - 10_000_000) + (i % block) as i64
488                })
489                .collect();
490            let full_roots = len / span;
491            for (mut values, root) in [
492                (vec![0; len], full_roots),
493                (parabola, rng.random(0..full_roots)),
494                (extremes, rng.random(0..full_roots)),
495            ] {
496                let mut data = RangeLinearAddRangeMin::from_keys(values.iter().copied());
497                let height = data.levels.len() - 1;
498                let target = root * span..((root + 1) * span).min(len);
499                let mut slope = 0;
500                for width in [0, 1, rng.random(2..target.len())] {
501                    if width != 0 {
502                        let l = rng.random(target.start..=target.end - width);
503                        let range = l..l + width;
504                        let b = rng.random(-3..=3);
505                        let c = rng.random(-1_000_000..=-100_000);
506                        data.update(range.clone(), b, c);
507                        for i in range {
508                            values[i] += b * i as i64 + c;
509                        }
510                        assert!(data.levels[height].envelopes[root].switches.is_empty());
511                    }
512                    // Changing the slope and querying this node in isolation forces a visit.
513                    for turn in 0..rng.random(limit + 2..=2 * limit) {
514                        let magnitude = rng.random(1i64..=49);
515                        let next = if slope > 0 { -magnitude } else { magnitude };
516                        let b = next - slope;
517                        let c = rng.random(-100..=100);
518                        data.update(.., b, c);
519                        for (i, x) in values.iter_mut().enumerate() {
520                            *x += b * i as i64 + c;
521                        }
522                        slope = next;
523                        let (l, r) = rng.random(WithEmptySegment(len));
524                        let query = if turn % 4 == 0 { 0..len } else { l..r };
525                        for range in [target.clone(), query] {
526                            let expected = values[range.clone()].iter().copied().min();
527                            assert_eq!(data.fold(range), expected.unwrap_or(i64::MAX));
528                        }
529                    }
530                    assert!(data.levels[height].envelopes[root].switches.len() > 1);
531                }
532            }
533        }
534    }
