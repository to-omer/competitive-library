48impl<M> LazySegmentTree<M>
49where
50    M: LazyMapMonoid,
51{
52    pub fn new(len: usize) -> Self {
53        let n = len.next_power_of_two();
54        let seg = vec![M::agg_unit(); 2 * n];
55        let lazy = vec![M::act_unit(); n];
56        Self { len, n, seg, lazy }
57    }
58    pub fn from_vec(v: Vec<LazyMapAgg<M>>) -> Self {
59        let len = v.len();
60        let n = len.next_power_of_two();
61        let mut seg = vec![M::agg_unit(); 2 * n];
62        for (i, x) in v.into_iter().enumerate() {
63            seg[i + n] = x;
64        }
65        for i in (1..n).rev() {
66            seg[i] = M::agg_operate(&seg[2 * i], &seg[2 * i + 1]);
67        }
68        let lazy = vec![M::act_unit(); n];
69        Self { len, n, seg, lazy }
70    }
71    pub fn from_keys(keys: impl ExactSizeIterator<Item = M::Key>) -> Self {
72        let len = keys.len();
73        let n = len.next_power_of_two();
74        let mut seg = vec![M::agg_unit(); 2 * n];
75        for (i, key) in keys.enumerate() {
76            seg[i + n] = M::single_agg(&key);
77        }
78        for i in (1..n).rev() {
79            seg[i] = M::agg_operate(&seg[2 * i], &seg[2 * i + 1]);
80        }
81        let lazy = vec![M::act_unit(); n];
82        Self { len, n, seg, lazy }
83    }
84    #[inline]
85    fn update_at(&mut self, k: usize, x: &LazyMapAct<M>) {
86        if M::is_act_unit(x) {
87            return;
88        }
89        let nx = M::act_agg(&self.seg[k], x);
90        if k < self.n {
91            self.lazy[k] = M::act_operate(&self.lazy[k], x);
92        }
93        if let Some(nx) = nx {
94            self.seg[k] = nx;
95        } else if k < self.n {
96            self.propagate_at(k);
97            self.recalc_at(k);
98        } else {
99            panic!("act failed on leaf");
100        }
101    }
102    #[inline]
103    fn recalc_at(&mut self, k: usize) {
104        self.seg[k] = M::agg_operate(&self.seg[2 * k], &self.seg[2 * k + 1]);
105    }
106    #[inline]
107    fn propagate_at(&mut self, k: usize) {
108        debug_assert!(k < self.n);
109        let x = replace(&mut self.lazy[k], M::act_unit());
110        if M::is_act_unit(&x) {
111            return;
112        }
113        self.update_at(2 * k, &x);
114        self.update_at(2 * k + 1, &x);
115    }
116    #[inline]
117    fn propagate(&mut self, k: usize) {
118        for i in (1..=self.n.trailing_zeros()).rev() {
119            self.propagate_at(k >> i);
120        }
121    }
122    #[inline]
123    fn recalc(&mut self, mut k: usize) {
124        while k > 1 {
125            k >>= 1;
126            self.recalc_at(k);
127        }
128    }
129    pub fn update<R>(&mut self, range: R, x: LazyMapAct<M>)
130    where
131        R: RangeBounds<usize>,
132    {
133        let range = range.to_range_bounded(0, self.len).expect("invalid range");
134        if range.is_empty() || M::is_act_unit(&x) {
135            return;
136        }
137        let mut a = range.start + self.n;
138        let mut b = range.end + self.n;
139        for i in (1..=self.n.trailing_zeros()).rev() {
140            if (a >> i) << i != a {
141                self.propagate_at(a >> i);
142            }
143            if (b >> i) << i != b {
144                self.propagate_at((b - 1) >> i);
145            }
146        }
147        while a < b {
148            if a & 1 != 0 {
149                self.update_at(a, &x);
150                a += 1;
151            }
152            if b & 1 != 0 {
153                b -= 1;
154                self.update_at(b, &x);
155            }
156            a /= 2;
157            b /= 2;
158        }
159        let a = range.start + self.n;
160        let b = range.end + self.n;
161        for i in 1..=self.n.trailing_zeros() {
162            if (a >> i) << i != a {
163                self.recalc_at(a >> i);
164            }
165            if (b >> i) << i != b {
166                self.recalc_at((b - 1) >> i);
167            }
168        }
169    }
170    pub fn fold<R>(&mut self, range: R) -> LazyMapAgg<M>
171    where
172        R: RangeBounds<usize>,
173    {
174        let range = range.to_range_bounded(0, self.len).expect("invalid range");
175        if range.is_empty() {
176            return M::agg_unit();
177        }
178        if let Some(result) = (|| {
179            let mut left_index = range.start + self.n - 1;
180            let mut right_index = range.end + self.n;
181            let mut left = M::agg_unit();
182            let mut right = M::agg_unit();
183            let mut has_left = false;
184            let mut has_right = false;
185            for _ in 0..(left_index ^ right_index).ilog2() {
186                if left_index & 1 == 0 {
187                    left = M::agg_operate(&left, &self.seg[left_index ^ 1]);
188                    has_left = true;
189                }
190                if right_index & 1 != 0 {
191                    right = M::agg_operate(&self.seg[right_index ^ 1], &right);
192                    has_right = true;
193                }
194                left_index >>= 1;
195                right_index >>= 1;
196                if has_left {
197                    left = M::act_agg(&left, &self.lazy[left_index])?;
198                }
199                if has_right && right_index < self.n {
200                    right = M::act_agg(&right, &self.lazy[right_index])?;
201                }
202            }
203            let mut result = M::agg_operate(&left, &right);
204            while left_index > 1 {
205                left_index >>= 1;
206                result = M::act_agg(&result, &self.lazy[left_index])?;
207            }
208            Some(result)
209        })() {
210            return result;
211        }
212        let mut l = range.start + self.n;
213        let mut r = range.end + self.n;
214        self.propagate(l);
215        self.propagate(r - 1);
216        let mut vl = M::agg_unit();
217        let mut vr = M::agg_unit();
218        while l < r {
219            if l & 1 != 0 {
220                vl = M::agg_operate(&vl, &self.seg[l]);
221                l += 1;
222            }
223            if r & 1 != 0 {
224                r -= 1;
225                vr = M::agg_operate(&self.seg[r], &vr);
226            }
227            l /= 2;
228            r /= 2;
229        }
230        M::agg_operate(&vl, &vr)
231    }
232    pub fn set(&mut self, k: usize, x: LazyMapAgg<M>) {
233        assert!(k < self.len);
234        let k = k + self.n;
235        self.propagate(k);
236        self.seg[k] = x;
237        self.recalc(k);
238    }
239    pub fn get(&mut self, k: usize) -> LazyMapAgg<M> {
240        self.fold(k..k + 1)
241    }
242    pub fn fold_all(&self) -> LazyMapAgg<M> {
243        self.seg[1].clone()
244    }
245    pub fn partition_point_acc<P>(&mut self, left: usize, mut pred: P) -> usize
246    where
247        P: FnMut(&LazyMapAgg<M>) -> bool,
248    {
249        let mut acc = M::agg_unit();
250        if left == self.len {
251            return self.len;
252        }
253        let mut k = left + self.n;
254        self.propagate(k);
255        loop {
256            while k & 1 == 0 {
257                k >>= 1;
258            }
259            let nacc = M::agg_operate(&acc, &self.seg[k]);
260            if !pred(&nacc) {
261                while k < self.n {
262                    self.propagate_at(k);
263                    k <<= 1;
264                    let nacc = M::agg_operate(&acc, &self.seg[k]);
265                    if pred(&nacc) {
266                        acc = nacc;
267                        k += 1;
268                    }
269                }
270                return k - self.n;
271            }
272            acc = nacc;
273            k += 1;
274            if k.is_power_of_two() {
275                return self.len;
276            }
277        }
278    }
279    pub fn rpartition_point_acc<P>(&mut self, right: usize, mut pred: P) -> usize
280    where
281        P: FnMut(&LazyMapAgg<M>) -> bool,
282    {
283        let mut acc = M::agg_unit();
284        if right == 0 {
285            return 0;
286        }
287        let mut k = right + self.n;
288        self.propagate(k - 1);
289        loop {
290            k -= 1;
291            while k > 1 && k & 1 != 0 {
292                k >>= 1;
293            }
294            let nacc = M::agg_operate(&self.seg[k], &acc);
295            if !pred(&nacc) {
296                while k < self.n {
297                    self.propagate_at(k);
298                    k = 2 * k + 1;
299                    let nacc = M::agg_operate(&self.seg[k], &acc);
300                    if pred(&nacc) {
301                        acc = nacc;
302                        k -= 1;
303                    }
304                }
305                return k + 1 - self.n;
306            }
307            acc = nacc;
308            if k.is_power_of_two() {
309                return 0;
310            }
311        }
312    }

332    crate::define_monoid!(
333        HistoryAction,
334        [i64; 4],
335        |x, y| {
336            let &[a, b, c, d] = x;
337            let &[e, f, g, h] = y;
338            [a * e, b * e + f, c + a * g, d + b * g + h]
339        },
340        [1, 0, 0, 0]
341    );

343    struct History;
344    impl LazyMapMonoid for History {
345        type Key = i64;
346        type AggMonoid = ArrayOperation<AdditiveOperation<i64>, 3>;
347        type ActMonoid = HistoryAction;
348
349        fn single_agg(key: &Self::Key) -> LazyMapAgg<Self> {
350            [1, *key, 0]
351        }
352
353        fn act_agg(x: &LazyMapAgg<Self>, act: &LazyMapAct<Self>) -> Option<LazyMapAgg<Self>> {
354            let &[n, current, history] = x;
355            let &[a, b, c, d] = act;
356            Some([n, a * current + b * n, history + c * current + d * n])
357        }
358    }

361    fn test_aggregate_only_action() {
362        let mut rng = Xorshift::default();
363        let inputs: Vec<_> = sequence_cases(&mut rng, -5i64..=5, 0..=65).collect();
364        for input in inputs {
365            let n = input.len();
366            let mut seg = LazySegmentTree::<History>::from_keys(input.iter().copied());
367            let mut values: Vec<_> = input.into_iter().map(|x| (x, 0)).collect();
368            for step in 0..64 {
369                let a = rng.random(0..=n);
370                let b = rng.random(0..=n);
371                let (l, r) = if step < 2 {
372                    (0, n)
373                } else {
374                    (a.min(b), a.max(b))
375                };
376                if step % 2 == 0 {
377                    let value = rng.random(-5..=5);
378                    seg.update(l..r, [0, value, 0, 0]);
379                    for x in &mut values[l..r] {
380                        x.0 = value;
381                    }
382                } else {
383                    seg.update(l..r, [1, 0, 1, 0]);
384                    for x in &mut values[l..r] {
385                        x.1 += x.0;
386                    }
387                }
388                let expected = [
389                    (r - l) as i64,
390                    values[l..r].iter().map(|x| x.0).sum(),
391                    values[l..r].iter().map(|x| x.1).sum(),
392                ];
393                assert_eq!(seg.fold(l..r), expected);
394                if n != 0 {
395                    let i = rng.random(0..n);
396                    assert_eq!(seg.get(i), [1, values[i].0, values[i].1]);
397                }
398            }
399        }

402    #[test]
403    fn test_lazy_segment_tree() {
404        let mut rng = Xorshift::default();
405        // Range Sum Query & Range Add Query
406        rand!(rng, mut arr: [-A..A; N]);
407        let mut seg =
408            LazySegmentTree::<RangeSumRangeAdd<_>>::from_vec(arr.iter().map(|&a| (a, 1)).collect());
409        for _ in 0..Q {
410            rand!(rng, (l, r): NotEmptySegment(N));
411            match rng.rand(3) {
412                0 => {
413                    // Range Add Query
414                    rand!(rng, x: -A..A);
415                    seg.update(l..r, x);
416                    for a in arr[l..r].iter_mut() {
417                        *a += x;
418                    }
419                }
420                1 => {
421                    // Point Set Query
422                    rand!(rng, k: 0..N, x: -A..A);
423                    seg.set(k, (x, 1));
424                    arr[k] = x;
425                }
426                _ => {
427                    // Range Sum Query
428                    let res = arr[l..r].iter().sum();
429                    assert_eq!(seg.fold(l..r).0, res);
430                }
431            }
432            rand!(rng, k: 0..N);
433            assert_eq!(seg.get(k).0, arr[k]);
434            assert_eq!(seg.fold_all().0, arr.iter().sum());
435        }
436
437        // Range Max Query & Range Update Query & Binary Search Query
438        rand!(rng, mut arr: [-A..A; N]);
439        let mut seg = LazySegmentTree::<RangeMaxRangeUpdate<_>>::from_vec(arr.clone());
440        for _ in 0..Q {
441            rand!(rng, ty: 0..5, (l, r): NotEmptySegment(N));
442            match ty {
443                0 => {
444                    // Range Update Query
445                    rand!(rng, x: -A..A);
446                    seg.update(l..r, Some(x));
447                    arr[l..r].iter_mut().for_each(|a| *a = x);
448                }
449                1 => {
450                    // Range Max Query
451                    let res = arr[l..r].iter().max().cloned().unwrap_or_default();
452                    assert_eq!(seg.fold(l..r), res);
453                }
454                2 => {
455                    // Binary Search Query
456                    rand!(rng, left: ..=N, x: -A..A);
457                    assert_eq!(
458                        seg.partition_point_acc(left, |&d| d < x),
459                        arr[left..]
460                            .iter()
461                            .scan(i64::MIN, |acc, &a| {
462                                *acc = a.max(*acc);
463                                Some(*acc)
464                            })
465                            .position(|acc| acc >= x)
466                            .map_or(N, |i| i + left),
467                    );
468                }
469                3 => {
470                    // Binary Search Query
471                    rand!(rng, right: ..=N, x: -A..A);
472                    assert_eq!(
473                        seg.rpartition_point_acc(right, |&d| d < x),
474                        arr[..right]
475                            .iter()
476                            .rev()
477                            .scan(i64::MIN, |acc, &a| {
478                                *acc = a.max(*acc);
479                                Some(*acc)
480                            })
481                            .position(|acc| acc >= x)
482                            .map_or(0, |i| right - i),
483                    );
484                }
485                _ => {
486                    // Point Set Query
487                    rand!(rng, k: 0..N, x: -A..A);
488                    seg.set(k, x);
489                    arr[k] = x;
490                }
491            }
492            rand!(rng, k: 0..N);
493            assert_eq!(seg.get(k), arr[k]);
494            assert_eq!(seg.fold_all(), *arr.iter().max().unwrap());
495        }
496
497        // Range Sum Query & Range Chmin/Chmax/Add Query
498        let mut arr = rng
499            .random_iter(-1_000..=1_000)
500            .map(Saturating)
501            .take(N)
502            .collect::<Vec<_>>();
503        let mut seg =
504            LazySegmentTree::<RangeSumRangeChminChmaxAdd<_>>::from_keys(arr.iter().copied());
505        for _ in 0..Q {
506            rand!(rng, ty: 0..4, (l, r): NotEmptySegment(N), x: -1_000..=1_000);
507            let x = Saturating(x);
508            match ty {
509                0 => {
510                    seg.update(l..r, RangeChminChmaxAdd::chmin(x));
511                    arr[l..r].iter_mut().for_each(|a| *a = (*a).min(x));
512                }
513                1 => {
514                    seg.update(l..r, RangeChminChmaxAdd::chmax(x));
515                    arr[l..r].iter_mut().for_each(|a| *a = (*a).max(x));
516                }
517                2 => {
518                    seg.update(l..r, RangeChminChmaxAdd::add(x));
519                    arr[l..r].iter_mut().for_each(|a| *a += x);
520                }
521                _ => assert_eq!(
522                    seg.fold(l..r).sum,
523                    arr[l..r].iter().copied().sum::<Saturating<i64>>()
524                ),
525            }
526        }
