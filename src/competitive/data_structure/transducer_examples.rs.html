14/// Counts and sums integers in `[lower, upper]` whose base-`radix` digit sum is `k`.
15/// Requires `2 <= radix <= 10` and `lower <= upper`. Zero has digit sum 0.
16///
17/// ```
18/// use competitive::data_structure::transducer_examples::digit_sum_count_and_sum;
19/// assert_eq!(digit_sum_count_and_sum(2, 2, 0, 31), (10, 124));
20/// ```
21pub fn digit_sum_count_and_sum(radix: usize, k: usize, lower: usize, upper: usize) -> (u128, u128) {
22    let upper = upper.to_digit_sequence_radix(radix);
23    let lower = lower.to_digit_sequence_radix_len(radix, upper.len());
24    let machine = transducer!(=> || 0usize, |&sum, &digit| {
25        (sum + digit <= k).then_some((sum + digit, digit))
26    }, |&sum| sum == k)
27    .intersection(transducer!((lower <=) & (<= upper)));
28    let mut dp =
29        machine.dp_with::<(AdditiveOperation<u128>, AdditiveOperation<u128>), _>((1, 0), || {
30            VecMap::with_len(4 * (k + 1), |&(sum, ((_, lower), (_, upper)))| {
31                4 * sum + 2 * usize::from(lower) + usize::from(upper)
32            })
33        });
34    dp.run_effect(
35        upper.len(),
36        |_| 0..radix,
37        |&(count, sum), (digit, _)| (count, radix as u128 * sum + count * digit as u128),
38    )
39}

41/// Counts positive multiples of `divisor` up to `upper` that contain `required`
42/// and use at most `max_distinct` decimal digits. Leading padding zeros are ignored.
43/// Requires `1 <= divisor <= 100`, `max_distinct <= 10`, and `required < 10`.
44///
45/// ```
46/// use competitive::data_structure::transducer_examples::count_multiples_with_digits;
47/// assert_eq!(count_multiples_with_digits(100, 7, 2, 3), 2); // 35 and 63
48/// ```
49pub fn count_multiples_with_digits(
50    upper: usize,
51    divisor: usize,
52    max_distinct: u32,
53    required: usize,
54) -> u64 {
55    let upper = upper.to_digit_sequence();
56    let machine = transducer!(=> || (false, 0usize, 0usize),
57    |&(started, remainder, mask), &digit| {
58        let started = started || digit != 0;
59        let mask = if started { mask | 1usize << digit } else { mask };
60        Some(((started, (10 * remainder + digit) % divisor, mask), ()))
61    },
62    |&(started, remainder, mask)| {
63        started && remainder == 0 && mask.count_ones() <= max_distinct && mask & (1 << required) != 0
64    })
65    .intersection(transducer!(<= upper));
66    machine
67        .counting_with(FibHashMap::default)
68        .run(upper.len(), |_| 0..10)
69}

71/// Counts words with digit sum `k` and selects a word for each `(lower, upper, rank)` query.
72/// Bounds contain base-`radix` digits, including leading zeros, with equal lengths at most `len`.
73/// Requires `2 <= radix <= 10`.
74/// Counts saturate at `u128::MAX`; the zero-based `rank` must be smaller. Missing ranks return `None`.
75/// Precomputation takes O(len * (k + 1) * radix) time and O(len * (k + 1)) space.
76///
77/// ```
78/// use competitive::data_structure::transducer_examples::digit_sum_queries;
79/// let lower = [0, 0, 1, 0, 0];
80/// let upper = [0, 1, 1, 1, 1];
81/// let answers = digit_sum_queries(2, 2, 5, &[(&lower, &upper, 2)]);
82/// assert_eq!(answers, [(5, Some(vec![0, 1, 0, 0, 1]))]);
83/// ```
84pub fn digit_sum_queries(
85    radix: usize,
86    k: usize,
87    len: usize,
88    queries: &[(&[usize], &[usize], u128)],
89) -> Vec<(u128, Option<Vec<usize>>)> {
90    let machine = transducer!(=> || 0usize, |&sum, &digit| {
91        (sum + digit <= k).then_some((sum + digit, ()))
92    }, |&sum| sum == k);
93    let table = TransitionTable::new(&machine, radix, k + 1);
94    let counts = table.suffix_counts::<Saturating<u128>>(len);
95    queries
96        .iter()
97        .map(|&(lower, upper, rank)| {
98            (
99                counts.count_between(lower, upper).0,
100                counts.kth_between(lower, upper, rank),
101            )
102        })
103        .collect()
104}

106/// Counts length-`len` base-`radix` words whose value is `remainder` modulo `divisor`.
107/// Includes leading zeros and returns the count modulo 998244353.
108/// Requires `2 <= radix <= 10`, `1 <= divisor <= 100`, and `remainder < divisor`.
109/// Takes O(divisor^3 log len) time.
110///
111/// ```
112/// use competitive::data_structure::transducer_examples::count_words_with_remainder;
113/// assert_eq!(count_words_with_remainder(2, 3, 0, 3), 3.into()); // 000, 011, 110
114/// ```
115pub fn count_words_with_remainder(radix: usize, divisor: usize, remainder: usize, len: usize) -> M {
116    let machine = transducer!(=> || 0usize, |&state, &digit| {
117        Some(((radix * state + digit) % divisor, ()))
118    }, |&state| state == remainder);
119    let table = TransitionTable::new(&machine, radix, divisor);
120    let matrix = table
121        .to_matrix::<AddMulOperation<M>>(|_| M::from(1))
122        .pow(len);
123    (0..table.state_size())
124        .filter(|&state| table.is_accepting(state))
125        .map(|state| matrix[(0, state)])
126        .sum()
127}
