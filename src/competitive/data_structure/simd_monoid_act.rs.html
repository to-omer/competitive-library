10    pub trait SimdMonoidAct: MonoidAct<Key: Copy, Act: Copy> {
11        /// Applies a broadcast tag to every key. Tags compose in old-then-later order.
12        ///
13        /// # Safety
14        /// The ISA contract is the same as for [`SimdMonoid::operate_lanes`].
15        /// The slice may be unaligned or have any length.
16        unsafe fn act_lanes<const ISA: u8>(keys: &mut [Self::Key], tag: Self::Act);
17
18        /// Appends a broadcast tag to each pending tag.
19        ///
20        /// # Safety
21        /// The ISA and slice contract is the same as for [`Self::act_lanes`].
22        unsafe fn compose_lanes<const ISA: u8>(tags: &mut [Self::Act], later: Self::Act) {
23            for tag in tags {
24                Self::operate_assign(tag, &later);
25            }
26        }
27    }
28
29    /// Acting on an ordered fold must equal acting on each key, lifting, and folding.
30    pub trait SimdAggregateAct<M>: SimdMonoidAct
31    where
32        M: SimdMonoid,
33    {
34        /// Enables aggregate queries to check for tags that replace the old fold.
35        const HAS_CONSTANT_TAGS: bool = false;
36
37        fn lift(key: Self::Key) -> M::T;
38
39        fn act_aggregate(value: M::T, tag: Self::Act, len: usize) -> M::T;
40
41        /// Returns the mapped aggregate if it depends only on the tag and length.
42        fn constant_aggregate(_tag: Self::Act, _len: usize) -> Option<M::T> {
43            None
44        }
45
46        /// Applies a broadcast tag to aggregates which all cover `len` keys.
47        ///
48        /// # Safety
49        /// The ISA and slice contract is the same as for [`Self::act_lanes`].
50        unsafe fn act_aggregate_lanes<const ISA: u8>(
51            values: &mut [M::T],
52            tag: Self::Act,
53            len: usize,
54        ) {
55            for value in values {
56                *value = Self::act_aggregate(*value, tag, len);
57            }
58        }
59    }

61    impl<T> SimdMonoidAct for UpdateAct<T>
62    where
63        T: Copy,
64    {
65        #[inline]
66        unsafe fn act_lanes<const ISA: u8>(keys: &mut [T], tag: Option<T>) {
67            if let Some(key) = tag {
68                keys.fill(key);
69            }
70        }
71
72        #[inline]
73        unsafe fn compose_lanes<const ISA: u8>(tags: &mut [Option<T>], later: Option<T>) {
74            if later.is_some() {
75                tags.fill(later);
76            }
77        }
78    }

80    impl<M> SimdAggregateAct<M> for UpdateAct<M::T>
81    where
82        M: SimdMonoid,
83    {
84        const HAS_CONSTANT_TAGS: bool = true;
85
86        #[inline]
87        fn lift(key: M::T) -> M::T {
88            key
89        }
90
91        #[inline]
92        fn act_aggregate(value: M::T, tag: Option<M::T>, len: usize) -> M::T {
93            <Self as SimdAggregateAct<M>>::constant_aggregate(tag, len).unwrap_or(value)
94        }
95
96        #[inline]
97        fn constant_aggregate(tag: Option<M::T>, len: usize) -> Option<M::T> {
98            tag.map(|key| M::pow(key, len))
99        }
100
101        #[inline]
102        unsafe fn act_aggregate_lanes<const ISA: u8>(
103            values: &mut [M::T],
104            tag: Option<M::T>,
105            len: usize,
106        ) {
107            if !values.is_empty()
108                && let Some(repeated) = <Self as SimdAggregateAct<M>>::constant_aggregate(tag, len)
109            {
110                values.fill(repeated);
111            }
112        }
113    }

169        fn test_shared_point_and_aggregate_action() {
170            let mut rng = Xorshift::default();
171            for len in [0_usize, 1, 15, 16, 17, 31, 255, 256, 257, 273, 4097] {
172                let mut keys: Vec<_> = (0..len).map(|_| Key(rng.rand64() as i64)).collect();
173                let mut dual = DaryDualSegmentTree::<Affine>::from_keys(keys.iter().copied());
174                let mut lazy =
175                    DaryLazySegmentTree::<(SimdAddI64, Affine)>::from_keys(keys.iter().copied());
176                for step in 0..384 {
177                    let x = rng.rand((len + 1) as u64) as usize;
178                    let y = rng.rand((len + 1) as u64) as usize;
179                    let range = match step % 8 {
180                        0 | 1 => 0..len,
181                        2 => len.saturating_sub(1)..len,
182                        3 => len / 16 * 16..len,
183                        _ => x.min(y)..x.max(y),
184                    };
185                    let (a, b) = match step % 8 {
186                        0 => (0, 5),
187                        1 => (2, 1),
188                        _ => (rng.rand64() as i64, rng.rand64() as i64),
189                    };
190                    dual.update(range.clone(), (Wrapping(a), Wrapping(b)));
191                    lazy.update(range.clone(), (Wrapping(a), Wrapping(b)));
192                    for key in &mut keys[range.clone()] {
193                        key.0 = a.wrapping_mul(key.0).wrapping_add(b);
194                    }
195                    if len != 0 {
196                        let i = if step % 2 == 0 {
197                            len - 1
198                        } else {
199                            rng.rand(len as u64) as usize
200                        };
201                        if step % 4 <= 1 {
202                            let key = Key(rng.rand64() as i64);
203                            keys[i] = key;
204                            dual.set(i, key);
205                            lazy.set(i, key);
206                        }
207                        assert_eq!(dual.get(i), keys[i]);
208                        assert_eq!(lazy.get(i), keys[i]);
209                        for j in [0, len / 2, len - 1] {
210                            assert_eq!(dual.get(j), keys[j]);
211                            assert_eq!(lazy.get(j), keys[j]);
212                        }
213                    }
214                    assert_eq!(

216                        keys[range]
217                            .iter()
218                            .fold(0_i64, |acc, key| acc.wrapping_add(key.0))
219                    );
220                    assert_eq!(
221                        lazy.fold_all().0,
222                        keys.iter().fold(0_i64, |acc, key| acc.wrapping_add(key.0))

224                    let query = x.min(y)..x.max(y);
225                    assert_eq!(
226                        lazy.fold(query.clone()).0,
227                        keys[query]
228                            .iter()
229                            .fold(0_i64, |acc, key| acc.wrapping_add(key.0))

236        fn test_shared_ordered_product_and_assignment() {
237            type M = LinearOperation<Wrapping<i32>>;
238            let mut functions: Vec<_> = (0..257).map(|i| (Wrapping(3), Wrapping(i))).collect();
239            let mut lazy =
240                DaryLazySegmentTree::<(M, UpdateAct<_>)>::from_keys(functions.iter().copied());
241            for (range, tag) in [(14..256, (2, 1)), (255..257, (0, -3)), (16..32, (1, 4))] {
242                let tag = (Wrapping(tag.0), Wrapping(tag.1));
243                lazy.update(range.clone(), Some(tag));
244                functions[range].fill(tag);
245                let seg = DarySegmentTree::<M>::from_keys(functions.iter().copied());
246                for query in [0..0, 0..257, 11..38, 254..257, 256..257] {
247                    let (a, b) = lazy.fold(query.clone());
248                    assert_eq!((a, b), seg.fold(query.clone()));
249                    for x in [Wrapping(0), Wrapping(1)] {
250                        let expected = functions[query.clone()]
251                            .iter()
252                            .fold(x, |v, &(a, b)| a * v + b);
253                        assert_eq!(a * x + b, expected);
254                    }
255                }

267    #[inline]
268    pub fn apply_selected<T, F>(
269        lanes: &mut [T],
270        range: Range<usize>,
271        outside: F,
272        inside: F,
273        mut apply: impl FnMut(&mut [T], F),
274    ) where
275        F: Copy,
276    {
277        if range.is_empty() {
278            apply(lanes, outside);
279        } else {
280            let (before, rest) = lanes.split_at_mut(range.start);
281            let (selected, after) = rest.split_at_mut(range.len());
282            apply(before, outside);
283            apply(selected, inside);
284            apply(after, outside);
285        }
286    }

288    pub fn leaf<T>(mut keys: impl Iterator<Item = T>) -> [T; LANES]
289    where
290        T: Copy,
291    {
292        let first = keys.next().expect("missing leaf key");
293        // Only the supplied lanes participate in queries and updates.
294        let mut block = [first; LANES];
295        for (lane, key) in keys.enumerate() {
296            block[lane + 1] = key;
297        }
298        block
299    }

301    /// # Safety
302    /// ISA must satisfy the contract of `A::act_lanes`. The slice and range contain only valid keys.
303    pub unsafe fn edit_keys<A, const ISA: u8>(
304        keys: &mut [A::Key],
305        range: Range<usize>,
306        tag: A::Act,
307        carry: A::Act,
308    ) where
309        A: SimdMonoidAct,
310    {
311        if range.is_empty() {
312            unsafe { A::act_lanes::<ISA>(keys, carry) };
313        } else {
314            let combined = A::operate(&carry, &tag);
315            apply_selected(keys, range, carry, combined, |keys, tag| unsafe {
316                A::act_lanes::<ISA>(keys, tag);
317            });
318        }
319    }
