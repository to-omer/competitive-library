3/// Unites corresponding vertices of two equal-length ranges in an indexed set.
4/// Uses O(n log n) space. Excluding callbacks, q range unions take O((n log n + q) α(n)) time.
5#[derive(Clone, Debug)]
6pub struct RangeParallelUnionFind {
7    n: usize,
8    vertices: UnionFind,
9    blocks: Vec<UnionFind>,
10    periods: Vec<PeriodicRange>,
11    recent: usize,
12}
13
14#[derive(Clone, Copy, Debug)]
15struct PeriodicRange {
16    start: usize,
17    end: usize,
18    period: usize,
19}
20
21impl RangeParallelUnionFind {
22    /// Creates `n` singleton components, with `n <= i32::MAX`.
23    pub fn new(n: usize) -> Self {
24        assert!(n <= i32::MAX as usize);
25        let blocks = std::iter::successors(Some(4usize), |w| w.checked_mul(4))
26            .take_while(|&w| w < n)
27            .map(|w| UnionFind::new(n - w + 1))
28            .collect();
29        Self {
30            n,
31            vertices: UnionFind::new(n),
32            blocks,
33            periods: (0..n)
34                .step_by(4)
35                .map(|mid| PeriodicRange {
36                    start: mid,
37                    end: mid + 1,
38                    period: 1,
39                })
40                .collect(),
41            recent: 0,
42        }
43    }
44
45    pub fn find_root(&mut self, x: usize) -> usize {
46        self.vertices.find_root(x)
47    }
48
49    pub fn same(&mut self, x: usize, y: usize) -> bool {
50        self.vertices.same(x, y)
51    }
52
53    /// Unites `a+i` with `b+i` for `0 <= i < len`, in increasing offset order.
54    /// Calls `merge(surviving_root, absorbed_root)` once per successful vertex union.
55    pub fn unite(&mut self, a: usize, b: usize, len: usize, mut merge: impl FnMut(usize, usize)) {
56        let (first, second) = (a.min(b), a.max(b));
57        assert!(len <= self.n && second <= self.n - len);
58        if len == 0 || a == b {
59            return;
60        }
61        if len < 4 {
62            for i in 0..len {
63                self.unite_block(a + i, b + i, 0, &mut merge);
64            }
65            return;
66        }
67        let distance = second - first;
68        let end = second + len;
69        let recent = self.periods[self.recent];
70        if first >= recent.start && end <= recent.end && distance.is_multiple_of(recent.period) {
71            return;
72        }
73
74        // The enclosing dyadic interval has width at least eight, so its midpoint is divisible by four.
75        let shift = (first ^ (end - 1)).ilog2();
76        let half = 1usize << shift;
77        let mid = (end - 1) >> shift << shift;
78        let left = mid - half;
79        let right = (mid + half).min(self.n);
80        let slot = mid / 4;
81        let mut range = self.periods[slot];
82        let remainder = distance % range.period;
83        let aligned = remainder == 0;
84
85        // Inside the certified interval each vertex is connected to its period-shifted copy.
86        let prefix = range.start.saturating_sub(first).min(len);
87        let suffix = range.end.saturating_sub(second).min(len);
88        if aligned && prefix == 0 && suffix == len {
89            self.recent = slot;
90            return;
91        }
92        if prefix < suffix {
93            let residues = if aligned {
94                0
95            } else {
96                range.period.min(suffix - prefix)
97            };
98            self.unite_blocks(a, b, prefix + residues, &mut merge);
99            self.unite_blocks(a + suffix, b + suffix, len - suffix, &mut merge);
100        } else {
101            self.unite_blocks(a, b, len, &mut merge);
102        }
103        // A full cycle of residue pairs in the overlap identifies residues modulo the gcd.
104        let overlap = range.end.min(end) - range.start.max(first);
105        if overlap >= distance && (aligned || overlap - distance >= range.period) {
106            if !aligned {
107                range.period = gcd(range.period as u64, remainder as u64) as usize;
108            }
109            range.start = range.start.min(first);
110            range.end = range.end.max(end);
111        } else if first <= range.start && range.end <= end && distance <= len {
112            range = PeriodicRange {
113                start: first,
114                end,
115                period: distance,
116            };
117        }
118        while range.start > left
119            && self
120                .vertices
121                .same(range.start - 1, range.start - 1 + range.period)
122        {
123            range.start -= 1;
124        }
125        while range.end < right && self.vertices.same(range.end - range.period, range.end) {
126            range.end += 1;
127        }
128        self.periods[slot] = range;
129        self.recent = slot;
130    }
131
132    fn unite_blocks(
133        &mut self,
134        a: usize,
135        b: usize,
136        len: usize,
137        merge: &mut impl FnMut(usize, usize),
138    ) {
139        if len == 0 {
140            return;
141        }
142        let shift = len.ilog2() & !1;
143        let width = 1 << shift;
144        let mut offset = 0;
145        while offset < len - width {
146            self.unite_block(a + offset, b + offset, shift, merge);
147            offset += width;
148        }
149        self.unite_block(a + len - width, b + len - width, shift, merge);
150    }
151
152    fn unite_block(
153        &mut self,
154        a: usize,
155        b: usize,
156        shift: u32,
157        merge: &mut impl FnMut(usize, usize),
158    ) {
159        if shift == 0 {
160            let x = self.vertices.find_root(a);
161            let y = self.vertices.find_root(b);
162            if x != y {
163                self.vertices.unite(x, y);
164                if self.vertices.find_root(x) == x {
165                    merge(x, y);
166                } else {
167                    merge(y, x);
168                }
169            }
170        } else {
171            let level = (shift / 2 - 1) as usize;
172            let x = self.blocks[level].find_root(a);
173            let y = self.blocks[level].find_root(b);
174            if x != y {
175                for i in 0..4 {
176                    let offset = i << (shift - 2);
177                    self.unite_block(a + offset, b + offset, shift - 2, merge);
178                }
179                self.blocks[level].unite(x, y);
180            }
181        }
182    }
183}

190    fn check_queries(n: usize, queries: impl IntoIterator<Item = (usize, usize, usize)>) {
191        let mut uf = RangeParallelUnionFind::new(n);
192        let mut naive = UnionFind::new(n);
193        let mut members: Vec<_> = (0..n).map(|v| 1u128 << v).collect();
194        let mut expected = vec![0u128; n];
195        let mut events = Vec::new();
196        for (a, b, len) in queries {
197            events.clear();
198            for i in 0..len {
199                let x = naive.find_root(a + i);
200                let y = naive.find_root(b + i);
201                if naive.unite(x, y) {
202                    if naive.find_root(x) == x {
203                        events.push((x, y));
204                    } else {
205                        events.push((y, x));
206                    }
207                }
208            }
209            let mut events = events.iter();
210            uf.unite(a, b, len, |to, from| {
211                assert_eq!(events.next(), Some(&(to, from)));
212                let absorbed = std::mem::take(&mut members[from]);
213                assert_ne!(absorbed, 0);
214                members[to] |= absorbed;
215            });
216            assert_eq!(events.next(), None);
217            expected.fill(0);
218            for v in 0..n {
219                expected[naive.find_root(v)] |= 1u128 << v;
220            }
221            assert_eq!(members, expected);
222            for v in 0..n {
223                let root = uf.find_root(v);
224                assert_eq!(root, naive.find_root(v));
225                let other = n - 1 - v;
226                assert_eq!(uf.same(v, other), naive.same(v, other));
227            }
228        }
229    }

231    #[test]
232    fn test_range_parallel_union_find() {
233        let mut rng = Xorshift::default();
234        check_queries(0, [(0, 0, 0)]);
235        for n in 1..=128 {
236            for period in [1, 2, 4, 8, 16, 32, 64] {
237                if period < n {
238                    let len = n - period;
239                    let mut queries =
240                        vec![(n, 0, 0), (0, 0, n), (0, period, len), (period, 0, len)];
241                    let mut step = period;
242                    while step > 1 {
243                        step /= 2;
244                        queries.push((step, 0, n - step));
245                    }
246                    check_queries(n, queries);
247                }
248            }
249            for _ in 0..45 {
250                let queries = (0..48).map(|i| {
251                    rand!(rng, exp: 0..=n.ilog2(), delta: 0..3usize);
252                    let power = 1usize << exp;
253                    rand!(rng, len: 0..=n);
254                    let len = match i % 3 {
255                        0 => (power + delta - 1).min(n),
256                        1 => len.min(4),
257                        _ => len,
258                    };
259                    rand!(rng, a: 0..=n - len, b: 0..=n - len);
260                    (a, b, len)
261                });
262                check_queries(n, queries);
263            }
264        }
