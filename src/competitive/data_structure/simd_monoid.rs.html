3/// A monoid with a lane-wise implementation that can be shared by SIMD trees.
4///
5/// Implement this trait to use the operation directly in a d-ary segment tree.
6/// Commutative operations can also be used in a SIMD binary indexed tree.
7pub trait SimdMonoid: Monoid<T: Copy> {
8    /// Replaces each left operand with its ordered product with the right operand.
9    ///
10    /// # Safety
11    /// The slices have equal lengths. ISA 0 requests portable execution, ISA 1
12    /// guarantees x86-64 AVX2, and ISA 2 guarantees x86-64 AVX-512F/DQ/CD/BW/VL.
13    /// Implementations support unaligned slices of any length, including tails,
14    /// and check any additional CPU requirements themselves.
15    unsafe fn operate_lanes<const ISA: u8>(lhs: &mut [Self::T], rhs: &[Self::T]);
16
17    /// Appends the same right operand to every lane from `start` onward.
18    ///
19    /// # Safety
20    /// The ISA and alignment contracts are the same as for [`Self::operate_lanes`].
21    /// `start` must be at most `values.len()`.
22    #[inline]
23    unsafe fn broadcast_suffix<const ISA: u8>(values: &mut [Self::T], start: usize, rhs: Self::T) {
24        let operands = [rhs; LANES];
25        for chunk in values[start..].chunks_mut(LANES) {
26            unsafe { Self::operate_lanes::<ISA>(chunk, &operands[..chunk.len()]) };
27        }
28    }
29
30    /// Folds the slice in order, returning the unit for an empty slice.
31    ///
32    /// # Safety
33    /// The ISA contract is the same as for [`Self::operate_lanes`].
34    /// The slice has at most 16 values.
35    #[inline]
36    unsafe fn reduce_lanes<const ISA: u8>(values: &[Self::T]) -> Self::T {
37        if ISA == 0 || values.len() <= 1 {
38            return Self::fold(values.iter().copied());
39        }
40
41        let mut left = [Self::unit(); LANES / 2];
42        let mut right = left;
43        let (input, rest) = values.as_chunks::<2>();
44        let mut pairs = input.len();
45        for (i, pair) in input.iter().enumerate() {
46            left[i] = pair[0];
47            right[i] = pair[1];
48        }
49        unsafe { Self::operate_lanes::<ISA>(&mut left[..pairs], &right[..pairs]) };
50        // Odd operands are suffixes of the preceding pairs.
51        let mut suffix = rest.first().copied().unwrap_or_else(Self::unit);
52        while pairs > 1 {
53            if pairs & 1 != 0 {
54                suffix = Self::operate(&left[pairs - 1], &suffix);
55            }
56            pairs /= 2;
57            for i in 0..pairs {
58                right[i] = left[2 * i + 1];
59                left[i] = left[2 * i];
60            }
61            unsafe { Self::operate_lanes::<ISA>(&mut left[..pairs], &right[..pairs]) };
62        }
63        Self::operate(&left[0], &suffix)
64    }
65}

87    static PACKED_CALLS: AtomicUsize = AtomicUsize::new(0);

89    impl Magma for Xor {
90        type T = u64;
91
92        fn operate(left: &u64, right: &u64) -> u64 {
93            left ^ right
94        }
95    }
96    impl Associative for Xor {}
97    impl Commutative for Xor {}
98    impl Unital for Xor {
99        fn unit() -> u64 {
100            0
101        }
102    }
103    impl Invertible for Xor {
104        fn inverse(x: &u64) -> u64 {
105            *x
106        }
107    }
108    impl SimdMonoid for Xor {
109        unsafe fn operate_lanes<const ISA: u8>(lhs: &mut [u64], rhs: &[u64]) {
110            unsafe { BitXorOperation::<u64>::operate_lanes::<ISA>(lhs, rhs) };
111            if ISA != 0 {
112                PACKED_CALLS.fetch_add(1, Ordering::Relaxed);
113            }
114        }
115    }

117    #[test]
118    fn test_shared_point_operations() {
119        let packed = crate::tools::simd_backend() != crate::tools::SimdBackend::Scalar;
120        let mut rng = Xorshift::default();
121        for n in [0, 1, 15, 16, 17, 255, 256, 257, 1025] {
122            let mut keys: Vec<_> = (0..n).map(|_| rng.rand64()).collect();
123            let mut seg = DarySegmentTree::<Xor>::from_keys(keys.iter().copied());
124            let mut bit = SimdBinaryIndexedTree::<Xor>::from_keys(keys.iter().copied());
125            let mut extrema: Vec<_> = keys.iter().map(|&key| key as i32).collect();
126            let mut maximum =
127                SimdBinaryIndexedTree::<MaxOperation<i32>>::from_keys(extrema.iter().copied());
128            let mut minimum =
129                DarySegmentTree::<MinOperation<i32>>::from_keys(extrema.iter().copied());
130            let mut sums: Vec<_> = keys.iter().map(|&x| x as i64).collect();
131            let mut prefix = DarySegmentTree::<DaryPrefixGroup<SimdAddI64>>::from_keys(
132                sums.iter().copied().map(Wrapping),
133            );
134            for step in 0..256 {
135                let left = rng.rand((n + 1) as u64) as usize;
136                let right = rng.rand((n + 1) as u64) as usize;
137                let (l, r) = if step % 8 == 0 {
138                    (0, n)
139                } else {
140                    (left.min(right), left.max(right))
141                };
142                let xor = keys[l..r].iter().fold(0, |acc, x| acc ^ x);
143                let sum = sums[l..r].iter().fold(0_i64, |acc, &x| acc.wrapping_add(x));
144                assert_eq!(seg.fold(l..r), xor);
145                assert_eq!(seg.fold_by_prefix(l..r), xor);
146                assert_eq!(bit.fold(l, r), xor);
147                assert_eq!(
148                    maximum.accumulate0(r),
149                    extrema[..r].iter().copied().max().unwrap_or(i32::MIN)
150                );
151                assert_eq!(
152                    minimum.fold(l..r),
153                    extrema[l..r].iter().copied().min().unwrap_or(i32::MAX)
154                );
155                assert_eq!(prefix.fold(l..r), Wrapping(sum));
156                assert_eq!(prefix.fold_by_prefix(l..r), Wrapping(sum));
157                if n == 0 {
158                    continue;
159                }
160                let i = if step % 8 <= 1 {
161                    n - 1
162                } else {
163                    rng.rand(n as u64) as usize
164                };
165                let x = rng.rand64();
166                if step % 2 == 0 {
167                    keys[i] ^= x;
168                    sums[i] = sums[i].wrapping_add(x as i64);
169                    seg.update(i, x);
170                    bit.update(i, x);
171                    prefix.update(i, Wrapping(x as i64));
172                } else {
173                    keys[i] = x;
174                    sums[i] = x as i64;
175                    seg.set(i, x);
176                    bit.set(i, x);
177                    prefix.set(i, Wrapping(x as i64));
178                }
179                maximum.update(i, x as i32);
180                extrema[i] = extrema[i].max(x as i32);
181                minimum.set(i, extrema[i]);
182                assert_eq!(bit.get(i), keys[i]);
183                assert_eq!(seg.get(i), keys[i]);
184                assert_eq!(prefix.get(i), Wrapping(sums[i]));
185            }
186        }
187
188        let mut maximum = SimdBinaryIndexedTree::<MaxOperation<i32>>::from_keys(0..257);
189        maximum.update(256, i32::MAX);
190        assert_eq!(maximum.accumulate0(256), 255);
191        assert_eq!(maximum.accumulate0(257), i32::MAX);
192
193        if packed {
194            // Check each public structure separately. Scalar results alone would not detect a bypass.
195            let mut seg = DarySegmentTree::<Xor>::from_keys((0..17).map(|i| i as u64));
196            let before = PACKED_CALLS.load(Ordering::Relaxed);
197            seg.set(0, 100);
198            assert!(PACKED_CALLS.load(Ordering::Relaxed) > before);
199            let mut bit = SimdBinaryIndexedTree::<Xor>::from_keys((0..17).map(|i| i as u64));
200            let before = PACKED_CALLS.load(Ordering::Relaxed);
201            bit.update(0, 100);
202            assert!(PACKED_CALLS.load(Ordering::Relaxed) > before);
203        }
204    }
