3    data: Vec<Usize>,
4    index: Vec<Usize>,
5}

7macro_rules! impl_sparse_set {
8    ($($ty:ty)*) => {
9        $(
10            impl<const FIXED: bool> SparseSet<$ty, FIXED> {
11                pub fn new(n: usize) -> Self {
12                    Self {
13                        data: Vec::with_capacity(n),
14                        index: vec![!0; n],
15                    }
16                }
17
18                pub unsafe fn insert_unchecked(&mut self, x: $ty) {
19                    if !FIXED && x as usize >= self.index.len() {
20                        self.index.resize(x as usize + 1, !0);
21                    }
22                    self.index[x as usize] = self.data.len() as _;
23                    self.data.push(x);
24                }
25
26                pub fn insert(&mut self, x: $ty) -> bool {
27                    if self.contains(x) {
28                        return false;
29                    }
30                    unsafe { self.insert_unchecked(x) };
31                    true
32                }
33
34                pub unsafe fn remove_unchecked(&mut self, x: $ty) {
35                    let n = self.data.len();
36                    let k = std::mem::replace(&mut self.index[x as usize], !0);
37                    if k as usize != n - 1 {
38                        self.data.swap(k as usize, n - 1);
39                        self.index[self.data[k as usize] as usize] = k;
40                    }
41                    self.data.pop();
42                }
43
44                pub fn remove(&mut self, x: $ty) -> bool {
45                    if !self.contains(x) {
46                        return false;
47                    }
48                    unsafe { self.remove_unchecked(x) };
49                    true
50                }
51
52                pub fn len(&self) -> usize {
53                    self.data.len()
54                }
55
56                pub fn is_empty(&self) -> bool {
57                    self.len() == 0
58                }
59
60                pub fn contains(&self, x: $ty) -> bool {
61                    if !FIXED && x as usize >= self.index.len() {
62                        false
63                    } else {
64                        // The last position of a full key domain shares the absent marker.
65                        self.index[x as usize] != !0
66                            || self.data.get(<$ty>::MAX as usize) == Some(&x)
67                    }
68                }
69
70                pub fn iter(&self) -> std::slice::Iter<'_, $ty> {
71                    self.data.iter()
72                }
73            }
74        )*
75    };
76}

80mod verification {

83    #[kani::proof]
84    #[kani::unwind(6)]
85    fn proof_fixed_set() {
86        let order: [u8; 4] = kani::any();
87        let len: usize = kani::any();
88        let key: u8 = kani::any();
89        let insert: bool = kani::any();
90        kani::assume(len <= order.len() && key < 4);
91        let mut set = SparseSet::<u8>::new(4);
92        let mut expected = [false; 4];
93        for &value in &order[..len] {
94            kani::assume(value < 4 && !expected[value as usize]);
95            set.insert(value);
96            expected[value as usize] = true;
97        }
98        let present = expected[key as usize];
99        if insert {
100            kani::assert(
101                set.insert(key) != present,
102                "For every ordered set of keys 0 through 3, fixed SparseSet<u8> insertion reports whether the key was absent",
103            );
104            expected[key as usize] = true;
105        } else {
106            kani::assert(
107                set.remove(key) == present,
108                "For every ordered set of keys 0 through 3, fixed SparseSet<u8> removal reports whether the key was present",
109            );
110            expected[key as usize] = false;
111        }
112        let mut actual = [false; 4];
113        for &value in set.iter() {
114            kani::assert(
115                value < 4 && !actual[value as usize],
116                "After insertion or removal in the four-key domain, sparse-set iteration contains valid keys without duplicates",
117            );
118            actual[value as usize] = true;
119        }
120        kani::assert(
121            actual == expected && set.len() == expected.iter().filter(|&&x| x).count(),
122            "After insertion or removal in the four-key domain, sparse-set iteration and length equal the mathematical set",
123        );
124        for value in 0..4 {
125            kani::assert(
126                set.contains(value as u8) == expected[value],
127                "After insertion or removal in the four-key domain, every sparse-set membership query agrees with the mathematical set",
128            );
129        }
130        kani::cover!(len == 4 && !insert);
131        kani::cover!(len == 0 && insert && key == 3);
132    }
133
134    #[kani::proof]
135    #[kani::unwind(6)]
136    fn proof_growing_set() {
137        let a: u8 = kani::any();
138        let b: u8 = kani::any();
139        let query: u8 = kani::any();
140        kani::assume(a < 4 && b < 4 && query < 4);
141        let mut set = SparseSet::<u8, false>::new(0);
142        kani::assert(
143            set.insert(a),
144            "A growing SparseSet<u8> accepts every first key from 0 through 3 starting at zero capacity",
145        );
146        kani::assert(
147            set.contains(query) == (query == a),
148            "After one insertion into a growing sparse set, holes and keys beyond the allocated index are absent",
149        );
150        kani::assert(
151            set.insert(b) == (a != b) && set.len() == 1 + usize::from(a != b),
152            "For all two-key insertion sequences on keys 0 through 3, a growing sparse set suppresses duplicates and tracks the exact size",
153        );
154        kani::assert(
155            set.contains(query) == (query == a || query == b),
156            "After every two-key insertion sequence on keys 0 through 3, growing sparse-set membership is exactly the inserted set",
157        );
158        set.remove(a);
159        kani::assert(
160            set.contains(query) == (a != b && query == b) && set.len() == usize::from(a != b),
161            "Removing the first key after two insertions preserves exactly the distinct second key in a growing sparse set",
162        );
163        kani::cover!(a == 0 && b == 3);
164        kani::cover!(a == 3 && b == 0);
165        kani::cover!(a == b);
166    }

174    fn test_full_key_domain() {
175        for removed in 0..=u8::MAX {
176            let mut set = SparseSet::<u8>::new(256);
177            for key in 0..=u8::MAX {
178                assert!(set.insert(key));
179            }
180            assert_eq!(set.len(), 256);
181            for key in 0..=u8::MAX {
182                assert!(set.contains(key));
183                assert!(!set.insert(key));
184            }
185            assert!(set.remove(removed));
186            assert_eq!(set.len(), 255);
187            for key in 0..=u8::MAX {
188                assert_eq!(set.contains(key), key != removed);
189            }
190            assert!(set.insert(removed));
191            assert_eq!(set.len(), 256);
192            for key in 0..=u8::MAX {
193                assert!(set.contains(key));
194            }
195        }
