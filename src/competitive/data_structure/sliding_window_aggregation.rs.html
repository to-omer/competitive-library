48impl<M> QueueAggregation<M>
49where
50    M: Monoid,
51{
52    pub fn new() -> Self {
53        Self::default()
54    }
55    pub fn len(&self) -> usize {
56        self.front_stack.len() + self.back_stack.len()
57    }
58    pub fn is_empty(&self) -> bool {
59        self.front_stack.is_empty() && self.back_stack.is_empty()
60    }
61    pub fn fold_all(&self) -> M::T {
62        M::operate(
63            self.front_stack.last().map(|t| &t.0).unwrap_or(&M::unit()),
64            self.back_stack.last().map(|t| &t.0).unwrap_or(&M::unit()),
65        )
66    }
67    pub fn last(&self) -> Option<&M::T> {
68        self.back_stack
69            .last()
70            .or_else(|| self.front_stack.first())
71            .map(|t| &t.1)
72    }
73    pub fn push(&mut self, value: M::T) {
74        let x = M::operate(
75            self.back_stack.last().map(|t| &t.0).unwrap_or(&M::unit()),
76            &value,
77        );
78        self.back_stack.push((x, value));
79    }
80    fn push_front(&mut self, value: M::T) {
81        let x = M::operate(
82            &value,
83            self.front_stack.last().map(|t| &t.0).unwrap_or(&M::unit()),
84        );
85        self.front_stack.push((x, value));
86    }
87    pub fn pop(&mut self) -> Option<M::T> {
88        if self.front_stack.is_empty() {
89            let mut back_stack = std::mem::take(&mut self.back_stack);
90            for x in back_stack.drain(..).map(|t| t.1).rev() {
91                self.push_front(x);
92            }
93            self.back_stack = back_stack;
94        }
95        self.front_stack.pop().map(|t| t.1)
96    }
97}

143impl<M> DequeAggregation<M>
144where
145    M: Monoid,
146{
147    pub fn new() -> Self {
148        Self::default()
149    }
150    pub fn len(&self) -> usize {
151        self.front_stack.len() + self.back_stack.len()
152    }
153    pub fn is_empty(&self) -> bool {
154        self.front_stack.is_empty() && self.back_stack.is_empty()
155    }
156    pub fn fold_all(&self) -> M::T {
157        M::operate(
158            self.front_stack.last().map(|t| &t.0).unwrap_or(&M::unit()),
159            self.back_stack.last().map(|t| &t.0).unwrap_or(&M::unit()),
160        )
161    }
162    pub fn front(&self) -> Option<&M::T> {
163        self.front_stack
164            .last()
165            .or_else(|| self.back_stack.first())
166            .map(|t| &t.1)
167    }
168    pub fn back(&self) -> Option<&M::T> {
169        self.back_stack
170            .last()
171            .or_else(|| self.front_stack.first())
172            .map(|t| &t.1)
173    }
174    pub fn push_front(&mut self, value: M::T) {
175        let x = M::operate(
176            &value,
177            self.front_stack.last().map(|t| &t.0).unwrap_or(&M::unit()),
178        );
179        self.front_stack.push((x, value));
180    }
181    pub fn push_back(&mut self, value: M::T) {
182        let x = M::operate(
183            self.back_stack.last().map(|t| &t.0).unwrap_or(&M::unit()),
184            &value,
185        );
186        self.back_stack.push((x, value));
187    }
188    pub fn pop_front(&mut self) -> Option<M::T> {
189        if self.front_stack.is_empty() {
190            let n = self.back_stack.len();
191            let mut back_stack = std::mem::take(&mut self.back_stack);
192            for x in back_stack.drain(..n.div_ceil(2)).map(|t| t.1).rev() {
193                self.push_front(x);
194            }
195            for x in back_stack.drain(..).map(|t| t.1) {
196                self.push_back(x);
197            }
198        }
199        self.front_stack.pop().map(|t| t.1)
200    }
201    pub fn pop_back(&mut self) -> Option<M::T> {
202        if self.back_stack.is_empty() {
203            let n = self.front_stack.len();
204            let mut front_stack = std::mem::take(&mut self.front_stack);
205            for x in front_stack.drain(..n.div_ceil(2)).map(|t| t.1).rev() {
206                self.push_back(x);
207            }
208            for x in front_stack.drain(..).map(|t| t.1) {
209                self.push_front(x);
210            }
211        }
212        self.back_stack.pop().map(|t| t.1)
213    }
214}

218    use super::*;

221    #[kani::proof]
222    #[kani::unwind(6)]
223    fn proof_queue_transfer() {
224        let values: [(u8, u8); 4] = kani::any();
225        let pops: usize = kani::any();
226        let input: u8 = kani::any();
227        kani::assume(pops <= 3);
228        let mut queue = QueueAggregation::<LogicalLinearOperation<u8>>::new();
229        kani::assert(
230            queue.pop().is_none(),
231            "An empty QueueAggregation returns None",
232        );
233        for &value in &values[..3] {
234            queue.push(value);
235        }
236        for &value in &values[..pops] {
237            kani::assert(
238                queue.pop() == Some(value),
239                "For any three u8 bitwise affine maps and 0..=3 removals, QueueAggregation preserves FIFO order through transfer to the front stack",
240            );
241        }
242        queue.push(values[3]);
243        kani::cover!(!queue.front_stack.is_empty() && !queue.back_stack.is_empty());
244        kani::assert(
245            queue.pop() == Some(values[pops]),
246            "Appending a fourth arbitrary map preserves the oldest remaining value, including when the queue was emptied",
247        );
248        let mut output = input;
249        for &(mask, xor) in &values[pops + 1..] {
250            output = (output & mask) ^ xor;
251        }
252        kani::assert(
253            queue.len() == 3 - pops
254                && queue.is_empty() == (pops == 3)
255                && queue.last() == (pops < 3).then_some(&values[3])
256                && LogicalLinearOperation::eval(&queue.fold_all(), &input) == output,
257            "After the bounded transfer/refill sequence, queue length, last value and noncommutative aggregate match the remaining FIFO values",
258        );
259        kani::cover!(pops == 0);
260        kani::cover!(pops == 3);
261    }
262
263    macro_rules! proof_deque_rebalance {
264        ($name:ident, $len:expr, $push:ident, $pop:ident, $initial_front:expr) => {
265            #[kani::proof]
266            #[kani::unwind(6)]
267            fn $name() {
268                let mut values: [(u8, u8); $len] = kani::any();
269                let added: (u8, u8) = kani::any();
270                let push_front: bool = kani::any();
271                let input: u8 = kani::any();
272                let mut deque = DequeAggregation::<LogicalLinearOperation<u8>>::new();
273                kani::assert(
274                    deque.pop_front().is_none() && deque.pop_back().is_none(),
275                    "Both removal directions return None on an empty DequeAggregation",
276                );
277                for &value in &values {
278                    deque.$push(value);
279                }
280                if $initial_front {
281                    values.reverse();
282                }
283                let (left, right) = if $initial_front {
284                    (0, $len - 1)
285                } else {
286                    (1, $len)
287                };
288                kani::assert(
289                    deque.$pop() == Some(values[if $initial_front { $len - 1 } else { 0 }]),
290                    concat!(
291                        "For all ", stringify!($len),
292                        " u8 bitwise affine maps inserted with ", stringify!($push),
293                        ", ", stringify!($pop), " rebalances and returns the opposite endpoint"
294                    ),
295                );
296                if push_front {
297                    deque.push_front(added);
298                } else {
299                    deque.push_back(added);
300                }
301                let mut output = input;
302                if push_front {
303                    output = (output & added.0) ^ added.1;
304                }
305                for &(mask, xor) in &values[left..right] {
306                    output = (output & mask) ^ xor;
307                }
308                if !push_front {
309                    output = (output & added.0) ^ added.1;
310                }
311                kani::assert(
312                    deque.len() == $len
313                        && !deque.is_empty()
314                        && deque.front() == Some(if push_front { &added } else { &values[left] })
315                        && deque.back() == Some(if push_front { &values[right - 1] } else { &added })
316                        && LogicalLinearOperation::eval(&deque.fold_all(), &input) == output,
317                    concat!(
318                        "After an endpoint removal and insertion from a ", stringify!($len),
319                        "-element deque, length, endpoints and noncommutative aggregate match the logical order"
320                    ),
321                );
322                kani::cover!(!deque.front_stack.is_empty() && !deque.back_stack.is_empty());
323                kani::cover!(
324                    ((input & values[0].0) ^ values[0].1) & values[1].0 ^ values[1].1
325                        != ((input & values[1].0) ^ values[1].1) & values[0].0 ^ values[0].1
326                );
327            }
328        };
329    }

331    // Odd and even lengths exercise both div_ceil rebalancing boundaries.
332    proof_deque_rebalance!(proof_deque_front_three, 3, push_back, pop_front, false);
333    proof_deque_rebalance!(proof_deque_front_four, 4, push_back, pop_front, false);
334    proof_deque_rebalance!(proof_deque_back_three, 3, push_front, pop_back, true);
335    proof_deque_rebalance!(proof_deque_back_four, 4, push_front, pop_back, true);
