19    /// Block operations for a binary indexed tree over a commutative monoid.
20    ///
21    /// B is a power of two greater than one. At each level, the B-th Fenwick cells
22    /// move to the next level. `block` receives at most B - 1 cells, not original keys,
23    /// and pads unused lanes with unit. A kernel may cache the local prefix values. `add` applies a value
24    /// to local Fenwick ancestors (lane, lane | (lane + 1), ...), including padding.
25    /// `prefix` combines cells at end - 1, (end & (end - 1)) - 1, ... in that block;
26    /// prefix(0) is unit. The tree never queries the last padding lane.
27    ///
28    /// # Safety
29    /// ISA is 0 (portable), 1 (AVX2), or 2 (AVX-512F/DQ/CD/BW/VL), supported by the CPU.
30    /// Additional instructions require the kernel's own check. Unsafe methods receive
31    /// lane < B or end < B. Portable implementations must work on non-x86 targets.
32    pub trait SimdFenwickKernel {
33        const B: usize;
34        type Key;
35        type Output;
36        type Value: Copy;
37        type Block;
38        fn unit() -> Self::Value;
39        fn encode(key: Self::Key) -> Self::Value;
40        fn decode(value: Self::Value) -> Self::Output;
41        fn combine(left: Self::Value, right: Self::Value) -> Self::Value;
42        fn block(values: impl Iterator<Item = Self::Value>) -> Self::Block;
43        /// # Safety
44        /// ISA and lane must satisfy the block contract.
45        unsafe fn add<const ISA: u8>(block: &mut Self::Block, lane: usize, value: Self::Value);
46        /// # Safety
47        /// ISA and end must satisfy the block contract.
48        unsafe fn prefix<const ISA: u8>(block: &Self::Block, end: usize) -> Self::Value;
49    }
50
51    /// Inverses for range folds and replacement of individual values.
52    pub trait SimdFenwickGroup: SimdFenwickKernel {
53        fn inverse(value: Self::Value) -> Self::Value;
54    }

88    impl<K> SimdBinaryIndexedTree<K>
89    where
90        K: SimdFenwickKernel,
91    {
92        pub fn new(len: usize) -> Self {
93            Self::build(std::iter::repeat_n(K::unit(), len))
94        }
95        pub fn from_keys(keys: impl ExactSizeIterator<Item = K::Key>) -> Self {
96            Self::build(keys.map(K::encode))
97        }
98        pub fn len(&self) -> usize {
99            self.len
100        }
101        pub fn is_empty(&self) -> bool {
102            self.len == 0
103        }
104
105        pub fn update(&mut self, index: usize, value: K::Key) {
106            assert!(index < self.len);
107            let value = K::encode(value);
108            // SAFETY: index is in bounds.
109            unsafe { self.add_value(index, value) };
110        }
111        pub fn accumulate0(&self, end: usize) -> K::Output {
112            assert!(end <= self.len);
113            K::decode(self.prefix(end))
114        }
115        pub fn accumulate(&self, index: usize) -> K::Output {
116            assert!(index < self.len);
117            self.accumulate0(index + 1)
118        }
119        pub fn fold(&self, left: usize, right: usize) -> K::Output
120        where
121            K: SimdFenwickGroup,
122        {
123            assert!(left <= right && right <= self.len);
124            K::decode(self.fold_value::<false>(left, right))
125        }
126        pub fn get(&self, index: usize) -> K::Output
127        where
128            K: SimdFenwickGroup,
129        {
130            assert!(index < self.len);
131            K::decode(self.fold_value::<true>(index, index + 1))
132        }
133        pub fn set(&mut self, index: usize, value: K::Key)
134        where
135            K: SimdFenwickGroup,
136        {
137            assert!(index < self.len);
138            let old = self.fold_value::<true>(index, index + 1);
139            let delta = K::combine(K::inverse(old), K::encode(value));
140            // SAFETY: index is in bounds.
141            unsafe { self.add_value(index, delta) };
142        }

144        /// Hints the cache for a future access; index need not be in bounds.
145        pub fn prefetch(&self, index: usize) {
146            #[cfg(target_arch = "x86_64")]
147            {
148                let mut index = index;
149                for level in self.levels.iter().take(2) {
150                    if let Some(block) = level.get(index / K::B) {
151                        prefetch_block(block);
152                    }
153                    index /= K::B;
154                }
155            }
156            let _ = index;
157        }
158
159        fn build(keys: impl ExactSizeIterator<Item = K::Value>) -> Self {
160            assert!(K::B >= 2 && K::B.is_power_of_two());
161            let mut values: Vec<_> = keys.collect();
162            let len = values.len();
163            for index in 0..len {
164                let parent = index | (index + 1);
165                if parent < len {
166                    values[parent] = K::combine(values[parent], values[index]);
167                }
168            }
169            let mut levels = Vec::new();
170            while !values.is_empty() {
171                // The extra block makes an end boundary valid even at a multiple of B.
172                let count = values.len() / K::B + 1;
173                let blocks = (0..count)
174                    .map(|i| {
175                        let begin = i * K::B;
176                        let end = begin + (values.len() - begin).min(K::B - 1);
177                        K::block(values[begin..end].iter().copied())
178                    })
179                    .collect();
180                levels.push(blocks);
181                values = values.into_iter().skip(K::B - 1).step_by(K::B).collect();
182            }
183            Self {
184                levels,
185                len,
186                backend: simd_backend(),
187            }
188        }
189        unsafe fn add_value(&mut self, index: usize, value: K::Value) {
190            // The backend was selected on this CPU.
191            match self.backend {
192                SimdBackend::Scalar => self.add::<0>(index, value),
193                SimdBackend::Avx2 => self.add::<1>(index, value),
194                SimdBackend::Avx512 => self.add::<2>(index, value),
195            }
196        }
197
198        unsafe fn add<const ISA: u8>(&mut self, mut index: usize, value: K::Value) {
199            for level in &mut self.levels {
200                K::add::<ISA>(level.get_unchecked_mut(index / K::B), index % K::B, value);
201                index /= K::B;
202            }
203        }
204        fn prefix(&self, end: usize) -> K::Value {
205            // SAFETY: public callers bound end; every level includes its end sentinel block.
206            unsafe {
207                match self.backend {
208                    SimdBackend::Scalar => self.prefix_with::<0>(end),
209                    SimdBackend::Avx2 => self.prefix_with::<1>(end),
210                    SimdBackend::Avx512 => self.prefix_with::<2>(end),
211                }
212            }
213        }
214        unsafe fn prefix_with<const ISA: u8>(&self, mut end: usize) -> K::Value {
215            let mut value = K::unit();
216            for level in &self.levels {
217                value = K::combine(
218                    value,
219                    K::prefix::<ISA>(level.get_unchecked(end / K::B), end % K::B),
220                );
221                end /= K::B;
222            }
223            value
224        }
225        fn fold_value<const POINT: bool>(&self, left: usize, right: usize) -> K::Value
226        where
227            K: SimdFenwickGroup,
228        {
229            // SAFETY: callers bound both endpoints; each level includes its end sentinel block.
230            unsafe {
231                match self.backend {
232                    SimdBackend::Scalar => self.difference::<0, POINT>(left, right),
233                    SimdBackend::Avx2 => self.difference::<1, POINT>(left, right),
234                    SimdBackend::Avx512 => self.difference::<2, POINT>(left, right),
235                }
236            }
237        }
238        unsafe fn difference<const ISA: u8, const POINT: bool>(
239            &self,
240            left: usize,
241            right: usize,
242        ) -> K::Value
243        where
244            K: SimdFenwickGroup,
245        {
246            if left == right {
247                return K::unit();
248            }
249            // Point access cancels the shared ancestors; range folds keep a predictable depth.
250            let depth = if POINT {
251                ((left ^ right).ilog2() / K::B.trailing_zeros() + 1) as usize
252            } else {
253                self.levels.len()
254            };
255            match depth {
256                1 => self.difference_depth::<ISA, 1>(left, right),
257                2 => self.difference_depth::<ISA, 2>(left, right),
258                3 => self.difference_depth::<ISA, 3>(left, right),
259                4 => self.difference_depth::<ISA, 4>(left, right),
260                5 => self.difference_depth::<ISA, 5>(left, right),
261                6 => self.difference_depth::<ISA, 6>(left, right),
262                7 => self.difference_depth::<ISA, 7>(left, right),
263                8 => self.difference_depth::<ISA, 8>(left, right),
264                _ => self.difference_depth::<ISA, 0>(left, right),
265            }
266        }
267        unsafe fn difference_depth<const ISA: u8, const D: usize>(
268            &self,
269            left: usize,
270            right: usize,
271        ) -> K::Value
272        where
273            K: SimdFenwickGroup,
274        {
275            let mut l = K::unit();
276            let mut r = K::unit();
277            let depth = if D == 0 { self.levels.len() } else { D };
278            for at in 0..depth {
279                let a = left >> (at * K::B.trailing_zeros() as usize);
280                let b = right >> (at * K::B.trailing_zeros() as usize);
281                let level = self.levels.get_unchecked(at);
282                l = K::combine(l, K::prefix::<ISA>(level.get_unchecked(a / K::B), a % K::B));
283                r = K::combine(r, K::prefix::<ISA>(level.get_unchecked(b / K::B), b % K::B));
284            }
285            K::combine(K::inverse(l), r)
286        }
287    }

378            struct Minimum<const B: usize>;
379            impl<const B: usize> SimdFenwickKernel for Minimum<B> {
380                const B: usize = B;
381                type Key = u32;
382                type Output = u32;
383                type Value = u32;
384                type Block = [u32; B];
385                fn unit() -> u32 {
386                    u32::MAX
387                }
388                fn encode(key: u32) -> u32 {
389                    key
390                }
391                fn decode(value: u32) -> u32 {
392                    value
393                }
394                fn combine(left: u32, right: u32) -> u32 {
395                    left.min(right)
396                }
397                fn block(values: impl Iterator<Item = u32>) -> Self::Block {
398                    let mut block = [u32::MAX; B];
399                    for (lane, value) in values.enumerate() {
400                        block[lane] = value;
401                    }
402                    block
403                }
404                unsafe fn add<const ISA: u8>(block: &mut Self::Block, mut lane: usize, value: u32) {
405                    while lane < B {
406                        block[lane] = block[lane].min(value);
407                        lane |= lane + 1;
408                    }
409                }
410                unsafe fn prefix<const ISA: u8>(block: &Self::Block, mut end: usize) -> u32 {
411                    let mut value = u32::MAX;
412                    while end != 0 {
413                        value = value.min(block[end - 1]);
414                        end &= end - 1;
415                    }
416                    value
417                }
418            }
419            fn check<const B: usize>() {
420                let mut rng = Xorshift::default();

484    impl<M> SimdFenwickKernel for M
485    where
486        M: SimdMonoid + Commutative,
487    {
488        const B: usize = LANES;
489        type Key = M::T;
490        type Output = M::T;
491        type Value = M::T;
492        type Block = [M::T; LANES];
493
494        fn unit() -> M::T {
495            M::unit()
496        }
497        fn encode(key: M::T) -> M::T {
498            key
499        }
500        fn decode(value: M::T) -> M::T {
501            value
502        }
503        fn combine(left: M::T, right: M::T) -> M::T {
504            M::operate(&left, &right)
505        }
506        fn block(values: impl Iterator<Item = M::T>) -> Self::Block {
507            let mut block = [M::unit(); LANES];
508            let mut values = values.fuse();
509            // Each cell completes the prefix at end using the earlier cached prefix.
510            for end in 1..LANES {
511                let cell = values.next().unwrap_or_else(M::unit);
512                let before = end & (end - 1);
513                block[end] = M::operate(&block[before], &cell);
514            }
515            block
516        }
517        unsafe fn add<const ISA: u8>(block: &mut Self::Block, lane: usize, value: M::T) {
518            unsafe { M::broadcast_suffix::<ISA>(block, lane + 1, value) };
519        }
520        unsafe fn prefix<const ISA: u8>(block: &Self::Block, end: usize) -> M::T {
521            block[end]
522        }
523    }
524
525    impl<M> SimdFenwickGroup for M
526    where
527        M: SimdMonoid + Commutative + Invertible,
528    {
529        fn inverse(value: M::T) -> M::T {
530            M::inverse(&value)
531        }
532    }

540    #![allow(unsafe_op_in_unsafe_fn)] // The tree selects an ISA supported by the CPU.
541    #[cfg(target_arch = "x86_64")]
542    use super::simd;
543    use super::{DaryAddKernelI32, DaryAddKernelI64};
544    use super::{SimdFenwickGroup, SimdFenwickKernel};

546    #[repr(C, align(64))]
547    #[derive(Clone, Debug)]
548    pub struct FenwickNumericBlock<T, const B: usize>([T; B]);
549
550    macro_rules! define_fenwick_kernel {
551        ($kernel:ty, $value:ty, $branch:expr, $unsigned:ty, $avx2:ident, $avx512:ident) => {
552            impl SimdFenwickKernel for $kernel {
553                const B: usize = $branch;
554                type Key = $value;
555                type Output = $value;
556                type Value = $value;
557                type Block = FenwickNumericBlock<$value, $branch>;
558                fn unit() -> $value {
559                    0
560                }
561                fn encode(key: $value) -> $value {
562                    key
563                }
564                fn decode(value: $value) -> $value {
565                    value
566                }
567                fn combine(left: $value, right: $value) -> $value {
568                    left.wrapping_add(right)
569                }
570                fn block(values: impl Iterator<Item = $value>) -> Self::Block {
571                    let mut block = [0; $branch];
572                    for (lane, value) in values.enumerate() {
573                        block[lane] = value;
574                    }
575                    // Cache local prefixes while retaining the Fenwick cell contract.
576                    for i in 0..$branch {
577                        let before = i & (i + 1);
578                        if before != 0 {
579                            block[i] = block[i].wrapping_add(block[before - 1]);
580                        }
581                    }
582                    FenwickNumericBlock(block)
583                }
584                unsafe fn add<const ISA: u8>(
585                    block: &mut Self::Block,
586                    mut lane: usize,
587                    value: $value,
588                ) {
589                    #[cfg(target_arch = "x86_64")]
590                    if ISA != 0 {
591                        let values = &mut *block.0.as_mut_ptr().cast::<[$unsigned; $branch]>();
592                        if ISA == 1 {
593                            simd::$avx2(values, lane, value as $unsigned);
594                        } else {
595                            simd::$avx512(values, lane, value as $unsigned);
596                        }
597                        return;
598                    }
599                    while lane < $branch {
600                        block.0[lane] = block.0[lane].wrapping_add(value);
601                        lane += 1;
602                    }
603                }
604                unsafe fn prefix<const ISA: u8>(block: &Self::Block, end: usize) -> $value {
605                    *block.0.get_unchecked(end.wrapping_sub(1) & ($branch - 1))
606                        & (0 as $value).wrapping_sub((end != 0) as $value)
607                }
608            }
609            impl SimdFenwickGroup for $kernel {
610                fn inverse(value: $value) -> $value {
611                    value.wrapping_neg()
612                }
613            }
614        };
615    }
