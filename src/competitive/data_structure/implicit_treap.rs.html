83impl<T> ImplicitTreapSpec<T>
84where
85    T: LazyMapMonoidWithKeyAct,
86{
87    fn update_act(mut node: BstDataMutRef<'_, Self>, act: &LazyMapAct<T>) {
88        if T::is_act_unit(act) {
89            return;
90        }
91        T::act_operate_assign(&mut node.data_mut().value.act, act);
92        node.data_mut().value.key = T::act_key(&node.reborrow().into_data().value.key, act);
93        if let Some(agg) = T::act_agg(&node.reborrow().into_data().value.agg, act) {
94            node.data_mut().value.agg = agg;
95        } else {
96            Self::top_down(node.reborrow_datamut());
97            Self::bottom_up(node);
98        }
99    }
100
101    fn reverse(mut node: BstDataMutRef<'_, Self>) {
102        node.swap_children();
103        let data = node.data_mut();
104        T::toggle(&mut data.value.agg);
105        data.rev ^= true;
106    }
107}

109impl<T> BstSpec for ImplicitTreapSpec<T>
110where
111    T: LazyMapMonoidWithKeyAct,
112{
113    type Parent = WithNoParent<Self::Data>;
114    type Data = ImplicitTreapData<T>;
115
116    fn top_down(mut node: BstDataMutRef<'_, Self>) {
117        if !T::is_act_unit(&node.reborrow().into_data().value.act) {
118            let act = replace(&mut node.data_mut().value.act, T::act_unit());
119            if let Ok(left) = node.reborrow_datamut().left().descend() {
120                Self::update_act(left, &act);
121            }
122            if let Ok(right) = node.reborrow_datamut().right().descend() {
123                Self::update_act(right, &act);
124            }
125        }
126        if node.reborrow().into_data().rev {
127            node.data_mut().rev = false;
128            if let Ok(left) = node.reborrow_datamut().left().descend() {
129                Self::reverse(left);
130            }
131            if let Ok(right) = node.reborrow_datamut().right().descend() {
132                Self::reverse(right);
133            }
134        }
135    }
136
137    fn bottom_up(mut node: BstDataMutRef<'_, Self>) {
138        let mut agg = T::single_agg(&node.reborrow().into_data().value.key);
139        let mut size = 1;
140        if let Ok(left) = node.reborrow().left().descend() {
141            let data = left.into_data();
142            agg = T::agg_operate(&data.value.agg, &agg);
143            size += data.size;
144        }
145        if let Ok(right) = node.reborrow().right().descend() {
146            let data = right.into_data();
147            agg = T::agg_operate(&agg, &data.value.agg);
148            size += data.size;
149        }
150        let data = node.data_mut();
151        data.value.agg = agg;
152        data.size = size;
153    }
154
155    fn merge(
156        left: Option<ImplicitTreapRoot<T>>,
157        right: Option<ImplicitTreapRoot<T>>,
158    ) -> Option<ImplicitTreapRoot<T>> {
159        match (left, right) {
160            (None, None) => None,
161            (None, Some(node)) | (Some(node), None) => Some(node),
162            (Some(mut left), Some(mut right)) => unsafe {
163                if left.reborrow().into_data().priority > right.reborrow().into_data().priority {
164                    Self::top_down(left.borrow_datamut());
165                    let lr = left.borrow_mut().right().take();
166                    let lr = Self::merge(lr, Some(right)).unwrap_unchecked();
167                    left.borrow_mut().right().set(lr);
168                    Self::bottom_up(left.borrow_datamut());
169                    Some(left)
170                } else {
171                    Self::top_down(right.borrow_datamut());
172                    let rl = right.borrow_mut().left().take();
173                    let rl = Self::merge(Some(left), rl).unwrap_unchecked();
174                    right.borrow_mut().left().set(rl);
175                    Self::bottom_up(right.borrow_datamut());
176                    Some(right)
177                }
178            },
179        }
180    }
181
182    fn split<Seeker>(
183        node: Option<ImplicitTreapRoot<T>>,
184        mut seeker: Seeker,
185        equal_side: EqualSide,
186    ) -> (Option<ImplicitTreapRoot<T>>, Option<ImplicitTreapRoot<T>>)
187    where
188        Seeker: BstSeeker<Spec = Self>,
189    {
190        match node {
191            None => (None, None),
192            Some(mut node) => {
193                Self::top_down(node.borrow_datamut());
194                if equal_side.goes_left(seeker.bst_seek(node.reborrow())) {
195                    unsafe {
196                        let right = node.borrow_mut().right().take();
197                        let (l, r) = Self::split(right, seeker, equal_side);
198                        if let Some(l) = l {
199                            node.borrow_mut().right().set(l);
200                        }
201                        Self::bottom_up(node.borrow_datamut());
202                        (Some(node), r)
203                    }
204                } else {
205                    unsafe {
206                        let left = node.borrow_mut().left().take();
207                        let (l, r) = Self::split(left, seeker, equal_side);
208                        if let Some(r) = r {
209                            node.borrow_mut().left().set(r);
210                        }
211                        Self::bottom_up(node.borrow_datamut());
212                        (l, Some(node))
213                    }
214                }
215            }
216        }
217    }
218}

282impl<T, A> ImplicitTreap<T, A>
283where
284    T: LazyMapMonoidWithKeyAct,
285    A: Allocator<ImplicitTreapNode<T>>,
286{
287    pub fn with_rng(mut self, rng: Xorshift) -> Self {
288        self.rng = rng;
289        self
290    }
291
292    fn node(&mut self, key: T::Key) -> ImplicitTreapRoot<T> {
293        unsafe {
294            BstRoot::from_data(
295                ImplicitTreapData {
296                    priority: self.rng.rand64(),
297                    value: LazyMapElement::from_key(key),
298                    size: 1,
299                    rev: false,
300                },
301                self.allocator.deref_mut(),
302            )
303        }
304    }
305
306    fn build<I>(&mut self, iter: I) -> (Option<ImplicitTreapRoot<T>>, usize)
307    where
308        I: IntoIterator<Item = T::Key>,
309    {
310        let mut stack = vec![];
311        let mut len = 0;
312        for key in iter {
313            let mut cur = self.node(key).node;
314            let mut left = None;
315            unsafe {
316                while stack
317                    .last()
318                    .is_some_and(|node: &NonNull<ImplicitTreapNode<T>>| {
319                        node.as_ref().data.priority < cur.as_ref().data.priority
320                    })
321                {
322                    left = stack.pop();
323                }
324                cur.as_mut().child[0] = left;
325                if let Some(parent) = stack.last_mut() {
326                    parent.as_mut().child[1] = Some(cur);
327                }
328            }
329            stack.push(cur);
330            len += 1;
331        }
332        let root = stack
333            .first()
334            .copied()
335            .map(|node| unsafe { BstRoot::new_unchecked(node) });
336        if let Some(mut root) = root {
337            Self::build_bottom_up(root.borrow_datamut());
338            (Some(root), len)
339        } else {
340            (None, len)
341        }
342    }
343
344    fn build_bottom_up(mut node: BstDataMutRef<'_, ImplicitTreapSpec<T>>) {
345        if let Ok(left) = node.reborrow_datamut().left().descend() {
346            Self::build_bottom_up(left);
347        }
348        if let Ok(right) = node.reborrow_datamut().right().descend() {
349            Self::build_bottom_up(right);
350        }
351        ImplicitTreapSpec::<T>::bottom_up(node);
352    }
353
354    pub fn len(&self) -> usize {
355        self.length
356    }
357
358    pub fn is_empty(&self) -> bool {
359        self.length == 0
360    }
361
362    pub fn update<R>(&mut self, range: R, x: LazyMapAct<T>)
363    where
364        R: RangeBounds<usize>,
365    {
366        let mut split = Split3::seek_by_size(&mut self.root, range);
367        if let Some(root) = split.mid_datamut() {
368            ImplicitTreapSpec::<T>::update_act(root, &x);
369        }
370    }
371
372    pub fn fold<R>(&mut self, range: R) -> LazyMapAgg<T>
373    where
374        R: RangeBounds<usize>,
375    {
376        let split = Split3::seek_by_size(&mut self.root, range);
377        split
378            .mid()
379            .map(|node| node.into_data().value.agg.clone())
380            .unwrap_or_else(T::agg_unit)
381    }
382
383    pub fn reverse<R>(&mut self, range: R)
384    where
385        R: RangeBounds<usize>,
386    {
387        let mut split = Split3::seek_by_size(&mut self.root, range);
388        if let Some(root) = split.mid_datamut() {
389            ImplicitTreapSpec::<T>::reverse(root);
390        }
391    }
392
393    pub fn get(&mut self, index: usize) -> Option<&T::Key> {
394        if index >= self.length {
395            return None;
396        }
397        let split = Split3::seek_by_size(&mut self.root, index..=index);
398        let node = split.mid()?.node;
399        drop(split);
400        Some(unsafe { &(*node.as_ptr()).data.value.key })
401    }
402
403    pub fn modify<F>(&mut self, index: usize, f: F)
404    where
405        F: FnOnce(&T::Key) -> T::Key,
406    {
407        assert!(index < self.length);
408        let mut split = Split3::seek_by_size(&mut self.root, index..=index);
409        let mut node = split.mid_datamut().unwrap();
410        ImplicitTreapSpec::<T>::top_down(node.reborrow_datamut());
411        {
412            let data = node.data_mut();
413            data.value.key = f(&data.value.key);
414        }
415        ImplicitTreapSpec::<T>::bottom_up(node);
416    }
417
418    pub fn insert(&mut self, index: usize, x: T::Key) {
419        assert!(index <= self.length);
420        let node = self.node(x);
421        if index == 0 {
422            self.root = ImplicitTreapSpec::<T>::merge(Some(node), self.root.take());
423        } else if index == self.length {
424            self.root = ImplicitTreapSpec::<T>::merge(self.root.take(), Some(node));
425        } else {
426            let mut node = Some(node);
427            let mut split = Split::new(&mut self.root, SeekBySize::new(index), EqualSide::Right);
428            split.manually_merge(|left, right| {
429                ImplicitTreapSpec::<T>::merge(
430                    ImplicitTreapSpec::<T>::merge(left, node.take()),
431                    right,
432                )
433            });
434        }
435        self.length += 1;
436    }
437
438    pub fn remove(&mut self, index: usize) -> Option<T::Key> {
439        if index >= self.length {
440            return None;
441        }
442        let mid;
443        if index == 0 {
444            let (left, right) = ImplicitTreapSpec::<T>::split(
445                self.root.take(),
446                SeekBySize::new(1),
447                EqualSide::Right,
448            );
449            mid = left;
450            self.root = right;
451        } else if index + 1 == self.length {
452            let (left, right) = ImplicitTreapSpec::<T>::split(
453                self.root.take(),
454                SeekBySize::new(index),
455                EqualSide::Right,
456            );
457            mid = right;
458            self.root = left;
459        } else {
460            let (left, rest) = ImplicitTreapSpec::<T>::split(
461                self.root.take(),
462                SeekBySize::new(index),
463                EqualSide::Right,
464            );
465            let (middle, right) =
466                ImplicitTreapSpec::<T>::split(rest, SeekBySize::new(1), EqualSide::Right);
467            mid = middle;
468            self.root = ImplicitTreapSpec::<T>::merge(left, right);
469        }
470        self.length -= 1;
471        let mut node = mid.unwrap();
472        ImplicitTreapSpec::<T>::top_down(node.borrow_datamut());
473        let data = unsafe { node.into_dying().into_data(self.allocator.deref_mut()) };
474        Some(data.value.key)
475    }
476
477    pub fn partition_point_acc<F>(&mut self, left: usize, mut pred: F) -> usize
478    where
479        F: FnMut(&LazyMapAgg<T>) -> bool,
480    {
481        let mut split3 = Split3::seek_by_size(&mut self.root, left..);
482        let front_size = split3
483            .left()
484            .map(|node| node.into_data().size)
485            .unwrap_or_default();
486        let split = split3.split_mid(SeekByAccCond::new(|acc| !pred(acc)), EqualSide::Right);
487        let index = split
488            .left()
489            .map(|node| node.into_data().size)
490            .unwrap_or_default();
491        front_size + index
492    }
493
494    pub fn rpartition_point_acc<F>(&mut self, right: usize, mut pred: F) -> usize
495    where
496        F: FnMut(&LazyMapAgg<T>) -> bool,
497    {
498        let mut split3 = Split3::seek_by_size(&mut self.root, ..right);
499        let split = split3.split_mid(SeekByRaccCond::new(|acc| !pred(acc)), EqualSide::Left);
500        split
501            .left()
502            .map(|node| node.into_data().size)
503            .unwrap_or_default()
504    }
505
506    pub fn rotate_left(&mut self, mid: usize) {
507        assert!(mid <= self.length);
508        if mid == 0 || mid == self.length {
509            return;
510        }
511        let (left, right) =
512            ImplicitTreapSpec::<T>::split(self.root.take(), SeekBySize::new(mid), EqualSide::Right);
513        self.root = ImplicitTreapSpec::<T>::merge(right, left);
514    }
515
516    pub fn rotate_right(&mut self, k: usize) {
517        assert!(k <= self.length);
518        self.rotate_left(self.length - k);
519    }
520}

658                }

660                    let i = rng.random(0..arr.len());
661                    if rng.random(0..2) == 0 {
662                        assert_eq!(arr.get(i), treap.get(i));
663                    } else {
664                        let x = Saturating(rng.random(0..=A));
665                        treap.modify(i, |_| x);
666                        arr[i] = x;
667                    }
668                }
669                _ => {}
670            }
671        }

674    #[test]
675    fn test_implicit_treap_sequence() {
676        const MAX_LEN: usize = 1024;
677        const MAX_VALUE: i64 = 1_000_000_000;
678        let mut input_rng = Xorshift::default();
679        let mut rng = Xorshift::default();
680        for (case, mut arr) in
681            sequence_cases(&mut input_rng, -MAX_VALUE..=MAX_VALUE, 0..=MAX_LEN).enumerate()
682        {
683            let initial_len = arr.len();
684            let mut tree = ImplicitTreap::<ConcatUpdate>::new().with_rng(Xorshift::default());
685            tree.extend(arr.iter().copied());
686            assert_eq!(tree.fold(..), arr, "case {case}");
687            assert_eq!(tree.remove(arr.len()), None);
688            for step in 0..initial_len / 2 + 10 {
689                let n = arr.len();
690                let kind = if step < 10 { step } else { rng.random(0..10) };
691                let index = rng.random(0..=n);
692                let (l, r) = rng.random(WithEmptySegment(n));
693                let value = rng.random(-MAX_VALUE..=MAX_VALUE);
694                match kind {
695                    0 if n < MAX_LEN => {
696                        tree.insert(index, value);
697                        arr.insert(index, value);
698                    }
699                    1 => {
700                        let index = index.min(n.saturating_sub(1));
701                        let expected = (index < n).then(|| arr.remove(index));
702                        assert_eq!(tree.remove(index), expected);
703                    }
704                    3 if n != 0 => {
705                        let index = index.min(n - 1);
706                        tree.modify(index, |_| value);
707                        arr[index] = value;
708                    }
709                    4 => assert_eq!(tree.fold(l..r), arr[l..r]),
710                    5 => {
711                        tree.update(l..r, Some(value));
712                        arr[l..r].fill(value);
713                    }
714                    6 => {
715                        tree.reverse(l..r);
716                        arr[l..r].reverse();
717                    }
718                    7 => {
719                        tree.rotate_left(index);
720                        arr.rotate_left(index);
721                    }
722                    8 => {
723                        tree.rotate_right(index);
724                        arr.rotate_right(index);
725                    }
726                    9 => {
727                        let target = &arr[l..r];
728                        assert_eq!(
729                            tree.partition_point_acc(l, |agg| target.starts_with(agg)),
730                            r
731                        );
732                        assert_eq!(tree.rpartition_point_acc(r, |agg| target.ends_with(agg)), l);
733                        assert_eq!(
734                            tree.partition_point_acc(l, |agg| agg.iter().all(|x| *x < value)),
735                            arr[l..]
736                                .iter()
737                                .position(|x| *x >= value)
738                                .map_or(n, |i| l + i),
739                        );
740                        assert_eq!(
741                            tree.rpartition_point_acc(r, |agg| agg.iter().all(|x| *x < value)),
742                            arr[..r]
743                                .iter()
744                                .rposition(|x| *x >= value)
745                                .map_or(0, |i| i + 1),
746                        );
747                    }
748                    _ => assert_eq!(tree.get(index), arr.get(index)),
749                }
750                assert_eq!((tree.len(), tree.is_empty()), (arr.len(), arr.is_empty()));
751                assert_eq!(tree.fold(..), arr, "case {case}, step {step}");
752            }
753            while !arr.is_empty() {
754                let index = match arr.len() % 3 {
755                    0 => 0,
756                    1 => arr.len() - 1,
757                    _ if arr.len() > 2 => rng.random(1..arr.len() - 1),
758                    _ => rng.random(0..arr.len()),
759                };
760                assert_eq!(tree.remove(index), Some(arr.remove(index)));
761                assert_eq!(tree.fold(..), arr);
762            }
763            assert_eq!((tree.len(), tree.is_empty()), (0, true));
764            assert_eq!(tree.get(0), None);
765            assert_eq!(tree.remove(0), None);
766            tree.update(.., Some(rng.random(-MAX_VALUE..=MAX_VALUE)));
767            tree.reverse(..);
768            tree.rotate_left(0);
769            tree.rotate_right(0);
770            assert_eq!(tree.fold(..), arr);
771            assert_eq!(tree.partition_point_acc(0, |_| true), 0);
772            assert_eq!(tree.rpartition_point_acc(0, |_| true), 0);
773            for _ in 0..initial_len.max(2) {
774                let index = match arr.len() % 3 {
775                    0 => 0,
776                    1 => arr.len(),
777                    _ => rng.random(1..arr.len()),
778                };
779                let value = rng.random(-MAX_VALUE..=MAX_VALUE);
780                tree.insert(index, value);
781                arr.insert(index, value);
782                assert_eq!(tree.fold(..), arr);
783            }
784            let first = rng.random(-MAX_VALUE..MAX_VALUE);
785            let second = rng.random(first + 1..=MAX_VALUE);
786            let split = rng.random(1..arr.len());
787            let index = rng.random(0..arr.len());
788            tree.modify(index, |_| first);
789            arr[index] = first;
790            assert_eq!(tree.get(index), arr.get(index));
791            tree.update(.., Some(first));
792            arr.fill(first);
793            tree.update(split.., Some(second));
794            arr[split..].fill(second);
795            tree.reverse(..);
796            arr.reverse();
797            tree.update(.., None);
798            assert_eq!(tree.fold(..), arr);
799            let middle = arr.len() - split;
800            for (l, r) in [
801                (0, 0),
802                (0, arr.len()),
803                (middle, middle + 1),
804                (middle - 1, middle + 1),
805            ] {
806                let target = &arr[l..r];
807                assert_eq!(tree.fold(l..r), target);
808                assert_eq!(
809                    tree.partition_point_acc(l, |agg| target.starts_with(agg)),
810                    r
811                );
812                assert_eq!(tree.rpartition_point_acc(r, |agg| target.ends_with(agg)), l);
813            }
814        }
815    }
