3/// rank_i(select_i(k)) = k
4/// rank_i(select_i(k) + 1) = k + 1
5pub trait RankSelectDictionaries {
6    fn bit_length(&self) -> usize;
7    /// get k-th bit
8    fn access(&self, k: usize) -> bool;
9    /// Returns the k-th bit and the number of ones before it.
10    fn access_rank1(&self, k: usize) -> (bool, usize) {
11        (self.access(k), self.rank1(k))
12    }
13    /// the number of 1 in [0, k)
14    fn rank1(&self, k: usize) -> usize {
15        (0..k).filter(|&i| self.access(i)).count()
16    }
17    /// the number of 0 in [0, k)
18    fn rank0(&self, k: usize) -> usize {
19        k - self.rank1(k)
20    }
21    /// index of k-th 1
22    fn select1(&self, k: usize) -> Option<usize> {
23        let n = self.bit_length();
24        if self.rank1(n) <= k {
25            return None;
26        }
27        let (mut l, mut r) = (0, n);
28        while r - l > 1 {
29            let m = l.midpoint(r);
30            if self.rank1(m) <= k {
31                l = m;
32            } else {
33                r = m;
34            }
35        }
36        Some(l)
37    }
38    /// index of k-th 0
39    fn select0(&self, k: usize) -> Option<usize> {
40        let n = self.bit_length();
41        if self.rank0(n) <= k {
42            return None;
43        }
44        let (mut l, mut r) = (0, n);
45        while r - l > 1 {
46            let m = l.midpoint(r);
47            if self.rank0(m) <= k {
48                l = m;
49            } else {
50                r = m;
51            }
52        }
53        Some(l)
54    }
55}

57macro_rules! impl_rank_select_for_bits {
58    ($($t:ty)*) => {$(
59        impl RankSelectDictionaries for $t {
60            fn bit_length(&self) -> usize {
61                const WORD_SIZE: usize = (0 as $t).count_zeros() as usize;
62                WORD_SIZE
63            }
64            fn access(&self, k: usize) -> bool {
65                const WORD_SIZE: usize = (0 as $t).count_zeros() as usize;
66                if k < WORD_SIZE {
67                    self & (1 as $t) << k != 0
68                } else {
69                    false
70                }
71            }
72            fn rank1(&self, k: usize) -> usize {
73                const WORD_SIZE: usize = (0 as $t).count_zeros() as usize;
74                if k < WORD_SIZE {
75                    (self & !(!(0 as $t) << k)).count_ones() as usize
76                } else {
77                    self.count_ones() as usize
78                }
79            }
80        }
81    )*};
82}

86#[inline]
87fn select_word_scalar(mut word: u64, mut rank: usize) -> usize {
88    let count = word.count_ones() as usize;
89    debug_assert!(rank < count);
90    if rank < 4 {
91        for _ in 0..rank {
92            word &= word - 1;
93        }
94        return word.trailing_zeros() as usize;
95    }
96    if count - rank <= 4 {
97        for _ in 0..count - rank - 1 {
98            word &= !(1 << (u64::BITS as usize - 1 - word.leading_zeros() as usize));
99        }
100        return u64::BITS as usize - 1 - word.leading_zeros() as usize;
101    }
102
103    let mut offset = 0;
104    let mut width = u64::BITS as usize / 2;
105    while width != 0 {
106        let mask = u64::MAX >> (u64::BITS as usize - width);
107        let count = (word & mask).count_ones() as usize;
108        if rank < count {
109            word &= mask;
110        } else {
111            word >>= width;
112            rank -= count;
113            offset += width;
114        }
115        width /= 2;
116    }
117    offset
118}

132/// An append-only rank/select dictionary with 256-word prefix blocks.
133///
134/// The layout reduces metadata and improves large or select-heavy workloads. A compact
135/// rank-only workload can be faster with an absolute prefix stored beside every word.
136#[derive(Debug, Clone)]
137pub struct BitVector {
138    words: Vec<u64>,
139    super_prefix: Vec<usize>,
140    sub_prefix: Vec<u16>,
141    sum: usize,
142    len: usize,
143    #[cfg(target_arch = "x86_64")]
144    bmi2: bool,
145}
146impl BitVector {
147    const WORD_SIZE: usize = u64::BITS as usize;
148    const SUPER_WORDS: usize = 256;
149
150    pub fn with_capacity(bits: usize) -> Self {
151        let words = bits.div_ceil(Self::WORD_SIZE);
152        let mut word_values = Vec::with_capacity(words + 1);
153        word_values.push(0);
154        let mut super_prefix = Vec::with_capacity(words / Self::SUPER_WORDS + 1);
155        super_prefix.push(0);
156        let mut sub_prefix = Vec::with_capacity(words + 1);
157        sub_prefix.push(0);
158        Self {
159            words: word_values,
160            super_prefix,
161            sub_prefix,
162            sum: 0,
163            len: 0,
164            #[cfg(target_arch = "x86_64")]
165            bmi2: is_x86_feature_detected!("bmi2"),
166        }
167    }
168
169    pub fn push(&mut self, bit: bool) {
170        let word = self.len / Self::WORD_SIZE;
171        let offset = self.len % Self::WORD_SIZE;
172        if offset == 0 {
173            self.words.push(0);
174            self.sub_prefix.push(0);
175        }
176        if bit {
177            self.words[word] |= 1u64 << offset;
178            self.sum += 1;
179        }
180        self.len += 1;
181        if self.len.is_multiple_of(Self::WORD_SIZE) {
182            let word = self.len / Self::WORD_SIZE;
183            if word.is_multiple_of(Self::SUPER_WORDS) {
184                if let Some(prefix) = self.super_prefix.get_mut(word / Self::SUPER_WORDS) {
185                    *prefix = self.sum;
186                } else {
187                    self.super_prefix.push(self.sum);
188                }
189            }
190            self.sub_prefix[word] = (self.sum - self.super_prefix[word / Self::SUPER_WORDS]) as u16;
191        }
192    }
193
194    fn from_words(mut words: Vec<u64>, len: usize) -> Self {
195        let mut super_prefix = Vec::with_capacity(words.len().div_ceil(Self::SUPER_WORDS));
196        let mut sub_prefix = Vec::with_capacity(words.len() + 1);
197        let mut sum = 0;
198        let mut super_sum = 0;
199        for index in 0..=words.len() {
200            if index.is_multiple_of(Self::SUPER_WORDS) {
201                if index < words.len() || len.is_multiple_of(Self::WORD_SIZE) {
202                    super_sum = sum;
203                    super_prefix.push(sum);
204                }
205                sub_prefix.push(0);
206            } else {
207                sub_prefix.push((sum - super_sum) as u16);
208            }
209            if let Some(&word) = words.get(index) {
210                sum += word.count_ones() as usize;
211            }
212        }
213        words.push(0);
214        Self {
215            words,
216            super_prefix,
217            sub_prefix,
218            sum,
219            len,
220            #[cfg(target_arch = "x86_64")]
221            bmi2: is_x86_feature_detected!("bmi2"),
222        }
223    }
224
225    #[inline]
226    fn select_word(&self, word: u64, rank: usize) -> usize {
227        #[cfg(target_arch = "x86_64")]
228        if self.bmi2 {
229            // SAFETY: support for BMI2 was cached during construction.
230            return unsafe { simd::select_word(word, rank) };
231        }
232        select_word_scalar(word, rank)
233    }
234
235    #[inline]
236    fn locate_one(&self, mut rank: usize) -> (usize, usize) {
237        let block = self.super_prefix.partition_point(|&sum| sum <= rank) - 1;
238        rank -= self.super_prefix[block];
239        let word_start = block * Self::SUPER_WORDS;
240        let word_end = (word_start + Self::SUPER_WORDS).min(self.words.len() - 1);
241        let lane =
242            self.sub_prefix[word_start..word_end].partition_point(|&sum| sum as usize <= rank) - 1;
243        let word = word_start + lane;
244        rank -= self.sub_prefix[word] as usize;
245        (word, rank)
246    }
247
248    #[inline]
249    fn locate_zero(&self, mut rank: usize) -> (usize, usize) {
250        let mut block = 0;
251        let mut right = self.super_prefix.len();
252        while right - block > 1 {
253            let middle = block.midpoint(right);
254            if middle * Self::SUPER_WORDS * Self::WORD_SIZE - self.super_prefix[middle] <= rank {
255                block = middle;
256            } else {
257                right = middle;
258            }
259        }
260        rank -= block * Self::SUPER_WORDS * Self::WORD_SIZE - self.super_prefix[block];
261        let word_start = block * Self::SUPER_WORDS;
262        let word_end = (word_start + Self::SUPER_WORDS).min(self.words.len() - 1);
263        let mut word = word_start;
264        let mut right = word_end;
265        while right - word > 1 {
266            let middle = word.midpoint(right);
267            let zeros = (middle - word_start) * Self::WORD_SIZE - self.sub_prefix[middle] as usize;
268            if zeros <= rank {
269                word = middle;
270            } else {
271                right = middle;
272            }
273        }
274        rank -= (word - word_start) * Self::WORD_SIZE - self.sub_prefix[word] as usize;
275        (word, rank)
276    }
277}
278impl RankSelectDictionaries for BitVector {
279    fn bit_length(&self) -> usize {
280        self.len
281    }
282
283    #[inline]
284    fn access(&self, k: usize) -> bool {
285        debug_assert!(k < self.len);
286        self.words[k / Self::WORD_SIZE] & (1u64 << (k % Self::WORD_SIZE)) != 0
287    }
288    fn access_rank1(&self, k: usize) -> (bool, usize) {
289        debug_assert!(k <= self.len);
290        let word = k / Self::WORD_SIZE;
291        let offset = k % Self::WORD_SIZE;
292        let bits = self.words[word];
293        (
294            bits & (1u64 << offset) != 0,
295            self.super_prefix[word / Self::SUPER_WORDS]
296                + self.sub_prefix[word] as usize
297                + (bits & !(u64::MAX << offset)).count_ones() as usize,
298        )
299    }
300    fn rank1(&self, k: usize) -> usize {
301        self.access_rank1(k).1
302    }
303    fn select1(&self, k: usize) -> Option<usize> {
304        if self.sum <= k {
305            return None;
306        }
307        let (word, rank) = self.locate_one(k);
308        Some(word * Self::WORD_SIZE + self.select_word(self.words[word], rank))
309    }
310    fn select0(&self, k: usize) -> Option<usize> {
311        if self.len - self.sum <= k {
312            return None;
313        }
314        let (word, rank) = self.locate_zero(k);
315        let mut bits = !self.words[word];
316        if word + 1 == self.words.len() - 1 && !self.len.is_multiple_of(Self::WORD_SIZE) {
317            bits &= u64::MAX >> (Self::WORD_SIZE - self.len % Self::WORD_SIZE);
318        }
319        Some(word * Self::WORD_SIZE + self.select_word(bits, rank))
320    }
321}
322impl FromIterator<bool> for BitVector {
323    fn from_iter<T: IntoIterator<Item = bool>>(iter: T) -> Self {
324        let iter = iter.into_iter();
325        let (lower, upper) = iter.size_hint();
326        let capacity = match upper {
327            Some(upper) => upper,
328            None => lower,
329        };
330        let mut words = Vec::with_capacity(capacity.div_ceil(Self::WORD_SIZE) + 1);
331        let mut word = 0u64;
332        let mut word_len = 0;
333        let mut len = 0;
334        for bit in iter {
335            word |= (bit as u64) << word_len;
336            word_len += 1;
337            len += 1;
338            if word_len == Self::WORD_SIZE {
339                words.push(word);
340                word = 0;
341                word_len = 0;
342            }
343        }
344        if word_len != 0 {
345            words.push(word);
346        }
347        Self::from_words(words, len)
348    }
349}

358    #[test]
359    fn test_rank_select_word() {
360        const WORD_SIZE: usize = u64::BITS as usize;
361        let mut rng = Xorshift::default();
362        for x in rng.random_iter(0u64..).take(Q) {
363            let word = x;
364            for k in 0..=WORD_SIZE {
365                assert_eq!(x.rank1(k), (0..k).filter(|&i| x.access(i)).count());
366                assert_eq!(x.rank0(k), (0..k).filter(|&i| !x.access(i)).count());
367                if k < word.count_ones() as usize {
368                    assert_eq!(select_word_scalar(word, k), word.select1(k).unwrap());
369                }
370                if let Some(i) = x.select1(k) {
371                    assert_eq!((0..i).filter(|&j| x.access(j)).count(), k);
372                    assert!(x.access(i));
373                } else {
374                    assert!(x.rank1(WORD_SIZE) <= k);
375                }
376                if let Some(i) = x.select0(k) {
377                    assert_eq!((0..i).filter(|&j| !x.access(j)).count(), k);
378                    assert!(!x.access(i));
379                } else {
380                    assert!(x.rank0(WORD_SIZE) <= k);
381                }
382            }
383        }
384    }

386    #[test]
387    fn test_rank_select_bit_vector() {
388        let mut rng = Xorshift::default();
389        for len in [
390            0,
391            1,
392            BitVector::WORD_SIZE - 1,
393            BitVector::WORD_SIZE,
394            BitVector::WORD_SIZE + 1,
395            BitVector::WORD_SIZE * BitVector::SUPER_WORDS - 1,
396            BitVector::WORD_SIZE * BitVector::SUPER_WORDS,
397            BitVector::WORD_SIZE * BitVector::SUPER_WORDS + 1,
398            BitVector::WORD_SIZE * (BitVector::SUPER_WORDS * 2 + 17),
399        ] {
400            for pattern in 0..3 {
401                let bits: Vec<_> = (0..len)
402                    .map(|index| match pattern {
403                        0 => rng.rand(5) != 0,
404                        1 => index.is_multiple_of(BitVector::WORD_SIZE * 3 + 1),
405                        _ => !index.is_multiple_of(BitVector::WORD_SIZE * 3 + 1),
406                    })
407                    .collect();
408                let mut pushed = BitVector::with_capacity(len);
409                for &bit in &bits {
410                    pushed.push(bit);
411                }
412                let collected: BitVector = bits.iter().copied().collect();
413                let split = len / 2;
414                let mut extended: BitVector = bits[..split].iter().copied().collect();
415                for &bit in &bits[split..] {
416                    extended.push(bit);
417                }
418                let split = len.saturating_sub(1);
419                let mut appended: BitVector = bits[..split].iter().copied().collect();
420                for &bit in &bits[split..] {
421                    appended.push(bit);
422                }
423                for actual in [pushed, collected, extended, appended] {
424                    let mut rank1 = 0;
425                    for (index, &bit) in bits.iter().enumerate() {
426                        assert_eq!(actual.access(index), bit);
427                        assert_eq!(actual.access_rank1(index), (bit, rank1));
428                        rank1 += bit as usize;
429                    }
430                    for end in [0, len / 3, len / 2, len] {
431                        assert_eq!(
432                            actual.rank1(end),
433                            bits[..end].iter().filter(|&&bit| bit).count()
434                        );
435                        assert_eq!(
436                            actual.rank0(end),
437                            bits[..end].iter().filter(|&&bit| !bit).count()
438                        );
439                    }
440                    let ones: Vec<_> = bits
441                        .iter()
442                        .enumerate()
443                        .filter_map(|(index, &bit)| bit.then_some(index))
444                        .collect();
445                    let zeros: Vec<_> = bits
446                        .iter()
447                        .enumerate()
448                        .filter_map(|(index, &bit)| (!bit).then_some(index))
449                        .collect();
450                    for rank in 0..=ones.len() {
451                        assert_eq!(actual.select1(rank), ones.get(rank).copied());
452                    }
453                    for rank in 0..=zeros.len() {
454                        assert_eq!(actual.select0(rank), zeros.get(rank).copied());
455                    }
456                }
457            }
458        }
459    }
