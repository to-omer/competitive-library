5#[repr(C, align(64))]
6#[derive(Clone, Debug)]
7struct PrefixBlock<T, const B: usize>([T; B]);
8
9macro_rules! define_dary_prefix_sum_tree {
10    (
11        $name:ident,
12        $value:ty,
13        $branch:expr,
14        $add_avx2:ident,
15        $first_gt_avx2:ident,
16        $add_avx512:ident,
17        $first_gt_avx512:ident
18    ) => {
19        /// A cache-line-oriented d-ary tree for point updates, prefix sums, and prefix searches.
20        ///
21        /// `BinaryIndexedTree` has a denser layout and can be preferable for small workloads.
22        /// Use this type for repeated updates and prefix searches, especially with SIMD support.
23        #[derive(Clone, Debug)]
24        pub struct $name {
25            levels: Vec<Vec<PrefixBlock<$value, $branch>>>,
26            len: usize,
27            total: $value,
28            partition_valid: bool,
29            #[cfg(target_arch = "x86_64")]
30            backend: SimdBackend,
31        }
32
33        impl $name {
34            pub fn new(len: usize) -> Self {
35                Self::zeroed(len, simd_backend())
36            }
37
38            pub fn from_slice(values: &[$value]) -> Self {
39                Self::build(values, simd_backend())
40            }
41
42            #[inline]
43            pub fn len(&self) -> usize {
44                self.len
45            }
46
47            #[inline]
48            pub fn is_empty(&self) -> bool {
49                self.len == 0
50            }
51
52            /// Adds `value` at `index`. Arithmetic is wrapping.
53            #[inline]
54            pub fn update(&mut self, index: usize, value: $value) {
55                assert!(index < self.len);
56                self.add(index, value);
57                self.partition_valid &= self.total.checked_add(value).is_some();
58                self.total = self.total.wrapping_add(value);
59            }
60
61            /// Replaces the value at `index`. Arithmetic is wrapping.
62            #[inline]
63            pub fn set(&mut self, index: usize, value: $value) {
64                let previous = self.get(index);
65                self.add(index, value.wrapping_sub(previous));
66                self.partition_valid &= self
67                    .total
68                    .checked_sub(previous)
69                    .and_then(|total| total.checked_add(value))
70                    .is_some();
71                self.total = self.total.wrapping_sub(previous).wrapping_add(value);
72            }
73
74            /// Returns the wrapping sum of `0..end`.
75            #[inline]
76            pub fn accumulate0(&self, mut end: usize) -> $value {
77                assert!(end <= self.len);
78                if end == self.len {
79                    return self.total;
80                }
81                let mut result: $value = 0;
82                for level in &self.levels {
83                    let block = end / $branch;
84                    let lane = end % $branch;
85                    if lane != 0 {
86                        // SAFETY: `end < len`; mapping it upward remains within every level.
87                        let value = unsafe { level.get_unchecked(block).0.get_unchecked(lane - 1) };
88                        result = result.wrapping_add(*value);
89                    }
90                    end = block;
91                }
92                result
93            }
94
95            /// Returns the wrapping sum of `0..=index`.
96            #[inline]
97            pub fn accumulate(&self, index: usize) -> $value {
98                self.accumulate0(index + 1)
99            }
100
101            /// Returns the wrapping sum of `left..right`.
102            #[inline]
103            pub fn fold(&self, left: usize, right: usize) -> $value {
104                assert!(left <= right && right <= self.len);
105                self.accumulate0(right).wrapping_sub(self.accumulate0(left))
106            }
107
108            #[inline]
109            pub fn get(&self, index: usize) -> $value {
110                self.fold(index, index + 1)
111            }
112
113            #[inline]
114            pub fn fold_all(&self) -> $value {
115                self.total
116            }
117
118            /// Returns the number of leading values whose inclusive prefix sum is at most `value`.
119            ///
120            /// # Panics
121            ///
122            /// Panics if a prefix sum has overflowed.
123            #[inline]
124            pub fn partition_point_acc(&self, value: $value) -> usize {
125                assert!(self.partition_valid, "prefix sum overflowed");
126                if value >= self.total {
127                    return self.len;
128                }
129                #[cfg(target_arch = "x86_64")]
130                return match self.backend {
131                    SimdBackend::Scalar => self.partition_point_scalar(value),
132                    // SAFETY: `simd_backend` only selects supported instruction sets. Tests and
133                    // standalone benchmarks pass supported backends to the private constructor.
134                    SimdBackend::Avx2 => unsafe { self.partition_point_avx2(value) },
135                    // SAFETY: same as above.
136                    SimdBackend::Avx512 => unsafe { self.partition_point_avx512(value) },
137                };
138                #[cfg(not(target_arch = "x86_64"))]
139                self.partition_point_scalar(value)
140            }
141
142            fn zeroed(len: usize, backend: SimdBackend) -> Self {
143                let _ = &backend;
144                let mut levels = Vec::new();
145                let mut level_len = len;
146                while level_len != 0 {
147                    level_len = level_len.div_ceil($branch);
148                    levels.push(vec![PrefixBlock([0; $branch]); level_len]);
149                    if level_len == 1 {
150                        break;
151                    }
152                }
153                Self {
154                    levels,
155                    len,
156                    total: 0,
157                    partition_valid: true,
158                    #[cfg(target_arch = "x86_64")]
159                    backend,
160                }
161            }
162
163            fn build(values: &[$value], backend: SimdBackend) -> Self {
164                let _ = &backend;
165                let mut levels = Vec::new();
166                let mut partition_valid = true;
167                let mut current = Vec::with_capacity(values.len().div_ceil($branch));
168                let mut blocks = Vec::with_capacity(current.capacity());
169                for chunk in values.chunks($branch) {
170                    let mut prefix = [0; $branch];
171                    let mut sum: $value = 0;
172                    for (index, &value) in chunk.iter().enumerate() {
173                        partition_valid &= sum.checked_add(value).is_some();
174                        sum = sum.wrapping_add(value);
175                        prefix[index] = sum;
176                    }
177                    prefix[chunk.len()..].fill(sum);
178                    blocks.push(PrefixBlock(prefix));
179                    current.push(sum);
180                }
181                if !blocks.is_empty() {
182                    levels.push(blocks);
183                }
184                while current.len() > 1 {
185                    let mut blocks = Vec::with_capacity(current.len().div_ceil($branch));
186                    for chunk in current.chunks($branch) {
187                        let mut prefix = [0; $branch];
188                        let mut sum: $value = 0;
189                        for (index, &value) in chunk.iter().enumerate() {
190                            partition_valid &= sum.checked_add(value).is_some();
191                            sum = sum.wrapping_add(value);
192                            prefix[index] = sum;
193                        }
194                        prefix[chunk.len()..].fill(sum);
195                        blocks.push(PrefixBlock(prefix));
196                    }
197                    current = blocks.iter().map(|block| block.0[$branch - 1]).collect();
198                    levels.push(blocks);
199                }
200                Self {
201                    levels,
202                    len: values.len(),
203                    total: current.first().copied().unwrap_or(0),
204                    partition_valid,
205                    #[cfg(target_arch = "x86_64")]
206                    backend,
207                }
208            }
209
210            #[inline]
211            fn add(&mut self, index: usize, value: $value) {
212                #[cfg(target_arch = "x86_64")]
213                match self.backend {
214                    SimdBackend::Scalar => self.add_scalar(index, value),
215                    // SAFETY: `simd_backend` only selects supported instruction sets. Tests and
216                    // standalone benchmarks pass supported backends to the private constructor.
217                    SimdBackend::Avx2 => unsafe { self.add_avx2(index, value) },
218                    // SAFETY: same as above.
219                    SimdBackend::Avx512 => unsafe { self.add_avx512(index, value) },
220                }
221                #[cfg(not(target_arch = "x86_64"))]
222                self.add_scalar(index, value);
223            }
224
225            #[inline]
226            fn add_scalar(&mut self, mut index: usize, value: $value) {
227                for level in &mut self.levels {
228                    let block = index / $branch;
229                    let lane = index % $branch;
230                    // SAFETY: the public methods validate the leaf index; construction fixes the
231                    // mapping from every child block to its parent.
232                    let prefix = &mut unsafe { level.get_unchecked_mut(block) }.0;
233                    for prefix in &mut prefix[lane..] {
234                        *prefix = prefix.wrapping_add(value);
235                    }
236                    index = block;
237                }
238            }
239
240            #[cfg(target_arch = "x86_64")]
241            #[target_feature(enable = "avx2")]
242            unsafe fn add_avx2(&mut self, mut index: usize, value: $value) {
243                for level in &mut self.levels {
244                    let block = index / $branch;
245                    let lane = index % $branch;
246                    let prefix = &mut unsafe { level.get_unchecked_mut(block) }.0;
247                    unsafe { simd::$add_avx2(prefix, lane, value) };
248                    index = block;
249                }
250            }
251
252            #[cfg(target_arch = "x86_64")]
253            #[target_feature(enable = "avx512f")]
254            unsafe fn add_avx512(&mut self, mut index: usize, value: $value) {
255                for level in &mut self.levels {
256                    let block = index / $branch;
257                    let lane = index % $branch;
258                    let prefix = &mut unsafe { level.get_unchecked_mut(block) }.0;
259                    unsafe { simd::$add_avx512(prefix, lane, value) };
260                    index = block;
261                }
262            }
263
264            #[inline(always)]
265            fn partition_point_by<F>(&self, mut value: $value, mut first_gt: F) -> usize
266            where
267                F: FnMut(&[$value; $branch], $value) -> usize,
268            {
269                let mut node = 0;
270                for level in self.levels.iter().rev() {
271                    // SAFETY: `value < total` and non-overflowing prefixes select a real child at
272                    // every level.
273                    let prefix = &unsafe { level.get_unchecked(node) }.0;
274                    let lane = first_gt(prefix, value);
275                    if lane != 0 {
276                        value = value.wrapping_sub(unsafe { *prefix.get_unchecked(lane - 1) });
277                    }
278                    node = node * $branch + lane;
279                }
280                node.min(self.len)
281            }
282
283            #[inline(always)]
284            fn partition_point_scalar(&self, value: $value) -> usize {
285                self.partition_point_by(value, |prefix, value| {
286                    prefix.partition_point(|&sum| sum <= value)
287                })
288            }
289
290            #[cfg(target_arch = "x86_64")]
291            #[target_feature(enable = "avx2")]
292            unsafe fn partition_point_avx2(&self, value: $value) -> usize {
293                self.partition_point_by(value, |prefix, value| unsafe {
294                    simd::$first_gt_avx2(prefix, value)
295                })
296            }
297
298            #[cfg(target_arch = "x86_64")]
299            #[target_feature(enable = "avx512f")]
300            unsafe fn partition_point_avx512(&self, value: $value) -> usize {
301                self.partition_point_by(value, |prefix, value| unsafe {
302                    simd::$first_gt_avx512(prefix, value)
303                })
304            }
305        }
306    };
307}

352    #[test]
353    fn test_dary_prefix_sum_tree() {
354        let mut rng = Xorshift::default();
355        for len in [0, 1, 7, 8, 15, 16, 17, 255, 256, 257, 4095, 4096, 4097] {
356            let values: Vec<_> = (0..len).map(|_| rng.rand(8) as u32).collect();
357            for backend in backends() {
358                let mut actual = DaryPrefixSumTreeU32::build(&values, backend);
359                let mut expected = values.clone();
360                for step in 0..500 {
361                    if len != 0 {
362                        let index = rng.rand(len as u64) as usize;
363                        if step % 3 == 0 {
364                            let value = rng.rand(32) as u32;
365                            actual.set(index, value);
366                            expected[index] = value;
367                        } else {
368                            let value = rng.rand(8) as u32;
369                            actual.update(index, value);
370                            expected[index] += value;
371                        }
372                    }
373                    for end in [0, len / 2, len] {
374                        assert_eq!(actual.accumulate0(end), expected[..end].iter().sum());
375                    }
376                    if len != 0 {
377                        let left = rng.rand(len as u64) as usize;
378                        let right = left + rng.rand((len - left + 1) as u64) as usize;
379                        assert_eq!(actual.fold(left, right), expected[left..right].iter().sum());
380                        assert_eq!(actual.get(left), expected[left]);
381                    }
382                    let mut sum = 0;
383                    let prefix: Vec<_> = expected
384                        .iter()
385                        .map(|&value| {
386                            sum += value;
387                            sum
388                        })
389                        .collect();
390                    for value in [0, sum / 2, sum] {
391                        assert_eq!(
392                            actual.partition_point_acc(value),
393                            prefix.partition_point(|&prefix| prefix <= value)
394                        );
395                    }
396                }
397            }
398        }
399
400        let values: Vec<_> = (0..513).map(|_| rng.rand(16)).collect();
401        for backend in backends() {
402            let mut actual = DaryPrefixSumTreeU64::build(&values, backend);
403            let mut expected = values.clone();
404            for _ in 0..1000 {
405                let index = rng.rand(expected.len() as u64) as usize;
406                let value = rng.rand(16);
407                actual.update(index, value);
408                expected[index] += value;
409                let end = rng.rand(expected.len() as u64 + 1) as usize;
410                assert_eq!(actual.accumulate0(end), expected[..end].iter().sum());
411            }
412            assert_eq!(actual.fold_all(), expected.iter().sum());
413        }
414    }
