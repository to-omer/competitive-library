#[cfg_attr(any(),rustfmt::skip)]#[doc=" binary gcd"]pub fn gcd(original_a:u64,original_b:u64)->u64{let(mut a,mut b)=(original_a,original_b);if a==0{return b;}if b==0{return a;}let u=a.trailing_zeros();let v=b.trailing_zeros();a>>=u;b>>=v;let k=u.min(v);while a!=b{if a<b{std::mem::swap(&mut a,&mut b);}a-=b;a>>=a.trailing_zeros();}a<<k}
