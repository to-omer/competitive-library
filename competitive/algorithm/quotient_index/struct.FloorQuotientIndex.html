292        #[cfg(verus_keep_ghost)]
293        proof! { floor_membership(self.num, value); }
294        if value == usize::MAX {
295            #[cfg(verus_keep_ghost)]
296            proof! {
297                assert forall|k: int| 1 <= k <= self.num implies
298                    ((k == 1) == (#[trigger] (self.num as int / k) == value)) by {
299                    if k > 1 { vstd::arithmetic::div_mod::lemma_div_decreases(self.num as int, k); }
300                }
301                #[verifier::proof_note("The maximum usize quotient occurs only for numerator usize::MAX and key 1")]
302                assert(self.num == usize::MAX && forall|k: int| 1 <= k <= self.num ==>
303                    ((k == 1) == (#[trigger] (self.num as int / k) == value)));
304            }
305            return 1..=1;
306        }
307        let start = (self.num / (value + 1)) + 1;
308        let end = self.num / value;
309        #[cfg(verus_keep_ghost)]
310        proof! {
311            vstd::arithmetic::div_mod::lemma_div_nonincreasing(self.num as int, value as int);
312            assert forall|k: int| 1 <= k <= self.num implies
313                ((start <= k <= end) == (#[trigger] (self.num as int / k) == value)) by {

316            #[verifier::proof_note("For every present floor quotient below usize::MAX, value_to_key_unchecked returns a nonempty interval containing exactly all keys producing that quotient")]

318                ((start <= k <= end) == (#[trigger] (self.num as int / k) == value)));
319        }
320        start..=end
321    }
