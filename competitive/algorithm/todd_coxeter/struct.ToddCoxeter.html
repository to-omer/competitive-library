pub fn solve(self) -> Vec<Vec<usize>>

Returns the right generator transitions table[state][generator]. State 0 is the identity; states follow breadth-first order with ascending generators. Use OperationTable::from_transitions to multiply arbitrary elements.

The presented monoid must be finite.

With P created states, K generators, and L the sum of 1 + lhs.len() + rhs.len() over the equations, time is O(P(1 + K + L) alpha(P)) and space O(P(1 + K) + L). P can greatly exceed the final number of states and depends on equation order.

use competitive::algorithm::ToddCoxeter;
let mut tc = ToddCoxeter::new(2);
tc.add_relation([0, 0], []);
tc.add_relation([1, 1, 1], []);
tc.add_relation([0, 1, 0, 1], []);
let table = tc.solve();
assert_eq!(table.len(), 6);
assert_eq!([1, 0, 1].into_iter().fold(0, |q, c| table[q][c]), table[0][0]);

54    pub fn solve(mut self) -> Vec<Vec<usize>> {
55        let relations = take(&mut self.relations);
56        let mut stack = Vec::new();
57        let alphabet_size = self.table[0].len();
58        let mut state = 0;
59        while state < self.table.len() {
60            if self.uf.find_root(state) == state {
61                for (lhs, rhs) in &relations {
62                    let (lc, lp) = lhs
63                        .split_last()
64                        .map_or((None, &[][..]), |(&c, p)| (Some(c), p));
65                    let (rc, rp) = rhs
66                        .split_last()
67                        .map_or((None, &[][..]), |(&c, p)| (Some(c), p));
68                    let left = self.follow(state, lp);
69                    let right = self.follow(state, rp);
70                    // Both prefixes must exist before reading either last edge (e.g. a = aa).
71                    let x = lc.map_or(left, |c| self.table[left][c]);
72                    let y = rc.map_or(right, |c| self.table[right][c]);
73                    if x == usize::MAX {
74                        let y = if y == usize::MAX {
75                            self.define(right, rc.unwrap())
76                        } else {
77                            y
78                        };
79                        self.table[left][lc.unwrap()] = y;
80                    } else if y == usize::MAX {
81                        self.table[right][rc.unwrap()] = x;
82                    } else {
83                        self.merge(x, y, &mut stack);
84                    }
85                }
86                for c in 0..alphabet_size {
87                    self.define(state, c);
88                }
89            }
90            state += 1;
91        }
92
93        let identity = self.uf.find_root(0);
94        let mut order = vec![identity];
95        let mut index = vec![usize::MAX; self.table.len()];
96        index[identity] = 0;
97        let mut table = Vec::new();
98        let mut i = 0;
99        while i < order.len() {
100            let mut row = Vec::with_capacity(alphabet_size);
101            for &target in &self.table[order[i]] {
102                let target = self.uf.find_root(target);
103                if index[target] == usize::MAX {
104                    index[target] = order.len();
105                    order.push(target);
106                }
107                row.push(index[target]);
108            }
109            table.push(row);
110            i += 1;
111        }
112        table
113    }

115    fn define(&mut self, state: usize, c: usize) -> usize {
116        let state = self.uf.find_root(state);
117        if self.table[state][c] == usize::MAX {
118            assert!(self.table.len() < i32::MAX as usize);
119            self.table[state][c] = self.table.len();
120            self.table.push(vec![usize::MAX; self.table[0].len()]);
121            self.uf.push();
122        }
123        self.uf.find_root(self.table[state][c])
124    }

126    fn follow(&mut self, state: usize, word: &[usize]) -> usize {
127        let mut state = self.uf.find_root(state);
128        for &c in word {
129            state = self.define(state, c);
130        }
131        state
132    }
