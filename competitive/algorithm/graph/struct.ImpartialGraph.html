333    pub fn best_move(&mut self, state: &[S]) -> Option<Neighbor<Vec<S>, (usize, L)>> {
334        if state.is_empty() {
335            return None;
336        }
337        if let [single] = state {
338            let mut query: GameAnalysis<'_, _, S, GameOutcome, MISERE> =
339                GameAnalysis::new(self.game, |g: &mut ImpartialGraph<'_, G>, s| {
340                    g.outcome(s, MISERE)
341                });
342            return query
343                .best_move(single)
344                .map(|next| Neighbor::new(vec![next.to], (0, next.label)));
345        }
346        let outcome = self.eval(state);
347        if outcome == GameOutcome::Lose {
348            return None;
349        }
350        let mut indices = self.game.sum_indices(state);
351        let game = &self.game;
352        if MISERE {
353            let key = GraphSums::canonical(indices.clone(), &game.index.as_ref().unwrap().edges);
354            let winner = game.sums.outcomes.winning[game.sums.indices[&key]];
355            for (i, &component) in state.iter().enumerate() {
356                let original = indices[i];
357                for next in game.graph.neighbors(component) {
358                    indices[i] = game.index_of(next.to);
359                    let key =
360                        GraphSums::canonical(indices.clone(), &game.index.as_ref().unwrap().edges);
361                    let id = game.sums.indices[&key];
362                    if Some(id) == winner
363                        || outcome == GameOutcome::Draw
364                            && game.sums.outcomes.values[id] == GameOutcome::Draw
365                    {
366                        let mut to = state.to_vec();
367                        to[i] = next.to;
368                        return Some(Neighbor::new(to, (i, next.label)));
369                    }
370                }
371                indices[i] = original;
372            }
373            return None;
374        }
375        let mut xor = 0;
376        let mut infinite = None;
377        let mut multiple = false;
378        for (i, &v) in indices.iter().enumerate() {
379            if let Some((value, _)) = game.generalized_grundy[v] {
380                xor ^= value;
381            } else if infinite.replace(i).is_some() {
382                multiple = true;
383                break;
384            }
385        }
386        if outcome == GameOutcome::Draw {
387            let i = infinite.unwrap();
388            let v = indices[i];
389            let edges = &game.index.as_ref().unwrap().edges;
390            let target = (!multiple).then(|| {
391                let values = edges[v]
392                    .iter()
393                    .filter_map(|&u| game.generalized_grundy[u].map(|(g, _)| g))
394                    .collect();
395                game_mex(values)
396            });
397            // Preserve two infinite components, or keep the finite XOR outside
398            // the remaining infinite component's finite follower values.
399            if target.is_none_or(|g| g == xor) {
400                let to = edges[v]
401                    .iter()
402                    .copied()
403                    .find(|&u| {
404                        game.generalized_grundy[u].is_none()
405                            && target.is_none_or(|g| {
406                                !edges[u].iter().any(|&w| {
407                                    game.generalized_grundy[w].is_some_and(|(value, _)| value == g)
408                                })
409                            })
410                    })
411                    .unwrap();
412                let next = game
413                    .graph
414                    .neighbors(state[i])
415                    .find(|next| game.index_of(next.to) == to)
416                    .unwrap();
417                let mut to = state.to_vec();
418                to[i] = next.to;
419                return Some(Neighbor::new(to, (i, next.label)));
420            }
421            // The current finite XOR exceeds this mex, so a finite component
422            // can decrease it to the mex without moving the infinite component.
423            xor ^= target.unwrap();
424            infinite = None;
425        }
426        // Reduce the largest (Grundy, counter) pair, then minimize the
427        // destination counter so repeated winning moves terminate.
428        let (i, target) = if let Some(i) = infinite {
429            (i, xor)
430        } else {
431            let (i, (g, _)) = indices
432                .iter()
433                .enumerate()
434                .filter_map(|(i, &v)| game.generalized_grundy[v].map(|g| (i, g)))
435                .filter(|(_, (g, _))| g ^ xor < *g)
436                .max_by_key(|&(i, g)| (g, Reverse(i)))?;
437            (i, g ^ xor)
438        };
439        let next = game
440            .graph
441            .neighbors(state[i])
442            .filter_map(|next| {
443                let (g, counter) = game.generalized_grundy[game.index_of(next.to)]?;
444                (g == target).then_some((next, counter))
445            })
446            .min_by_key(|&(_, counter)| counter)?
447            .0;
448        let mut to = state.to_vec();
449        to[i] = next.to;
450        Some(Neighbor::new(to, (i, next.label)))
451    }

458    fn grundy(&mut self, state: &G::Vertex) -> u64 {
459        self.prepare();
460        let root = self.index_of(*state);
461        let index = self.index.as_ref().unwrap();
462        self.grundy.resize(index.edges.len(), None);
463        if let Some(value) = self.grundy[root] {
464            return value;
465        }
466        let mut active = HashSet::from([root]);
467        let mut stack = vec![(root, index.edges[root].iter())];
468        while let Some((v, moves)) = stack.last_mut() {
469            if let Some(&u) = moves.next() {
470                if self.grundy[u].is_none() {
471                    assert!(active.insert(u), "cycle in a finite game");
472                    stack.push((u, index.edges[u].iter()));
473                }
474            } else {
475                let values = index.edges[*v]
476                    .iter()
477                    .map(|&u| self.grundy[u].unwrap())
478                    .collect();
479                self.grundy[*v] = Some(game_mex(values));
480                active.remove(v);
481                stack.pop();
482            }
483        }
484        self.grundy[root].unwrap()
485    }
486
487    fn outcome(&mut self, state: &G::Vertex, misere: bool) -> GameOutcome {
488        self.prepare();
489        let index = self.index.as_ref().unwrap();
490        let outcome = &mut self.outcomes[usize::from(misere)];
491        if outcome.values.is_empty() {
492            outcome.extend(&index.edges, &index.reverse, misere);
493        }
494        self.outcomes[usize::from(misere)].values[self.index_of(*state)]
495    }

498        state.iter().fold(0, |value, s| value ^ self.grundy(s))

462        self.grundy.resize(index.edges.len(), None);

446            .min_by_key(|&(_, counter)| counter)?
447            .0;

449        to[i] = next.to;
