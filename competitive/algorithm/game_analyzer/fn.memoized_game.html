213fn memoized_game<S, V, I, A, T>(
214    state: &S,
215    cache: &mut HashMap<S, V>,
216    active: &mut HashSet<S>,
217    expand: &mut impl FnMut(&S) -> ControlFlow<V, (I, A)>,
218    advance: &mut impl FnMut(&S, &mut I) -> Option<(S, T)>,
219    feed: &mut impl FnMut(&mut A, T, V) -> ControlFlow<V>,
220    finish: &mut impl FnMut(&S, A) -> V,
221) -> V
222where
223    S: Clone + Eq + Hash,
224    V: Clone,
225{
226    if let Some(value) = cache.get(state) {
227        return value.clone();
228    }
229    assert!(active.insert(state.clone()), "cycle in a finite game");
230    let value = match expand(state) {
231        ControlFlow::Break(value) => value,
232        ControlFlow::Continue((mut moves, mut accumulator)) => 'search: {
233            while let Some((next, edge)) = advance(state, &mut moves) {
234                let child = memoized_game(&next, cache, active, expand, advance, feed, finish);
235                if let ControlFlow::Break(value) = feed(&mut accumulator, edge, child) {
236                    break 'search value;
237                }
238            }
239            finish(state, accumulator)
240        }
241    };
242    active.remove(state);
243    cache.insert(state.clone(), value.clone());
244    value
245}

233    /// Undominated moves for the selected player; incomparable alternatives remain.
234    pub fn undominated_moves(&mut self, state: &S, player: Player) -> Vec<Neighbor<S, L>> {
235        let moves = self.moves(state, player);
236        let preferred = if player == Player::Left {
237            Ordering::Greater
238        } else {
239            Ordering::Less
240        };
241        let best = moves.iter().try_fold(None, |best, m| {
242            let (q, k, n) = m.value.as_number_up_nimber()?;
243            let pair = (q, k);
244            Some(Some(match best {
245                Some((previous, nimber)) => match pair.cmp(&previous) {
246                    Ordering::Equal => (previous, if nimber == Some(n) { nimber } else { None }),
247                    order if order == preferred => (pair, Some(n)),
248                    _ => (previous, nimber),
249                },
250                None => (pair, Some(n)),
251            }))
252        });
253        if let Some(best) = best {
254            return moves
255                .into_iter()
256                .filter(|m| {
257                    let (q, k, n) = m.value.as_number_up_nimber().unwrap();
258                    best.is_some_and(|((b, l), nimber)| {
259                        q == b && (k == l || k.abs_diff(l) == 1 && nimber == Some(n ^ 1))
260                    })
261                })
262                .map(|m| m.next)
263                .collect();
264        }
265        let mut dominated = vec![false; moves.len()];
266        for i in 0..moves.len() {
267            if dominated[i] {
268                continue;
269            }
270            for j in i + 1..moves.len() {
271                if dominated[j] {
272                    continue;
273                }
274                let order = moves[i].value.partial_cmp(&moves[j].value);
275                if order == Some(preferred) {
276                    dominated[j] = true;
277                } else if order == Some(preferred.reverse()) {
278                    dominated[i] = true;
279                    break;
280                }
281            }
282        }
283        moves
284            .into_iter()
285            .zip(dominated)
286            .filter(|(_, d)| !d)
287            .map(|(m, _)| m.next)
288            .collect()
289    }

400            &mut |(left, right), player, value| {
401                if player == Player::Left {
402                    left.push(value)
403                } else {
404                    right.push(value)
405                }
406                ControlFlow::Continue(())
407            },
408            &mut |_, (left, right)| GameValue::from_options(left, right),
409        )
410    }
