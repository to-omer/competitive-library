443    /// `index` must satisfy `index < self.len()`.
444    pub unsafe fn index_to_key_unchecked(&self, index: usize) -> RangeInclusive<usize> {
445        if index < self.pivot {
446            if index == 0 {
447                return self.num..=self.num;
448            }
449            let start = self.num.div_ceil(index + 1);
450            let end = (self.num - 1) / index;
451            start..=end
452        } else {
453            let key = self.len() - index;
454            key..=key
455        }
456    }
