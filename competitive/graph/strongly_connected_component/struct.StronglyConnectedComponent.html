111    fn decompose(
112        vertices: &mut [usize],
113        tail: usize,
114        rows: &mut [(usize, usize)],
115        mut roots: Vec<usize>,
116        next_edge: impl Fn(usize) -> (usize, usize),
117    ) -> Vec<usize> {
118        let n = vertices.len();
119        let mut starts = Vec::new();
120        roots.clear();
121        let mut branches = Vec::new();
122        let mut open = tail;
123        let mut end = n;
124        for root in 0..n {
125            if rows[root].0 > n {
126                continue;
127            }
128            let mut v = root;
129            loop {
130                if v < n {
131                    let (first, rest) = rows[v];
132                    if first <= n {
133                        // The first edge is consumed on discovery. Its slot then holds the
134                        // complemented one-based stack position, or MAX for a closed SCC.
135                        // These marks exceed n since rows holds two words per vertex.
136                        vertices[open] = v;
137                        roots.push(open);
138                        open += 1;
139                        rows[v].0 = !open;
140                        if rest != n {
141                            branches.push((open, rest));
142                        }
143                        v = first;
144                        continue;
145                    }
146                    if first != usize::MAX {
147                        let rank = !first;
148                        while roots.last().is_some_and(|&i| i >= rank) {
149                            roots.pop();
150                        }
151                    }
152                }
153                // Roots deeper than the nearest ancestor with unexamined edges
154                // are complete. A vertex's last edge needs no return frame.
155                let parent = branches.last().map_or(0, |&(rank, _)| rank);
156                while roots.last().is_some_and(|&i| i >= parent) {
157                    let first = roots.pop().unwrap();
158                    let len = open - first;
159                    end -= len;
160                    vertices.copy_within(first..open, end);
161                    for &u in &vertices[end..end + len] {
162                        rows[u].0 = usize::MAX;
163                    }
164                    open = first;
165                    starts.push(end);
166                }
167                let Some((_, cursor)) = branches.last_mut() else {
168                    break;
169                };
170                let (to, rest) = next_edge(*cursor);
171                if rest == n {
172                    branches.pop();
173                } else {
174                    *cursor = rest;
175                }
176                v = to;
177            }
178        }
179        starts.reverse();
180        starts
181    }
182}

15    pub fn from_edges(n: usize, edges: impl Iterator<Item = (usize, usize)>) -> Self {
16        // Each block stores the previous cursor followed by neighbors in insertion order.
17        const BLOCK: usize = 4;
18        let mut degree = vec![0usize; n];
19        let mut rows = vec![(n, n); n];
20        let mut overflow = Vec::new();
21        for (u, v) in edges {
22            degree[v] += 1;
23            let (first, rest) = rows[u];
24            let rest = if rest == n {
25                first
26            } else if rest > n && !(rest - n).is_multiple_of(BLOCK) {
27                overflow[rest - n] = first;
28                rest + 1
29            } else {
30                let i = overflow.len();
31                overflow.extend_from_slice(&[0; BLOCK]);
32                overflow[i] = rest;
33                overflow[i + 1] = first;
34                n + i + 2
35            };
36            rows[u] = (v, rest);
37        }
38        // A cursor below n is the last neighbor, n is the end, and larger
39        // values link to earlier edges of the same vertex.
40        let next_edge = |cursor: usize| {
41            if cursor <= n {
42                (cursor, n)
43            } else {
44                let i = cursor - n - 1;
45                let next = if i % BLOCK == 1 {
46                    overflow[i - 1]
47                } else {
48                    cursor - 1
49                };
50                (overflow[i], next)
51            }
52        };
53        let mut vertices = vec![0; n];
54        let mut tail = 0;
55        for (i, &d) in degree.iter().enumerate() {
56            vertices[tail] = i;
57            tail += usize::from(d == 0);
58        }
59        let mut head = 0;
60        while head < tail {
61            let u = vertices[head];
62            head += 1;
63            let (mut v, mut rest) = rows[u];
64            while v < n {
65                degree[v] -= 1;
66                // This edge keeps its target out of the queue until it is processed.
67                vertices[tail] = v;
68                tail += usize::from(degree[v] == 0);
69                (v, rest) = next_edge(rest);
70            }
71            rows[u].0 = usize::MAX;
72        }
73        let mut starts = if tail == n {
74            Vec::new()
75        } else {
76            Self::decompose(&mut vertices, tail, &mut rows, degree, next_edge)
77        };
78        starts.push(n);
79        Self { vertices, starts }
80    }
