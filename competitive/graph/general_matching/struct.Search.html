170    fn complete(mate: &mut [usize], mut count: usize, edges: &[(usize, usize)]) -> (usize, usize) {
171        let size = mate.len();
172        let mut degree = vec![0; size];
173        let mut neighbors_xor = vec![0; size];
174        for &(u, v) in edges {
175            degree[u] += 1;
176            degree[v] += 1;
177            neighbors_xor[u] ^= v;
178            neighbors_xor[v] ^= u;
179        }
180        let upper = degree.iter().filter(|&&d| d != 0).count() / 2;
181        if count == upper {
182            return (count, upper);
183        }
184        if !degree.contains(&1) && degree.contains(&2) {
185            let mut candidate = vec![size; size];
186            let mut candidate_size = 0;
187            for &(u, v) in edges {
188                if degree[u] != 2 || degree[v] != 2 || candidate[u] != size || candidate[v] != size
189                {
190                    continue;
191                }
192                // On a-u-v-b, prefer two disjoint end edges to the middle edge.
193                let a = neighbors_xor[u] ^ v;
194                let b = neighbors_xor[v] ^ u;
195                if a != v && b != u && a != b && candidate[a] == size && candidate[b] == size {
196                    candidate[a] = u;
197                    candidate[u] = a;
198                    candidate[b] = v;
199                    candidate[v] = b;
200                    candidate_size += 2;
201                } else {
202                    candidate[u] = v;
203                    candidate[v] = u;
204                    candidate_size += 1;
205                }
206            }
207            if candidate_size > count {
208                mate.copy_from_slice(&candidate);
209                count = candidate_size;
210                if count == upper {
211                    return (count, upper);
212                }
213            }
214        }
215        for root in 0..size {
216            let mut u = root;
217            while degree[u] == 1 {
218                let v = neighbors_xor[u];
219                if mate[u] == size {
220                    let old = mate[v];
221                    // A partner already peeled from the graph fixes this matching edge.
222                    if old == size || degree[old] != 0 {
223                        if old == size {
224                            count += 1;
225                        } else {
226                            mate[old] = size;
227                        }
228                        mate[u] = v;
229                        mate[v] = u;
230                    }
231                }
232                degree[u] = 0;
233                degree[v] -= 1;
234                neighbors_xor[v] ^= u;
235                u = v;
236            }
237        }
238        let mut free = (0..size)
239            .filter(|&v| mate[v] == size && degree[v] != 0)
240            .count();
241        if free < 2 {
242            return (count, upper);
243        }
244        for &(u, v) in edges.iter().rev() {
245            if mate[u] == size && mate[v] == size {
246                mate[u] = v;
247                mate[v] = u;
248                count += 1;
249                free -= 2;
250                if free < 2 {
251                    return (count, upper);
252                }
253            }
254        }
255        if Self::certifies_maximum(mate, &degree, edges) {
256            return (count, upper);
257        }
258
259        let active: Vec<_> = (0..size)
260            .map(|v| degree[v] != 0 && (mate[v] == size || degree[mate[v]] != 0))
261            .collect();
262        let mut start = vec![0; size + 1];
263        for &(u, v) in edges {
264            if active[u] && active[v] {
265                start[u] += 1;
266                start[v] += 1;
267            }
268        }
269        for v in 0..size {
270            start[v + 1] += start[v];
271        }
272        let mut adj = vec![0; start[size]];
273        for &(u, v) in edges.iter().rev() {
274            if active[u] && active[v] {
275                start[u] -= 1;
276                adj[start[u]] = v;
277                start[v] -= 1;
278                adj[start[v]] = u;
279            }
280        }
281        let mut search = Search {
282            size,
283            start: &start,
284            adj: &adj,
285            mate,
286            matching_size: count,
287            parent: vec![size; size + 1],
288            base: (0..=size).collect(),
289            state: vec![Search::UNSEEN; size],
290            seen: vec![0; size],
291            queue: Vec::new(),
292            timestamp: 0,
293        };
294        search.compute();
295        (search.matching_size, upper)
296    }

458    fn augment_from(&mut self, root: usize) -> bool {
459        self.queue.clear();
460        self.state[root] = Self::OUTER;
461        self.queue.push(root);
462        let mut head = 0;
463        while head < self.queue.len() {
464            let u = self.queue[head];
465            head += 1;
466            for edge in self.start[u]..self.start[u + 1] {
467                let v = self.adj[edge];
468                if self.state[v] == Self::UNSEEN {
469                    self.parent[v] = u;
470                    if self.mate[v] == self.size {
471                        self.augment(u, v);
472                        return true;
473                    }
474                    self.state[v] = Self::INNER;
475                    let v = self.mate[v];
476                    self.state[v] = Self::OUTER;
477                    self.queue.push(v);
478                    if let Some(&end) = self.adj[self.start[v]..self.start[v + 1]]
479                        .iter()
480                        .find(|&&w| self.mate[w] == self.size && self.state[w] == Self::UNSEEN)
481                    {
482                        self.augment(v, end);
483                        return true;
484                    }
485                } else if self.state[v] == Self::OUTER
486                    && self.base[u] != self.base[v]
487                    && find(&mut self.base, u) != find(&mut self.base, v)
488                {
489                    let ancestor = self.lca(u, v);
490                    self.contract(u, v, ancestor);
491                    self.contract(v, u, ancestor);
492                }
493            }
494        }
495        // Edges from this tree to unsearched vertices leave inner vertices.
496        // An alternating path entering it cannot leave through another inner vertex.
497        for &v in &self.queue {
498            self.state[v] = Self::INNER;
499        }
500        false
501    }

358    fn compute(&mut self) {
359        for u in 0..self.size {
360            if self.mate[u] != self.size {
361                continue;
362            }
363            if let Some(&v) = self.adj[self.start[u]..self.start[u + 1]]
364                .iter()
365                .find(|&&v| self.mate[v] == self.size)
366            {
367                self.mate[u] = v;
368                self.mate[v] = u;
369                self.matching_size += 1;
370            } else if let Some((&v, rest)) =
371                self.adj[self.start[u]..self.start[u + 1]].split_first()
372                && rest.iter().all(|&w| w == v)
373            {
374                let old = self.mate[v];
375                self.mate[old] = self.size;
376                self.mate[v] = u;
377                self.mate[u] = v;
378            }
379        }
380        let mut free = (0..self.size)
381            .filter(|&v| self.mate[v] == self.size && self.start[v] != self.start[v + 1])
382            .count();
383        if free < 2 {
384            return;
385        }
386
387        for v in 0..self.size {
388            if self.mate[v] == self.size && self.start[v] != self.start[v + 1] {
389                free -= 1;
390                if self.augment_from(v) {
391                    self.matching_size += 1;
392                    free -= 1;
393                }
394                if free < 2 {
395                    break;
396                }
397            }
398        }
