298    fn certifies_maximum(mate: &[usize], degree: &[usize], edges: &[(usize, usize)]) -> bool {
299        use std::cmp::Ordering::{Greater, Less};
300
301        let size = mate.len();
302        // Deleting the higher-degree end of each unequal pair removes one matched edge
303        // per deleted vertex. The bound is tight if each remaining component has at most
304        // one exposed vertex.
305        let role: Vec<_> = (0..size)
306            .map(|v| {
307                if mate[v] == size {
308                    Less
309                } else {
310                    degree[v].cmp(&degree[mate[v]])
311                }
312            })
313            .collect();
314        let mut base: Vec<_> = (0..size).collect();
315        for &(u, v) in edges {
316            if role[u] == Greater || role[v] == Greater || base[u] == base[v] {
317                continue;
318            }
319            let u = find(&mut base, u);
320            let v = find(&mut base, v);
321            if u == v {
322                continue;
323            }
324            // A component's root is exposed in the remaining matching if any vertex is.
325            match (role[u], role[v]) {
326                (Less, Less) => return false,
327                (_, Less) => base[u] = v,
328                _ => base[v] = u,
329            }
330        }
331        true
332    }

335fn find(base: &mut [usize], mut v: usize) -> usize {
336    while base[v] != v {
337        let parent = base[v];
338        base[v] = base[parent];
339        v = base[v];
340    }
341    v

358    fn compute(&mut self) {
359        for u in 0..self.size {
360            if self.mate[u] != self.size {
361                continue;
362            }
363            if let Some(&v) = self.adj[self.start[u]..self.start[u + 1]]
364                .iter()
365                .find(|&&v| self.mate[v] == self.size)
366            {
367                self.mate[u] = v;
368                self.mate[v] = u;
369                self.matching_size += 1;
370            } else if let Some((&v, rest)) =
371                self.adj[self.start[u]..self.start[u + 1]].split_first()
372                && rest.iter().all(|&w| w == v)
373            {
374                let old = self.mate[v];
375                self.mate[old] = self.size;
376                self.mate[v] = u;
377                self.mate[u] = v;
378            }
379        }
380        let mut free = (0..self.size)
381            .filter(|&v| self.mate[v] == self.size && self.start[v] != self.start[v + 1])
382            .count();
383        if free < 2 {
384            return;
385        }
386
387        for v in 0..self.size {
388            if self.mate[v] == self.size && self.start[v] != self.start[v + 1] {
389                free -= 1;
390                if self.augment_from(v) {
391                    self.matching_size += 1;
392                    free -= 1;
393                }
394                if free < 2 {
395                    break;
396                }
397            }
398        }
399    }

405    fn lca(&mut self, mut u: usize, mut v: usize) -> usize {
406        self.timestamp += 1;
407        u = find(&mut self.base, u);
408        v = find(&mut self.base, v);
409        loop {
410            if u != self.size {
411                if self.seen[u] == self.timestamp {
412                    return u;
413                }
414                self.seen[u] = self.timestamp;
415                u = find(&mut self.base, self.parent[self.mate[u]]);
416            }
417            swap(&mut u, &mut v);
418        }
419    }

421    fn contract(&mut self, mut v: usize, mut child: usize, ancestor: usize) {
422        while find(&mut self.base, v) != ancestor {
423            self.parent[v] = child;
424            child = self.mate[v];
425            if self.state[child] == Self::INNER {
426                self.state[child] = Self::OUTER;
427                self.queue.push(child);
428            }
429            if self.base[v] == v {
430                self.base[v] = ancestor;
431            }
432            if self.base[child] == child {
433                self.base[child] = ancestor;
434            }
435            v = self.parent[child];
436        }
437    }

439    fn augment(&mut self, mut u: usize, mut v: usize) {
440        for &outer in &self.queue {
441            self.base[outer] = outer;
442            self.state[outer] = Self::UNSEEN;
443            let inner = self.mate[outer];
444            if inner != self.size {
445                self.base[inner] = inner;
446                self.state[inner] = Self::UNSEEN;
447            }
448        }
449        while u != self.size {
450            let next = self.mate[u];
451            self.mate[u] = v;
452            self.mate[v] = u;
453            v = next;
454            u = self.parent[v];
455        }
456    }

458    fn augment_from(&mut self, root: usize) -> bool {
459        self.queue.clear();
460        self.state[root] = Self::OUTER;
461        self.queue.push(root);
462        let mut head = 0;
463        while head < self.queue.len() {
464            let u = self.queue[head];
465            head += 1;
466            for edge in self.start[u]..self.start[u + 1] {
467                let v = self.adj[edge];
468                if self.state[v] == Self::UNSEEN {
469                    self.parent[v] = u;
470                    if self.mate[v] == self.size {
471                        self.augment(u, v);
472                        return true;
473                    }
474                    self.state[v] = Self::INNER;
475                    let v = self.mate[v];
476                    self.state[v] = Self::OUTER;
477                    self.queue.push(v);
478                    if let Some(&end) = self.adj[self.start[v]..self.start[v + 1]]
479                        .iter()
480                        .find(|&&w| self.mate[w] == self.size && self.state[w] == Self::UNSEEN)
481                    {
482                        self.augment(v, end);
483                        return true;
484                    }
485                } else if self.state[v] == Self::OUTER
486                    && self.base[u] != self.base[v]
487                    && find(&mut self.base, u) != find(&mut self.base, v)
488                {
489                    let ancestor = self.lca(u, v);
490                    self.contract(u, v, ancestor);
491                    self.contract(v, u, ancestor);
492                }
493            }
494        }
495        // Edges from this tree to unsearched vertices leave inner vertices.
496        // An alternating path entering it cannot leave through another inner vertex.
497        for &v in &self.queue {
498            self.state[v] = Self::INNER;
499        }
500        false
501    }
