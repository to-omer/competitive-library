114    pub fn fold_abelian(&self, mut l: usize, mut r: usize) -> G::T
115    where
116        G: AbelianGroup,
117    {
118        debug_assert!(l <= self.n && r <= self.n);
119        if l == r {
120            return G::unit();
121        }
122        // The prefix above the highest differing bit cancels in an Abelian group.
123        let common = l & !(usize::MAX >> (l ^ r).leading_zeros());
124        let mut left = G::unit();
125        let mut right = G::unit();
126        while l != common {
127            G::operate_assign(&mut left, &self.bit[l]);
128            l &= l - 1;
129        }
130        while r != common {
131            G::operate_assign(&mut right, &self.bit[r]);
132            r &= r - 1;
133        }
134        G::rinv_operate(&right, &left)
