313            } else {

316        }

333                    R::sub_assign(x, &R::mul(&factor, y));
334                }
335            }
336        }

425    pub fn characteristic_polynomial(&mut self) -> Vec<R::T> {
426        let n = self.shape.0;
427        if n == 0 {
428            return vec![R::one()];
429        }
430        assert!(self.data.iter().all(|a| a.len() == n));
431        for j in 0..(n - 1) {
432            if let Some(x) = ((j + 1)..n).find(|&x| !R::is_zero(&self[x][j])) {
433                self.data.swap(j + 1, x);
434                self.data.iter_mut().for_each(|a| a.swap(j + 1, x));
435                let inv = R::inv(&self[j + 1][j]);
436                let mut v = vec![];
437                let src = std::mem::take(&mut self[j + 1]);
438                for a in self.data[(j + 2)..].iter_mut() {
439                    let mul = R::mul(&a[j], &inv);
440                    for (a, src) in a[j..].iter_mut().zip(src[j..].iter()) {
441                        R::sub_assign(a, &R::mul(&mul, src));
442                    }
443                    v.push(mul);
444                }
445                self[j + 1] = src;
446                for a in self.data.iter_mut() {
447                    let v = a[(j + 2)..]
448                        .iter()
449                        .zip(v.iter())
450                        .fold(R::zero(), |s, a| R::add(&s, &R::mul(a.0, a.1)));
451                    R::add_assign(&mut a[j + 1], &v);
452                }
453            }
454        }
455        let mut dp = vec![vec![R::one()]];
456        for i in 0..n {
457            let mut next = vec![R::zero(); i + 2];
458            for (j, dp) in dp[i].iter().enumerate() {
459                R::sub_assign(&mut next[j], &R::mul(dp, &self[i][i]));
460                R::add_assign(&mut next[j + 1], dp);
461            }
462            let mut mul = R::one();
463            for j in (0..i).rev() {
464                mul = R::mul(&mul, &self[j + 1][j]);

467                    R::sub_assign(next, &R::mul(&c, dp));
468                }
