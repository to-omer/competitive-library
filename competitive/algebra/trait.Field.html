342            .collect();

344            let c = pivots[k];
345            let inv = R::inv(&a[k][c]);
346            R::mul_assign(&mut b[k], &inv);
347            let pivot_b = b[k].clone();
348            let (upper, lower) = coefficients.split_at_mut(k);
349            let pivot_coefficients = &mut lower[0];
350            for x in pivot_coefficients.iter_mut() {
351                R::mul_assign(x, &inv);
352            }
353            for ((row, value), coefficients) in a.data[..k].iter_mut().zip(&mut b[..k]).zip(upper) {
354                if R::is_zero(&row[c]) {
355                    continue;
356                }
357                let factor = row[c].clone();
358                row[c] = R::zero();
359                R::sub_assign(value, &R::mul(&factor, &pivot_b));
360                R::add_scaled_assign(coefficients, pivot_coefficients, &R::neg(&factor));
361            }
362        }
363
364        let mut particular = vec![R::zero(); m];
365        for i in 0..rank {
366            particular[pivots[i]] = b[i].clone();
367        }
368        let mut basis = Vec::with_capacity(free.len());
369        for (j, &c) in free.iter().enumerate() {
370            let mut vector = vec![R::zero(); m];
371            vector[c] = R::one();
372            for i in 0..rank {
373                vector[pivots[i]] = R::neg(&coefficients[i][j]);
374            }
375            basis.push(vector);
376        }
377        Some(SystemOfLinearEquationsSolution { particular, basis })

498                    R::add_assign(&mut a[j + 1], &v);
499                }
500            }
501        }
502        // dp[k][j - k] stores [x^k] det(xI - A[..j, ..j]).
503        let mut dp: Vec<Vec<R::T>> = (0..=n).map(|i| Vec::with_capacity(n + 1 - i)).collect();
504        dp[0].push(R::one());
505        for i in 0..n {
506            let mut c = vec![R::zero(); i + 1];
507            c[i] = R::neg(&self[i][i]);
508            let mut mul = R::one();
509            for j in (0..i).rev() {
510                mul = R::mul(&mul, &self[j + 1][j]);
511                c[j] = R::neg(&R::mul(&mul, &self[j][i]));
512            }
513            for k in (0..=i).rev() {
514                let mut value = R::dot_product(&dp[k], &c[k..]);
515                if k > 0 {
516                    R::add_assign(&mut value, dp[k - 1].last().unwrap());
517                }
518                dp[k].push(value);
519            }

521        }

100    pub fn determinant(&self) -> R::T {
101        assert_eq!(self.shape.0, self.shape.1);
102        let n = self.shape.0;
103        let mut columns = vec![Vec::<(usize, R::T)>::new(); n];
104        for &(i, j, ref value) in &self.nonzero {
105            columns[j].push((i, value.clone()));
106        }
107        let mut degrees = vec![0; n];
108        for column in &mut columns {
109            column.sort_unstable_by_key(|&(i, _)| i);
110            let mut merged: Vec<(usize, R::T)> = Vec::with_capacity(column.len());
111            for (i, value) in column.drain(..) {
112                if let Some((_, x)) = merged.last_mut().filter(|(last, _)| *last == i) {
113                    R::add_assign(x, &value);
114                } else {
115                    merged.push((i, value));
116                }
117            }
118            merged.retain(|(i, value)| {
119                if R::is_zero(value) {
120                    false
121                } else {
122                    degrees[*i] += 1;
123                    true
124                }
125            });
126            *column = merged;
127        }
128        let mut order: Vec<_> = (0..n).collect();
129        order.sort_unstable_by_key(|&j| columns[j].len());
130        let mut lower: Vec<Vec<(usize, R::T)>> = Vec::with_capacity(n);
131        let mut pivots: Vec<Option<usize>> = vec![None; n];
132        let mut x = vec![R::zero(); n];
133        let mut seen = vec![0; n];
134        let mut stack = Vec::new();
135        let mut support = Vec::new();
136        let mut determinant = R::one();
137        for (k, &j) in order.iter().enumerate() {
138            support.clear();
139            for &(i, _) in &columns[j] {
140                if seen[i] == k + 1 {
141                    continue;
142                }
143                seen[i] = k + 1;
144                x[i] = R::zero();
145                stack.push((i, 0));
146                while let Some((i, next)) = stack.last_mut() {
147                    if let Some(pivot) = pivots[*i]
148                        && *next < lower[pivot].len()
149                    {
150                        let row = lower[pivot][*next].0;
151                        *next += 1;
152                        if seen[row] != k + 1 {
153                            seen[row] = k + 1;
154                            x[row] = R::zero();
155                            stack.push((row, 0));
156                        }
157                        continue;
158                    }
159                    support.push(*i);
160                    stack.pop();
161                }
162            }
163            for &(i, ref value) in &columns[j] {
164                x[i] = value.clone();
165            }
166            let mut pivot = None;
167            for &i in support.iter().rev() {
168                if let Some(p) = pivots[i] {
169                    let factor = R::neg(&x[i]);
170                    for &(row, ref value) in &lower[p] {
171                        R::add_assign(&mut x[row], &R::mul(&factor, value));
172                    }
173                } else if !R::is_zero(&x[i]) && pivot.is_none_or(|p| degrees[i] < degrees[p]) {
174                    pivot = Some(i);
175                }
176            }
177            let Some(pivot) = pivot else { return R::zero() };
178            R::mul_assign(&mut determinant, &x[pivot]);
179            let inv = R::inv(&x[pivot]);
180            pivots[pivot] = Some(k);
181            lower.push(
182                support
183                    .iter()
184                    .filter(|&&i| pivots[i].is_none() && !R::is_zero(&x[i]))
185                    .map(|&i| (i, R::mul(&x[i], &inv)))
186                    .collect(),
187            );
188        }
189        for mut permutation in [order, pivots.into_iter().map(Option::unwrap).collect()] {
190            for i in 0..n {
191                while permutation[i] != i {
192                    let j = permutation[i];
193                    permutation.swap(i, j);
194                    determinant = R::neg(&determinant);
195                }
196            }
197        }
198        determinant
