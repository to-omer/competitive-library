260    pub fn solve_system_of_linear_equations(
261        &self,
262        b: &[R::T],
263    ) -> Option<SystemOfLinearEquationsSolution<R>> {
264        assert_eq!(self.shape.0, b.len());
265        let (n, m) = self.shape;
266        let mut a = self.clone();
267        let mut b = b.to_vec();
268        let mut pivots = Vec::with_capacity(n.min(m));
269        let mut rank = 0;
270        for c in 0..m {
271            if rank == n {
272                break;
273            }
274            let Some(pivot) = (rank..n).find(|&i| !R::is_zero(&a[i][c])) else {
275                continue;
276            };
277            a.data.swap(rank, pivot);
278            b.swap(rank, pivot);
279
280            let d = R::inv(&a[rank][c]);
281            a[rank][c] = R::one();
282            for x in &mut a[rank][(c + 1)..] {
283                R::mul_assign(x, &d);
284            }
285            R::mul_assign(&mut b[rank], &d);
286
287            let (upper, lower) = a.data.split_at_mut(rank + 1);
288            let pivot = &upper[rank];
289            let pivot_b = b[rank].clone();
290            for (row, value) in lower.iter_mut().zip(&mut b[(rank + 1)..]) {
291                if R::is_zero(&row[c]) {
292                    continue;
293                }
294                let factor = row[c].clone();
295                row[c] = R::zero();
296                for (x, y) in row[(c + 1)..].iter_mut().zip(&pivot[(c + 1)..]) {
297                    R::sub_assign(x, &R::mul(&factor, y));
298                }
299                R::sub_assign(value, &R::mul(&factor, &pivot_b));
300            }
301            pivots.push(c);
302            rank += 1;
303        }
304        if b[rank..].iter().any(|x| !R::is_zero(x)) {
305            return None;
306        }
307
308        let mut free = Vec::with_capacity(m - rank);
309        let mut pivot = 0;
310        for c in 0..m {
311            if pivot < rank && pivots[pivot] == c {
312                pivot += 1;
313            } else {
314                free.push(c);
315            }
316        }
317        let mut coefficients: Vec<Vec<_>> = (0..rank)
318            .map(|i| free.iter().map(|&c| a[i][c].clone()).collect())
319            .collect();
320        for k in (0..rank).rev() {
321            let c = pivots[k];
322            let pivot_b = b[k].clone();
323            let (upper, lower) = coefficients.split_at_mut(k);
324            let pivot_coefficients = &lower[0];
325            for ((row, value), coefficients) in a.data[..k].iter_mut().zip(&mut b[..k]).zip(upper) {
326                if R::is_zero(&row[c]) {
327                    continue;
328                }
329                let factor = row[c].clone();
330                row[c] = R::zero();
331                R::sub_assign(value, &R::mul(&factor, &pivot_b));
332                for (x, y) in coefficients.iter_mut().zip(pivot_coefficients) {
333                    R::sub_assign(x, &R::mul(&factor, y));
334                }
335            }
336        }
337
338        let mut particular = vec![R::zero(); m];
339        for i in 0..rank {
340            particular[pivots[i]] = b[i].clone();
341        }
342        let mut basis = Vec::with_capacity(free.len());
343        for (j, &c) in free.iter().enumerate() {
344            let mut vector = vec![R::zero(); m];
345            vector[c] = R::one();
346            for i in 0..rank {
347                vector[pivots[i]] = R::neg(&coefficients[i][j]);
348            }
349            basis.push(vector);
350        }
351        Some(SystemOfLinearEquationsSolution { particular, basis })
352    }

355        assert_eq!(self.shape.0, self.shape.1);
356        let n = self.shape.0;
357        let mut a = self.clone();
358        let mut inverse = Self::eye((n, n));
359        let mut ranges: Vec<_> = (0..n).map(|i| (i, i + 1)).collect();
360        for r in 0..n {
361            let pivot = (r..n).find(|&i| !R::is_zero(&a[i][r]))?;
362            a.data.swap(r, pivot);
363            inverse.data.swap(r, pivot);
364            ranges.swap(r, pivot);
365
366            let d = R::inv(&a[r][r]);
367            for x in &mut a[r][r..] {
368                R::mul_assign(x, &d);
369            }
370            let (left, right) = ranges[r];
371            for x in &mut inverse[r][left..right] {
372                R::mul_assign(x, &d);
373            }
374
375            let (a_upper, a_lower) = a.data.split_at_mut(r + 1);
376            let pivot_a = &a_upper[r];

384                }

409                }

425    pub fn characteristic_polynomial(&mut self) -> Vec<R::T> {
426        let n = self.shape.0;
427        if n == 0 {
428            return vec![R::one()];
429        }
430        assert!(self.data.iter().all(|a| a.len() == n));
431        for j in 0..(n - 1) {
432            if let Some(x) = ((j + 1)..n).find(|&x| !R::is_zero(&self[x][j])) {
433                self.data.swap(j + 1, x);
434                self.data.iter_mut().for_each(|a| a.swap(j + 1, x));
435                let inv = R::inv(&self[j + 1][j]);
436                let mut v = vec![];
437                let src = std::mem::take(&mut self[j + 1]);
438                for a in self.data[(j + 2)..].iter_mut() {
439                    let mul = R::mul(&a[j], &inv);
440                    for (a, src) in a[j..].iter_mut().zip(src[j..].iter()) {
441                        R::sub_assign(a, &R::mul(&mul, src));
442                    }
443                    v.push(mul);
444                }
445                self[j + 1] = src;
446                for a in self.data.iter_mut() {
447                    let v = a[(j + 2)..]
448                        .iter()
449                        .zip(v.iter())
450                        .fold(R::zero(), |s, a| R::add(&s, &R::mul(a.0, a.1)));
451                    R::add_assign(&mut a[j + 1], &v);
452                }
453            }
454        }
455        let mut dp = vec![vec![R::one()]];
456        for i in 0..n {
457            let mut next = vec![R::zero(); i + 2];
458            for (j, dp) in dp[i].iter().enumerate() {
459                R::sub_assign(&mut next[j], &R::mul(dp, &self[i][i]));
460                R::add_assign(&mut next[j + 1], dp);
461            }
462            let mut mul = R::one();
463            for j in (0..i).rev() {
464                mul = R::mul(&mul, &self[j + 1][j]);
465                let c = R::mul(&mul, &self[j][i]);
466                for (next, dp) in next.iter_mut().zip(dp[j].iter()) {
467                    R::sub_assign(next, &R::mul(&c, dp));
468                }
469            }
470            dp.push(next);
471        }
472        dp.pop().unwrap()
473    }

460                R::add_assign(&mut next[j + 1], dp);
461            }
462            let mut mul = R::one();
463            for j in (0..i).rev() {
464                mul = R::mul(&mul, &self[j + 1][j]);
465                let c = R::mul(&mul, &self[j][i]);
466                for (next, dp) in next.iter_mut().zip(dp[j].iter()) {
467                    R::sub_assign(next, &R::mul(&c, dp));
468                }
469            }
470            dp.push(next);
471        }
472        dp.pop().unwrap()
473    }

626fn strassen_rec<R: Ring>(
627    a: &[R::T],
628    b: &[R::T],
629    c: &mut [R::T],
630    n: usize,
631    stride_a: usize,
632    stride_b: usize,
633) {
634    fn add_block<R: Ring>(
635        a: &[R::T],
636        b: &[R::T],
637        out: &mut [R::T],
638        n: usize,
639        stride_a: usize,
640        stride_b: usize,
641    ) {
642        for ((a, b), c) in a
643            .chunks(stride_a)
644            .zip(b.chunks(stride_b))
645            .zip(out.chunks_exact_mut(n))
646        {
647            for ((a, b), c) in a.iter().zip(b.iter()).zip(c.iter_mut()) {
648                *c = R::add(a, b);
649            }
650        }
651    }
652
653    fn sub_block<R: Ring>(
654        a: &[R::T],
655        b: &[R::T],
656        out: &mut [R::T],
657        n: usize,
658        stride_a: usize,
659        stride_b: usize,
660    ) {
661        for ((a, b), c) in a
662            .chunks(stride_a)
663            .zip(b.chunks(stride_b))
664            .zip(out.chunks_exact_mut(n))
665        {
666            for ((a, b), c) in a.iter().zip(b.iter()).zip(c.iter_mut()) {
667                *c = R::sub(a, b);
668            }
669        }
670    }
671
672    if n <= 16 {
673        for (a, c) in a.chunks(stride_a).zip(c.chunks_exact_mut(n)) {
674            for (a, b) in a.iter().zip(b.chunks(stride_b)).take(n) {
675                for (b, c) in b.iter().zip(c.iter_mut()) {
676                    R::add_assign(c, &R::mul(a, b));
677                }
678            }
679        }
680        return;
681    }
682    let h = n / 2;
683    let a11 = 0;
684    let a12 = h;
685    let a21 = h * stride_a;
686    let a22 = a21 + h;
687    let b11 = 0;
688    let b12 = h;
689    let b21 = h * stride_b;
690    let b22 = b21 + h;
691
692    let block = h * h;
693    let mut buf = vec![R::zero(); block * 9];
694    let (s_buf, m_buf) = buf.split_at_mut(block * 2);
695    let (s1, s2) = s_buf.split_at_mut(block);
696    let (m1, rest) = m_buf.split_at_mut(block);
697    let (m2, rest) = rest.split_at_mut(block);
698    let (m3, rest) = rest.split_at_mut(block);
699    let (m4, rest) = rest.split_at_mut(block);
700    let (m5, rest) = rest.split_at_mut(block);
701    let (m6, m7) = rest.split_at_mut(block);
702
703    // (A11 + A22)(B11 + B22)
704    add_block::<R>(&a[a11..], &a[a22..], s1, h, stride_a, stride_a);
705    add_block::<R>(&b[b11..], &b[b22..], s2, h, stride_b, stride_b);
706    strassen_rec::<R>(s1, s2, m1, h, h, h);
707
708    // (A21 + A22) B11
709    add_block::<R>(&a[a21..], &a[a22..], s1, h, stride_a, stride_a);
710    strassen_rec::<R>(s1, &b[b11..], m2, h, h, stride_b);
711
712    // A11 (B12 - B22)
713    sub_block::<R>(&b[b12..], &b[b22..], s1, h, stride_b, stride_b);
714    strassen_rec::<R>(&a[a11..], s1, m3, h, stride_a, h);
715
716    // A22 (B21 - B11)
717    sub_block::<R>(&b[b21..], &b[b11..], s1, h, stride_b, stride_b);
718    strassen_rec::<R>(&a[a22..], s1, m4, h, stride_a, h);
719
720    // (A11 + A12) B22
721    add_block::<R>(&a[a11..], &a[a12..], s1, h, stride_a, stride_a);
722    strassen_rec::<R>(s1, &b[b22..], m5, h, h, stride_b);
723
724    // (A21 - A11)(B11 + B12)
725    sub_block::<R>(&a[a21..], &a[a11..], s1, h, stride_a, stride_a);
726    add_block::<R>(&b[b11..], &b[b12..], s2, h, stride_b, stride_b);
727    strassen_rec::<R>(s1, s2, m6, h, h, h);
728
729    // (A12 - A22)(B21 + B22)
730    sub_block::<R>(&a[a12..], &a[a22..], s1, h, stride_a, stride_a);
731    add_block::<R>(&b[b21..], &b[b22..], s2, h, stride_b, stride_b);
732    strassen_rec::<R>(s1, s2, m7, h, h, h);
733
734    let c11 = 0;
735    let c12 = h;
736    let c21 = h * n;
737    let c22 = c21 + h;
738    for ((((m1, m4), m5), m7), c) in m1
739        .iter()
740        .zip(m4.iter())
741        .zip(m5.iter())
742        .zip(m7.iter())
743        .zip(c[c11..].chunks_mut(n).flat_map(|c| c.iter_mut().take(h)))
744    {
745        *c = R::add(m1, m4);
746        R::sub_assign(c, m5);
747        R::add_assign(c, m7);
748    }
749    for ((m3, m5), c) in m3
750        .iter()
751        .zip(m5.iter())
752        .zip(c[c12..].chunks_mut(n).flat_map(|c| c.iter_mut().take(h)))
753    {
754        *c = R::add(m3, m5);
755    }
756    for ((m2, m4), c) in m2
757        .iter()
758        .zip(m4.iter())
759        .zip(c[c21..].chunks_mut(n).flat_map(|c| c.iter_mut().take(h)))
760    {
761        *c = R::add(m2, m4);
762    }
763    for ((((m1, m2), m3), m6), c) in m1
764        .iter()
765        .zip(m2.iter())
766        .zip(m3.iter())
767        .zip(m6.iter())
768        .zip(c[c22..].chunks_mut(n).flat_map(|c| c.iter_mut().take(h)))
769    {
770        *c = R::sub(m1, m2);
771        R::add_assign(c, m3);
772        R::add_assign(c, m6);
773    }
774}

776impl<R> Matrix<R>
777where
778    R: Ring,
779{
780    pub fn mul_strassen(&self, rhs: &Matrix<R>) -> Matrix<R> {
781        assert_eq!(self.shape.1, rhs.shape.0);
782        let (n, m) = self.shape;
783        let p = rhs.shape.1;
784        if n == 0 || m == 0 || p == 0 {
785            return Matrix::zeros((n, p));
786        }
787        let max_dim = n.max(m).max(p);
788        if max_dim <= 64 {
789            return self * rhs;
790        }
791        let size = max_dim.next_power_of_two();
792        if size * size * size > n * m * p * 2 {
793            return self * rhs;
794        }
795        let mut a = vec![R::zero(); size * size];
796        for (a, data) in a.chunks_exact_mut(size).zip(&self.data) {
797            a[..m].clone_from_slice(data);
798        }
799        let mut b = vec![R::zero(); size * size];
800        for (b, data) in b.chunks_exact_mut(size).zip(&rhs.data) {
801            b[..p].clone_from_slice(data);
802        }
803        let mut c = vec![R::zero(); size * size];
804        strassen_rec::<R>(&a, &b, &mut c, size, size, size);
805        let mut res = Matrix::zeros((n, p));
806        for (data, c) in res.data.iter_mut().zip(c.chunks_exact(size)) {
807            data.clone_from_slice(&c[..p]);
808        }
809        res
810    }
811}
