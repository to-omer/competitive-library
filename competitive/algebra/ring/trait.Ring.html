425    pub fn characteristic_polynomial(&mut self) -> Vec<R::T> {
426        let n = self.shape.0;
427        if n == 0 {
428            return vec![R::one()];
429        }
430        assert!(self.data.iter().all(|a| a.len() == n));
431        for j in 0..(n - 1) {
432            if let Some(x) = ((j + 1)..n).find(|&x| !R::is_zero(&self[x][j])) {
433                self.data.swap(j + 1, x);
434                self.data.iter_mut().for_each(|a| a.swap(j + 1, x));
435                let inv = R::inv(&self[j + 1][j]);
436                let mut v = vec![];
437                let src = std::mem::take(&mut self[j + 1]);
438                for a in self.data[(j + 2)..].iter_mut() {
439                    let mul = R::mul(&a[j], &inv);
440                    for (a, src) in a[j..].iter_mut().zip(src[j..].iter()) {
441                        R::sub_assign(a, &R::mul(&mul, src));
442                    }
443                    v.push(mul);
444                }
445                self[j + 1] = src;
446                for a in self.data.iter_mut() {
447                    let v = a[(j + 2)..]
448                        .iter()
449                        .zip(v.iter())
450                        .fold(R::zero(), |s, a| R::add(&s, &R::mul(a.0, a.1)));
451                    R::add_assign(&mut a[j + 1], &v);
452                }
453            }
454        }
455        let mut dp = vec![vec![R::one()]];
456        for i in 0..n {
457            let mut next = vec![R::zero(); i + 2];
458            for (j, dp) in dp[i].iter().enumerate() {
459                R::sub_assign(&mut next[j], &R::mul(dp, &self[i][i]));
460                R::add_assign(&mut next[j + 1], dp);
461            }
462            let mut mul = R::one();
463            for j in (0..i).rev() {
464                mul = R::mul(&mul, &self[j + 1][j]);
465                let c = R::mul(&mul, &self[j][i]);
466                for (next, dp) in next.iter_mut().zip(dp[j].iter()) {
467                    R::sub_assign(next, &R::mul(&c, dp));
468                }
469            }
470            dp.push(next);
471        }
472        dp.pop().unwrap()
473    }

780    pub fn mul_strassen(&self, rhs: &Matrix<R>) -> Matrix<R> {
781        assert_eq!(self.shape.1, rhs.shape.0);
782        let (n, m) = self.shape;
783        let p = rhs.shape.1;
784        if n == 0 || m == 0 || p == 0 {
785            return Matrix::zeros((n, p));
786        }
787        let max_dim = n.max(m).max(p);
788        if max_dim <= 64 {
789            return self * rhs;
790        }
791        let size = max_dim.next_power_of_two();
792        if size * size * size > n * m * p * 2 {
793            return self * rhs;
794        }
795        let mut a = vec![R::zero(); size * size];
796        for (a, data) in a.chunks_exact_mut(size).zip(&self.data) {
797            a[..m].clone_from_slice(data);
798        }
799        let mut b = vec![R::zero(); size * size];
800        for (b, data) in b.chunks_exact_mut(size).zip(&rhs.data) {
801            b[..p].clone_from_slice(data);
802        }
803        let mut c = vec![R::zero(); size * size];
804        strassen_rec::<R>(&a, &b, &mut c, size, size, size);
805        let mut res = Matrix::zeros((n, p));
806        for (data, c) in res.data.iter_mut().zip(c.chunks_exact(size)) {
807            data.clone_from_slice(&c[..p]);
808        }
