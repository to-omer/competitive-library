148fn generate_frobenius_block<M>(
149    a: &Matrix<AddMulOperation<MInt<M>>>,
150    mut v: Vec<MInt<M>>,
151    rows: &mut Vec<EchelonRow<M>>,
152    t: &mut Vec<Vec<MInt<M>>>,
153) -> Polynomial<M>
154where
155    M: MIntBase,
156{
157    let n = a.shape.0;
158    loop {
159        let mut row = vec![MInt::zero(); 2 * n + 1];
160        let (x, c) = row.split_at_mut(n);
161        x.copy_from_slice(&v);
162        c[rows.len()] = MInt::one();
163        for r in rows.iter() {
164            r.reduce(&mut row);
165        }
166        if let Some(pivot) = row[..n].iter().position(|x| !x.is_zero()) {
167            t.push(v);
168            let u = t.last().unwrap();
169            v = a.data.iter().map(|row| MInt::dot_product(u, row)).collect();
170            rows.push(EchelonRow {
171                pivot,
172                inv: row[pivot].inv(),
173                row,
174            });
175        } else {
176            let mut p = row.split_off(n);
177            while p.last().is_some_and(|x| x.is_zero()) {
178                p.pop();
179            }
180            return Polynomial(p);
181        }
