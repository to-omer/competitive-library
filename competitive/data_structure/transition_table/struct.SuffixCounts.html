Expand description

Counts accepted words, including leading zeros. Bound queries use the bound’s length, digits in 0..radix, and at most the precomputed length.

192    pub fn count_between(&self, lower: &[usize], upper: &[usize]) -> V {

231    /// The zero-based `k`th accepted word in lexicographic order, or `None` if absent.
232    /// Requires `k < U::MAX` because counts may saturate.
233    pub fn kth(&self, len: usize, mut k: U) -> Option<Vec<usize>> {
234        assert!(k < U::maximum());
235        if self.count(0, len).0 <= k {
236            return None;
237        }
238        let mut word = Vec::with_capacity(len);
239        let mut state = 0;
240        for remaining in (0..len).rev() {
241            for (digit, edge) in self.table.transitions(state).iter().enumerate() {
242                if let Some((next, _)) = edge {
243                    let count = self.count(*next, remaining).0;
244                    if count <= k {
245                        k -= count;
246                    } else {
247                        word.push(digit);
248                        state = *next;
249                        break;
250                    }
251                }
252            }
253        }
254        Some(word)
255    }

258    pub fn kth_between(&self, lower: &[usize], upper: &[usize], mut k: U) -> Option<Vec<usize>>
259    where
260        Saturating<U>: std::ops::Add<Output = Saturating<U>>,
261    {
262        assert!(k < U::maximum());
263        assert_eq!(lower.len(), upper.len());
264        if lower > upper {
265            return None;
266        }
267        let lower_counts = self.bound_counts::<false>(lower);
268        let upper_counts = self.bound_counts::<true>(upper);
269        let mut state = 0;
270        let (mut tight_lower, mut tight_upper) = (true, true);
271        let mut word = Vec::with_capacity(lower.len());
272        for i in 0..lower.len() {
273            let edges = self.table.transitions(state);
274            let lo = if tight_lower { lower[i] } else { 0 };
275            let hi = if tight_upper {
276                upper[i]
277            } else {
278                self.table.radix() - 1
279            };
280            if tight_lower && tight_upper && lo == hi {
281                state = edges[lo].as_ref()?.0;
282                word.push(lo);
283                continue;
284            }
285            let mut selected = None;
286            for (digit, edge) in edges.iter().enumerate().take(hi + 1).skip(lo) {
287                if let Some((next, _)) = edge {
288                    let count = if tight_lower && digit == lo {
289                        lower_counts[i + 1].0
290                    } else if tight_upper && digit == hi {
291                        upper_counts[i + 1].0
292                    } else {
293                        self.count(*next, lower.len() - i - 1).0
294                    };
295                    if count <= k {
296                        k -= count;
297                    } else {
298                        selected = Some((digit, *next));
299                        break;
300                    }
301                }
302            }
303            let (digit, next) = selected?;
304            tight_lower &= digit == lower[i];
305            tight_upper &= digit == upper[i];
306            word.push(digit);
307            state = next;
308        }
309        (k.is_zero() && self.table.is_accepting(state)).then_some(word)
310    }

317    fn count_bound<const LESS: bool>(
318        &self,
319        mut state: usize,
320        bound: &[usize],
321        inclusive: bool,
322    ) -> V {
323        assert!(bound.len() < self.counts.len());
324        let mut sum = V::zero();
325        for (i, &digit) in bound.iter().enumerate() {
326            let edges = self.table.transitions(state);
327            let range = if LESS {
328                0..digit
329            } else {
330                digit + 1..self.table.radix()
331            };
332            for (next, _) in edges[range].iter().flatten() {
333                sum = sum + self.count(*next, bound.len() - i - 1).clone();
334            }
335            match &edges[digit] {
336                Some((next, _)) => state = *next,
337                None => return sum,
338            }
339        }
340        if inclusive {
341            sum = sum + self.count(state, 0).clone();
342        }
343        sum
344    }

346    fn bound_counts<const LESS: bool>(&self, bound: &[usize]) -> Vec<V> {
347        assert!(bound.len() < self.counts.len());
348        let mut counts = vec![V::zero(); bound.len() + 1];
349        let mut state = 0;
350        let mut matched = 0;
351        for (i, &digit) in bound.iter().enumerate() {
352            let edges = self.table.transitions(state);
353            let range = if LESS {
354                0..digit
355            } else {
356                digit + 1..self.table.radix()
357            };
358            counts[i] = edges[range]
359                .iter()
360                .flatten()
361                .fold(V::zero(), |sum, (next, _)| {
362                    sum + self.count(*next, bound.len() - i - 1).clone()
363                });
364            let Some((next, _)) = &edges[digit] else {
365                break;
366            };
367            state = *next;
368            matched += 1;
369        }
370        if matched == bound.len() {
371            counts[matched] = self.count(state, 0).clone();
372        }
373        let mut sum = counts[matched].clone();
374        for count in counts[..matched].iter_mut().rev() {
375            sum = sum + count.clone();
376            *count = sum.clone();
377        }
378        counts
379    }
