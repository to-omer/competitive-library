205    pub fn rank_batch(&self, queries: impl IntoIterator<Item = (T, Range<usize>)>) -> Vec<usize> {
206        let queries: Vec<_> = queries.into_iter().collect();
207        if queries.len() < 8 || self.len > u32::MAX as usize {
208            return queries
209                .into_iter()
210                .map(|(value, range)| self.rank(value, range))
211                .collect();
212        }
213        let mut result = vec![0; queries.len()];
214        let mut active = Vec::new();
215        for (output, (value, range)) in queries.into_iter().enumerate() {
216            if let Some(index) = self.compress.index_exact(&value) {
217                active.push((output, index, range));
218            }
219        }
220        for queries in active.chunks(16) {
221            if queries.len() < 8 {
222                for (output, index, range) in queries {
223                    result[*output] = self.rank_by_index(*index, range.clone());
224                }
225                continue;
226            }
227            let mut states = [[0_u32; 3]; 16];
228            for (state, (_, index, range)) in states.iter_mut().zip(queries) {
229                *state = [range.start as u32, range.end as u32, *index as u32];
230            }
231            for d in (0..self.bit_length).rev() {
232                let level = self.level(d);
233                for state in &mut states[..queries.len()] {
234                    let start = state[0] as usize;
235                    let end = state[1] as usize;
236                    let start1 = self.rank1(level, start);
237                    let end1 = self.rank1(level, end);
238                    if ((state[2] >> d) & 1) != 0 {
239                        state[0] = (self.zeros[level] + start1) as u32;
240                        state[1] = (self.zeros[level] + end1) as u32;
241                    } else {
242                        state[0] = (start - start1) as u32;
243                        state[1] = (end - end1) as u32;
244                    }
245                }
246            }
247            for ((output, _, _), state) in queries.iter().zip(&states) {
248                result[*output] = (state[1] - state[0]) as usize;

305    pub fn quantile_batch(
306        &self,
307        queries: impl IntoIterator<Item = (Range<usize>, usize)>,
308    ) -> Vec<T> {
309        let queries: Vec<_> = queries.into_iter().collect();
310        if queries.len() < 8 || self.len > u32::MAX as usize {
311            return queries
312                .into_iter()
313                .map(|(range, k)| self.quantile(range, k))
314                .collect();
315        }
316        let mut result = Vec::with_capacity(queries.len());
317        for queries in queries.chunks(16) {
318            if queries.len() < 8 {
319                result.extend(
320                    queries
321                        .iter()
322                        .map(|(range, k)| self.quantile(range.clone(), *k)),
323                );
324                continue;
325            }
326            let mut states = [[0_u32; 4]; 16];
327            for (state, (range, k)) in states.iter_mut().zip(queries) {
328                *state = [range.start as u32, range.end as u32, *k as u32, 0];
329            }
330            for d in (0..self.bit_length).rev() {
331                let level = self.level(d);
332                for state in &mut states[..queries.len()] {
333                    let start = state[0] as usize;
334                    let end = state[1] as usize;
335                    let start1 = self.rank1(level, start);
336                    let end1 = self.rank1(level, end);
337                    let start0 = (start - start1) as u32;
338                    let end0 = (end - end1) as u32;
339                    let zeros = end0 - start0;
340                    let mask = 0u32.wrapping_sub((state[2] >= zeros) as u32);
341                    state[0] =
342                        (start0 & !mask) | ((self.zeros[level] as u32 + start1 as u32) & mask);
343                    state[1] = (end0 & !mask) | ((self.zeros[level] as u32 + end1 as u32) & mask);
344                    state[2] -= zeros & mask;
345                    state[3] |= (1u32 << d) & mask;
346                }
347            }
348            result.extend(
349                states[..queries.len()]
350                    .iter()
351                    .map(|state| self.compress.values()[state[3] as usize].clone()),
352            );
353        }
