180            #[target_feature(enable = $feature)]
181            pub unsafe fn edit_sum(
182                values: &mut [u32; B],
183                mut tags: Option<(&mut [u32; B], &mut [u32; B])>,
184                len: usize,
185                width: usize,
186                full: Range<usize>,
187                action: Affine,
188                carry: Affine,
189            ) -> u32 {
190                let modulus = $set(Mod::MOD as i32);
191
192                let combined = compose(carry, action);
193                let ca_r = $set(carry.0.wrapping_mul(Mod::R) as i32);
194                let fa_r = $set(combined.0.wrapping_mul(Mod::R) as i32);
195                let ca = $set(carry.0 as i32);
196                let cb = $set(carry.1 as i32);
197                let fa = $set(combined.0 as i32);
198                let fb = $set(combined.1 as i32);
199                let count = ((len - 1) >> width.trailing_zeros()) + 1;
200                let scaled = |value: u32, count: usize| {
201                    ((value as u64 * (count as u64 % Mod::MOD as u64)) % Mod::MOD as u64) as i32
202                };
203                let old_offset = $set(scaled(carry.1, width));
204                let new_offset = $set(scaled(combined.1, width));
205                let tail = len - (count - 1) * width;
206                let (old_tail, new_tail) = if tail == width {
207                    (old_offset, new_offset)
208                } else {
209                    ($set(scaled(carry.1, tail)), $set(scaled(combined.1, tail)))
210                };
211                let mut selected_sum = $set(0);
212                for base in (0..B).step_by($lanes) {
213                    let selected = mask(full.start, full.end, base);
214                    let a = blend(selected, ca, fa);
215                    let a_r = blend(selected, ca_r, fa_r);
216                    let b = blend(selected, cb, fb);
217                    let offset = blend(selected, old_offset, new_offset);
218                    let offset = if tail == width {
219                        offset
220                    } else {
221                        blend(
222                            mask(count - 1, count, base),
223                            offset,
224                            blend(selected, old_tail, new_tail),
225                        )
226                    };
227                    let offset = if count == B {
228                        offset
229                    } else {
230                        blend(mask(0, count, base), $set(0), offset)
231                    };
232                    let old = $load(values.as_ptr().add(base).cast());
233                    selected_sum = $add(selected_sum, blend(selected, $set(0), old), modulus);
234                    let new = $add(mul_prepared(old, a, a_r, modulus), offset, modulus);
235                    $store(values.as_mut_ptr().add(base).cast(), new);
236                    if let Some((mul, add)) = &mut tags {
237                        let old_mul = $load(mul.as_ptr().add(base).cast());
238                        let old_add = $load(add.as_ptr().add(base).cast());
239                        $store(
240                            mul.as_mut_ptr().add(base).cast(),
241                            mul_prepared(old_mul, a, a_r, modulus),
242                        );
243                        $store(
244                            add.as_mut_ptr().add(base).cast(),
245                            $add(mul_prepared(old_add, a, a_r, modulus), b, modulus),
246                        );
247                    }
248                }
249                let previous = reduce(selected_sum);
250                let covered = if full.is_empty() {
251                    0
252                } else {
253                    if full.end == count {
254                        len - full.start * width
255                    } else {
256                        (full.end - full.start) * width
257                    }
258                };
259                mod_add(
260                    mod_mul(previous, mod_sub(combined.0, carry.0)),
261                    scaled(mod_sub(combined.1, carry.1), covered) as u32,
262                )
263            }
264        };
265    }

271        unsafe fn reduce(value: __m256i) -> u32 {
272            let even = _mm256_and_si256(value, _mm256_set1_epi64x(u32::MAX as i64));
273            let pairs = _mm256_add_epi64(even, _mm256_srli_epi64::<32>(value));
274            let halves = _mm_add_epi64(
275                _mm256_castsi256_si128(pairs),
276                _mm256_extracti128_si256::<1>(pairs),
277            );
278            let sum = (_mm_cvtsi128_si64(halves) as u64) + (_mm_extract_epi64::<1>(halves) as u64);
279            (sum % Mod::MOD as u64) as u32
280        }

335        #[inline]
336        unsafe fn mul_prepared(a: __m512i, b: __m512i, br: __m512i, p: __m512i) -> __m512i {
337            let a13 = _mm512_srli_epi64::<32>(a);
338            let b13 = _mm512_srli_epi64::<32>(b);
339            let br13 = _mm512_srli_epi64::<32>(br);
340            let t02 = _mm512_mul_epu32(a, b);
341            let t13 = _mm512_mul_epu32(a13, b13);
342            let m02 = mul_epu32_512(a, br);
343            let m13 = mul_epu32_512(a13, br13);
344            let u02 = _mm512_add_epi64(t02, _mm512_mul_epu32(m02, p));
345            let u13 = _mm512_add_epi64(t13, _mm512_mul_epu32(m13, p));
346            _mm512_or_si512(_mm512_srli_epi64::<32>(u02), u13)
347        }

407    pub unsafe fn refold<const ISA: u8>(node: &CompositeLeaf) -> Affine {
408        #[cfg(target_arch = "x86_64")]
409        if ISA != 0 {
410            return bitrev_vector::fold::<ISA, true>(node, 0..B);
411        }
412        let pair =
413            |a: usize, b: usize| compose((node.mul[a], node.add[a]), (node.mul[b], node.add[b]));
414        let quarter = |i: usize| {
415            let [a, b, c, d] = REV.as_chunks::<4>().0[i];
416            compose(pair(a, b), pair(c, d))
417        };
418        compose(
419            compose(quarter(0), quarter(1)),
420            compose(quarter(2), quarter(3)),
421        )
422    }

424    pub unsafe fn fold<const ISA: u8>(node: &CompositeLeaf, range: Range<usize>) -> Affine {
425        // Short fragments avoid the permutation cost of a vector reduction.
426        #[cfg(target_arch = "x86_64")]
427        if ISA != 0 && range.len() > 3 {
428            return bitrev_vector::fold::<ISA, false>(node, range);
429        }
430        if range.is_empty() {
431            return (ONE, 0);
432        }
433        let first = REV[range.start];
434        (range.start + 1..range.end).fold((node.mul[first], node.add[first]), |value, lane| {
435            compose(value, (node.mul[REV[lane]], node.add[REV[lane]]))
436        })
437    }

459    unsafe fn even(x: __m256i, y: __m256i) -> __m256i {
460        let t = _mm256_mul_epu32(x, y);
461        let m = _mm256_mul_epu32(t, _mm256_set1_epi32(Mod::R as i32));
462        _mm256_srli_epi64::<32>(_mm256_add_epi64(
463            t,

469    unsafe fn pairs(first: __m256i, second: __m256i) -> __m256i {
470        let factor = _mm256_shuffle_epi32::<0xa0>(second);
471        let shift = _mm256_blend_epi32::<0xaa>(_mm256_setzero_si256(), second);
472        add_mod_256(

485    unsafe fn avx2<const FULL: bool>(node: &CompositeLeaf, range: Range<usize>) -> Affine {
486        let p = _mm256_set1_epi32(Mod::MOD as i32);
487        let p2 = _mm256_add_epi32(p, p);
488        let inv = _mm256_set1_epi32(Mod::R as i32);
489        let one = _mm256_set1_epi32(ONE as i32);
490        let zero = _mm256_setzero_si256();
491        let mut a = [zero; 2];
492        let mut b = [zero; 2];
493        for base in [0, 8] {
494            let lanes = _mm256_add_epi32(
495                _mm256_setr_epi32(0, 8, 4, 12, 2, 10, 6, 14),
496                _mm256_set1_epi32((base / 8) as i32),
497            );
498            let mask = if FULL {
499                _mm256_set1_epi32(-1)
500            } else {
501                _mm256_and_si256(
502                    _mm256_cmpgt_epi32(lanes, _mm256_set1_epi32(range.start as i32 - 1)),
503                    _mm256_cmpgt_epi32(_mm256_set1_epi32(range.end as i32), lanes),
504                )
505            };
506            a[base / 8] = _mm256_blendv_epi8(
507                one,
508                _mm256_loadu_si256(node.mul.as_ptr().add(base).cast()),
509                mask,
510            );
511            b[base / 8] = _mm256_blendv_epi8(
512                zero,
513                _mm256_loadu_si256(node.add.as_ptr().add(base).cast()),
514                mask,
515            );
516        }
517        if !FULL && (range.end <= 8 || range.start >= 8) {
518            if range.start >= 8 {
519                for i in 0..2 {
520                    a[i] = _mm256_srli_epi64::<32>(a[i]);
521                    b[i] = _mm256_srli_epi64::<32>(b[i]);
522                }
523            }
524            let mut x = even(a[0], a[1]);
525            let mut y = add_mod_256(even(b[0], a[1]), b[1], p2);
526            let rx = _mm256_permute2x128_si256::<1>(x, x);
527            let ry = _mm256_permute2x128_si256::<1>(y, y);
528            x = even(x, rx);
529            y = add_mod_256(even(y, rx), ry, p2);
530            let rx = _mm256_shuffle_epi32::<0xee>(x);
531            let ry = _mm256_shuffle_epi32::<0xee>(y);
532            x = even(x, rx);
533            y = add_mod_256(even(y, rx), ry, p2);
534            return (
535                _mm256_extract_epi32::<0>(x) as u32,
536                _mm256_extract_epi32::<0>(y) as u32,
537            );
538        }
539        if !FULL {
540            let x = montgomery_mul_256(a[0], a[1], inv, p);
541            let y = add_mod_256(montgomery_mul_256(b[0], a[1], inv, p), b[1], p2);
542            let low = _mm256_unpacklo_epi32(x, y);
543            let high = _mm256_unpackhi_epi32(x, y);
544            let first = _mm256_permute2x128_si256::<0x20>(low, high);
545            let second = _mm256_permute2x128_si256::<0x31>(low, high);
546            let quads = pairs(first, second);
547            let halves = pairs(quads, _mm256_permute2x128_si256::<1>(quads, quads));
548            let last = _mm256_castsi256_si128(halves);
549            return compose(
550                (
551                    _mm_cvtsi128_si32(last) as u32,
552                    _mm_extract_epi32::<1>(last) as u32,
553                ),
554                (
555                    _mm_extract_epi32::<2>(last) as u32,
556                    _mm_extract_epi32::<3>(last) as u32,
557                ),
558            );
559        }
560        reduce256(a, b)
561    }

563    #[target_feature(enable = "avx2")]
564    pub unsafe fn reduce256(a: [__m256i; 2], b: [__m256i; 2]) -> Affine {
565        let p = _mm256_set1_epi32(Mod::MOD as i32);
566        let p2 = _mm256_add_epi32(p, p);
567        let inv = _mm256_set1_epi32(Mod::R as i32);
568        let mut x = montgomery_mul_256(a[0], a[1], inv, p);
569        let mut y = add_mod_256(montgomery_mul_256(b[0], a[1], inv, p), b[1], p2);
570        for step in 0..3 {
571            let (rx, ry) = match step {
572                0 => (
573                    _mm256_permute2x128_si256::<1>(x, x),
574                    _mm256_permute2x128_si256::<1>(y, y),
575                ),
576                1 => (
577                    _mm256_shuffle_epi32::<0xee>(x),
578                    _mm256_shuffle_epi32::<0xee>(y),
579                ),
580                _ => (
581                    _mm256_shuffle_epi32::<0x55>(x),
582                    _mm256_shuffle_epi32::<0x55>(y),
583                ),
584            };
585            x = montgomery_mul_256(x, rx, inv, p);
586            y = add_mod_256(montgomery_mul_256(y, rx, inv, p), ry, p2);
587        }
588        (
589            _mm256_extract_epi32::<0>(x) as u32,
590            _mm256_extract_epi32::<0>(y) as u32,
591        )
592    }

594    #[target_feature(enable = "avx512f,avx512dq,avx512cd,avx512bw,avx512vl")]
595    unsafe fn avx512<const FULL: bool>(node: &CompositeLeaf, range: Range<usize>) -> Affine {
596        let p = _mm512_set1_epi32(Mod::MOD as i32);
597        let p2 = _mm512_add_epi32(p, p);
598        let inv = _mm512_set1_epi32(Mod::R as i32);
599        let mask = if FULL {
600            u16::MAX
601        } else {
602            PREFIX_MASK[range.end] ^ PREFIX_MASK[range.start]
603        };
604        let mut a = _mm512_mask_blend_epi32(
605            mask,
606            _mm512_set1_epi32(ONE as i32),
607            _mm512_loadu_si512(node.mul.as_ptr().cast()),
608        );
609        let mut b = _mm512_maskz_mov_epi32(mask, _mm512_loadu_si512(node.add.as_ptr().cast()));
610        for shift in [8, 4, 2, 1] {
611            let indices: [i32; 16] = std::array::from_fn(|i| (i ^ shift) as i32);
612            let index = _mm512_loadu_si512(indices.as_ptr().cast());
613            let ra = _mm512_permutexvar_epi32(index, a);
614            let rb = _mm512_permutexvar_epi32(index, b);
615            a = montgomery_mul_512(a, ra, inv, p);
616            b = add_mod_512(montgomery_mul_512(b, ra, inv, p), rb, p2);
617        }
618        (
619            _mm_cvtsi128_si32(_mm512_castsi512_si128(a)) as u32,
620            _mm_cvtsi128_si32(_mm512_castsi512_si128(b)) as u32,
621        )
622    }
623}

647    /// Affine updates and sums modulo 998244353, with Montgomery residues in [0, 2p).
648    #[derive(Clone, Debug)]
649    pub struct DaryAffineSumKernel;
650
651    impl DaryLazyKernel for DaryAffineSumKernel {
652        const B: usize = B;
653        const PAIRED_UPDATES: bool = true;
654        type Key = M;
655        type EncodedKey = u32;
656        type Output = M;
657        type Action = (M, M);
658        type State = ();
659        fn new_state(_len: usize) {}
660        fn encode(key: M) -> u32 {
661            <Mod as MIntConvert<_>>::from(key.inner())
662        }
663        fn decode(value: u32) -> M {
664            M::new_unchecked(Mod::mod_inner(value))
665        }
666        type Summary = u32;
667        type Tag = Affine;
668        type Prepared = Affine;
669        type Change = u32;
670        type Leaf = SumLeaf;
671        type Branch = SumBranch;
672
673        fn unit() -> u32 {
674            0
675        }
676        fn tag_unit() -> Affine {
677            (ONE, 0)
678        }
679        fn combine(left: u32, right: u32) -> u32 {
680            mod_add(left, right)
681        }
682        fn compose(old: Affine, later: Affine) -> Affine {
683            compose(old, later)
684        }
685        fn prepare(
686            _state: &mut (),
687            (a, b): (M, M),
688            _range: Range<usize>,
689            _len: usize,
690            _branches: &[Vec<Self::Branch>],
691        ) -> Affine {
692            (
693                <Mod as MIntConvert<_>>::from(a.inner()),
694                <Mod as MIntConvert<_>>::from(b.inner()),
695            )
696        }
697        fn map(_state: &(), value: u32, tag: Affine, len: usize) -> u32 {
698            if tag == (ONE, 0) {
699                return value;
700            }
701            // Multiplication by an integer preserves the Montgomery representation.
702            let offset = ((tag.1 as u64 * (len as u64 % Mod::MOD as u64)) % Mod::MOD as u64) as u32;
703            mod_add(mod_mul(value, tag.0), offset)
704        }
705        fn repair(value: u32, change: u32) -> u32 {
706            mod_add(value, change)
707        }
708        fn leaf(keys: impl Iterator<Item = u32>) -> SumLeaf {
709            dary_sum_leaf::leaf(keys)
710        }
711        fn branch(values: &[u32]) -> SumBranch {
712            SumBranch {
713                values: Self::leaf(values.iter().copied()).0,
714                mul: [ONE; B],
715                add: [0; B],
716            }
717        }
718        unsafe fn fold_leaf<const ISA: u8>(
719            _state: &(),
720            node: &SumLeaf,
721            range: Range<usize>,
722        ) -> u32 {
723            #[cfg(target_arch = "x86_64")]
724            if ISA != 0 {
725                return dary_sum_vector::sum::<ISA>(&node.0, range);
726            }
727            (node.0[range].iter().map(|&v| v as u64).sum::<u64>() % Mod::MOD as u64) as u32
728        }
729        unsafe fn fold_branch<const ISA: u8>(
730            _state: &(),
731            node: &SumBranch,
732            range: Range<usize>,
733        ) -> u32 {
734            #[cfg(target_arch = "x86_64")]
735            if ISA != 0 {
736                return dary_sum_vector::sum::<ISA>(&node.values, range);
737            }
738            (node.values[range].iter().map(|&v| v as u64).sum::<u64>() % Mod::MOD as u64) as u32
739        }
740        fn tag(node: &SumBranch, lane: usize) -> Affine {
741            (node.mul[lane], node.add[lane])
742        }
743        unsafe fn edit_leaf<const ISA: u8>(
744            _state: &mut (),
745            node: &mut SumLeaf,
746            len: usize,
747            range: Range<usize>,
748            action: &Affine,
749            carry: Affine,
750        ) -> u32 {
751            #[cfg(target_arch = "x86_64")]
752            if ISA != 0 {
753                return dary_sum_vector::edit_sum_leaf::<ISA>(node, len, range, *action, carry);
754            }
755            if carry != (ONE, 0) {
756                for value in &mut node.0[..len] {
757                    *value = Self::map(&(), *value, carry, 1);
758                }
759            }
760            let mut delta = 0i64;
761            for value in &mut node.0[range] {
762                let old = *value;
763                *value = Self::map(&(), old, *action, 1);
764                delta += i64::from(*value) - i64::from(old);
765            }
766            delta.rem_euclid(i64::from(2 * Mod::MOD)) as u32
767        }
768        unsafe fn edit_branch<const ISA: u8>(
769            _state: &mut (),
770            node: &mut SumBranch,
771            len: usize,
772            width: usize,
773            full: Range<usize>,
774            edges: &[Option<(usize, u32)>; 2],
775            action: &Affine,
776            carry: Affine,
777        ) -> u32 {
778            #[cfg(target_arch = "x86_64")]
779            if ISA != 0 {
780                return edit_sum_branch::<ISA>(node, len, width, full, edges, *action, carry);
781            }
782            if carry != (ONE, 0) {
783                let count = ((len - 1) >> width.trailing_zeros()) + 1;
784                for lane in 0..count {
785                    let span = width.min(len - lane * width);
786                    node.values[lane] = Self::map(&(), node.values[lane], carry, span);
787                    let tag = compose(Self::tag(node, lane), carry);
788                    node.mul[lane] = tag.0;
789                    node.add[lane] = tag.1;
790                }
791            }
792            let mut delta = 0i64;
793            for lane in full {
794                let old = node.values[lane];
795                let span = width.min(len - lane * width);
796                node.values[lane] = Self::map(&(), old, *action, span);
797                let tag = compose(Self::tag(node, lane), *action);
798                node.mul[lane] = tag.0;
799                node.add[lane] = tag.1;
800                delta += i64::from(node.values[lane]) - i64::from(old);
801            }
802            for &(lane, change) in edges.iter().flatten() {
803                node.values[lane] = mod_add(node.values[lane], change);
804                node.mul[lane] = ONE;
805                node.add[lane] = 0;
806                delta += i64::from(change);
807            }
808            delta.rem_euclid(i64::from(2 * Mod::MOD)) as u32
809        }
810    }
811
812    #[cfg(target_arch = "x86_64")]
813    #[inline]
814    unsafe fn edit_sum_branch<const ISA: u8>(
815        node: &mut SumBranch,
816        len: usize,
817        width: usize,
818        full: Range<usize>,
819        edges: &[Option<(usize, u32)>; 2],
820        action: Affine,
821        carry: Affine,
822    ) -> u32 {
823        let tags = Some((&mut node.mul, &mut node.add));
824        let mut delta = if full.is_empty() && carry == (ONE, 0) {
825            0
826        } else if ISA == 1 {
827            dary_sum_vector::avx2::edit_sum(&mut node.values, tags, len, width, full, action, carry)
828        } else {
829            dary_sum_vector::avx512::edit_sum(
830                &mut node.values,
831                tags,
832                len,
833                width,
834                full,
835                action,
836                carry,
837            )
838        };
839        for &(lane, change) in edges.iter().flatten() {
840            node.values[lane] = mod_add(node.values[lane], change);
841            node.mul[lane] = ONE;
842            node.add[lane] = 0;
843            delta = mod_add(delta, change);
844        }
845        delta
846    }
