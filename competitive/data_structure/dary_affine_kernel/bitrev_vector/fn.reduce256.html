564    pub unsafe fn reduce256(a: [__m256i; 2], b: [__m256i; 2]) -> Affine {
565        let p = _mm256_set1_epi32(Mod::MOD as i32);
566        let p2 = _mm256_add_epi32(p, p);
567        let inv = _mm256_set1_epi32(Mod::R as i32);
568        let mut x = montgomery_mul_256(a[0], a[1], inv, p);
569        let mut y = add_mod_256(montgomery_mul_256(b[0], a[1], inv, p), b[1], p2);
570        for step in 0..3 {
571            let (rx, ry) = match step {
572                0 => (
573                    _mm256_permute2x128_si256::<1>(x, x),
574                    _mm256_permute2x128_si256::<1>(y, y),
575                ),
576                1 => (
577                    _mm256_shuffle_epi32::<0xee>(x),
578                    _mm256_shuffle_epi32::<0xee>(y),
579                ),
580                _ => (
581                    _mm256_shuffle_epi32::<0x55>(x),
582                    _mm256_shuffle_epi32::<0x55>(y),
583                ),
584            };
585            x = montgomery_mul_256(x, rx, inv, p);
586            y = add_mod_256(montgomery_mul_256(y, rx, inv, p), ry, p2);
587        }
588        (
589            _mm256_extract_epi32::<0>(x) as u32,
590            _mm256_extract_epi32::<0>(y) as u32,
591        )

597        let p2 = _mm512_add_epi32(p, p);
598        let inv = _mm512_set1_epi32(Mod::R as i32);
599        let mask = if FULL {
600            u16::MAX
601        } else {
602            PREFIX_MASK[range.end] ^ PREFIX_MASK[range.start]
603        };

605            mask,
606            _mm512_set1_epi32(ONE as i32),
607            _mm512_loadu_si512(node.mul.as_ptr().cast()),
608        );
609        let mut b = _mm512_maskz_mov_epi32(mask, _mm512_loadu_si512(node.add.as_ptr().cast()));

611            let indices: [i32; 16] = std::array::from_fn(|i| (i ^ shift) as i32);
612            let index = _mm512_loadu_si512(indices.as_ptr().cast());

616            b = add_mod_512(montgomery_mul_512(b, ra, inv, p), rb, p2);
617        }
618        (
619            _mm_cvtsi128_si32(_mm512_castsi512_si128(a)) as u32,
620            _mm_cvtsi128_si32(_mm512_castsi512_si128(b)) as u32,

1021    impl AssignmentPowerStore {
1022        fn new(len: usize) -> Self {
1023            let width = len.max(1).ilog2() as usize + 1;
1024            let mut slots = 0;
1025            let mut count = len.div_ceil(B);
1026            while count > 1 {
1027                count = count.div_ceil(B);
1028                slots += count * B;
1029            }
1030            // A sweep leaves more than slots free records, even when every tag is live.
1031            let capacity = 2 * slots + 2;
1032            assert!(u32::try_from(capacity).is_ok());
1033            Self {
1034                powers: vec![(ONE, 0); capacity * width],
1035                free: (1..capacity as u32).rev().collect(),
1036                live: vec![false; capacity],
1037                width,
1038                lasting: false,
1039            }

1042        fn insert(&mut self, value: Affine, len: usize, tags: impl Iterator<Item = u32>) -> u32 {
1043            if self.free.is_empty() {
1044                self.live.fill(false);
1045                for tag in tags {
1046                    self.live[tag as usize] = true;
1047                }
1048                self.free.extend(
1049                    (1..self.live.len())
1050                        .rev()
1051                        .filter(|&i| !self.live[i])
1052                        .map(|i| i as u32),
1053                );
1054            }
1055            let tag = self.free.pop().expect("live tag capacity");
1056            self.lasting = false;
1057            let start = tag as usize * self.width;
1058            let record = &mut self.powers[start..][..=len.ilog2() as usize];
1059            let mut power = value;
1060            record[0] = power;
1061            for next in &mut record[1..] {
1062                let (a, b) = power;
1063                power = (mod_mul(a, a), mod_mul(mod_add(a, ONE), b));
1064                *next = power;
1065            }
1066            tag
1067        }

1069        fn retain(&mut self) {
1070            self.lasting = true;
1071        }
1072
1073        fn finish(&mut self, tag: u32) {
1074            if !self.lasting {
1075                self.free.push(tag);
1076            }
1077        }
1078
1079        fn step(&self, tag: u32, shift: usize) -> Affine {
1080            self.powers[tag as usize * self.width + shift]
1081        }
1082
1083        #[inline]
1084        fn power(&self, tag: u32, mut count: usize) -> Affine {
1085            if count == 0 {
1086                return (ONE, 0);
1087            }
1088            let start = tag as usize * self.width;
1089            let mut value = self.powers[start + count.trailing_zeros() as usize];
1090            count &= count - 1;
1091            while count != 0 {
1092                value = compose(value, self.powers[start + count.trailing_zeros() as usize]);
1093                count &= count - 1;
1094            }
1095            value
1096        }
1097    }

1342        fn masks(len: usize, range: Range<usize>) -> [(u16, u16); 2] {
1343            [
1344                (
1345                    PREFIX_MASK[len],
1346                    PREFIX_MASK[range.end] ^ PREFIX_MASK[range.start],
1347                ),
1348                (
1349                    ((1u32 << len) - 1) as u16,
1350                    ((1u32 << range.end) - (1u32 << range.start)) as u16,
1351                ),
1352            ]
1353        }

1355        #[inline]
1356        #[target_feature(enable = "avx2")]
1357        unsafe fn expand(mask: u16, base: usize) -> __m256i {
1358            let bits = _mm256_setr_epi32(1, 2, 4, 8, 16, 32, 64, 128);
1359            let selected = _mm256_and_si256(_mm256_set1_epi32((mask >> base) as i32), bits);
1360            _mm256_cmpeq_epi32(selected, bits)
1361        }

1363        pub unsafe fn assign<const ISA: u8>(
1364            node: &mut CompositeLeaf,
1365            mut tags: Option<&mut [u32; B]>,
1366            len: usize,
1367            range: Range<usize>,
1368            value: Affine,
1369            carry: Option<Affine>,
1370            action_tag: u32,
1371            carry_tag: u32,
1372            tail: Option<(usize, Affine)>,
1373            edges: &[Option<(usize, Affine)>; 2],
1374        ) -> Affine {
1375            if ISA == 1 || is_x86_feature_detected!("avx2") {
1376                return avx2(
1377                    node, tags, len, range, value, carry, action_tag, carry_tag, tail, edges,
1378                );
1379            }
1380            avx512(
1381                node,
1382                tags.as_deref_mut(),
1383                len,
1384                range,
1385                value,
1386                carry,
1387                action_tag,
1388                carry_tag,
1389            );

1391            dary_composite_leaf::refold::<ISA>(node)
1392        }
1393        #[inline]
1394        #[target_feature(enable = "avx2")]
1395        unsafe fn avx2(
1396            node: &mut CompositeLeaf,
1397            mut tags: Option<&mut [u32; B]>,
1398            len: usize,
1399            range: Range<usize>,
1400            value: Affine,
1401            carry: Option<Affine>,
1402            action_tag: u32,
1403            carry_tag: u32,
1404            tail: Option<(usize, Affine)>,
1405            edges: &[Option<(usize, Affine)>; 2],
1406        ) -> Affine {
1407            let [(valid, selected), (tag_valid, tag_selected)] = masks(len, range);
1408            let a = _mm256_set1_epi32(value.0 as i32);
1409            let b = _mm256_set1_epi32(value.1 as i32);
1410            let tag = _mm256_set1_epi32(action_tag as i32);
1411            let mut all_a = [_mm256_setzero_si256(); 2];
1412            let mut all_b = all_a;
1413            for base in [0, 8] {
1414                let selected = expand(selected, base);
1415                let pa = node.mul.as_mut_ptr().add(base).cast();
1416                let pb = node.add.as_mut_ptr().add(base).cast();
1417                let mut x = _mm256_loadu_si256(pa);
1418                let mut y = _mm256_loadu_si256(pb);
1419                if let Some((a, b)) = carry {
1420                    let valid = expand(valid, base);
1421                    x = _mm256_blendv_epi8(x, _mm256_set1_epi32(a as i32), valid);
1422                    y = _mm256_blendv_epi8(y, _mm256_set1_epi32(b as i32), valid);
1423                }
1424                x = _mm256_blendv_epi8(x, a, selected);
1425                y = _mm256_blendv_epi8(y, b, selected);
1426                for &(lane, (a, b)) in tail.iter().chain(edges.iter().flatten()) {
1427                    let selected = expand(1 << REV[lane], base);
1428                    x = _mm256_blendv_epi8(x, _mm256_set1_epi32(a as i32), selected);
1429                    y = _mm256_blendv_epi8(y, _mm256_set1_epi32(b as i32), selected);
1430                }
1431                _mm256_storeu_si256(pa, x);
1432                _mm256_storeu_si256(pb, y);
1433                all_a[base / 8] = x;
1434                all_b[base / 8] = y;
1435                if let Some(tags) = &mut tags {
1436                    let tag_selected = expand(tag_selected, base);
1437                    let p = tags.as_mut_ptr().add(base).cast();
1438                    let mut old = _mm256_loadu_si256(p);
1439                    if carry.is_some() {
1440                        let tag_valid = expand(tag_valid, base);
1441                        old =
1442                            _mm256_blendv_epi8(old, _mm256_set1_epi32(carry_tag as i32), tag_valid);
1443                    }
1444                    _mm256_storeu_si256(p, _mm256_blendv_epi8(old, tag, tag_selected));
1445                }
1446            }
1447            if let Some(tags) = tags {
1448                for &(lane, _) in edges.iter().flatten() {
1449                    tags[lane] = 0;
1450                }
1451            }
1452            bitrev_vector::reduce256(all_a, all_b)
1453        }
