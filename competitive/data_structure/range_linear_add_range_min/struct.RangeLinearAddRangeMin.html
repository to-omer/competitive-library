pub struct RangeLinearAddRangeMin {
    values: Vec<i64>,
    levels: Vec<Level>,
}
Expand description

Range minimum queries with additions of an affine function of the index.

For i64 values, construction and q operations take O(n + q log² n) time and O(n) space.

149            }

175                let first = k * FANOUT;

177                let (mut lower, mut upper) = (i128::MAX, i128::MAX);
178                for child in &nodes[first..(first + FANOUT).min(nodes.len())] {
179                    lower = lower.min(child.left.min(child.right));
180                    upper = upper.min(child.left.max(child.right) + child.spread);
181                }
182                let envelope = &mut self.levels[height].envelopes[k];
183                envelope.visits = 0;
184                envelope.switches.clear();

189            node.left = lower + tag.0 * start as i128 + tag.1;
190            node.right = node.left + (tag.0 << shift);
191            node.spread = upper - lower;
192        }

204    fn fold_at(
205        &mut self,
206        height: usize,
207        range: Range<usize>,
208        ancestors: Line,
209        mut best: i128,
210    ) -> i128 {
211        let shift = self.levels[height].shift;
212        let width = 1 << shift;
213        let (edges, full) = self.split(range, shift);
214        let step = ancestors.0 << shift;
215        let initial = ancestors.0 * (full.start << shift) as i128 + ancestors.1;
216        if height != 0 {
217            let mut delta = initial;
218            for node in &self.levels[height].nodes[full.clone()] {
219                let left = node.left + delta;
220                delta += step;
221                best = best.min(left.max(node.right + delta) + node.spread);
222            }
223        }
224        for edge in edges {
225            if edge.is_empty() {
226                continue;
227            }
228            let k = edge.start >> shift;
229            let tag = self.levels[height].nodes[k].lazy;
230            let line = (ancestors.0 + tag.0, ancestors.1 + tag.1);
231            best = if height == 0 {
232                best.min(self.minimum(edge, line) as i128)
233            } else {
234                self.fold_at(height - 1, edge, line, best)
235            };
236        }
237        let mut delta = initial;
238        for k in full {
239            let node = &self.levels[height].nodes[k];
240            let left = node.left + delta;
241            delta += step;
242            let right = node.right + delta;
243            if left.min(right) >= best {
244                continue;
245            }
246            if left == right && node.spread == 0 {
247                best = left;
248                continue;
249            }
250            let start = k << shift;
251            let end = (start + width).min(self.values.len());
252            let line = (ancestors.0 + node.lazy.0, ancestors.1 + node.lazy.1);
253            let upper = left.max(right) + node.spread;
254            let minimum = if height == 0 {
255                i128::from(self.minimum(start..end, line))
256            } else if self.levels[height].envelopes[k].visits < SCAN_LIMIT {
257                self.levels[height].envelopes[k].visits += 1;
258                // Use this node's own upper bound so the cached minimum is exact.
259                self.fold_at(height - 1, start..end, line, upper)
260            } else {
261                self.ensure_envelope(height, k, line.0);
262                let (block, line) = self.winning_block(height, k, ancestors);
263                i128::from(self.minimum(block, line))
264            };
265            best = best.min(minimum);
266            let node = &mut self.levels[height].nodes[k];
267            node.right = minimum - delta;
268            node.left = node.right + step;
269            node.spread = 0;
270        }
271        best
272    }

119    fn split(&self, range: Range<usize>, shift: u32) -> ([Range<usize>; 2], Range<usize>) {
120        let mask = (1 << shift) - 1;
121        let n = self.values.len();
122        let l = (range.start + mask) >> shift;
123        let r = (range.end + if range.end == n { mask } else { 0 }) >> shift;
124        if l > r {
125            return ([range, 0..0], l..l);
126        }
127        (
128            [
129                range.start..(l << shift).min(n),
130                (r << shift).min(n)..range.end,
131            ],
132            l..r,
133        )
134    }
