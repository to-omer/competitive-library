174    fn endpoint<const MAX: bool>(&self) -> Option<(bool, usize)>
175    where
176        T: Ord,
177    {
178        let i = (self.reserved_len != 0).then_some(if MAX { self.max_rank } else { self.min_rank });
179        let mut range = self.remaining.clone();
180        let j = if MAX { range.next_back() } else { range.next() };
181        if let Some(j) = j
182            && i.is_none_or(|i| {
183                if MAX {
184                    self.initial[j] >= self.keys[self.handles[i]]
185                } else {
186                    self.initial[j] <= self.keys[self.handles[i]]
187                }
188            })
189        {
190            return Some((true, j));
191        }
192        i.map(|i| (false, i))
193    }

195    fn pop<const MAX: bool>(&mut self) -> Option<&T>
196    where
197        T: Ord,
198    {
199        let (initial, mut i) = self.endpoint::<MAX>()?;
200        if initial {
201            if MAX {
202                self.remaining.end -= 1;
203            } else {
204                self.remaining.start += 1;
205            }
206            return Some(&self.initial[i]);
207        }
208        let value = &self.keys[self.handles[i]];
209        self.reserved_len -= 1;
210        for depth in 0..self.levels.len() {
211            let word = &mut self.levels[depth][i / 64];
212            *word &= !(1 << (i % 64));
213            i /= 64;
214            if *word != 0 {
215                // The next extreme belongs to the first ancestor that is still occupied.
216                for level in self.levels[..=depth].iter().rev() {
217                    let word = level[i];
218                    let bit = if MAX {
219                        63 - word.leading_zeros()
220                    } else {
221                        word.trailing_zeros()
222                    };
223                    i = i * 64 + bit as usize;
224                }
225                if MAX {
226                    self.max_rank = i;
227                } else {
228                    self.min_rank = i;
229                }
230                return Some(value);
231            }
232        }
233        self.min_rank = usize::MAX;
234        self.max_rank = 0;
235        Some(value)
236    }
