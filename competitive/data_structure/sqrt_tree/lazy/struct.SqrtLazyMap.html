pub struct SqrtLazyMap<K, L>
where
    L: LazyMapMonoidWithKeyAct,
{
    inner: Buckets<(K, L::Key), LazyState<L, MapValue>>,
}
Expand description

An ordered map aggregating and updating values in key order, without changing keys. Value access can materialize a bucket and takes O(sqrt(N)). Range operations have the same costs and act_agg fallback as SqrtLazyVec.

386    pub fn iter(&mut self) -> impl DoubleEndedIterator<Item = (&K, &L::Key)> {
387        for bucket in &mut self.inner.buckets {
388            bucket.state.push(&mut bucket.values);
389        }
390        self.inner
391            .buckets
392            .iter()
393            .flat_map(|b| b.values.iter().map(|(k, v)| (k, v)))
394    }
