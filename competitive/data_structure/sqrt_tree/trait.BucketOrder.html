69    fn fold(&mut self, range: Range<usize>) -> MapAgg<L> {
70        let mut result = L::agg_unit();
71        assert!(range.start <= range.end);
72        if range.is_empty() {
73            return result;
74        }
75        let (start, index) = self.locate(range.start);
76        let mut offset = range.start - index;
77        for Bucket { values, state } in &mut self.buckets[start..] {
78            let end = offset + values.len();
79            if range.end <= offset {
80                break;
81            }
82            if range.start <= offset && end <= range.end {
83                L::agg_operate_assign(&mut result, &state.agg);
84            } else if range.start < end {
85                let l = range.start.saturating_sub(offset);
86                let r = range.end.min(end) - offset;
87                let fold = |initial, values: &[T], reversed| {
88                    if reversed {
89                        P::fold::<L>(initial, &values[values.len() - r..values.len() - l], true)
90                    } else {
91                        P::fold::<L>(initial, &values[l..r], false)
92                    }
93                };
94                if L::is_act_unit(&state.act) {
95                    result = fold(result, values, state.order.is_reversed());
96                } else {
97                    let agg = fold(L::agg_unit(), values, state.order.is_reversed());
98                    if let Some(agg) = L::act_agg(&agg, &state.act) {
99                        L::agg_operate_assign(&mut result, &agg);
100                    } else {
101                        state.push(values);
102                        result = fold(result, values, false);
103                    }
104                }
105            }
106            offset = end;
107        }
108        result
109    }
110    fn update(&mut self, range: Range<usize>, act: LazyMapAct<L>) {
111        assert!(range.start <= range.end);
112        if range.is_empty() || L::is_act_unit(&act) {
113            return;
114        }
115        let (start, index) = self.locate(range.start);
116        let mut offset = range.start - index;
117        for Bucket { values, state } in &mut self.buckets[start..] {
118            let end = offset + values.len();
119            if range.end <= offset {
120                break;
121            }
122            if range.start <= offset
123                && end <= range.end
124                && let Some(agg) = L::act_agg(&state.agg, &act)
125            {
126                state.agg = agg;
127                L::act_operate_assign(&mut state.act, &act);
128                offset = end;
129                continue;
130            }
131            if range.start < end {
132                state.order.push(values);
133                let l = range.start.saturating_sub(offset);
134                let r = range.end.min(end) - offset;
135                if L::is_act_unit(&state.act) {
136                    for value in &mut values[l..r] {
137                        let value = P::get_mut(value);
138                        *value = L::act_key(value, &act);
139                    }
140                    *state = LazyState::new(values);
141                    offset = end;
142                    continue;
143                }
144                let combined = L::act_operate(&state.act, &act);
145                let (prefix, rest) = values.split_at_mut(l);
146                let (middle, suffix) = rest.split_at_mut(r - l);
147                let mut agg = L::agg_unit();
148                for (values, action) in [
149                    (prefix, &state.act),
150                    (middle, &combined),
151                    (suffix, &state.act),
152                ] {
153                    if L::is_act_unit(action) {
154                        for value in values {
155                            agg = L::agg_operate_key(&agg, P::get(value));
156                        }
157                    } else {
158                        let (chunks, rest) = values.as_chunks_mut::<4>();
159                        for chunk in chunks {
160                            for value in &mut *chunk {
161                                let value = P::get_mut(value);
162                                *value = L::act_key(value, action);
163                            }
164                            agg = P::fold::<L>(agg, chunk, false);
165                        }
166                        for value in rest {
167                            let value = P::get_mut(value);
168                            *value = L::act_key(value, action);
169                            agg = L::agg_operate_key(&agg, value);
170                        }
171                    }
172                }
173                state.agg = agg;
174                state.act = L::act_unit();
175            }
176            offset = end;
177        }
178    }
