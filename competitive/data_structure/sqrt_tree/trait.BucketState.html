118    fn build(len: usize, mut values: impl Iterator<Item = T>) -> Self {
119        if len == 0 {
120            return Self::new();
121        }
122        let bucket_size = (len - 1).isqrt() + 1;
123        let mut buckets = Vec::with_capacity(len.div_ceil(bucket_size));
124        let mut sizes = Vec::with_capacity(len.div_ceil(bucket_size));
125        while let Some(first) = values.next() {
126            let mut chunk = Vec::with_capacity(bucket_size);
127            chunk.push(first);
128            chunk.extend(values.by_ref().take(bucket_size - 1));
129            sizes.push(chunk.len());
130            let state = S::new(&chunk);
131            buckets.push(Bucket {
132                values: chunk,
133                state,
134            });
135        }
136        Self {
137            buckets,
138            sizes,
139            len,
140            base_len: len,
141            bucket_size,
142        }
143    }

165    fn insert(&mut self, (mut i, mut j): (usize, usize), value: T) {
166        if self.len == 0 {
167            *self = Self::build(1, std::iter::once(value));
168            return;
169        }
170        if i == self.buckets.len() {

204    fn rebuild_if_needed(&mut self) {
205        // Size alone does not bound the number of sparse buckets under insert/remove churn.
206        if self.len < self.base_len.div_ceil(2)
207            || self.len > self.base_len.saturating_mul(2)

212                .flat_map(|mut bucket| {

214                    bucket.values
