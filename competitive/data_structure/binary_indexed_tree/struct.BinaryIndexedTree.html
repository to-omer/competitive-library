86    where
87        P: FnMut(&M::T) -> bool,

96                if pred(&nacc) {

261    fn query(&self, left: u32, right: u32) -> usize {
262        if left >= right {
263            return 0;
264        }
265        let (left, right) = (left as usize, right as usize - 1);
266        let (li, lj) = (left / 64, left % 64);
267        let (ri, rj) = (right / 64, right % 64);
268        let rj_r = 63 - rj;
269        if li == ri {
270            (self.data[li] << rj_r >> (lj + rj_r)).count_ones() as usize
271        } else {
272            let mut ans = self.bit.fold(li + 1, ri) as usize;
273            ans += (self.data[li] >> lj).count_ones() as usize;
274            ans += (self.data[ri] << rj_r).count_ones() as usize;
275            ans
276        }
277    }
