285            let level = self.level(d);

293                idx |= 1 << d;

295                range.end = self.zeros[level] + end1;

299            }
300        }

305        &self,

324            }

326            for (state, (range, k)) in states.iter_mut().zip(queries) {
327                *state = [range.start as u32, range.end as u32, *k as u32, 0];
328            }
329            for d in (0..self.bit_length).rev() {
330                let level = self.level(d);
331                for state in &mut states[..queries.len()] {
332                    let start = state[0] as usize;
333                    let end = state[1] as usize;
334                    let start1 = self.rank1(level, start);
335                    let end1 = self.rank1(level, end);
336                    let start0 = (start - start1) as u32;
337                    let end0 = (end - end1) as u32;
338                    let zeros = end0 - start0;
339                    let mask = 0u32.wrapping_sub((state[2] >= zeros) as u32);
340                    state[0] =

343                    state[2] -= zeros & mask;
344                    state[3] |= (1u32 << d) & mask;
345                }
346            }
347            result.extend(
348                states[..queries.len()]
349                    .iter()
350                    .map(|state| self.compress.values()[state[3] as usize].clone()),

352        }

203    /// Returns the number of exact matches for each `(value, range)` query.
204    pub fn rank_batch(&self, queries: impl IntoIterator<Item = (T, Range<usize>)>) -> Vec<usize> {
205        let queries: Vec<_> = queries.into_iter().collect();
206        if queries.len() < 8 || self.len > u32::MAX as usize {
207            return queries
208                .into_iter()
209                .map(|(value, range)| self.rank(value, range))
210                .collect();
211        }
212        let mut result = vec![0; queries.len()];
213        let mut active = Vec::new();
214        for (output, (value, range)) in queries.into_iter().enumerate() {
215            if let Some(index) = self.compress.index_exact(&value) {
216                active.push((output, index, range));
217            }
218        }
219        for queries in active.chunks(16) {
220            if queries.len() < 8 {
221                for (output, index, range) in queries {
222                    result[*output] = self.rank_by_index(*index, range.clone());
223                }
224                continue;
225            }
226            let mut states = [[0_u32; 3]; 16];
227            for (state, (_, index, range)) in states.iter_mut().zip(queries) {
228                *state = [range.start as u32, range.end as u32, *index as u32];
229            }
230            for d in (0..self.bit_length).rev() {
231                let level = self.level(d);
232                for state in &mut states[..queries.len()] {
233                    let start = state[0] as usize;
234                    let end = state[1] as usize;
235                    let start1 = self.rank1(level, start);
236                    let end1 = self.rank1(level, end);
237                    if ((state[2] >> d) & 1) != 0 {
238                        state[0] = (self.zeros[level] + start1) as u32;
239                        state[1] = (self.zeros[level] + end1) as u32;
240                    } else {
241                        state[0] = (start - start1) as u32;
242                        state[1] = (end - end1) as u32;
243                    }
244                }
245            }
246            for ((output, _, _), state) in queries.iter().zip(&states) {
247                result[*output] = (state[1] - state[0]) as usize;
248            }
249        }
250        result
251    }

253    /// index of k-th val
254    pub fn select(&self, val: T, k: usize) -> Option<usize> {
255        let idx = self.compress.index_exact(&val)?;
256        if self.rank_by_index(idx, 0..self.len) <= k {
257            return None;
258        }
259        let mut i = 0;
260        for d in (0..self.bit_length).rev() {
261            let level = self.level(d);
262            if ((idx >> d) & 1) != 0 {
263                i = self.zeros[level] + self.rank1(level, i);
264            } else {
265                i = self.rank0(level, i);
266            }

273                    .unwrap();
274            } else {
275                i = self.bit_vectors[level].select0(i).unwrap();

277        }

281    /// get k-th smallest value in range
282    pub fn quantile(&self, mut range: Range<usize>, mut k: usize) -> T {
283        let mut idx = 0;
284        for d in (0..self.bit_length).rev() {
285            let level = self.level(d);
286            let start1 = self.rank1(level, range.start);
287            let end1 = self.rank1(level, range.end);
288            let start0 = range.start - start1;
289            let end0 = range.end - end1;
290            let z = end0 - start0;
291            if z <= k {
292                k -= z;
293                idx |= 1 << d;
294                range.start = self.zeros[level] + start1;
295                range.end = self.zeros[level] + end1;
296            } else {
297                range.start = start0;
298                range.end = end0;
299            }
300        }
301        self.compress.values()[idx].clone()
302    }

304    pub fn quantile_batch(
305        &self,
306        queries: impl IntoIterator<Item = (Range<usize>, usize)>,
307    ) -> Vec<T> {
308        let queries: Vec<_> = queries.into_iter().collect();
309        if queries.len() < 8 || self.len > u32::MAX as usize {
310            return queries
311                .into_iter()
312                .map(|(range, k)| self.quantile(range, k))
313                .collect();
314        }
315        let mut result = Vec::with_capacity(queries.len());
316        for queries in queries.chunks(16) {
317            if queries.len() < 8 {
318                result.extend(
319                    queries
320                        .iter()
321                        .map(|(range, k)| self.quantile(range.clone(), *k)),
322                );
323                continue;
324            }
325            let mut states = [[0_u32; 4]; 16];
326            for (state, (range, k)) in states.iter_mut().zip(queries) {
327                *state = [range.start as u32, range.end as u32, *k as u32, 0];
328            }
329            for d in (0..self.bit_length).rev() {
330                let level = self.level(d);
331                for state in &mut states[..queries.len()] {
332                    let start = state[0] as usize;
333                    let end = state[1] as usize;
334                    let start1 = self.rank1(level, start);
335                    let end1 = self.rank1(level, end);
336                    let start0 = (start - start1) as u32;
337                    let end0 = (end - end1) as u32;
338                    let zeros = end0 - start0;
339                    let mask = 0u32.wrapping_sub((state[2] >= zeros) as u32);
340                    state[0] =
341                        (start0 & !mask) | ((self.zeros[level] as u32 + start1 as u32) & mask);
342                    state[1] = (end0 & !mask) | ((self.zeros[level] as u32 + end1 as u32) & mask);
343                    state[2] -= zeros & mask;
344                    state[3] |= (1u32 << d) & mask;
345                }
346            }
347            result.extend(
348                states[..queries.len()]
349                    .iter()
350                    .map(|state| self.compress.values()[state[3] as usize].clone()),
351            );
352        }
353        result
354    }

357    pub fn quantile_outer(&self, mut range: Range<usize>, mut k: usize) -> T {
358        let mut idx = 0;
359        let mut orange = 0..self.len;
360        for d in (0..self.bit_length).rev() {
361            let level = self.level(d);
362            let range_start1 = self.rank1(level, range.start);
363            let range_end1 = self.rank1(level, range.end);
364            let outer_start1 = self.rank1(level, orange.start);
365            let outer_end1 = self.rank1(level, orange.end);
366            let range_start0 = range.start - range_start1;
367            let range_end0 = range.end - range_end1;
368            let outer_start0 = orange.start - outer_start1;
369            let outer_end0 = orange.end - outer_end1;
370            let z = (outer_end0 - outer_start0) - (range_end0 - range_start0);
371            if z <= k {
372                k -= z;
373                idx |= 1 << d;
374                range.start = self.zeros[level] + range_start1;
375                range.end = self.zeros[level] + range_end1;
376                orange.start = self.zeros[level] + outer_start1;

378            } else {
379                range.start = range_start0;
380                range.end = range_end0;
381                orange.start = outer_start0;
382                orange.end = outer_end0;
383            }
384        }
385        self.compress.values()[idx].clone()

435    pub fn build_fold<M>(&self, weights: &[M::T]) -> WaveletMatrixFold<'_, T, M>
436    where
437        M: AbelianGroup,
438    {
439        let len = self.len;
440        assert_eq!(weights.len(), len);
441        let mut prefix = Vec::with_capacity((self.bit_length + 1) * (len + 1));
442        let mut current: Vec<M::T> = weights.to_vec();
443        for level in 0..self.bit_length {
444            let mut acc = M::unit();
445            prefix.push(acc.clone());
446            current = self.reorder(level, current, |w| {
447                acc = M::operate(&acc, w);
448                prefix.push(acc.clone());
449            });
450        }
451        let mut acc = M::unit();
452        prefix.push(acc.clone());
453        for w in current.into_iter() {
454            acc = M::operate(&acc, &w);
455            prefix.push(acc.clone());
456        }
457        WaveletMatrixFold {
458            wavelet_matrix: self,
459            prefix,
460        }
461    }

561    pub fn fold_range_with_count(
562        &self,
563        valrange: Range<T>,
564        mut range: Range<usize>,
565    ) -> (usize, M::T) {
566        debug_assert!(range.end <= self.wavelet_matrix.len);
567        let lower = self
568            .wavelet_matrix
569            .compress
570            .index_lower_bound(&valrange.start);
571        let upper = self
572            .wavelet_matrix
573            .compress
574            .index_lower_bound(&valrange.end);
575        if lower >= upper {
576            return (0, M::unit());
577        }
578        for d in (0..self.wavelet_matrix.bit_length).rev() {
579            let level = self.wavelet_matrix.level(d);
580            let start1 = self.wavelet_matrix.bit_vectors[level].rank1(range.start);
581            let end1 = self.wavelet_matrix.bit_vectors[level].rank1(range.end);
582            let start0 = range.start - start1;
583            let end0 = range.end - end1;
584            if ((lower >> d) & 1) == ((upper >> d) & 1) {
585                if ((lower >> d) & 1) == 0 {
586                    range = start0..end0;
587                } else {
588                    range = self.wavelet_matrix.zeros[level] + start1
589                        ..self.wavelet_matrix.zeros[level] + end1;
590                }
591                continue;
592            }
593            let zero_range = start0..end0;
594            let one_range =
595                self.wavelet_matrix.zeros[level] + start1..self.wavelet_matrix.zeros[level] + end1;

598            let (upper_count, upper_sum) = self.fold_lessthan_index_with_count(upper, one_range, d);
599            let zero_sum = self.range_sum(level + 1, zero_range.clone());

602                M::operate(&M::rinv_operate(&zero_sum, &lower_sum), &upper_sum),
