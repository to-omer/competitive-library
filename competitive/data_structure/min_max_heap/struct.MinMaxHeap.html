190    fn sift_down<const MAX: bool>(&mut self, mut i: usize) {
191        let n = self.len();
192        while i < n / 2 {
193            let child = 2 * i + 1;
194            let grandchild = 4 * i + 3;
195            let better = |a, b| Self::precedes::<MAX>(&self.data[a], &self.data[b]);
196            let best = if grandchild + 3 < n {
197                let [a, b, c, d] = self.data[grandchild..].first_chunk().unwrap();
198                let left = Self::precedes::<MAX>(b, a);
199                let right = Self::precedes::<MAX>(d, c);
200                let left_index = grandchild + usize::from(left);
201                let right_index = grandchild + 2 + usize::from(right);
202                if Self::precedes::<MAX>(if left { b } else { a }, if right { d } else { c }) {
203                    left_index
204                } else {
205                    right_index
206                }
207            } else {
208                let mut best = child;
209                for j in (child + 1..(child + 2).min(n)).chain(grandchild..(grandchild + 4).min(n))
210                {
211                    if better(j, best) {
212                        best = j;
213                    }
214                }
215                best
216            };
217            if !better(best, i) {
218                break;
219            }
220            self.data.swap(i, best);
221            if best < grandchild {
222                break;
223            }
224            let parent = (best - 1) / 2;
225            if Self::precedes::<MAX>(&self.data[parent], &self.data[best]) {
226                self.data.swap(parent, best);
227            }
228            i = best;
