238        unsafe fn difference<const ISA: u8, const POINT: bool>(
239            &self,
240            left: usize,
241            right: usize,
242        ) -> K::Value
243        where
244            K: SimdFenwickGroup,
245        {
246            if left == right {
247                return K::unit();
248            }
249            // Point access cancels the shared ancestors; range folds keep a predictable depth.
250            let depth = if POINT {
251                ((left ^ right).ilog2() / K::B.trailing_zeros() + 1) as usize
252            } else {
253                self.levels.len()
254            };
255            match depth {
256                1 => self.difference_depth::<ISA, 1>(left, right),
257                2 => self.difference_depth::<ISA, 2>(left, right),
258                3 => self.difference_depth::<ISA, 3>(left, right),
259                4 => self.difference_depth::<ISA, 4>(left, right),
260                5 => self.difference_depth::<ISA, 5>(left, right),
261                6 => self.difference_depth::<ISA, 6>(left, right),
262                7 => self.difference_depth::<ISA, 7>(left, right),
263                8 => self.difference_depth::<ISA, 8>(left, right),
264                _ => self.difference_depth::<ISA, 0>(left, right),
265            }
266        }
