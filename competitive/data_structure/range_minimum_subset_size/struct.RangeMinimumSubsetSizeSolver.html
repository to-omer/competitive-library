114    fn dfs(&mut self, lo: usize, hi: usize, eids: &mut [usize], buf: &mut [usize]) {
115        if eids.is_empty() {
116            return;
117        }
118        if lo + 1 == hi {
119            if self.xs[lo] > 0 {
120                let x = i128::from(self.xs[lo]);
121                for &eid in eids.iter() {
122                    if let RangeMinimumSubsetSizeEvent::Query(id) = self.events[eid] {
123                        let remaining = self.remaining[id];
124                        self.count[id] += ((remaining - 1) / x + 1) as usize;
125                        self.remaining[id] = 0;
126                    }
127                }
128            }
129            return;
130        }
131        let mid = (lo + hi) / 2;
132        let mut nl = 0;
133        for &eid in eids.iter() {
134            match self.events[eid] {
135                RangeMinimumSubsetSizeEvent::Update(p, v, delta) => {
136                    self.right[eid] = v >= mid;
137                    if self.right[eid] {
138                        self.bit
139                            .update(p, (delta, i128::from(delta) * i128::from(self.xs[v])));
140                    } else {
141                        nl += 1;
142                    }
143                }
144                RangeMinimumSubsetSizeEvent::Query(id) => {
145                    let range = &self.ranges[id];
146                    let (count, sum) = self.bit.fold(range.start, range.end);
147                    self.right[eid] = sum >= self.remaining[id];
148                    if !self.right[eid] {
149                        self.remaining[id] -= sum;
150                        self.count[id] += count as usize;
151                        nl += 1;
152                    }
153                }
154            }
155        }
156        let (mut il, mut ir) = (0, nl);
157        for &eid in eids.iter() {
158            if let RangeMinimumSubsetSizeEvent::Update(p, v, delta) = self.events[eid]
159                && self.right[eid]
160            {
161                self.bit
162                    .update(p, (-delta, -i128::from(delta) * i128::from(self.xs[v])));
163            }
164            if self.right[eid] {
165                buf[ir] = eid;
166                ir += 1;
167            } else {
168                buf[il] = eid;
169                il += 1;
170            }
171        }
172        let (el, er) = buf.split_at_mut(nl);
173        let (bl, br) = eids.split_at_mut(nl);
174        self.dfs(lo, mid, el, bl);
175        self.dfs(mid, hi, er, br);
176    }
