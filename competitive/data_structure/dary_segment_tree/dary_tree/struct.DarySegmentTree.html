46pub fn point_set_range_composite_dary(reader: impl Read, writer: impl Write) {
47    prepare_io!(reader, writer);
48    sc!(n, q, initial: [u32; 2 * n; capacity 2 * n], queries: [u32; 4 * q; capacity 4 * q]);
49    let (initial, _): (&[[u32; 2]], _) = initial.as_chunks();
50    let (queries, _): (&[[u32; 4]], _) = queries.as_chunks();
51    let count = queries.iter().filter(|query| query[0] == 0).count();
52    let mut keys = Vec::with_capacity(n + count);
53    keys.extend(
54        initial
55            .iter()
56            .map(|ab| InvertibleAffine::new(M::new_unchecked(ab[0]), M::new_unchecked(ab[1]))),
57    );
58    keys.extend(queries.iter().filter(|query| query[0] == 0).map(|query| {
59        InvertibleAffine::new(M::new_unchecked(query[2]), M::new_unchecked(query[3]))
60    }));
61    InvertibleAffine::prepare(&mut keys);
62    let mut seg =
63        DarySegmentTree::<DaryInvertibleCompositeKernel>::from_keys(keys[..n].iter().copied());
64    let mut update = n;
65    let answers =
66        AffineFraction::apply_many(queries.iter().enumerate().filter_map(|(i, query)| {
67            if let Some(next) = queries.get(i + 4) {
68                seg.prefetch(next[1] as usize);
69                if next[0] == 1 {
70                    seg.prefetch(next[2] as usize);
71                }
72            }
73            if query[0] == 0 {
74                seg.set(query[1] as usize, keys[update]);
75                update += 1;
76                None
77            } else {
78                Some((
79                    seg.fold_by_prefix(query[1] as usize..query[2] as usize),
80                    M::new_unchecked(query[3]),
81                ))
82            }
83        }));
84    for value in answers {
85        pp!(value);
86    }

332        unsafe fn prefix_difference<const ISA: u8>(&self, left: usize, right: usize) -> K::Value
333        where
334            K: DarySegmentGroup,
335        {
336            if right == self.len {
337                let mut index = left;
338                let mut value = K::unit();
339                for level in &self.levels {
340                    value = K::combine(
341                        K::fold::<ISA>(level.get_unchecked(index / K::B), 0..index % K::B),

344                    index /= K::B;
345                }

347            }
