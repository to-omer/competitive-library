106/// Counts length-`len` base-`radix` words whose value is `remainder` modulo `divisor`.
107/// Includes leading zeros and returns the count modulo 998244353.
108/// Requires `2 <= radix <= 10`, `1 <= divisor <= 100`, and `remainder < divisor`.
109/// Takes O(divisor^3 log len) time.
110///
111/// ```
112/// use competitive::data_structure::transducer_examples::count_words_with_remainder;
113/// assert_eq!(count_words_with_remainder(2, 3, 0, 3), 3.into()); // 000, 011, 110
114/// ```
115pub fn count_words_with_remainder(radix: usize, divisor: usize, remainder: usize, len: usize) -> M {
116    let machine = transducer!(=> || 0usize, |&state, &digit| {
117        Some(((radix * state + digit) % divisor, ()))
118    }, |&state| state == remainder);
119    let table = TransitionTable::new(&machine, radix, divisor);
120    let matrix = table
121        .to_matrix::<AddMulOperation<M>>(|_| M::from(1))
122        .pow(len);
123    (0..table.state_size())
124        .filter(|&state| table.is_accepting(state))

185    }

189    }
190
191    /// Counts the inclusive interval `[lower, upper]` with equal-length bounds.
192    pub fn count_between(&self, lower: &[usize], upper: &[usize]) -> V {
193        assert_eq!(lower.len(), upper.len());
194        if lower > upper {
195            return V::zero();
196        }
197        assert!(lower.len() < self.counts.len());
198        let mut state = 0;
199        for (i, (&lo, &hi)) in lower.iter().zip(upper).enumerate() {
200            let edges = self.table.transitions(state);
201            if lo == hi {
202                match &edges[lo] {
203                    Some((next, _)) => state = *next,
204                    None => return V::zero(),
205                }
206                continue;
207            }
208            let mut sum = V::zero();
209            for (digit, edge) in edges.iter().enumerate().take(hi + 1).skip(lo) {
210                if let Some((next, _)) = edge {
211                    let count = if digit == lo {
212                        self.count_bound::<false>(*next, &lower[i + 1..], true)
213                    } else if digit == hi {
214                        self.count_bound::<true>(*next, &upper[i + 1..], true)
215                    } else {
216                        self.count(*next, lower.len() - i - 1).clone()
217                    };
218                    sum = sum + count;
219                }
220            }
221            return sum;
222        }
223        self.count(state, 0).clone()
224    }

227impl<U, S, O> SuffixCounts<'_, Saturating<U>, S, O>
228where
229    U: Unsigned,
230{
231    /// The zero-based `k`th accepted word in lexicographic order, or `None` if absent.
232    /// Requires `k < U::MAX` because counts may saturate.
233    pub fn kth(&self, len: usize, mut k: U) -> Option<Vec<usize>> {
234        assert!(k < U::maximum());
235        if self.count(0, len).0 <= k {
236            return None;
237        }
238        let mut word = Vec::with_capacity(len);
239        let mut state = 0;
240        for remaining in (0..len).rev() {
241            for (digit, edge) in self.table.transitions(state).iter().enumerate() {
242                if let Some((next, _)) = edge {
243                    let count = self.count(*next, remaining).0;
244                    if count <= k {
245                        k -= count;
246                    } else {
247                        word.push(digit);
248                        state = *next;
249                        break;
250                    }
251                }
252            }
253        }
254        Some(word)
255    }
256
257    /// Like `kth`, within `[lower, upper]` with equal-length bounds.
258    pub fn kth_between(&self, lower: &[usize], upper: &[usize], mut k: U) -> Option<Vec<usize>>
259    where
260        Saturating<U>: std::ops::Add<Output = Saturating<U>>,
261    {
262        assert!(k < U::maximum());
263        assert_eq!(lower.len(), upper.len());
264        if lower > upper {
265            return None;
266        }
267        let lower_counts = self.bound_counts::<false>(lower);
268        let upper_counts = self.bound_counts::<true>(upper);
269        let mut state = 0;
270        let (mut tight_lower, mut tight_upper) = (true, true);
271        let mut word = Vec::with_capacity(lower.len());
272        for i in 0..lower.len() {
273            let edges = self.table.transitions(state);
274            let lo = if tight_lower { lower[i] } else { 0 };
275            let hi = if tight_upper {
276                upper[i]
277            } else {
278                self.table.radix() - 1
279            };
280            if tight_lower && tight_upper && lo == hi {
281                state = edges[lo].as_ref()?.0;
282                word.push(lo);
283                continue;
284            }
285            let mut selected = None;
286            for (digit, edge) in edges.iter().enumerate().take(hi + 1).skip(lo) {
287                if let Some((next, _)) = edge {
288                    let count = if tight_lower && digit == lo {
289                        lower_counts[i + 1].0
290                    } else if tight_upper && digit == hi {
291                        upper_counts[i + 1].0
292                    } else {
293                        self.count(*next, lower.len() - i - 1).0
294                    };
295                    if count <= k {
296                        k -= count;
297                    } else {
298                        selected = Some((digit, *next));
299                        break;
300                    }
301                }
302            }
303            let (digit, next) = selected?;
304            tight_lower &= digit == lower[i];
305            tight_upper &= digit == upper[i];
306            word.push(digit);
307            state = next;
308        }
309        (k.is_zero() && self.table.is_accepting(state)).then_some(word)
310    }
