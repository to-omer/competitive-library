240    fn bitop_assign<const OP: u8>(&mut self, rhs: &Self) {
241        assert_eq!(
242            self.size, rhs.size,
243            "bitwise operations require equal lengths"
244        );
245        #[cfg(target_arch = "x86_64")]
246        if self.bits.len() >= SIMD_MIN_BLOCKS {
247            if avx512_enabled() && is_x86_feature_detected!("avx512f") {
248                // SAFETY: blocks are 64-byte aligned and feature detection checked AVX-512F.
249                unsafe { simd::bitop_avx512::<OP>(&mut self.bits, &rhs.bits) };
250                return;
251            }
252            if is_x86_feature_detected!("avx2") {
253                // SAFETY: 64-byte alignment also satisfies AVX2 and feature detection checked it.
254                unsafe { simd::bitop_avx2::<OP>(&mut self.bits, &rhs.bits) };
255                return;
256            }
257        }
258        for (lhs, &rhs) in self.words_mut().iter_mut().zip(rhs.words()) {
259            *lhs = match OP {
260                BIT_AND => *lhs & rhs,
261                BIT_OR => *lhs | rhs,
262                BIT_XOR => *lhs ^ rhs,
263                _ => unreachable!(),
264            };
265        }
