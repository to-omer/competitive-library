878    fn build(values: &[K], backend: SimdBackend, direct: bool) -> Self {
879        assert!(matches!(K::BITS, 8 | 16 | 32 | 64 | 128));
880        assert!(values.windows(2).all(|pair| pair[0] <= pair[1]));
881        let len = values.len();
882        let storage = match K::BITS {
883            8 => StaticSearchStorage::Direct(DirectStaticSearch::build(values, K::BITS)),
884            16 => {
885                if direct {
886                    StaticSearchStorage::Direct(DirectStaticSearch::build(values, K::BITS))
887                } else {
888                    StaticSearchStorage::U16(StaticSearchTree::build(
889                        values,
890                        u16::MAX,
891                        u16::MAX as u128,
892                        |value| value as u16,
893                        backend,
894                    ))
895                }
896            }
897            32 => StaticSearchStorage::U32(StaticSearchTree::build(
898                values,
899                u32::MAX,
900                u32::MAX as u128,
901                |value| value as u32,
902                backend,
903            )),
904            64 => StaticSearchStorage::U64(StaticSearchTree::build(
905                values,
906                u64::MAX,
907                u64::MAX as u128,
908                |value| value as u64,
909                backend,
910            )),
911            128 => StaticSearchStorage::U128(StaticSearchTree::build(
912                values,
913                u128::MAX,
914                u128::MAX,
915                |value| value,
916                backend,
917            )),
918            _ => unreachable!(),
919        };
920        Self {
921            storage,
922            len,
923            marker: PhantomData,
924        }
925    }
