281pub unsafe fn max_index_u64x8_avx2(values: &[u64; 8]) -> usize {
282    let sign = _mm256_set1_epi64x(i64::MIN);
283    let low = _mm256_xor_si256(_mm256_loadu_si256(values.as_ptr().cast()), sign);
284    let high = _mm256_xor_si256(_mm256_loadu_si256(values.as_ptr().add(4).cast()), sign);
285    let mut maximum = max_i64x4(low, high);
286    maximum = max_i64x4(maximum, _mm256_permute4x64_epi64::<0x4e>(maximum));
287    maximum = max_i64x4(maximum, _mm256_permute4x64_epi64::<0xb1>(maximum));
288    let low = _mm256_movemask_pd(_mm256_castsi256_pd(_mm256_cmpeq_epi64(low, maximum)));
289    let high = _mm256_movemask_pd(_mm256_castsi256_pd(_mm256_cmpeq_epi64(high, maximum)));
290    ((low as u32 | ((high as u32) << 4)) as u8).trailing_zeros() as usize
291}

305pub unsafe fn max_index_u128x4_avx2(low: &[u64; 4], high: &[u64; 4]) -> usize {
306    let low = _mm256_loadu_si256(low.as_ptr().cast());
307    let high = _mm256_loadu_si256(high.as_ptr().cast());
308    let mut maximum_high = high;
309    maximum_high = max_epu64(maximum_high, _mm256_permute4x64_epi64::<0x4e>(maximum_high));
310    maximum_high = max_epu64(maximum_high, _mm256_permute4x64_epi64::<0xb1>(maximum_high));
311    let high_equal = _mm256_cmpeq_epi64(high, maximum_high);
312    let high_mask = _mm256_movemask_pd(_mm256_castsi256_pd(high_equal)) as u32 & 15;
313    if high_mask.is_power_of_two() {
314        return high_mask.trailing_zeros() as usize;
315    }
316    let mut maximum_low = _mm256_and_si256(low, high_equal);
317    maximum_low = max_epu64(maximum_low, _mm256_permute4x64_epi64::<0x4e>(maximum_low));
318    maximum_low = max_epu64(maximum_low, _mm256_permute4x64_epi64::<0xb1>(maximum_low));
319    let both = _mm256_and_si256(high_equal, _mm256_cmpeq_epi64(low, maximum_low));
320    (_mm256_movemask_pd(_mm256_castsi256_pd(both)) as u32 & 15).trailing_zeros() as usize
321}
