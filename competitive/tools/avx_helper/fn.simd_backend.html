16fn batch_ntt_simd_backend(len: usize, width: usize) -> SimdBackend {

50const fn reduce(z: u64, p: u32, r: u32) -> u32 {
51    let mut z = ((z + r.wrapping_mul(z as u32) as u64 * p as u64) >> 32) as u32;
52    if z >= p {
53        z -= p;
54    }
55    z
56}
57const fn mod_mul(x: u32, y: u32, p: u32, r: u32) -> u32 {
58    reduce(x as u64 * y as u64, p, r)
59}
60const fn mod_pow(mut x: u32, mut y: u32, p: u32, r: u32, mut z: u32) -> u32 {
61    while y > 0 {
62        if y & 1 == 1 {
63            z = mod_mul(z, x, p, r);
64        }
65        x = mod_mul(x, x, p, r);
66        y >>= 1;
67    }
68    z
69}

71pub trait Montgomery32NttModulus: Sized + MontgomeryReduction32 {
72    const PRIMITIVE_ROOT: u32 = {
73        let mut g = 3u32;
74        loop {
75            let mut ok = true;
76            let mut d = 1u32;
77            while d * d < Self::MOD {
78                if (Self::MOD - 1) % d == 0 {
79                    let ds = [d, (Self::MOD - 1) / d];
80                    let mut i = 0;
81                    while i < 2 {
82                        ok &= ds[i] == Self::MOD - 1
83                            || mod_pow(
84                                reduce(g as u64 * Self::N2 as u64, Self::MOD, Self::R),
85                                ds[i],
86                                Self::MOD,
87                                Self::R,
88                                Self::N1,
89                            ) != Self::N1;
90                        i += 1;
91                    }
92                }
93                d += 1;
94            }
95            if ok {
96                break;
97            }
98            g += 2;
99        }
100        g
101    };
102    const RANK: u32 = (Self::MOD - 1).trailing_zeros();
103    const INFO: NttInfo = NttInfo::new::<Self>();
104}

107pub struct NttInfo {
108    root: [u32; 32],
109    inv_root: [u32; 32],
110    rate3: [u32; 32],
111    inv_rate3: [u32; 32],
112    rate3_packed: [[u32; 8]; 32],
113    inv_rate3_packed: [[u32; 8]; 32],
114}
115impl NttInfo {
116    const fn new<M>() -> Self
117    where
118        M: Montgomery32NttModulus,
119    {
120        let mut root = [0; 32];
121        let mut inv_root = [0; 32];
122        let mut rate3 = [0; 32];
123        let mut inv_rate3 = [0; 32];
124        let mut rate3_packed = [[0; 8]; 32];
125        let mut inv_rate3_packed = [[0; 8]; 32];
126        let rank = M::RANK as usize;
127
128        let g = reduce(M::PRIMITIVE_ROOT as u64 * M::N2 as u64, M::MOD, M::R);
129        root[rank] = mod_pow(g, (M::MOD - 1) >> rank, M::MOD, M::R, M::N1);
130        inv_root[rank] = mod_pow(root[rank], M::MOD - 2, M::MOD, M::R, M::N1);
131        let mut i = rank - 1;
132        loop {
133            root[i] = mod_mul(root[i + 1], root[i + 1], M::MOD, M::R);
134            inv_root[i] = mod_mul(inv_root[i + 1], inv_root[i + 1], M::MOD, M::R);
135            if i == 0 {
136                break;
137            }
138            i -= 1;
139        }
140
141        let (mut i, mut prod, mut inv_prod) = (0, M::N1, M::N1);
142        while i < rank - 2 {
143            rate3[i] = mod_mul(root[i + 3], prod, M::MOD, M::R);
144            let rate3_2 = mod_mul(rate3[i], rate3[i], M::MOD, M::R);
145            let rate3_3 = mod_mul(rate3_2, rate3[i], M::MOD, M::R);
146            inv_rate3[i] = mod_mul(inv_root[i + 3], inv_prod, M::MOD, M::R);
147            let inv_rate3_2 = mod_mul(inv_rate3[i], inv_rate3[i], M::MOD, M::R);
148            let inv_rate3_3 = mod_mul(inv_rate3_2, inv_rate3[i], M::MOD, M::R);
149            rate3_packed[i] = [
150                rate3[i].wrapping_mul(M::R),
151                rate3[i],
152                rate3_2.wrapping_mul(M::R),
153                rate3_2,
154                rate3_3.wrapping_mul(M::R),
155                rate3_3,
156                0,
157                0,
158            ];
159            inv_rate3_packed[i] = [
160                inv_rate3[i].wrapping_mul(M::R),
161                inv_rate3[i],
162                inv_rate3_2.wrapping_mul(M::R),
163                inv_rate3_2,
164                inv_rate3_3.wrapping_mul(M::R),
165                inv_rate3_3,
166                0,
167                0,
168            ];
169            prod = mod_mul(prod, inv_root[i + 3], M::MOD, M::R);
170            inv_prod = mod_mul(inv_prod, root[i + 3], M::MOD, M::R);
171            i += 1;
172        }
173
174        NttInfo {
175            root,
176            inv_root,
177            rate3,
178            inv_rate3,
179            rate3_packed,
180            inv_rate3_packed,
181        }
182    }
183}

215fn ntt_batch_scalar<M>(a: &mut [MInt<M>], width: usize)
216where
217    M: Montgomery32NttModulus,
218{
219    let n = a.len() / width;
220    if n <= 1 {
221        return;
222    }
223    let mut v = n / 2;
224    if n.trailing_zeros() & 1 == 1 {
225        let (l, r) = a.split_at_mut(v * width);
226        for (x0, x1) in l.iter_mut().zip(r) {
227            let a0 = *x0;
228            let a1 = *x1;
229            *x0 = a0 + a1;
230            *x1 = a0 - a1;
231        }
232        v >>= 1;
233    }
234    let imag = MInt::<M>::new_unchecked(M::INFO.root[2]);
235    while v > 1 {
236        let mut w1 = MInt::<M>::one();
237        for (s, a) in a.chunks_exact_mut((v << 1) * width).enumerate() {
238            let (l, r) = a.split_at_mut(v * width);
239            let (ll, lr) = l.split_at_mut((v >> 1) * width);
240            let (rl, rr) = r.split_at_mut((v >> 1) * width);
241            let w2 = w1 * w1;
242            let w3 = w1 * w2;
243            for (((x0, x1), x2), x3) in ll.iter_mut().zip(lr).zip(rl).zip(rr) {
244                let a0 = *x0;
245                let a1 = *x1 * w1;
246                let a2 = *x2 * w2;
247                let a3 = *x3 * w3;
248                let a0pa2 = a0 + a2;
249                let a0na2 = a0 - a2;
250                let a1pa3 = a1 + a3;
251                let a1na3imag = (a1 - a3) * imag;
252                *x0 = a0pa2 + a1pa3;
253                *x1 = a0pa2 - a1pa3;
254                *x2 = a0na2 + a1na3imag;
255                *x3 = a0na2 - a1na3imag;
256            }
257            w1 *= MInt::<M>::new_unchecked(M::INFO.rate3[s.trailing_ones() as usize]);
258        }
259        v >>= 2;
260    }
261}

293fn intt_batch_scalar<M>(a: &mut [MInt<M>], width: usize)
294where
295    M: Montgomery32NttModulus,
296{
297    let n = a.len() / width;
298    if n <= 1 {
299        return;
300    }
301    let mut v = 1;
302    let limit = if n.trailing_zeros() & 1 == 1 {
303        n / 2
304    } else {
305        n
306    };
307    let iimag = MInt::<M>::new_unchecked(M::INFO.inv_root[2]);
308    while v < limit {
309        let mut w1 = MInt::<M>::one();
310        for (s, a) in a.chunks_exact_mut((v << 2) * width).enumerate() {
311            let (l, r) = a.split_at_mut((v << 1) * width);
312            let (ll, lr) = l.split_at_mut(v * width);
313            let (rl, rr) = r.split_at_mut(v * width);
314            let w2 = w1 * w1;
315            let w3 = w1 * w2;
316            for (((x0, x1), x2), x3) in ll.iter_mut().zip(lr).zip(rl).zip(rr) {
317                let a0 = *x0;
318                let a1 = *x1;
319                let a2 = *x2;
320                let a3 = *x3;
321                let a0pa1 = a0 + a1;
322                let a0na1 = a0 - a1;
323                let a2pa3 = a2 + a3;
324                let a2na3iimag = (a2 - a3) * iimag;
325                *x0 = a0pa1 + a2pa3;
326                *x1 = (a0na1 + a2na3iimag) * w1;
327                *x2 = (a0pa1 - a2pa3) * w2;
328                *x3 = (a0na1 - a2na3iimag) * w3;
329            }
330            w1 *= MInt::<M>::new_unchecked(M::INFO.inv_rate3[s.trailing_ones() as usize]);
331        }
332        v <<= 2;
333    }
334    if n.trailing_zeros() & 1 == 1 {
335        let (l, r) = a.split_at_mut(n / 2 * width);
336        for (x0, x1) in l.iter_mut().zip(r) {
337            let a0 = *x0;
338            let a1 = *x1;
339            *x0 = a0 + a1;
340            *x1 = a0 - a1;
341        }
342    }
343    let inv = MInt::<M>::from(n as u32).inv();
344    for a in a {
345        *a *= inv;
346    }
347}
