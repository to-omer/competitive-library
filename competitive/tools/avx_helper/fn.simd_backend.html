49const fn reduce(z: u64, p: u32, r: u32) -> u32 {
50    let mut z = ((z + r.wrapping_mul(z as u32) as u64 * p as u64) >> 32) as u32;
51    if z >= p {
52        z -= p;
53    }
54    z
55}

59const fn mod_pow(mut x: u32, mut y: u32, p: u32, r: u32, mut z: u32) -> u32 {

61        if y & 1 == 1 {
62            z = mod_mul(z, x, p, r);
63        }
64        x = mod_mul(x, x, p, r);
65        y >>= 1;
66    }
67    z

70pub trait Montgomery32NttModulus: Sized + MontgomeryReduction32 {
71    const PRIMITIVE_ROOT: u32 = {
72        let mut g = 3u32;
73        loop {
74            let mut ok = true;
75            let mut d = 1u32;
76            while d * d < Self::MOD {
77                if (Self::MOD - 1) % d == 0 {
78                    let ds = [d, (Self::MOD - 1) / d];
79                    let mut i = 0;
80                    while i < 2 {
81                        ok &= ds[i] == Self::MOD - 1
82                            || mod_pow(
83                                reduce(g as u64 * Self::N2 as u64, Self::MOD, Self::R),
84                                ds[i],
85                                Self::MOD,
86                                Self::R,
87                                Self::N1,
88                            ) != Self::N1;
89                        i += 1;
90                    }
91                }
92                d += 1;
93            }
94            if ok {
95                break;
96            }
97            g += 2;
98        }
99        g
100    };
101    const RANK: u32 = (Self::MOD - 1).trailing_zeros();
102    const INFO: NttInfo = NttInfo::new::<Self>();
103}

250fn ntt_batch_scalar<M>(a: &mut [MInt<M>], width: usize)
251where
252    M: Montgomery32NttModulus,
253{
254    let n = a.len() / width;
255    if n <= 1 {
256        return;
257    }
258    let mut v = n / 2;
259    if n.trailing_zeros() & 1 == 1 {
260        let (l, r) = a.split_at_mut(v * width);
261        for (x0, x1) in l.iter_mut().zip(r) {
262            let a0 = *x0;
263            let a1 = *x1;
264            *x0 = a0 + a1;
265            *x1 = a0 - a1;
266        }
267        v >>= 1;
268    }
269    let imag = MInt::<M>::new_unchecked(M::INFO.root[2]);
270    while v > 1 {
271        let mut w1 = MInt::<M>::one();
272        let mut w2 = w1;
273        let mut w3 = w1;
274        for (s, a) in a.chunks_exact_mut((v << 1) * width).enumerate() {
275            let (l, r) = a.split_at_mut(v * width);
276            let (ll, lr) = l.split_at_mut((v >> 1) * width);
277            let (rl, rr) = r.split_at_mut((v >> 1) * width);
278            for (((x0, x1), x2), x3) in ll.iter_mut().zip(lr).zip(rl).zip(rr) {
279                let a0 = *x0;
280                let a1 = *x1 * w1;
281                let a2 = *x2 * w2;
282                let a3 = *x3 * w3;
283                let a0pa2 = a0 + a2;
284                let a0na2 = a0 - a2;
285                let a1pa3 = a1 + a3;
286                let a1na3imag = (a1 - a3) * imag;
287                *x0 = a0pa2 + a1pa3;
288                *x1 = a0pa2 - a1pa3;
289                *x2 = a0na2 + a1na3imag;
290                *x3 = a0na2 - a1na3imag;
291            }
292            let rate = &M::INFO.rate3_packed[s.trailing_ones() as usize];
293            w1 *= MInt::<M>::new_unchecked(rate[1]);
294            w2 *= MInt::<M>::new_unchecked(rate[3]);
295            w3 *= MInt::<M>::new_unchecked(rate[5]);
296        }
297        v >>= 2;
298    }

332where
333    M: Montgomery32NttModulus,
334{
335    let n = a.len() / width;
336    if n <= 1 {
337        return;
338    }

340    // normalized before the typed slice is used again.

344        n / 2
345    } else {
346        n
347    };
348    let iimag = M::INFO.inv_root[2];
349    while v < limit {
350        let mut w1 = M::N1;
351        let mut w2 = w1;
352        let mut w3 = w1;
353        for (s, a) in a.chunks_exact_mut((v << 2) * width).enumerate() {
354            let (l, r) = a.split_at_mut((v << 1) * width);
355            let (ll, lr) = l.split_at_mut(v * width);

368                *x2 = mul_scalar::<M>(sub_scalar::<M>(a0pa1, a2pa3), w2);
369                *x3 = mul_scalar::<M>(sub_scalar::<M>(a0na1, a2na3iimag), w3);
370            }
371            let rate = &M::INFO.inv_rate3_packed[s.trailing_ones() as usize];
372            w1 = M::mod_mul(w1, rate[1]);

374            w3 = M::mod_mul(w3, rate[5]);
375        }
376        v <<= 2;
377    }
378    if n.trailing_zeros() & 1 == 1 {
379        let (l, r) = a.split_at_mut(n / 2 * width);
380        for (x0, x1) in l.iter_mut().zip(r) {

385        }

388    for a in a {
389        *a = M::mod_mul(*a, inv);
390    }
391}

427            let ng3 = montgomery_simd::montgomery_mul_256(x3, nr3, r, modulus);
428            let g2 = montgomery_simd::montgomery_mul_256(x2, r2, r, modulus);
429            let g0 = shrink_avx2(x0, modulus2);
430            let h3 = montgomery_simd::montgomery_mul_256_fixed(
431                _mm256_add_epi32(g1, ng3),
432                imag,
433                imag_r,
434                modulus,
435            );
436            let h1 = montgomery_simd::sub_mod_256(g1, ng3, modulus2);
437            let h0 = montgomery_simd::add_mod_256(g0, g2, modulus2);
438            let h2 = montgomery_simd::sub_mod_256(g0, g2, modulus2);
439            store(i, _mm256_add_epi32(h0, h1));
440            store(i + 1, lazy_sub_avx2(h0, h1, modulus2));
441            store(i + 2, _mm256_add_epi32(h2, h3));
442            store(i + 3, lazy_sub_avx2(h2, h3, modulus2));
443            i += 4;
444        }
445        roots[0] = root;
446
447        tile += tile_len;
448        if tile < n {
449            stage_log = (tile.trailing_zeros() as usize & !1).min(max_stage_log);
450        }
451    }
452}
453
454#[target_feature(enable = "avx2")]
455unsafe fn intt_blocks_avx2<M>(a: *mut u32, n: usize)
456where
457    M: Montgomery32NttModulus,
458{
459    let inv = M::mod_inv(<M as MIntConvert<u32>>::from(n as u32));
460    let modulus = _mm256_set1_epi32(M::MOD as i32);
461    let modulus2 = _mm256_set1_epi32(M::MOD.wrapping_mul(2) as i32);
462    let wide = matches!(simd_backend(), SimdBackend::Avx512);
463
464    let initial_root = _mm256_setr_epi32(M::N1 as i32, 0, M::N1 as i32, 0, M::N1 as i32, 0, 0, 0);
465    let log_n = n.trailing_zeros() as usize;
466    let mut roots = [initial_root; 16];
467    let nn = n >> (log_n & 1);
468    let tile_len = n.min(64);
469    let inv_vec = _mm256_set1_epi32(inv as i32);
470
471    roots[0] = inv_vec;
472    let mut tile = 0;
473    while tile < n {
474        let max_stage_log = (tile + tile_len).trailing_zeros() as usize;
475        let mut stage_log = 2usize;
476        let mut group_len = 4usize;
477        let mut quarter = 1usize;
478        while stage_log <= max_stage_log {
479            let offset = tile + tile_len - group_len.max(tile_len);
480            let base = a.add(offset << 3);
481            let mut i = 0;
482            let root_slot = (stage_log - 2) >> 1;
483            let mut root = roots[root_slot];
484            let mut group = (tile + i) >> stage_log;
485            while i < tile_len {
486                let current = root;
487                root = update_root_avx2(
488                    root,
489                    packed_rate_avx2::<M>((!group).trailing_zeros() as usize, true),
490                    modulus,
491                );
492                let base = base.add(i << 3);
493                let last = group_len == n;
494                if wide && quarter > 1 {
495                    intt_butterfly_avx512::<M>(base, quarter, current, inv, group, last);
496                } else {
497                    intt_butterfly_avx2::<M>(base, quarter, current, inv, group, last);
498                }
499                i += group_len;
500                group += 1;
501            }
502            roots[root_slot] = root;
503            quarter = group_len;
504            group_len <<= 2;
505            stage_log += 2;
506        }
507        tile += tile_len;
508    }
509    if nn != n {
510        let mut i = 0;
511        while i < nn {
512            let x0 = load_block_avx2(a, i);
513            let x1 = load_block_avx2(a, nn + i);
514            store_block_avx2(
515                a,
516                i,
517                shrink_avx2(montgomery_simd::add_mod_256(x0, x1, modulus2), modulus),
518            );
519            store_block_avx2(
520                a,
521                nn + i,
522                shrink_avx2(montgomery_simd::sub_mod_256(x0, x1, modulus2), modulus),
523            );
524            i += 1;
525        }
526    }
527}
