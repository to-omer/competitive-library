201        #[cfg(target_arch = "x86_64")]

208                // SAFETY: blocks are 64-byte aligned and feature detection checked AVX-512F.

210            } else if is_x86_feature_detected!("avx2") {

367        if rhs == 0 {
368            return;
369        }
370        if rhs >= self.size {
371            if !OR_ASSIGN {
372                self.reset();
373            }
374            return;
375        }
376        #[cfg(target_arch = "x86_64")]
377        if self.bits.len() >= SIMD_MIN_BLOCKS {
378            if avx512_enabled()
379                && is_x86_feature_detected!("avx512f")
380                && is_x86_feature_detected!("avx512vbmi2")
381            {
382                // SAFETY: blocks are aligned and feature detection checked AVX-512F and VBMI2.
383                unsafe { simd::shift_left_avx512::<OR_ASSIGN>(&mut self.bits, rhs) };
384                if self.size & 511 != 0 {
385                    self.trim();
386                }
387                return;
388            }
389            if is_x86_feature_detected!("avx2") {
390                // SAFETY: feature detection checked AVX2 support.
391                unsafe { simd::shift_left_avx2::<OR_ASSIGN>(self.words_mut(), rhs) };
392                if self.size & 63 != 0 {
393                    self.trim();
394                }
395                return;
396            }
397        }
398
399        let bits = self.words_mut();
400        let word_shift = rhs >> 6;
401        let bit_shift = rhs & 63;
402        if bit_shift == 0 {
403            for i in (0..bits.len() - word_shift).rev() {
404                if OR_ASSIGN {
405                    bits[i + word_shift] |= bits[i];
406                } else {
407                    bits[i + word_shift] = bits[i];
408                }
409            }
410        } else {
411            for i in (1..bits.len() - word_shift).rev() {
412                let value = (bits[i] << bit_shift) | (bits[i - 1] >> (64 - bit_shift));
413                if OR_ASSIGN {
414                    bits[i + word_shift] |= value;
415                } else {
416                    bits[i + word_shift] = value;
417                }
418            }
419            if OR_ASSIGN {
420                bits[word_shift] |= bits[0] << bit_shift;
421            } else {
422                bits[word_shift] = bits[0] << bit_shift;
423            }
424        }
425        if !OR_ASSIGN {
426            bits[..word_shift].fill(0);
427        }
428        if self.size & 63 != 0 {
429            self.trim();
430        }

208        a: &[Vec<Self>],
209        b: &[Vec<Self>],
210        scale: u32,
211    ) -> Vec<Vec<Self>> {
212        let (n, m, p) = (a.len(), b.len(), b.first().map_or(0, Vec::len));
213        assert!(a.iter().all(|row| row.len() == m));
214        assert!(b.iter().all(|row| row.len() == p));
215        let modulus = M::get_mod();
216        let alignment = if n.min(m).min(p) <= 64 { 8 } else { 32 };
217        let (nn, mm, pp) = (
218            n.div_ceil(alignment) * alignment,
219            m.div_ceil(alignment) * alignment,
220            p.div_ceil(alignment) * alignment,
221        );
222        let mut depth = 0;
223        let (mut x, mut y, mut z) = (nn, mm, pp);
224        while x.min(y).min(z) > 64 && x % 16 == 0 && y % 16 == 0 && z % 16 == 0 {
225            depth += 1;
226            x /= 2;
227            y /= 2;
228            z /= 2;
229        }

231        // A recursive level uses one quarter of its parent's storage; siblings reuse it.
232        let mut data = if entries + entries / 3 >= 1 << 20 {
233            let mut data = Vec::with_capacity(entries + entries / 3);
234            advise_huge_pages(&mut data);
235            data.resize(entries + entries / 3, 0u32);
236            data
237        } else {
238            vec![0u32; entries + entries / 3]
239        };

241        let mut inverse = 1u32;
242        for _ in 0..5 {
243            inverse = inverse.wrapping_mul(2u32.wrapping_sub(modulus.wrapping_mul(inverse)));
244        }
245        let inverse = inverse.wrapping_neg();

247            let (nr, nc) = (nn >> depth, mm >> depth);
248            for i in row..(row + nr).min(n) {

251                for j in col..(col + nc).min(m) {

254                    let x = x.wrapping_mul(scale).wrapping_sub(q.wrapping_mul(modulus));

256                }

258        }

260            let (nr, nc) = (mm >> depth, pp >> depth);
261            for i in row..(row + nr).min(m) {
262                // SAFETY: MInt is transparent over u32 and row lengths were checked above.
263                let values: &[u32] = unsafe { std::slice::from_raw_parts(b[i].as_ptr().cast(), p) };
