pub struct FastInput<'input> {
    ptr: *const u8,
    end: *const u8,
    _marker: PhantomData<&'input [u8]>,
}
Expand description

Token reader for little-endian targets. Integer reads require a decimal representation that fits the requested type, with at most that type’s maximum number of digits and an optional - for signed types.

Every read, including iterator reads, must start at an available token. Reads consume exactly one trailing ASCII whitespace byte. parse and ScanSource require UTF-8. Slices returned by bytes must not outlive the input allocation. bytes and parse may also read an empty field at a delimiter.

158    pub unsafe fn u32(&mut self) -> u32 {
159        unsafe {
160            let mut x = u64::from_le(ptr::read_unaligned(self.ptr.cast())) ^ 0x3030303030303030;
161            let mask = x & 0xf0f0f0f0f0f0f0f0;
162            if mask != 0 {
163                self.fetch_ud8()
164            } else {
165                x = x.wrapping_mul(10).wrapping_add(x >> 8) & 0x00ff00ff00ff00ff;
166                x = x.wrapping_mul(100).wrapping_add(x >> 16) & 0x0000ffff0000ffff;
167                x = x.wrapping_mul(10000).wrapping_add(x >> 32) & 0xffffffff;
168                let y = u16::from_le(ptr::read_unaligned(self.ptr.add(8).cast())) ^ 0x3030;
169                if y & 0xf0f0 == 0 {
170                    self.ptr = self.ptr.add(11);
171                    x as u32 * 100 + ((y.wrapping_mul(10).wrapping_add(y >> 8)) & 0xff) as u32
172                } else if y & 0xf0 == 0 {
173                    self.ptr = self.ptr.add(10);
174                    x as u32 * 10 + (y & 0xff) as u32
175                } else {
176                    self.ptr = self.ptr.add(9);
177                    x as u32
178                }
179            }
180        }

203    unsafe fn extend_u32_avx2(&mut self, values: &mut Vec<u32>, count: usize) {
204        use std::arch::x86_64::*;
205        const CHUNK: usize = 1 << 17;
206        const CAPACITY: usize = CHUNK / 8 + 16;
207        #[inline]
208        #[target_feature(enable = "avx2")]
209        unsafe fn two_tokens(input: &mut *const u8) -> __m256i {
210            unsafe {
211                let bytes = _mm256_loadu_si256((*input).cast());
212                let separators =
213                    _mm256_movemask_epi8(_mm256_cmpgt_epi8(_mm256_set1_epi8(33), bytes)) as u32;
214                let first = separators.trailing_zeros() as usize;
215                let second = (separators & (separators - 1)).trailing_zeros() as usize;
216                let windows = _mm256_inserti128_si256::<1>(
217                    bytes,
218                    _mm_loadu_si128((*input).add(first + 1).cast()),
219                );
220                let rows = _mm256_set_m128i(
221                    _mm_loadu_si128(FastInput::SHUFFLE.as_ptr().add(second - first - 1).cast()),
222                    _mm_loadu_si128(FastInput::SHUFFLE.as_ptr().add(first).cast()),
223                );
224                *input = (*input).add(second + 1);
225                let digits = _mm256_shuffle_epi8(
226                    _mm256_subs_epu8(windows, _mm256_set1_epi8(b'0' as i8)),
227                    rows,
228                );
229                _mm256_madd_epi16(
230                    _mm256_maddubs_epi16(digits, _mm256_set1_epi16(0x010a)),
231                    _mm256_set1_epi32(0x00010064),
232                )
233            }
234        }
235        unsafe {
236            let start = values.len();
237            let mut scratch = Vec::<u32>::with_capacity(3 * CAPACITY);
238            let scratch = scratch.as_mut_ptr();
239            let mut written = 0;
240            while count - written >= 544 {
241                // At least two bytes per token; the extra 32 tokens cover boundary alignment.
242                let chunk = CHUNK.min(2 * (count - written - 32)) & !3;
243                let mut end = [self.ptr; 4];
244                for (k, end) in end.iter_mut().enumerate() {
245                    let mut p = self.ptr.add(chunk * (k + 1) / 4);
246                    while *p > b' ' {
247                        p = p.add(1);
248                    }
249                    *end = p.add(1);
250                }
251                let mut input = [self.ptr, end[0], end[1], end[2]];
252                let output = [
253                    values.as_mut_ptr().add(start + written),
254                    scratch,
255                    scratch.add(CAPACITY),
256                    scratch.add(2 * CAPACITY),
257                ];
258                let mut done = 0;
259                loop {
260                    let left = (0..4).map(|k| end[k].offset_from(input[k])).min().unwrap();
261                    if left < 33 {
262                        break;
263                    }
264                    // Each pair consumes at most 22 bytes; leave 32 bytes for its SIMD load.
265                    for _ in 0..(left - 32) / 22 + 1 {
266                        let g0 = two_tokens(&mut input[0]);
267                        let g1 = two_tokens(&mut input[1]);
268                        let g2 = two_tokens(&mut input[2]);
269                        let g3 = two_tokens(&mut input[3]);
270                        let k = _mm256_set1_epi32(0x00012710);
271                        let h01 =
272                            _mm256_castsi256_ps(_mm256_madd_epi16(_mm256_packus_epi32(g0, g1), k));
273                        let h23 =
274                            _mm256_castsi256_ps(_mm256_madd_epi16(_mm256_packus_epi32(g2, g3), k));
275                        let high = _mm256_castps_si256(_mm256_shuffle_ps::<0x88>(h01, h23));
276                        let low = _mm256_castps_si256(_mm256_shuffle_ps::<0xdd>(h01, h23));
277                        let v = _mm256_add_epi32(
278                            _mm256_mullo_epi32(high, _mm256_set1_epi32(100000000)),
279                            low,
280                        );
281                        let q = _mm256_permutevar8x32_epi32(
282                            v,
283                            _mm256_setr_epi32(0, 4, 1, 5, 2, 6, 3, 7),
284                        );
285                        ptr::write_unaligned(
286                            output[0].add(done).cast::<u64>(),
287                            _mm256_extract_epi64::<0>(q) as u64,
288                        );
289                        ptr::write_unaligned(
290                            output[1].add(done).cast::<u64>(),
291                            _mm256_extract_epi64::<1>(q) as u64,
292                        );
293                        ptr::write_unaligned(
294                            output[2].add(done).cast::<u64>(),
295                            _mm256_extract_epi64::<2>(q) as u64,
296                        );
297                        ptr::write_unaligned(
298                            output[3].add(done).cast::<u64>(),
299                            _mm256_extract_epi64::<3>(q) as u64,
300                        );
301                        done += 2;
302                    }
303                }
304                let mut lengths = [done; 4];
305                for k in 0..4 {
306                    self.ptr = input[k];
307                    while self.ptr < end[k] {
308                        output[k].add(lengths[k]).write(self.u32());
309                        lengths[k] += 1;
310                    }
311                }
312                written += lengths[0];
313                for k in 1..4 {
314                    ptr::copy_nonoverlapping(
315                        output[k],
316                        values.as_mut_ptr().add(start + written),
317                        lengths[k],
318                    );
319                    written += lengths[k];
320                }
321                self.ptr = end[3];
322            }
323            values.set_len(start + written);
324        }
325    }

335    const SHUFFLE: [u8; 32] = const {
336        let mut table = [128; 32];
337        let mut i = 16;
338        while i < 32 {
339            table[i] = (i - 16) as u8;
340            i += 1;
341        }
342        table

347    unsafe fn parse_ud16(digits: std::arch::x86_64::__m128i) -> u64 {
348        use std::arch::x86_64::*;
349        unsafe {
350            let pairs = _mm_maddubs_epi16(digits, _mm_set1_epi16(0x010a));
351            let quads = _mm_madd_epi16(pairs, _mm_set1_epi32(0x00010064));
352            let octets = _mm_add_epi64(
353                _mm_mul_epu32(quads, _mm_set1_epi64x(10000)),
354                _mm_srli_epi64::<32>(quads),
355            );
356            (_mm_cvtsi128_si64(octets) as u64) * 100000000
357                + (_mm_cvtsi128_si64(_mm_srli_si128::<8>(octets)) as u64)
358        }
359    }

363    unsafe fn u64_simd<const MAX_DIGITS: usize>(&mut self) -> u64 {
364        use std::arch::x86_64::*;
365        unsafe {
366            let mut digits =
367                _mm_sub_epi8(_mm_loadu_si128(self.ptr.cast()), _mm_set1_epi8(b'0' as i8));
368            let mask = _mm_movemask_epi8(digits) as u32;
369            if mask != 0 {
370                let len = mask.trailing_zeros();
371                digits = _mm_shuffle_epi8(
372                    digits,
373                    _mm_loadu_si128(Self::SHUFFLE.as_ptr().add(len as usize).cast()),
374                );
375                self.ptr = self.ptr.add(len as usize + 1);
376                return Self::parse_ud16(digits);
377            }
378            self.ptr = self.ptr.add(16);
379            let mut res = Self::parse_ud16(digits);
380            let mut rem = ptr::read_unaligned(self.ptr.cast::<u32>()) ^ 0x30303030;
381            if (rem & 0xf0f0f0) == 0 {
382                if MAX_DIGITS == 19 {
383                    res = res.wrapping_mul(1000).wrapping_add(
384                        ((rem & 0xff) as u64)
385                            .wrapping_mul(100)
386                            .wrapping_add((((rem.wrapping_mul(2561)) & 0xff0000) >> 16) as u64),
387                    );
388                    self.ptr = self.ptr.add(4);
389                } else {
390                    let four = (rem & 0xf0f0f0f0) == 0;
391                    rem = rem.wrapping_shl((!four as u32) << 3);
392                    rem = rem.wrapping_mul(10).wrapping_add(rem >> 8) & 0x00ff00ff;
393                    rem = rem.wrapping_mul(100).wrapping_add(rem >> 16) & 0x0000ffff;
394                    res = res
395                        .wrapping_mul(1000 + 9000 * four as u64)
396                        .wrapping_add(rem as u64);
397                    self.ptr = self.ptr.add(4 + four as usize);
398                }
399            } else if (rem & 0xf0f0) == 0 {
400                res = res
401                    .wrapping_mul(100)
402                    .wrapping_add((((rem >> 8).wrapping_add(rem.wrapping_mul(10))) & 0xff) as u64);
403                self.ptr = self.ptr.add(3);
404            } else if (rem & 0xf0) == 0 {
405                res = res.wrapping_mul(10).wrapping_add((rem & 0x0000000f) as u64);
406                self.ptr = self.ptr.add(2);
407            } else {
408                self.ptr = self.ptr.add(1);
409            }
410            res
411        }
412    }

267                        let g1 = two_tokens(&mut input[1]);
268                        let g2 = two_tokens(&mut input[2]);
269                        let g3 = two_tokens(&mut input[3]);
270                        let k = _mm256_set1_epi32(0x00012710);
271                        let h01 =
272                            _mm256_castsi256_ps(_mm256_madd_epi16(_mm256_packus_epi32(g0, g1), k));
273                        let h23 =
274                            _mm256_castsi256_ps(_mm256_madd_epi16(_mm256_packus_epi32(g2, g3), k));
275                        let high = _mm256_castps_si256(_mm256_shuffle_ps::<0x88>(h01, h23));
276                        let low = _mm256_castps_si256(_mm256_shuffle_ps::<0xdd>(h01, h23));
277                        let v = _mm256_add_epi32(
278                            _mm256_mullo_epi32(high, _mm256_set1_epi32(100000000)),
279                            low,
280                        );
281                        let q = _mm256_permutevar8x32_epi32(
282                            v,
283                            _mm256_setr_epi32(0, 4, 1, 5, 2, 6, 3, 7),
284                        );
285                        ptr::write_unaligned(
286                            output[0].add(done).cast::<u64>(),
287                            _mm256_extract_epi64::<0>(q) as u64,
288                        );
289                        ptr::write_unaligned(
290                            output[1].add(done).cast::<u64>(),
291                            _mm256_extract_epi64::<1>(q) as u64,
292                        );
293                        ptr::write_unaligned(
294                            output[2].add(done).cast::<u64>(),
295                            _mm256_extract_epi64::<2>(q) as u64,
296                        );
297                        ptr::write_unaligned(
298                            output[3].add(done).cast::<u64>(),
299                            _mm256_extract_epi64::<3>(q) as u64,
300                        );
301                        done += 2;
302                    }
303                }
304                let mut lengths = [done; 4];
305                for k in 0..4 {
306                    self.ptr = input[k];
307                    while self.ptr < end[k] {
308                        output[k].add(lengths[k]).write(self.u32());
309                        lengths[k] += 1;
310                    }
311                }
312                written += lengths[0];
313                for k in 1..4 {
314                    ptr::copy_nonoverlapping(
315                        output[k],
316                        values.as_mut_ptr().add(start + written),
317                        lengths[k],
318                    );
319                    written += lengths[k];
320                }
321                self.ptr = end[3];
322            }
323            values.set_len(start + written);
324        }

347    unsafe fn parse_ud16(digits: std::arch::x86_64::__m128i) -> u64 {
348        use std::arch::x86_64::*;
349        unsafe {
350            let pairs = _mm_maddubs_epi16(digits, _mm_set1_epi16(0x010a));
351            let quads = _mm_madd_epi16(pairs, _mm_set1_epi32(0x00010064));
352            let octets = _mm_add_epi64(
353                _mm_mul_epu32(quads, _mm_set1_epi64x(10000)),
354                _mm_srli_epi64::<32>(quads),
355            );
356            (_mm_cvtsi128_si64(octets) as u64) * 100000000
357                + (_mm_cvtsi128_si64(_mm_srli_si128::<8>(octets)) as u64)

363    unsafe fn u64_simd<const MAX_DIGITS: usize>(&mut self) -> u64 {
364        use std::arch::x86_64::*;
365        unsafe {
366            let mut digits =
367                _mm_sub_epi8(_mm_loadu_si128(self.ptr.cast()), _mm_set1_epi8(b'0' as i8));
368            let mask = _mm_movemask_epi8(digits) as u32;
369            if mask != 0 {
370                let len = mask.trailing_zeros();
371                digits = _mm_shuffle_epi8(
372                    digits,
373                    _mm_loadu_si128(Self::SHUFFLE.as_ptr().add(len as usize).cast()),
374                );
375                self.ptr = self.ptr.add(len as usize + 1);
376                return Self::parse_ud16(digits);
377            }
378            self.ptr = self.ptr.add(16);
379            let mut res = Self::parse_ud16(digits);
380            let mut rem = ptr::read_unaligned(self.ptr.cast::<u32>()) ^ 0x30303030;
381            if (rem & 0xf0f0f0) == 0 {
382                if MAX_DIGITS == 19 {
383                    res = res.wrapping_mul(1000).wrapping_add(
384                        ((rem & 0xff) as u64)
385                            .wrapping_mul(100)
386                            .wrapping_add((((rem.wrapping_mul(2561)) & 0xff0000) >> 16) as u64),
387                    );
388                    self.ptr = self.ptr.add(4);
389                } else {
390                    let four = (rem & 0xf0f0f0f0) == 0;
391                    rem = rem.wrapping_shl((!four as u32) << 3);
392                    rem = rem.wrapping_mul(10).wrapping_add(rem >> 8) & 0x00ff00ff;
393                    rem = rem.wrapping_mul(100).wrapping_add(rem >> 16) & 0x0000ffff;
394                    res = res
395                        .wrapping_mul(1000 + 9000 * four as u64)
396                        .wrapping_add(rem as u64);
397                    self.ptr = self.ptr.add(4 + four as usize);
398                }
399            } else if (rem & 0xf0f0) == 0 {
400                res = res
401                    .wrapping_mul(100)
402                    .wrapping_add((((rem >> 8).wrapping_add(rem.wrapping_mul(10))) & 0xff) as u64);
403                self.ptr = self.ptr.add(3);
404            } else if (rem & 0xf0) == 0 {
405                res = res.wrapping_mul(10).wrapping_add((rem & 0x0000000f) as u64);
406                self.ptr = self.ptr.add(2);
407            } else {
408                self.ptr = self.ptr.add(1);
409            }
410            res
411        }

491    pub unsafe fn u128(&mut self) -> u128 {
492        use std::arch::x86_64::*;
493        const POW10: [u64; 16] = const {
494            let mut powers = [1; 16];
495            let mut i = 1;
496            while i < 16 {
497                powers[i] = powers[i - 1] * 10;
498                i += 1;
499            }
500            powers
501        };
502        unsafe {
503            let first = _mm_sub_epi8(_mm_loadu_si128(self.ptr.cast()), _mm_set1_epi8(b'0' as i8));
504            let mask = _mm_movemask_epi8(first) as u32;
505            if mask != 0 {
506                let len = mask.trailing_zeros() as usize;
507                self.ptr = self.ptr.add(len + 1);
508                return Self::parse_ud16(_mm_shuffle_epi8(
509                    first,
510                    _mm_loadu_si128(Self::SHUFFLE.as_ptr().add(len).cast()),
511                )) as u128;
512            }
513            let second = _mm_sub_epi8(
514                _mm_loadu_si128(self.ptr.add(16).cast()),
515                _mm_set1_epi8(b'0' as i8),
516            );
517            let mask = _mm_movemask_epi8(second) as u32;
518            if mask != 0 {
519                let len = mask.trailing_zeros() as usize;
520                self.ptr = self.ptr.add(17 + len);
521                return Self::parse_ud16(first) as u128 * POW10[len] as u128
522                    + Self::parse_ud16(_mm_shuffle_epi8(
523                        second,
524                        _mm_loadu_si128(Self::SHUFFLE.as_ptr().add(len).cast()),
525                    )) as u128;
526            }
527            let mut tail = ptr::read_unaligned(self.ptr.add(32).cast::<u64>()) ^ 0x3030303030303030;
528            let len = (tail & 0xf0f0f0f0f0f0f0f0).trailing_zeros() as usize / 8;
529            tail = if len == 0 { 0 } else { tail << ((8 - len) * 8) };
530            tail = tail.wrapping_mul(10).wrapping_add(tail >> 8) & 0x00ff00ff00ff00ff;
531            tail = tail.wrapping_mul(100).wrapping_add(tail >> 16) & 0x0000ffff0000ffff;
532            tail = tail.wrapping_mul(10000).wrapping_add(tail >> 32) & 0xffffffff;
533            self.ptr = self.ptr.add(33 + len);
534            (Self::parse_ud16(first) as u128 * 10_000_000_000_000_000
535                + Self::parse_ud16(second) as u128)
536                * POW10[len] as u128
537                + tail as u128
538        }
539    }

766/// Tokens may span reads and borrowed tokens remain valid until the next scan.

778    pub unsafe fn new(reader: R) -> Self {
779        let buffer = vec![b' '; (1 << 16) + 16];
780        let input = FastInput {
781            ptr: buffer.as_ptr(),
782            end: buffer.as_ptr(),
783            _marker: PhantomData,

794    #[cold]
795    fn refill(&mut self) -> bool {
796        if self.eof {
797            return false;
798        }
799        let position = unsafe { self.input.ptr.offset_from(self.buffer.as_ptr()) as usize };
800        self.buffer.copy_within(position..self.filled, 0);
801        self.filled -= position;
802        self.input = FastInput {
803            ptr: self.buffer.as_ptr(),
804            end: self.buffer.as_ptr(),
805            _marker: PhantomData,
806        };
807        loop {
808            if self.filled == self.buffer.len() - 16 {
809                self.buffer.resize(2 * self.filled + 16, b' ');
810                self.input = FastInput {
811                    ptr: self.buffer.as_ptr(),
812                    end: self.buffer.as_ptr(),
813                    _marker: PhantomData,
814                };
815            }
816            let end = self.buffer.len() - 16;
817            let start = self.filled;
818            match self.reader.read(&mut self.buffer[self.filled..end]) {
819                Ok(0) => {
820                    self.eof = true;
821                    if self.filled != 0 {
822                        self.buffer[self.filled] = b' ';
823                        self.filled += 1;
824                    }
825                }
826                Ok(n) => self.filled += n,
827                Err(e) if e.kind() == std::io::ErrorKind::Interrupted => continue,
828                Err(e) => panic!("io error: {e}"),
829            }
830            let (prefix, chunks) = self.buffer[start..self.filled].as_rchunks();
831            let end = prefix.len()
832                + 8 * chunks
833                    .iter()
834                    .rposition(|&chunk| {
835                        let x = u64::from_le_bytes(chunk);
836                        x.wrapping_sub(0x2121_2121_2121_2121) & !x & 0x8080_8080_8080_8080 != 0
837                    })
838                    .map_or(0, |i| i + 1);
839            if let Some(end) = self.buffer[start..start + end]
840                .iter()
841                .rposition(u8::is_ascii_whitespace)
842            {
843                self.buffer[self.filled..self.filled + 16].fill(b' ');
844                self.input = FastInput {
845                    ptr: self.buffer.as_ptr(),
846                    end: unsafe { self.buffer.as_ptr().add(start + end + 1) },
847                    _marker: PhantomData,
848                };
849                return true;
850            }
851            if self.eof {
852                self.input = FastInput {
853                    ptr: self.buffer.as_ptr(),
854                    end: self.buffer.as_ptr(),
855                    _marker: PhantomData,
856                };
857                return false;
858            }
859        }
860    }
