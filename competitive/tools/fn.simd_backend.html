54const fn reduce(z: u64, p: u32, r: u32) -> u32 {
55    let mut z = ((z + r.wrapping_mul(z as u32) as u64 * p as u64) >> 32) as u32;
56    if z >= p {
57        z -= p;
58    }
59    z
60}

64const fn mod_pow(mut x: u32, mut y: u32, p: u32, r: u32, mut z: u32) -> u32 {
65    while y > 0 {
66        if y & 1 == 1 {
67            z = mod_mul(z, x, p, r);
68        }
69        x = mod_mul(x, x, p, r);
70        y >>= 1;
71    }
72    z

75pub trait Montgomery32NttModulus: Sized + MontgomeryReduction32 {
76    const PRIMITIVE_ROOT: u32 = {
77        let mut g = 3u32;
78        loop {
79            let mut ok = true;
80            let mut d = 1u32;
81            while d * d < Self::MOD {
82                if (Self::MOD - 1) % d == 0 {
83                    let ds = [d, (Self::MOD - 1) / d];
84                    let mut i = 0;
85                    while i < 2 {
86                        ok &= ds[i] == Self::MOD - 1
87                            || mod_pow(
88                                reduce(g as u64 * Self::N2 as u64, Self::MOD, Self::R),
89                                ds[i],
90                                Self::MOD,
91                                Self::R,
92                                Self::N1,
93                            ) != Self::N1;
94                        i += 1;
95                    }
96                }
97                d += 1;
98            }
99            if ok {
100                break;
101            }
102            g += 2;
103        }
104        g
105    };
106    const RANK: u32 = (Self::MOD - 1).trailing_zeros();
107    const INFO: NttInfo = NttInfo::new::<Self>();
108}

259fn ntt_batch_scalar<M>(a: &mut [MInt<M>], width: usize)
260where
261    M: Montgomery32NttModulus,
262{
263    let n = a.len() / width;
264    if n <= 1 {
265        return;
266    }
267    let mut v = n / 2;
268    if n.trailing_zeros() & 1 == 1 {
269        let (l, r) = a.split_at_mut(v * width);
270        for (x0, x1) in l.iter_mut().zip(r) {
271            let a0 = *x0;
272            let a1 = *x1;
273            *x0 = a0 + a1;
274            *x1 = a0 - a1;
275        }
276        v >>= 1;
277    }
278    let imag = MInt::<M>::new_unchecked(M::INFO.root[2]);
279    while v > 1 {
280        let mut w1 = MInt::<M>::one();
281        let mut w2 = w1;
282        let mut w3 = w1;
283        for (s, a) in a.chunks_exact_mut((v << 1) * width).enumerate() {
284            let (l, r) = a.split_at_mut(v * width);
285            let (ll, lr) = l.split_at_mut((v >> 1) * width);
286            let (rl, rr) = r.split_at_mut((v >> 1) * width);
287            for (((x0, x1), x2), x3) in ll.iter_mut().zip(lr).zip(rl).zip(rr) {
288                let a0 = *x0;
289                let a1 = *x1 * w1;
290                let a2 = *x2 * w2;
291                let a3 = *x3 * w3;
292                let a0pa2 = a0 + a2;
293                let a0na2 = a0 - a2;
294                let a1pa3 = a1 + a3;
295                let a1na3imag = (a1 - a3) * imag;
296                *x0 = a0pa2 + a1pa3;
297                *x1 = a0pa2 - a1pa3;
298                *x2 = a0na2 + a1na3imag;
299                *x3 = a0na2 - a1na3imag;
300            }
301            let rate = &M::INFO.rate3_packed[s.trailing_ones() as usize];
302            w1 *= MInt::<M>::new_unchecked(rate[1]);
303            w2 *= MInt::<M>::new_unchecked(rate[3]);
304            w3 *= MInt::<M>::new_unchecked(rate[5]);
305        }
306        v >>= 2;
307    }
308}

340fn intt_batch_scalar<M>(a: &mut [MInt<M>], width: usize)
341where
342    M: Montgomery32NttModulus,
343{
344    let n = a.len() / width;
345    if n <= 1 {
346        return;
347    }
348    // MInt is transparent over u32; lazy residues stay below 2 * MOD and are
349    // normalized before the typed slice is used again.
350    let a = unsafe { std::slice::from_raw_parts_mut(a.as_mut_ptr().cast::<u32>(), a.len()) };
351    let mut v = 1;
352    let limit = if n.trailing_zeros() & 1 == 1 {
353        n / 2
354    } else {
355        n
356    };
357    let iimag = M::INFO.inv_root[2];
358    while v < limit {
359        let mut w1 = M::N1;
360        let mut w2 = w1;
361        let mut w3 = w1;
362        for (s, a) in a.chunks_exact_mut((v << 2) * width).enumerate() {
363            let (l, r) = a.split_at_mut((v << 1) * width);
364            let (ll, lr) = l.split_at_mut(v * width);
365            let (rl, rr) = r.split_at_mut(v * width);
366            for (((x0, x1), x2), x3) in ll.iter_mut().zip(lr).zip(rl).zip(rr) {
367                let a0 = *x0;
368                let a1 = *x1;
369                let a2 = *x2;
370                let a3 = *x3;
371                let a0pa1 = add_scalar::<M>(a0, a1);
372                let a0na1 = sub_scalar::<M>(a0, a1);
373                let a2pa3 = add_scalar::<M>(a2, a3);
374                let a2na3iimag = mul_scalar::<M>(sub_scalar::<M>(a2, a3), iimag);
375                *x0 = add_scalar::<M>(a0pa1, a2pa3);
376                *x1 = mul_scalar::<M>(add_scalar::<M>(a0na1, a2na3iimag), w1);
377                *x2 = mul_scalar::<M>(sub_scalar::<M>(a0pa1, a2pa3), w2);
378                *x3 = mul_scalar::<M>(sub_scalar::<M>(a0na1, a2na3iimag), w3);
379            }
380            let rate = &M::INFO.inv_rate3_packed[s.trailing_ones() as usize];
381            w1 = M::mod_mul(w1, rate[1]);
382            w2 = M::mod_mul(w2, rate[3]);
383            w3 = M::mod_mul(w3, rate[5]);
384        }
385        v <<= 2;
386    }
387    if n.trailing_zeros() & 1 == 1 {
388        let (l, r) = a.split_at_mut(n / 2 * width);
389        for (x0, x1) in l.iter_mut().zip(r) {
390            let a0 = *x0;
391            let a1 = *x1;
392            *x0 = add_scalar::<M>(a0, a1);
393            *x1 = sub_scalar::<M>(a0, a1);
394        }
395    }
396    let inv = M::mod_inv(<M as MIntConvert<u32>>::from(n as u32));
397    for a in a {
398        *a = M::mod_mul(*a, inv);
399    }
400}

505        let mut zero = 0;
506        for (chunk, &word) in indices.chunks(64).zip(words) {
507            if word == 0 {
508                next[zero..zero + chunk.len()].copy_from_slice(chunk);
509                zero += chunk.len();
510            } else if word == u64::MAX {
511                next[one..one + chunk.len()].copy_from_slice(chunk);
512                one += chunk.len();

516                    next[if bit { one } else { zero }] = index;
517                    zero += !bit as usize;
