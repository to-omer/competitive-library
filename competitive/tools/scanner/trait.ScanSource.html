207/// scan a value with Scanner
208///
209/// - `scan_value!(scanner, ELEMENT)`
210///
211/// ELEMENT :=
212/// - `$ty`: Scan
213/// - `@$expr`: MarkedScan
214/// - `$ty = $expr`: MarkedScan
215/// - `[ELEMENT; $expr]`: vector
216/// - `[ELEMENT; const $expr]`: array
217/// - `[ELEMENT]`: iterator
218/// - `[ELEMENT; iter $expr]`: iterator of the specified length
219/// - `($(ELEMENT)*,)`: tuple
220#[macro_export]
221macro_rules! scan_value {
222    (@repeat $scanner:expr, [$($t:tt)*] $len:expr)                             => { { $crate::scan_value!(@iter $scanner, [$($t)*] $len).collect::<Vec<_>>() } };
