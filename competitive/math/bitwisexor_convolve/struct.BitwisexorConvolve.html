131        let n = a.len().ilog2() as usize;
132        let width = n - 1;
133        let ranks = std::array::from_fn(|i| i.count_ones() as usize);
134        let zero = std::array::from_fn(|_| R::zero());
135        let mut x = vec![zero.clone(); a.len() / (8 * W) * width];
136        let mut y = x.clone();
137        let mut local = vec![zero.clone(); W.ilog2() as usize + 4];
138        let mut result = vec![R::zero(); a.len()];
139
140        // |A| + |B| - |A xor B| = 2 |A & B|. The rank sum also fixes the parity
141        // of A xor B, so its lowest bit can be omitted from the Walsh transform.
142        for frequency in 0..4 {
143            local.fill(zero.clone());
144            Self::project_ranks::<W, false>(a, &mut x, &mut local, frequency, width, &ranks);
145            Self::project_ranks::<W, true>(b, &mut y, &mut local, frequency, width, &ranks);
146            for (x, y) in x.chunks_exact_mut(width).zip(y.chunks_exact(width)) {
147                for k in (0..width).rev() {
148                    x[k] = R::dot_product_batch(&x[..=k], &y[width - 1 - k..]);
149                }
150            }
151            BitwisexorConvolve::<R::Additive>::hadamard_transform_rows(
152                x.as_flattened_mut(),
153                width * W,
154            );
155            for (middle, (block, output)) in x
156                .chunks_exact(width)
157                .zip(result.chunks_exact_mut(8 * W))
158                .enumerate()
159            {
160                let base = middle.count_ones() as usize;
161                // Omitting both constant terms leaves product ranks 2 through n.
162                for (k, row) in local.iter_mut().enumerate() {
163                    if base + k < 2 {
164                        row.fill(R::zero());
165                    } else {
166                        row.clone_from(&block[base + k - 2]);
167                        BitwisexorConvolve::<R::Additive>::hadamard_transform(row);
168                    }
169                }
170                for (j, (out, &rank)) in output
171                    .as_chunks_mut::<8>()
172                    .0
173                    .iter_mut()
174                    .zip(&ranks)
175                    .enumerate()
176                {
177                    let low_ranks = [0, 1, 1, 2, 1, 2, 2, 3];
178                    for ((i, value), low_rank) in out.iter_mut().enumerate().zip(low_ranks) {
179                        let term = &local[rank + low_rank][j];
180                        if frequency == 0 {
181                            value.clone_from(term);
182                        } else if [false, true, true, false][frequency & (i >> 1)] {
183                            R::sub_assign(value, term);
184                        } else {
185                            R::add_assign(value, term);
186                        }
187                    }
188                }
189            }
190        }
191        let mut scale = R::one();
192        for _ in 1..n {
193            R::mul_assign(&mut scale, &inverse_two);
194        }
195        result[0] = R::mul(&a[0], &b[0]);
196        for i in 1..a.len() {
197            result[i] = R::dot_product(
198                &[result[i].clone(), a[0].clone(), a[i].clone()],
199                &[scale.clone(), b[i].clone(), b[0].clone()],

206    fn project_ranks<const W: usize, const REVERSE: bool>(
207        input: &[R::T],
208        transformed: &mut [[R::T; W]],
209        local: &mut [[R::T; W]],
210        frequency: usize,
211        width: usize,
212        ranks: &[usize; W],
213    ) {
214        let first = |x: &R::T, y: &R::T| {
215            if frequency & 1 == 0 {
216                R::add(x, y)
217            } else {
218                R::sub(x, y)
219            }
220        };
221        let second = |x: &R::T, y: &R::T| {
222            if frequency & 2 == 0 {
223                R::add(x, y)
224            } else {
225                R::sub(x, y)
226            }
227        };
228        let zero = std::array::from_fn(|_| R::zero());
229        for (middle, (src, block)) in input
230            .chunks_exact(8 * W)
231            .zip(transformed.chunks_exact_mut(width))
232            .enumerate()
233        {
234            for (j, (s, &k)) in src.as_chunks::<8>().0.iter().zip(ranks).enumerate() {
235                local[k][j].clone_from(&s[0]);
236                local[k + 1][j] = second(&first(&s[1], &s[2]), &s[4]);
237                let low = if frequency & 1 == 0 {
238                    s[3].clone()
239                } else {
240                    R::neg(&s[3])
241                };
242                local[k + 2][j] = second(&low, &first(&s[5], &s[6]));
243                local[k + 3][j] = if frequency == 1 || frequency == 2 {
244                    R::neg(&s[7])
245                } else {
246                    s[7].clone()
247                };
248            }
249            let base = middle.count_ones() as usize;
250            let end = (base + local.len() - 1).min(width);
251            block.fill(zero.clone());
252            for rank in base.max(1)..=end {
253                let source = &local[rank - base];
254                let index = if REVERSE { width - rank } else { rank - 1 };
255                block[index].clone_from(source);
256                BitwisexorConvolve::<R::Additive>::hadamard_transform(&mut block[index]);
257            }
258        }
259        BitwisexorConvolve::<R::Additive>::hadamard_transform_rows(
260            transformed.as_flattened_mut(),
261            width * W,
262        );
263    }
