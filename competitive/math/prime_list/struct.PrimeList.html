48pub fn count_primes(n: u64) -> u64 {
49    if n < 1 << 16 {
50        return PrimeList::new(n as u32).len() as u64;
51    }
52    let root = n.isqrt() as u32;
53    let y = (4 * (n as f64).cbrt() as u32).min(root);
54    let table = PrimeCounts::new(root);
55    let primes = &table.primes[..table.count(y)];
56    let factors = signed_factors(y, primes);
57    let split = table.count(root.isqrt());
58
59    // Meissel's formula: pi(n) = pi(y) - 1 + phi(n, pi(y)) - P2(n, pi(y)).
60    let mut sum = primes.len() as i128 - 1 + i128::from(tiny_phi(n));
61    for (i, &factor) in factors.iter().enumerate().skip(1) {
62        if factor != 0 {
63            sum += i128::from(factor.signum()) * i128::from(tiny_phi(n / (2 * i + 1) as u64));
64        }
65    }
66    sum += easy_leaves(n, &table, primes.len(), split);
67    sum += sieved_leaves_minus_semiprimes(n, y, &table, &factors, split);
68    sum.try_into().unwrap()
69}

96fn easy_leaves(n: u64, table: &PrimeCounts, a: usize, split: usize) -> i128 {
97    let remaining = (a - split) as i128;
98    let mut sum = remaining * (remaining - 1) / 2;
99    for (i, &p) in table.primes[..a].iter().enumerate().skip(split) {
100        let t = n / u64::from(p);
101        let mid = table.count(t.isqrt() as u32);
102        if mid <= i {
103            break;
104        }
105        let end = mid.min(a);
106        let count = table.count((t / u64::from(p)) as u32);
107        sum += (count.min(a) - end) as i128;
108        for &q in &table.primes[i + 1..end] {
109            let count = table.count((t / u64::from(q)) as u32);
110            sum += (count - i + count.min(a) - end) as i128;
111        }
112    }
113    sum
114}

116fn sieved_leaves_minus_semiprimes(
117    n: u64,
118    y: u32,
119    table: &PrimeCounts,
120    factors: &[i32],
121    split: usize,
122) -> i128 {
123    let z = n / u64::from(y);
124    // Every full segment starts at the same phase of the presieve.
125    const PAGE_SIZE: u64 = 240 * 4 * PATTERN.len() as u64;
126    let mut partial_counts = vec![0u64; split];
127    let mut prime_count = (SMALL_PRIMES.len() - 1) as u64;
128    let a = table.count(y);
129    let mut upper = table.primes.len();
130    let mut sum = 0;
131    for low in (0..=z).step_by(PAGE_SIZE as usize) {
132        let high = z.min(low + PAGE_SIZE - 1);
133        let mut segment = Segment::new(low, high);
134        let sieve_end = table.count(high.isqrt() as u32);
135        let partial_end = split.min(table.count((n / low.max(1)).isqrt() as u32));
136        for (i, base) in partial_counts[..partial_end]
137            .iter_mut()
138            .enumerate()
139            .skip(SMALL_PRIMES.len())
140        {
141            let prime = table.primes[i];
142            let p = u64::from(prime);
143            let t = n / p;
144            let count = segment.build_prefix(high.min(t / p));
145            // Sieving from p^2 retains the earlier primes as well as 1.
146            let offset = *base as i64 + SMALL_PRIMES.len() as i64 - i as i64;
147            sum += table.special_leaves(&segment, t, prime, y, factors, offset);
148            *base += u64::from(count);
149            segment.cross_off(p);
150        }
151        for &p in table.primes[..sieve_end]
152            .iter()
153            .skip(partial_end.max(SMALL_PRIMES.len()))
154        {
155            segment.cross_off(u64::from(p));
156        }
157        let count = segment.build_prefix(high);
158        while upper > a {
159            let x = n / u64::from(table.primes[upper - 1]);
160            if x > high {
161                break;
162            }
163            upper -= 1;
164            let pi = prime_count + u64::from(segment.count(x));
165            sum -= i128::from(pi) - upper as i128;
166        }
167        prime_count += u64::from(count);
168    }
169    sum
170}

121    /// to those of `f * g`, multiplying coefficients of `f` from the left.
122    /// Uses $O(n^{2/3})$ ring operations.
123    pub fn dirichlet_convolve<R>(&self, rhs: &Self) -> Self
124    where
125        T: Clone,
126        R: Ring<T = T, Additive: Invertible>,
127    {
128        assert_eq!(self.n, rhs.n);
129        let f = self.dense_coefficients::<R>();
130        let g = rhs.dense_coefficients::<R>();
131        let mut result = vec![R::zero(); self.root];
132        for (i, weight) in g.iter().enumerate() {
133            let cells = result[i..].iter_mut().step_by(i + 1);
134            for (cell, left) in cells.zip(&f) {
135                R::add_assign(cell, &R::mul(left, weight));
136            }
137        }
138        for i in 1..self.root {
139            result[i] = R::add(&result[i - 1], &result[i]);
140        }
141        self.dirichlet_prefixes::<R>(DirichletOperand::Factor(rhs), f, g, result)
142    }

144    /// Converts inclusive prefix sums of `f` to those of its Dirichlet inverse
145    /// in $O(n^{2/3})$ ring operations.
146    /// `first_inverse` must be the two-sided multiplicative inverse of `self[1]`.
147    pub fn dirichlet_inverse<R>(&self, first_inverse: T) -> Self
148    where
149        T: Clone,
150        R: Ring<T = T, Additive: Invertible>,
151    {
152        let f = self.dense_coefficients::<R>();
153        let mut g = vec![R::zero(); self.root];
154        g[0] = first_inverse.clone();
155        let mut result = Vec::with_capacity(self.data.len());
156        let mut sum = R::zero();
157        for i in 1..=self.root {
158            let (weight, rest) = g[i - 1..].split_first_mut().unwrap();
159            if i > 1 {
160                *weight = R::mul(&first_inverse, &R::neg(weight));
161            }
162            let cells = rest.iter_mut().skip(i - 1).step_by(i);
163            for (cell, left) in cells.zip(&f[1..]) {
164                R::add_assign(cell, &R::mul(left, weight));
165            }
166            R::add_assign(&mut sum, weight);
167            result.push(sum.clone());
168        }
169        self.dirichlet_prefixes::<R>(DirichletOperand::Inverse(first_inverse), f, g, result)
170    }

172    fn smooth_sum<R>(
173        &self,
174        powers: &[Vec<(u64, T)>],
175        d: u64,
176        limit: u64,
177        cutoff: usize,
178        before: &T,
179    ) -> T
180    where
181        R: Ring<T = T, Additive: Invertible>,
182    {
183        let mut sum = R::one();
184        if limit > cutoff as u64 {
185            let i = if limit <= self.root as u64 {
186                limit as usize - 1
187            } else {
188                self.data.len() - d as usize
189            };
190            R::add_assign(&mut sum, &R::sub(&self.data[i], before));
191        }
192        for (j, powers_of_p) in powers.iter().enumerate() {
193            if powers_of_p[0].0 > limit {
194                break;
195            }
196            for (power, weight) in powers_of_p {
197                if *power > limit {
198                    break;
199                }
200                let rest = self.smooth_sum::<R>(
201                    &powers[j + 1..],
202                    d * power,
203                    limit / power,
204                    cutoff,
205                    before,
206                );
207                R::add_assign(&mut sum, &R::mul(weight, &rest));
208            }
209        }
210        sum

213    fn sieve<G, const REMOVE: bool>(
214        &mut self,
215        cutoff: usize,
216        mul: impl Fn(&T, &T) -> T,
217        mut next: impl FnMut(u64, u32, &T, &T) -> T,
218    ) where
219        T: Clone,
220        G: AbelianGroup<T = T>,
221    {
222        let primes: Vec<_> = PrimeList::new(self.root as u32)
223            .primes()
224            .map(|p| {
225                let i = p as usize - 1;
226                (
227                    u64::from(p),
228                    G::rinv_operate(&self.data[i], &self.data[i - 1]),
229                )
230            })
231            .collect();
232        let fourth = self.root.isqrt();
233        let small_problem = self.root <= 128;
234        let dense = if small_problem {
235            self.root
236        } else {
237            (fourth.isqrt() / 2).max(2)
238        };
239        let end = if small_problem {
240            primes.len()
241        } else {
242            primes.partition_point(|x| x.0 <= fourth as u64)
243        };
244        let (small, large) = primes.split_at(end);
245        if !REMOVE && !large.is_empty() {
246            self.large_primes::<G, REMOVE>(large, &mul, &mut next);
247        }
248        let max_k = self.cofactor_limit(dense as u64 + 1);
249        let mut least = Vec::new();
250        let large_len = self.data.len() - self.root;
251        let quotients: Vec<_> = (1..=large_len).map(|d| self.n / d as u64).collect();
252        let mut powers = Vec::new();
253        let bands = small.last().map_or(0, |x| x.0.ilog2());
254        for i in 1..=bands {
255            let shift = if REMOVE { i } else { bands + 1 - i };
256            let first = 1u64 << shift;
257            let start = small.partition_point(|x| x.0 < first || x.0 <= cutoff as u64);
258            let end = small.partition_point(|x| x.0 < first * 2);
259            if start >= end {
260                continue;
261            }
262            let band = &small[start..end];
263            if first <= dense as u64 {
264                for i in 0..band.len() {
265                    let (p, weight) = &band[if REMOVE { i } else { band.len() - 1 - i }];
266                    powers.clear();
267                    powers.push((*p, weight.clone()));
268                    if !REMOVE {
269                        while let Some((power, old)) = powers.last() {
270                            if *power > self.n / p {
271                                break;
272                            }
273                            powers
274                                .push((power * p, next(*p, powers.len() as u32 + 1, weight, old)));
275                        }
276                    }
277                    self.dense_prime::<G, REMOVE>(&powers, &quotients, &mul);
278                }
279            } else {
280                if least.is_empty() {
281                    // A composite cofactor up to sqrt(n) has a least prime factor smaller than 2^16.
282                    least.resize(max_k + 1, u16::MAX);
283                    for &(p, _) in small.iter().rev() {
284                        if p * p <= max_k as u64 {
285                            for k in (p as usize * p as usize..=max_k).step_by(p as usize) {
286                                least[k] = p as u16;
287                            }
288                        }
289                    }
290                }
291                let mut groups = band.chunks(32);
292                while let Some(batch) = if REMOVE {
293                    groups.next()
294                } else {
295                    groups.next_back()
296                } {
297                    self.prime_band::<G, REMOVE>(batch, &least, &mul, &mut next);
298                }
299            }
300        }
301        if REMOVE && !large.is_empty() {
302            self.large_primes::<G, REMOVE>(large, &mul, &mut next);
303        }
304    }
