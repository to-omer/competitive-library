45pub fn euler_phi_sum(n: u64) -> u128 {
46    if n == 0 {
47        return 0;
48    }
49    let root = n.isqrt() as usize;
50    let limit = ((n as f64).cbrt().powi(2) * 0.5) as usize;
51    let limit = limit.max(root);
52    let mobius = sieved_mobius(limit);
53    let mut small = Vec::with_capacity(root + 1);
54    let mut sum = 0i32;
55    for i in 0..=root {
56        if i % 2 == 1 {
57            sum += i32::from(mobius[i / 2]);
58        }
59        small.push((WHEEL.count(i as u64) as i32, sum));
60    }
61    let mut large = vec![(0i64, 0i64); root + 1];
62    let mut pos = root.div_ceil(2);
63    let mut sum = i64::from(sum);
64    for (i, entry) in large.iter_mut().enumerate().skip(1).rev() {
65        let x = n / i as u64;
66        let end = (x.min(limit as u64) as usize).div_ceil(2);
67        sum += mobius[pos..end].iter().map(|&v| i64::from(v)).sum::<i64>();
68        pos = end;
69        *entry = (WHEEL.count(x) as i64, sum);
70    }
71
72    let rough: Vec<_> = (2..=root)
73        .filter(|&d| small[d].0 != small[d - 1].0)
74        .map(|d| (d as u32, mobius[d / 2], u64::MAX / d as u64 + 1))
75        .collect();
76    // On integers coprime to the wheel, mu * 1 is the unit for Dirichlet convolution.
77    for i in (1..=(n / (limit as u64 + 1)) as usize).rev() {
78        let x = n / i as u64;
79        let b = x.isqrt() as usize;
80        let (count, mertens) = small[b];
81        let mut sum = 1 - large[i].0 + i64::from(count) * i64::from(mertens);
82        let split = small[root / i].0 as usize - 1;
83        let end = count as usize - 1;
84        for &(d, mu, _) in &rough[..split] {
85            let (count, mertens) = large[i * d as usize];
86            sum = sum.wrapping_sub(mertens + i64::from(mu) * count);
87        }
88        // ceil(2^64/d) gives the exact quotient when x*d < 2^64.
89        let use_reciprocal = x <= u64::MAX / b as u64;
90        for &(d, mu, reciprocal) in &rough[split..end] {
91            let q = if use_reciprocal {
92                ((u128::from(x) * u128::from(reciprocal)) >> 64) as usize
93            } else {
94                (x / u64::from(d)) as usize
95            };
96            let (count, mertens) = small[q];
97            sum = sum.wrapping_sub(i64::from(mertens + i32::from(mu) * count));
98        }
99        large[i].1 = sum;
100    }

102    // phi = restricted mu * h, where h(e) = e * product_{p | gcd(e, PERIOD)} (1 - 1/p).
103    let mut total = WHEEL.weight_sum(n) as i128;
104    for (e, &(_, mertens)) in large.iter().enumerate().skip(1) {
105        total = total.wrapping_add(i128::from(WHEEL.weight(e)) * i128::from(mertens));
106    }
107    for &(d, mu, _) in &rough {
108        total = total.wrapping_add(i128::from(mu) * WHEEL.weight_sum(n / u64::from(d)) as i128);
109    }
110    total.wrapping_sub(WHEEL.weight_sum(root as u64) as i128 * i128::from(small[root].1)) as u128
