761fn reconstruct_mint_crt<M, N1, N2, N3>(f: (MVec<N1>, MVec<N2>, MVec<N3>)) -> MVec<M>
762where
763    M: MIntConvert + MIntConvert<u32>,
764    N1: Montgomery32NttModulus,
765    N2: Montgomery32NttModulus,
766    N3: Montgomery32NttModulus,
767{
768    let t1 = MInt::<N2>::new(N1::get_mod()).inv();
769    let m1_3 = MInt::<N3>::new(N1::get_mod());
770    let t2 = (m1_3 * MInt::<N3>::new(N2::get_mod())).inv();
771    let modulus = <M as MIntConvert<u32>>::mod_into() as u64;
772    let m1 = N1::get_mod() as u64;
773    let m2 = m1 * N2::get_mod() as u64 % modulus;
774    let fits_u64 = (N1::get_mod() - 1) as u128
775        + (N2::get_mod() - 1) as u128 * m1 as u128
776        + (N3::get_mod() - 1) as u128 * m2 as u128
777        <= u64::MAX as u128;
778    f.0.into_iter()
779        .zip(f.1)
780        .zip(f.2)
781        .map(|((c1, c2), c3)| {
782            let d1 = c1.inner();
783            let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
784            let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
785            let d3 = ((c3 - x) * t2).inner();
786            let value = if fits_u64 {
787                (d1 as u64 + d2 as u64 * m1 + d3 as u64 * m2) % modulus
788            } else {
789                ((d1 as u128 + d2 as u128 * m1 as u128 + d3 as u128 * m2 as u128) % modulus as u128)
790                    as u64
791            };
792            MInt::<M>::from(value as u32)
793        })
794        .collect()
795}

868        }

881        let a_len = a.len();
882        let b_len = b.len();

491            C::multiply(&mut values, &kernel);
492            let mut values = C::inverse_transform(values, block * 2);
493            values.truncate(len);
494            result.extend(values);
495        }
496        Self::from_vec(result)
497    }

499    fn exp_newton(&self, deg: usize, indices: &[T], inv: &[T]) -> (Self, Self, C::F)
500    where
501        C: NttReuse<T = Vec<T>>,
502        C::F: Clone,
503    {
504        if deg == 1 {
505            let one = Self::one();
506            return (one.clone(), one.clone(), C::transform_ntt(one.data, 1));
507        }
508        let mut f = Self::from_vec(vec![T::one(), self.coeff(1)]);
509        let mut inverse = Self::one();
510        let mut inverse_fft = C::transform_ntt(inverse.data.clone(), 2);
511        let mut m = 2;
512        while m < deg {
513            let f_fft = C::transform_ntt(f.data.clone(), 2 * m);
514
515            let previous_inverse_fft = inverse_fft;
516            let mut error_fft = previous_inverse_fft.clone();
517            C::multiply_prefix(&mut error_fft, &f_fft);
518            let mut error = C::inverse_transform_ntt(error_fft, m);
519            error[..m / 2].fill(T::zero());
520            let mut error_fft = C::transform_ntt(error, m);
521            C::multiply_prefix(&mut error_fft, &previous_inverse_fft);
522            let error = C::inverse_transform_ntt(error_fft, m);
523            inverse
524                .data
525                .extend(error.into_iter().skip(m / 2).map(Neg::neg));
526            inverse_fft = C::transform_ntt(inverse.data.clone(), 2 * m);
527
528            let mut delta = Self::from_vec(
529                self.data
530                    .iter()
531                    .take(m)
532                    .enumerate()
533                    .skip(1)
534                    .map(|(i, value)| value.clone() * &indices[i])
535                    .collect(),
536            );
537            delta.resize(m);
538            let mut delta_fft = C::transform_ntt(delta.data, m);
539            C::multiply_prefix(&mut delta_fft, &f_fft);
540            let mut delta = Self::from_vec(C::inverse_transform_ntt(delta_fft, m));
541            for i in 1..f.length() {
542                delta[i - 1] -= f[i].clone() * &indices[i];
543            }
544            delta.resize(2 * m);
545            for i in (0..m - 1).rev() {
546                delta.data[m + i] = delta.data[i].clone();
547            }
548            delta.data[..m - 1].fill(T::zero());
549            let mut delta_fft = C::transform_ntt(delta.data, 2 * m);
550            C::multiply_prefix(&mut delta_fft, &inverse_fft);
551            let mut delta = C::inverse_transform_ntt(delta_fft, 2 * m);
552            delta.pop();
553            delta.push(T::zero());
554            let target = (2 * m).min(deg);
555            for i in (1..target).rev() {
556                delta[i] = delta[i - 1].clone() * &inv[i];
557            }
558            delta[0] = T::zero();
559            delta[target..].fill(T::zero());
560            for i in m..(2 * m).min(self.length()) {
561                delta[i] += self[i].clone();
562            }
563            delta[..m].fill(T::zero());
564            let mut delta_fft = C::transform_ntt(delta, 2 * m);
565            C::multiply_prefix(&mut delta_fft, &f_fft);
566            let delta = C::inverse_transform_ntt(delta_fft, 2 * m);
567            f.data
568                .extend(delta.into_iter().skip(m).take((deg - m).min(m)));
569            m *= 2;
570        }
571        (f, inverse, inverse_fft)
572    }
573    pub fn log(&self, deg: usize) -> Self {
574        if deg == 0 {
575            return Self::zero();
576        }
577        debug_assert!(!self[0].is_zero());
578        if deg == 1 {
579            return Self::zeros(1);
580        }
581        if let Some(step) = self.sparse_stride(deg, 2) {
582            let pos: Vec<_> = self
583                .iter()
584                .take(deg)
585                .enumerate()
586                .skip(1)
587                .filter_map(|(i, x)| (!x.is_zero()).then_some(i))
588                .collect();
589            let mut derivative = Self::zeros(deg);
590            let inverse = T::one() / self[0].clone();
591            for i in (pos.first().copied().unwrap_or(deg)..deg).step_by(step) {
592                let mut value = self.coeff(i) * T::from(i);
593                for &j in &pos {
594                    if j >= i {
595                        break;
596                    }
597                    value -= self[j].clone() * &derivative[i - j];
598                }
599                derivative[i] = value * &inverse;
600            }
601            if pos.is_empty() {
602                return derivative;
603            }
604            derivative.data.remove(0);
605            return derivative.integral();
606        }
607        let n = deg - 1;
608        if n <= 64 {
609            return (self.inv(deg) * self.prefix_ref(deg).diff())
610                .prefix(n)
611                .integral();
612        }
613        let half = n.next_power_of_two() / 2;
614        let derivative = self.prefix_ref(deg).diff();
615        let inverse = C::transform(self.inv(half).data, half * 2);
616        let mut quotient = C::transform(derivative.prefix_ref(half).data, half * 2);
617        C::multiply(&mut quotient, &inverse);
618        let mut result = C::inverse_transform(quotient, half * 2);
619        result.truncate(half);
620        if n - half <= 4 {
621            let inverse = T::one() / self[0].clone();
622            for i in half..n {
623                let mut value = derivative.coeff(i);
624                for j in 1..=i.min(self.length() - 1) {
625                    value -= self[j].clone() * &result[i - j];
626                }
627                result.push(value * &inverse);
628            }
629            return Self::from_vec(result).integral();
630        }
631        let quotient = C::transform(result.clone(), half * 2);
632        let mut error = C::transform(self.prefix_ref(n).data, half * 2);
633        C::multiply(&mut error, &quotient);
634        let mut error = C::inverse_transform(error, half * 2);
635        for i in 0..n - half {
636            error[i] = derivative.coeff(half + i) - &error[half + i];
637        }
638        error.truncate(n - half);
639        let mut error = C::transform(error, half * 2);
640        C::multiply(&mut error, &inverse);
641        let error = C::inverse_transform(error, half * 2);
642        result.extend(error.into_iter().take(n - half));
643        Self::from_vec(result).integral()
644    }
645    pub fn pow(&self, rhs: usize, deg: usize) -> Self
646    where
647        C: NttReuse<T = Vec<T>>,
648        C::F: Clone,
649    {
650        if rhs == 0 {
651            return Self::from_vec(
652                once(T::one())
653                    .chain(repeat_with(T::zero))
654                    .take(deg)
655                    .collect(),
656            );
657        }
658        if rhs == 1 {
659            return self.prefix_ref(deg).resized(deg);
660        }
661        if let Some(k) = self
662            .iter()
663            .take(deg.div_ceil(rhs))
664            .position(|x| !x.is_zero())
665        {
666            let deg = deg - k * rhs;
667            let x0 = self[k].clone();
668            let mut f = (self.prefix_ref(k + deg) >> k) / &x0;
669            if let Some(step) = f.sparse_stride(deg, 12) {
670                f = f.pow_sparse1(T::from(rhs), deg, step);
671            } else if rhs <= 4 {
672                let squared = (&f * &f).prefix(deg);
673                f = match rhs {
674                    2 => squared,
675                    3 => (squared * f).prefix(deg),
676                    _ => (&squared * &squared).prefix(deg),
677                }
678                .resized(deg);
679            } else {
680                f = f.exp_or_pow(Some(T::from(rhs)), deg);
681            }
682            f *= x0.pow(rhs);
683            f <<= k * rhs;
684            f
685        } else {
686            Self::zeros(deg)
687        }
688    }
689    fn pow_sparse1(&self, rhs: T, deg: usize, step: usize) -> Self {
690        debug_assert!(!self[0].is_zero());
691        let mut pos: Vec<_> = self
692            .data
693            .iter()
694            .take(deg)
695            .enumerate()
696            .skip(1)
697            .filter(|(_, x)| !x.is_zero())
698            .map(|(i, x)| (i, x.clone(), T::from(i) * &rhs * x))
699            .collect();
700        let mut f = Self::zeros(deg);
701        f[0] = T::one();
702        if pos.is_empty() {
703            return f;
704        }
705        let mf = T::memorized_factorial(deg);
706        for (_, coefficient, _) in &mut pos {
707            *coefficient *= T::from(step);
708        }
709        for i in (pos.first().map_or(deg, |x| x.0)..deg).step_by(step) {
710            let mut tot = T::zero();
711            for (j, coefficient, weight) in &mut pos {
712                if *j > i {
713                    break;
714                }
715                tot += weight.clone() * &f[i - *j];
716                *weight -= &*coefficient;
717            }
718            f[i] = tot * T::memorized_inv(&mf, i);
719        }
720        f
721    }

723    fn sparse_fold(&self, sparse: impl IntoIterator<Item = (usize, T)>, deg: usize) -> T {
724        sparse
725            .into_iter()
726            .take_while(|&(i, _)| i <= deg)
727            .fold(T::zero(), |sum, (i, x)| sum + x * self.coeff(deg - i))
728    }

730    /// solve: $X(QF)'=\alpha P'(QF)+\beta P(Q'F)$ in $O(deg * max(nz(P), nz(Q), nz(X)))$
731    pub fn solve_sparse_differential2(
732        p: &Self,
733        q: &Self,
734        x: &Self,
735        alpha: T,
736        beta: T,
737        deg: usize,
738    ) -> Self {
739        if deg == 0 {
740            return Self::zero();
741        }
742        let collect_sparse = |p: &Self| -> Vec<(usize, T)> {
743            p.iter()
744                .enumerate()
745                .filter(|&(_, x)| !x.is_zero())
746                .map(|(i, x)| (i, x.clone()))
747                .collect()
748        };
749        assert!(q.coeff(0).is_one());
750        assert!(x.coeff(0).is_one());
751        let p = collect_sparse(p);
752        let q = collect_sparse(q);
753        let x = collect_sparse(x);
754        let diff = |p: &[(usize, T)]| -> Vec<(usize, T)> {
755            p.iter()
756                .filter(|&&(i, _)| i > 0)
757                .map(|&(i, ref x)| (i - 1, x.clone() * T::from(i)))
758                .collect()
759        };
760        let dp = diff(&p);
761        let dq = diff(&q);
762
763        let mf = T::memorized_factorial(deg);
764        let mut f = Self::zeros(deg);
765        let mut qf = Self::zeros(deg);
766        let mut dq_f = Self::zeros(deg);
767        let mut d_qf = Self::zeros(deg);
768        f[0] = T::one();
769        for i in 0..deg - 1 {
770            qf[i] = f.sparse_fold(q.iter().cloned(), i);
771            dq_f[i] = f.sparse_fold(dq.iter().cloned(), i);
772            let dp_qf_i = qf.sparse_fold(dp.iter().cloned(), i);
773            let p_dq_f_i = dq_f.sparse_fold(p.iter().cloned(), i);
774            let x_d_qf_i = d_qf.sparse_fold(
775                x.iter()
776                    .map(|&(i, ref x)| (i, x.clone() - T::from((i == 0) as usize))),
777                i,
778            );
779            d_qf[i] = alpha.clone() * dp_qf_i + beta.clone() * p_dq_f_i - x_d_qf_i;
780
781            let mut f_ip1 = d_qf[i].clone();
782            for &(j, ref q) in q.iter().take_while(|&&(j, _)| j <= i) {
783                if j > 0 {
784                    f_ip1 -= q.clone() * &f[i - (j - 1)] * T::from(i - (j - 1));
785                }
786            }
787            f[i + 1] = f_ip1 * T::memorized_inv(&mf, i + 1);
788        }
789        f
790    }

792    /// P^exp_p * Q^exp_q
793    pub fn mul_of_pow_sparse(&self, q: &Self, exp_p: isize, exp_q: isize, deg: usize) -> Self {
794        if deg == 0 {
795            return Self::zero();
796        }
797        if exp_p == 0 && exp_q == 0 {
798            return Self::from_vec(
799                once(T::one())
800                    .chain(repeat_with(T::zero))
801                    .take(deg)
802                    .collect(),
803            );
804        }
805        if exp_p != 0 && self.iter().all(|x| x.is_zero()) {
806            assert!(exp_p > 0);
807            return Self::zeros(deg);
808        }
809        if exp_q != 0 && q.iter().all(|x| x.is_zero()) {
810            assert!(exp_q > 0);
811            return Self::zeros(deg);
812        }
813
814        let normalize = |f: &Self, exp: isize| {
815            if exp == 0 {
816                return (0usize, T::one(), Self::from_vec(vec![T::one()]));
817            }
818            let k = f.iter().position(|value| !value.is_zero()).unwrap();
819            assert!(
820                exp >= 0 || k == 0,
821                "Negative exponent with zero constant term"
822            );
823            let c = f[k].clone();
824            let f = (f.clone() >> k) / &c;
825            (k, c, f)
826        };
827        let (sp, cp, mut p) = normalize(self, exp_p);
828        let (sq, cq, mut q) = normalize(q, exp_q);
829
830        let shift = exp_p
831            .saturating_mul(sp as _)
832            .saturating_add(exp_q.saturating_mul(sq as _)) as usize;
833        if shift >= deg {
834            return Self::zeros(deg);
835        }
836        p.truncate(deg - shift);
837        q.truncate(deg - shift);
838
839        let mut f = Self::solve_sparse_differential2(
840            &p,
841            &q,
842            &p,
843            T::from(exp_p),
844            T::from(exp_q),
845            deg - shift,
846        );
847        f *= cp.signed_pow(exp_p) * cq.signed_pow(exp_q);
848        if shift > 0 {
849            f <<= shift;
850        }
851        f.prefix(deg)
852    }

854    /// exp(P/Q)
855    pub fn exp_of_div_sparse(&self, q: &Self, deg: usize) -> Self {
856        if deg == 0 {
857            return Self::zero();
858        }
859        let shift_q = q
860            .iter()
861            .position(|value| !value.is_zero())
862            .expect("Zero denominator");
863        let shift_p = self.iter().position(|value| !value.is_zero()).unwrap_or(!0);
864        assert!(shift_p > shift_q);
865
866        let mut p = self >> shift_q;
867        let mut q = q >> shift_q;
868        assert!(!q.coeff(0).is_zero());
869
870        let c = q[0].clone();
871        p /= c.clone();
872        q /= c;
873
874        Self::solve_sparse_differential2(&p, &q, &q, T::one(), -T::one(), deg)
875    }

878impl<T, C> FormalPowerSeries<T, C>
879where
880    T: FormalPowerSeriesCoefficientSqrt,
881    C: ConvolveSteps<T = Vec<T>>,
882{
883    pub fn sqrt(&self, deg: usize) -> Option<Self> {
884        if self[0].is_zero() {
885            if let Some(k) = self.iter().position(|x| !x.is_zero()) {
886                if k % 2 != 0 {
887                    return None;
888                } else if deg > k / 2 {
889                    return Some((self >> k).sqrt(deg - k / 2)? << (k / 2));
890                }
891            }
892        } else {
893            let s = self[0].sqrt_coefficient()?;
894            if deg <= 1 {
895                return Some(Self::from(s).prefix(deg));
896            }
897            if let Some(step) = self.sparse_stride(deg, 4) {
898                let t = self[0].clone();
899                let mut f = self.prefix_ref(deg) / t;
900                f = f.pow_sparse1(T::one() / T::from(2usize), deg, step);
901                f *= s;
902                return Some(f);
903            }
904
905            let mut f = Self::from(s);
906            let inv2 = T::one() / (T::one() + T::one());
907            let inv2s = inv2.clone() / &f[0];
908            let extend = |f: &mut Self, end| {
909                for i in f.length()..end {
910                    let mut value = self.coeff(i);
911                    for j in 1..i {
912                        value -= f[j].clone() * &f[i - j];
913                    }
914                    f.data.push(value * &inv2s);
915                }
916            };
917            extend(&mut f, deg.min(32));
918            f.truncate(deg);
919            if f.length() == deg {
920                return Some(f);
921            }
922            let mut inverse = f.inv(f.length());
923            let mut i = f.length();
924            while i < deg {
925                if deg - i <= 4 {
926                    extend(&mut f, deg);
927                    break;
928                }
929                let len = (i * 2).min(deg);
930                let factor = C::transform(inverse.data.clone(), i * 2);
931                let error = if !C::CYCLIC || i < 128 {
932                    (self.prefix_ref(len) - &f * &f) >> i
933                } else {
934                    let square = C::square(f.data.clone(), i);
935                    // The cyclic square folds its high half into the already known low half.
936                    Self::from_vec(
937                        square
938                            .into_iter()
939                            .take(len - i)
940                            .enumerate()
941                            .map(|(j, value)| self.coeff(i + j) + self.coeff(j) - value)
942                            .collect(),
943                    )
944                };
945                let mut error_fft = C::transform(error.data, i * 2);
946                C::multiply(&mut error_fft, &factor);
947                let delta = C::inverse_transform(error_fft, i * 2);
948                f.data
949                    .extend(delta.into_iter().take(len - i).map(|x| x * &inv2));
950                if i * 2 + 4 < deg {
951                    let mut error_fft = C::transform(f.data.clone(), i * 2);
952                    C::multiply(&mut error_fft, &factor);
953                    let error = C::inverse_transform(error_fft, i * 2);
954                    let mut error_fft = C::transform(error.into_iter().skip(i).collect(), i * 2);
955                    C::multiply(&mut error_fft, &factor);
956                    let error = C::inverse_transform(error_fft, i * 2);
957                    inverse.data.extend(error.into_iter().take(i).map(Neg::neg));
958                }
959                i *= 2;
960            }
961            f.truncate(deg);
962            return Some(f);
963        }
964        Some(Self::zeros(deg))
965    }
966}

1011    /// [x^n] P(x) / Q(x)
1012    pub fn bostan_mori(mut self, mut rhs: Self, mut n: usize) -> T
1013    where
1014        C: NttReuse<T = Vec<T>>,
1015    {
1016        let mut res = T::zero();
1017        rhs.trim_tail_zeros();
1018        if self.length() >= rhs.length() {
1019            let r = &self / &rhs;
1020            if n < r.length() {
1021                res = r[n].clone();
1022            }
1023            self -= r * &rhs;
1024            self.trim_tail_zeros();
1025        }
1026        let mut k = rhs.length().next_power_of_two();
1027        let mut p = C::transform_ntt(self.data, k * 2);
1028        let mut q = C::transform_ntt(rhs.data, k * 2);
1029        while n > 0 {
1030            let t = C::even_mul_normal_neg(&q, &q);
1031            p = if n.is_multiple_of(2) {
1032                C::even_mul_normal_neg(&p, &q)
1033            } else {
1034                C::odd_mul_normal_neg(&p, &q)
1035            };
1036            q = t;
1037            n /= 2;
1038            if n != 0 {
1039                if n < k / 2 {
1040                    p = C::transform_ntt(C::inverse_transform_ntt(p, k / 2), k);
1041                    q = C::transform_ntt(C::inverse_transform_ntt(q, k / 2), k);
1042                    k /= 2;
1043                } else if C::MULTIPLE {
1044                    p = C::transform_ntt(C::inverse_transform_ntt(p, k), k * 2);
1045                    q = C::transform_ntt(C::inverse_transform_ntt(q, k), k * 2);
1046                } else {
1047                    p = C::ntt_doubling(p, false);
1048                    q = C::ntt_doubling(q, false);
1049                }
1050            }
1051        }
1052        let p = C::inverse_transform_ntt(p, k);
1053        let q = C::inverse_transform_ntt(q, k);
1054        res + p[0].clone() / q[0].clone()
1055    }
1056    /// return F(x) where [x^n] P(x) / Q(x) = [x^d-1] P(x) F(x)
1057    pub fn bostan_mori_msb(self, n: usize) -> Self {
1058        let d = self.length() - 1;
1059        if n == 0 {
1060            return (Self::one() << (d - 1)) / self[0].clone();
1061        }
1062        let q = self;
1063        let mq = q.clone().parity_inversion();
1064        let w = (q * &mq).even().bostan_mori_msb(n / 2);
1065        let mut s = Self::zeros(w.length() * 2 - (n % 2));
1066        for (i, x) in w.iter().enumerate() {
1067            s[i * 2 + (1 - n % 2)] = x.clone();
1068        }
1069        let len = 2 * d + 1;
1070        let ts = C::transform(s.prefix(len).data, len);
1071        mq.reversed().middle_product(&ts, len).prefix(d + 1)
1072    }
1073    /// x^n mod self

773    let m2 = m1 * N2::get_mod() as u64 % modulus;
774    let fits_u64 = (N1::get_mod() - 1) as u128
775        + (N2::get_mod() - 1) as u128 * m1 as u128
776        + (N3::get_mod() - 1) as u128 * m2 as u128
777        <= u64::MAX as u128;
778    f.0.into_iter()
779        .zip(f.1)
780        .zip(f.2)
781        .map(|((c1, c2), c3)| {
782            let d1 = c1.inner();
783            let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
784            let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
785            let d3 = ((c3 - x) * t2).inner();
786            let value = if fits_u64 {
787                (d1 as u64 + d2 as u64 * m1 + d3 as u64 * m2) % modulus
788            } else {
789                ((d1 as u128 + d2 as u128 * m1 as u128 + d3 as u128 * m2 as u128) % modulus as u128)
790                    as u64
791            };
792            MInt::<M>::from(value as u32)
793        })
794        .collect()
795}

797impl<M, N1, N2, N3> ConvolveSteps for Convolve<(M, (N1, N2, N3))>
798where
799    M: MIntConvert + MIntConvert<u32>,
800    N1: Montgomery32NttModulus,
801    N2: Montgomery32NttModulus,
802    N3: Montgomery32NttModulus,
803{
804    type T = MVec<M>;
805    type F = (MVec<N1>, MVec<N2>, MVec<N3>);
806    fn length(t: &Self::T) -> usize {
807        t.len()
808    }
809    fn transform(t: Self::T, len: usize) -> Self::F {
810        let npot = len.max(1).next_power_of_two();
811        let f = convert_crt_input(t, npot);
812        (
813            Convolve::<N1>::transform(f.0, npot),
814            Convolve::<N2>::transform(f.1, npot),
815            Convolve::<N3>::transform(f.2, npot),
816        )
817    }
818    fn inverse_transform(f: Self::F, len: usize) -> Self::T {
819        reconstruct_mint_crt((
820            Convolve::<N1>::inverse_transform(f.0, len),
821            Convolve::<N2>::inverse_transform(f.1, len),
822            Convolve::<N3>::inverse_transform(f.2, len),
823        ))
824    }
825    fn multiply(f: &mut Self::F, g: &Self::F) {
826        Convolve::<N1>::multiply(&mut f.0, &g.0);
827        Convolve::<N2>::multiply(&mut f.1, &g.1);
828        Convolve::<N3>::multiply(&mut f.2, &g.2);
829    }
830    fn convolve(a: Self::T, b: Self::T) -> Self::T {
831        let max_len = Self::length(&a).max(Self::length(&b));
832        let min_len = Self::length(&a).min(Self::length(&b));
833        let (balanced, short) = crate::avx_helper!(@dispatch_avx2_fma (30, 10), (384, 128));
834        if max_len <= balanced || min_len <= short {
835            return convolve_karatsuba(&a, &b);
836        }
837        // Limit coefficient growth to leave headroom for FFT roundoff.
838        let fft_limit = crate::avx_helper!(@dispatch_avx2_fma
839            1usize << ((1u64 << 50) / <M as MIntConvert<u32>>::mod_into() as u64).ilog2().min(20), 0);
840        let convolve = |a: Self::T, b: Self::T| {
841            let fft_len = (a.len() + b.len() - 1).next_power_of_two();
842            if fft_len <= 256 && a.len() * b.len() <= fft_len * 8 {
843                return convolve_karatsuba(&a, &b);
844            }
845            if fft_len <= fft_limit {
846                crate::avx_helper!(@dispatch_avx2_fma return unsafe {
847                    super::mint_fft_convolve::convolve_mint_avx2(a, b)
848                }, ());
849            }
850            convolve_mint_crt::<M, N1, N2, N3>(a, b)
851        };
852        let block_len = min_len.next_power_of_two() * 8 - min_len + 1;
853        let block_len = if min_len <= fft_limit / 2 {
854            block_len.min(fft_limit - min_len + 1)
855        } else {
856            block_len
857        };
858        if max_len <= block_len {
859            return convolve(a, b);
860        }
861        let (a, b) = if a.len() >= b.len() { (a, b) } else { (b, a) };
862        let mut result = vec![MInt::<M>::zero(); a.len() + b.len() - 1];
863        for (i, a) in a.chunks(block_len).enumerate() {
864            let product = convolve(a.to_vec(), b.clone());
865            for (value, product) in result[i * block_len..].iter_mut().zip(product) {
866                *value += product;
867            }
868        }
869        result
870    }
871}

873fn convolve_mint_crt<M, N1, N2, N3>(a: MVec<M>, b: MVec<M>) -> MVec<M>
874where
875    M: MIntConvert + MIntConvert<u32>,
876    N1: Montgomery32NttModulus,
877    N2: Montgomery32NttModulus,
878    N3: Montgomery32NttModulus,
879{
880    let convolve = |a: MVec<M>, b: MVec<M>| {
881        let a_len = a.len();
882        let b_len = b.len();
883        let a = convert_crt_input(a, a_len);
884        let b = convert_crt_input(b, b_len);
885        reconstruct_mint_crt((
886            Convolve::<N1>::convolve(a.0, b.0),
887            Convolve::<N2>::convolve(a.1, b.1),
888            Convolve::<N3>::convolve(a.2, b.2),
889        ))
890    };
891    let modulus = <M as MIntConvert<u32>>::mod_into() as u128;
892    let capacity = N1::MOD as u128 * N2::MOD as u128 * N3::MOD as u128;
893    if a.len().min(b.len()) as u128 * (modulus - 1).pow(2) < capacity {
894        return convolve(a, b);
895    }
896    let block_len = ((capacity - 1) / (modulus - 1).pow(2)) as usize;
897    if block_len == 0 {
898        return convolve_naive(&a, &b);
899    }
900    let mut result = vec![MInt::<M>::zero(); a.len() + b.len() - 1];
901    for (i, a) in a.chunks(block_len).enumerate() {
902        for (j, b) in b.chunks(block_len).enumerate() {
903            let product = convolve(a.to_vec(), b.to_vec());
904            for (value, product) in result[(i + j) * block_len..].iter_mut().zip(product) {
905                *value += product;
906            }
907        }
908    }
909    result
910}

912impl<N1, N2, N3> ConvolveSteps for Convolve<(u64, (N1, N2, N3))>
913where
914    N1: Montgomery32NttModulus,
915    N2: Montgomery32NttModulus,
916    N3: Montgomery32NttModulus,
917{
918    type T = Vec<u64>;
919    type F = ([MVec<N1>; 3], [MVec<N2>; 3], [MVec<N3>; 3]);
920
921    fn length(t: &Self::T) -> usize {
922        t.len()
923    }
924
925    fn transform(t: Self::T, len: usize) -> Self::F {
926        let npot = len.max(1).next_power_of_two();
927        assert!(npot <= 1usize << N1::RANK.min(N2::RANK).min(N3::RANK));
928        // The 22-bit fallback needs room for three limb products per coefficient.
929        assert!(
930            3 * npot as u128 * ((1u128 << 22) - 1).pow(2)
931                < N1::MOD as u128 * N2::MOD as u128 * N3::MOD as u128
932        );
933        let bits = if 2 * npot as u128 * (u32::MAX as u128).pow(2)
934            < N1::MOD as u128 * N2::MOD as u128 * N3::MOD as u128
935        {
936            32
937        } else {
938            22
939        };
940        let parts = if bits == 32 && t.iter().all(|&value| value <= u32::MAX as u64) {
941            1
942        } else {
943            64usize.div_ceil(bits)
944        };
945        fn split<M: Montgomery32NttModulus>(
946            t: &[u64],
947            len: usize,
948            bits: usize,
949            parts: usize,
950        ) -> [MVec<M>; 3] {
951            std::array::from_fn(|part| {
952                if part >= parts {
953                    return Vec::new();
954                }
955                Convolve::<M>::transform(
956                    t.iter()
957                        .map(|&t| MInt::from((t >> (part * bits)) & ((1u64 << bits) - 1)))
958                        .collect(),
959                    len,
960                )
961            })
962        }
963        (
964            split(&t, npot, bits, parts),
965            split(&t, npot, bits, parts),
966            split(&t, npot, bits, parts),
967        )
968    }
969
970    fn inverse_transform(f: Self::F, len: usize) -> Self::T {
971        let bits = if f.0[2].is_empty() { 32 } else { 22 };
972        let t1 = MInt::<N2>::new(N1::get_mod()).inv();
973        let m1 = N1::get_mod() as u64;
974        let m1_3 = MInt::<N3>::new(N1::get_mod());
975        let t2 = (m1_3 * MInt::<N3>::new(N2::get_mod())).inv();
976        let m2 = m1 * N2::get_mod() as u64;
977        let mut result = vec![0u64; len.min(f.0[0].len())];
978        for (part, ((f1, f2), f3)) in f.0.into_iter().zip(f.1).zip(f.2).enumerate() {
979            if f1.is_empty() {
980                continue;
981            }
982            for (value, ((c1, c2), c3)) in result.iter_mut().zip(
983                Convolve::<N1>::inverse_transform(f1, len)
984                    .into_iter()
985                    .zip(Convolve::<N2>::inverse_transform(f2, len))
986                    .zip(Convolve::<N3>::inverse_transform(f3, len)),
987            ) {
988                let d1 = c1.inner();
989                let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
990                let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
991                let d3 = ((c3 - x) * t2).inner();
992                let limb = (d1 as u64)
993                    .wrapping_add((d2 as u64).wrapping_mul(m1))
994                    .wrapping_add((d3 as u64).wrapping_mul(m2));
995                *value = value.wrapping_add(limb << (part * bits));
996            }
997        }
998        result
999    }
1000
1001    fn multiply(f: &mut Self::F, g: &Self::F) {
1002        fn multiply<M: Montgomery32NttModulus>(f: &mut [MVec<M>; 3], g: &[MVec<M>; 3]) {
1003            assert_eq!(f[0].len(), g[0].len());
1004            if f[1].is_empty() || g[1].is_empty() {
1005                if f[1].is_empty() && !g[1].is_empty() {
1006                    f[1] = f[0].clone();
1007                    Convolve::<M>::multiply(&mut f[1], &g[1]);
1008                } else if !f[1].is_empty() {
1009                    Convolve::<M>::multiply(&mut f[1], &g[0]);
1010                }
1011                Convolve::<M>::multiply(&mut f[0], &g[0]);
1012                return;
1013            }
1014            #[cfg(target_arch = "x86_64")]
1015            if use_block_ntt::<M>(f[0].len()) {
1016                for part in (1..if f[2].is_empty() { 2 } else { 3 }).rev() {
1017                    let mut sum = f[0].clone();
1018                    Convolve::<M>::multiply(&mut sum, &g[part]);
1019                    for left in 1..=part {
1020                        let mut product = f[left].clone();
1021                        Convolve::<M>::multiply(&mut product, &g[part - left]);
1022                        for (value, product) in sum.iter_mut().zip(product) {
1023                            // Block products contain lazy Montgomery residues.
1024                            *value = MInt::new(value.inner() + product.inner());
1025                        }
1026                    }
1027                    f[part] = sum;
1028                }
1029                Convolve::<M>::multiply(&mut f[0], &g[0]);
1030                return;
1031            }
1032            if f[2].is_empty() {
1033                for i in 0..f[0].len() {
1034                    f[1][i] = f[0][i] * g[1][i] + f[1][i] * g[0][i];
1035                    f[0][i] *= g[0][i];
1036                }
1037                return;
1038            }
1039            for i in 0..f[0].len() {
1040                f[2][i] = f[0][i] * g[2][i] + f[1][i] * g[1][i] + f[2][i] * g[0][i];
1041                f[1][i] = f[0][i] * g[1][i] + f[1][i] * g[0][i];
1042                f[0][i] *= g[0][i];
1043            }
1044        }
1045        multiply(&mut f.0, &g.0);
1046        multiply(&mut f.1, &g.1);
1047        multiply(&mut f.2, &g.2);
1048    }
1049
1050    fn square(t: Self::T, len: usize) -> Self::T {
1051        let mut f = Self::transform(t, len);
1052        let g = f.clone();
1053        Self::multiply(&mut f, &g);
1054        Self::inverse_transform(f, len)
1055    }
1056
1057    fn convolve(a: Self::T, b: Self::T) -> Self::T {
1058        let max_len = Self::length(&a).max(Self::length(&b));
1059        let min_len = Self::length(&a).min(Self::length(&b));
1060        let (balanced, short) = crate::avx_helper!(@dispatch_avx2_fma (300, 64), (1536, 512));
1061        if max_len <= balanced || min_len <= short {
1062            let a_wrapping: &[Wrapping<u64>] =
1063                unsafe { std::slice::from_raw_parts(a.as_ptr().cast(), a.len()) };
1064            let b_wrapping: &[Wrapping<u64>] =
1065                unsafe { std::slice::from_raw_parts(b.as_ptr().cast(), b.len()) };
1066            let mut c = std::mem::ManuallyDrop::new(if max_len <= 300 || min_len > 60 {
1067                convolve_karatsuba(a_wrapping, b_wrapping)
1068            } else {
1069                convolve_naive(a_wrapping, b_wrapping)
1070            });
1071            return unsafe { Vec::from_raw_parts(c.as_mut_ptr().cast(), c.len(), c.capacity()) };
1072        }
1073        let len = (Self::length(&a) + Self::length(&b)).saturating_sub(1);
1074        let block_len = if min_len >= 1 << 20 {
1075            1 << 20
1076        } else {
1077            (min_len.next_power_of_two() * 8).min(1 << 21) - min_len + 1
1078        };
1079        if max_len <= block_len {
1080            return convolve_u64_fft(a, b);
1081        }
1082        let mut result = vec![0u64; len];
1083        for (i, a) in a.chunks(block_len).enumerate() {
1084            for (j, b) in b.chunks(block_len).enumerate() {
1085                if a.len().min(b.len()) <= 60 {
1086                    for (x, &a) in a.iter().enumerate() {
1087                        for (y, &b) in b.iter().enumerate() {
1088                            let value = &mut result[(i + j) * block_len + x + y];
1089                            *value = value.wrapping_add(a.wrapping_mul(b));
1090                        }
1091                    }
1092                    continue;
1093                }
1094                let product = convolve_u64_fft(a.to_vec(), b.to_vec());
1095                for (value, product) in result[(i + j) * block_len..].iter_mut().zip(product) {
1096                    *value = value.wrapping_add(product);
1097                }
1098            }
1099        }
1100        result
1101    }
1102}

1105    // Keep limb convolutions below 2^47 at the 2^21 FFT limit.

1107        super::mint_fft_convolve::convolve_u64_avx2(a, b)
1108    }, ());
1109    convolve_u64_fft_scalar(a, b)
1110}

1112fn convolve_u64_fft_scalar(a: Vec<u64>, b: Vec<u64>) -> Vec<u64> {
1113    fn split(values: &[u64]) -> [Vec<i64>; 5] {
1114        let mut result = std::array::from_fn(|_| Vec::with_capacity(values.len()));
1115        for mut value in values.iter().copied() {
1116            for part in &mut result {
1117                let digit = ((value << 51) as i64) >> 51;
1118                part.push(digit);
1119                value = (value >> 13).wrapping_add(u64::from(digit < 0));
1120            }
1121        }
1122        result
1123    }
1124
1125    let len = a.len() + b.len() - 1;
1126    let transform = |values: &[u64]| {
1127        if values.iter().any(|&value| value > u32::MAX as u64) {
1128            return split(values).map(|part| ConvolveRealFft::transform(part, len));
1129        }
1130        let [a, b, c, _, _] = split(values);
1131        let a = ConvolveRealFft::transform(a, len);
1132        let size = a.len();
1133        [
1134            a,
1135            ConvolveRealFft::transform(b, len),
1136            ConvolveRealFft::transform(c, len),
1137            vec![Zero::zero(); size],
1138            vec![Zero::zero(); size],
1139        ]
1140    };
1141    let fa = transform(&a);
1142    drop(a);
1143    let fb = transform(&b);
1144    drop(b);
1145    let values: [Vec<i64>; 5] = std::array::from_fn(|part| {
1146        let mut sum = fa[0].clone();
1147        ConvolveRealFft::multiply(&mut sum, &fb[part]);
1148        for left in 1..=part {
1149            let mut product = fa[left].clone();
1150            ConvolveRealFft::multiply(&mut product, &fb[part - left]);
1151            for (sum, product) in sum.iter_mut().zip(product) {
1152                *sum += product;
1153            }
1154        }
1155        ConvolveRealFft::inverse_transform(sum, len)
1156    });
1157    (0..len)
1158        .map(|i| {
1159            (values[0][i] as u64)
1160                .wrapping_add((values[1][i] as u64) << 13)
1161                .wrapping_add((values[2][i] as u64) << 26)
1162                .wrapping_add((values[3][i] as u64) << 39)
1163                .wrapping_add((values[4][i] as u64) << 52)
1164        })
1165        .collect()
1166}

1168pub trait NttReuse: ConvolveSteps {
1169    const MULTIPLE: bool = true;
1170
1171    /// Transforms coefficients into the usual NTT frequency order.
1172    fn transform_ntt(t: Self::T, len: usize) -> Self::F {
1173        Self::transform(t, len)
1174    }
1175
1176    /// Inverts a value produced by `transform_ntt`.
1177    fn inverse_transform_ntt(f: Self::F, len: usize) -> Self::T {
1178        Self::inverse_transform(f, len)
1179    }
1180
1181    /// Extends a value produced by `transform_ntt` to twice its length.
1182    /// If `monic`, the input represents a monic degree-`n` polynomial modulo
1183    /// `x^n - 1`, where `n` is the transform length.
1184    fn ntt_doubling(f: Self::F, monic: bool) -> Self::F;
1185
1186    /// Extracts the even coefficients of `a(x) * b(-x)` in the usual NTT frequency order.
1187    fn even_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F;
1188
1189    /// Extracts the odd coefficients of `a(x) * b(-x)` in the usual NTT frequency order.
1190    fn odd_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F;
1191
1192    /// Multiplies a usual NTT transform by the corresponding prefix of another one.
1193    fn multiply_prefix(f: &mut Self::F, g: &Self::F);
1194
1195    /// Adds the pointwise product of two usual NTT transforms to `sum`.
1196    fn multiply_add(sum: &mut Self::F, f: &Self::F, g: &Self::F);
1197
1198    /// Maximum number of products that can be summed before reconstruction.
1199    /// Both factors must transform canonical coefficients at the supplied transform's length,
1200    /// and each cyclic product must itself be reconstructible.
1201    fn max_product_sum_count(_f: &Self::F) -> usize {
1202        if Self::MULTIPLE { 1 } else { usize::MAX }
1203    }
1204
1205    fn power_projection_step(
1206        p_flat: Self::T,
1207        q_flat: Self::T,
1208        n: usize,
1209        py: usize,
1210        qy: usize,
1211    ) -> (Self::T, Self::T) {
1212        let base = n * 2;
1213        let len_p = base * py;
1214        let len_q = base * qy;
1215        let len = (len_p + len_q - 1).max(len_q + len_q - 1);
1216        let half = len.max(1).next_power_of_two() / 2;
1217
1218        let p_fft = Self::transform_ntt(p_flat, len);
1219        let q_fft = Self::transform_ntt(q_flat, len);
1220        let pr_fft = Self::odd_mul_normal_neg(&p_fft, &q_fft);
1221        let qr_fft = Self::even_mul_normal_neg(&q_fft, &q_fft);
1222        (
1223            Self::inverse_transform_ntt(pr_fft, half),
1224            Self::inverse_transform_ntt(qr_fft, half),
1225        )
1226    }
1227}

1233impl<M> NttReuse for Convolve<M>
1234where
1235    M: Montgomery32NttModulus,

1238
1239    fn transform_ntt(mut t: Self::T, len: usize) -> Self::F {
1240        t.resize_with(len.max(1).next_power_of_two(), Zero::zero);
1241        ntt(&mut t);
1242        t
1243    }

1245    fn inverse_transform_ntt(mut f: Self::F, len: usize) -> Self::T {
1246        intt(&mut f);
1247        f.truncate(len);
1248        f
1249    }
1250
1251    fn ntt_doubling(mut f: Self::F, monic: bool) -> Self::F {
1252        let n = f.len();
1253        let k = n.trailing_zeros() as usize;
1254        let mut a = Self::inverse_transform_ntt(f.clone(), n);
1255        if monic {
1256            a[0] -= MInt::<M>::from(2);
1257        }
1258        let zeta = MInt::<M>::new_unchecked(M::INFO.root[k + 1]);
1259        let zeta2 = zeta * zeta;
1260        let mut rot = [MInt::one(), zeta, zeta2, zeta2 * zeta];
1261        let step = zeta2 * zeta2;
1262        for a in a.chunks_mut(4) {
1263            for (a, rot) in a.iter_mut().zip(&mut rot) {
1264                *a *= *rot;
1265                *rot *= step;
1266            }
1267        }
1268        f.extend(Self::transform_ntt(a, n));
1269        f
1270    }
1271
1272    fn even_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F {
1273        assert_eq!(f.len(), g.len());
