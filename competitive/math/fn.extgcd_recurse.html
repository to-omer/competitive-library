40pub fn extgcd_recurse(a: i64, b: i64) -> (i64, i64, i64) {
41    if b == 0 {
42        (a, 1, 0)
43    } else {
44        let (g, x, y) = extgcd_recurse(b, a % b);
45        (g, y, x - (a / b) * y)
46    }
47}

50pub fn extgcd(mut a: i64, mut b: i64) -> (i64, i64, i64) {
51    let (mut u, mut v, mut x, mut y) = (1, 0, 0, 1);
52    while a != 0 {
53        let k = b / a;
54        x -= k * u;
55        y -= k * v;
56        b -= k * a;
57        std::mem::swap(&mut x, &mut u);
58        std::mem::swap(&mut y, &mut v);
59        std::mem::swap(&mut b, &mut a);
60    }
61    (b, x, y)
62}

64/// Requires nonnegative inputs.
65pub fn extgcd_binary(a: i64, b: i64) -> (i64, i64, i64) {
66    if b == 0 {
67        return (a, 1, 0);
68    }
69    if a == 0 {
70        return (b, 0, 1);
71    }
72    let k = (a | b).trailing_zeros();
73    let (mut c, mut d) = ((a >> k) as u64, (b >> k) as u64);
74    let swapped = d & 1 == 0;
75    if swapped {
76        std::mem::swap(&mut c, &mut d);
77    }
78    let (mut a, mut b) = (c, d);
79    // c*u == a and c*s == b modulo the odd d, with 0 <= u,s <= d.
80    let (mut u, mut s) = (1u64, 0u64);
81    let shift = a.trailing_zeros();
82    a >>= shift;
83    for _ in 0..shift {
84        if u & 1 == 1 {
85            u += d;
86        }
87        u /= 2;
88    }
89    while a != b {
90        if b < a {
91            std::mem::swap(&mut a, &mut b);
92            std::mem::swap(&mut u, &mut s);
93        }
94        b -= a;
95        if s < u {
96            s += d;
97        }
98        s -= u;
99        let shift = b.trailing_zeros();
100        b >>= shift;
101        for _ in 0..shift {
102            if s & 1 == 1 {
103                s += d;
104            }
105            s /= 2;
106        }
107    }
108    let x = s as i64;
109    let y = ((a as i128 - c as i128 * s as i128) / d as i128) as i64;
110    let g = (a << k) as i64;
111    if swapped { (g, y, x) } else { (g, x, y) }
112}
