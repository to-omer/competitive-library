215    fn convolve(a: Self::T, b: Self::T) -> Self::T {
216        let len = (a.len() + b.len()).saturating_sub(1);
217        // Keep accumulation overflow-free and exact in the FFT's f64 representation.
218        if (a.len().min(b.len()) <= 32 || {
219            let size = len.next_power_of_two();
220            let log = size.ilog2();
221            let limit = crate::avx_helper!(@dispatch simd_backend, SimdBackend;
222                3 * log + 16, 2 * log + 16, 4 * log + 16
223            );
224            2 * a.len() as u128 * b.len() as u128 <= limit as u128 * size as u128
225        }) && a.iter().map(|x| x.unsigned_abs()).max().unwrap_or(0) as u128
226            * b.iter().map(|x| x.unsigned_abs()).max().unwrap_or(0) as u128
227            <= (1u128 << 53) / a.len().min(b.len()).max(1) as u128
228        {
229            return crate::avx_helper!(@dispatch simd_backend, SimdBackend;
230                unsafe {
231                    if a.len().max(b.len()) < 32 {
232                        simd::convolve_i64_avx2(a, b, len)
233                    } else {
234                        simd::convolve_i64_avx512(a, b, len)
235                    }
236                },
237                unsafe { simd::convolve_i64_avx2(a, b, len) },
238                convolve_i64_naive(a, b, len)
239            );
240        }
241        if !a.is_empty() && !b.is_empty() {
242            crate::avx_helper!(@dispatch_avx2_fma return unsafe {
243                simd::convolve_f64_avx2(
244                    a.into_iter().map(|x| x as f64),
245                    b.into_iter().map(|x| x as f64),
246                    0..len,
247                )
248                .into_iter()
249                .map(|x| x.round() as i64)
250                .collect()
251            }, ());
252        }
253        let mut a = Self::transform(a, len);
254        let b = Self::transform(b, len);
255        Self::multiply(&mut a, &b);
256        Self::inverse_transform(a, len)
257    }
