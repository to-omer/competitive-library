96fn bit_reverse<T>(f: &mut [T]) {
97    let mut ip = vec![0u32];
98    let mut k = f.len();
99    let mut m = 1;
100    while 2 * m < k {
101        k /= 2;
102        for j in 0..m {
103            ip.push(ip[j] + k as u32);
104        }
105        m *= 2;
106    }
107    if m == k {
108        for i in 1..m {
109            for j in 0..i {
110                let ji = j + ip[i] as usize;
111                let ij = i + ip[j] as usize;
112                f.swap(ji, ij);
113            }
114        }
115    } else {
116        for i in 1..m {
117            for j in 0..i {
118                let ji = j + ip[i] as usize;
119                let ij = i + ip[j] as usize;
120                f.swap(ji, ij);
121                f.swap(ji + m, ij + m);
122            }
123        }
124    }
125}

127fn real_twiddles(n: usize, inverse: bool, mut f: impl FnMut(usize, Complex<f64>)) {
128    const BLOCK: usize = 256;
129    let sign = if inverse { 1.0 } else { -1.0 };
130    let step = Complex::primitive_nth_root_of_unity(sign * n as f64);
131    for start in (1..n / 4).step_by(BLOCK) {
132        let mut w = Complex::polar(1.0, sign * std::f64::consts::TAU * start as f64 / n as f64);
133        for k in start..(start + BLOCK).min(n / 4) {
134            f(k, w);
135            w *= step;
136        }
137    }
138}

140pub fn transform_real(t: impl IntoIterator<Item = f64>, len: usize) -> Vec<Complex<f64>> {
141    let n = len.max(4).next_power_of_two();
142    let mut f = vec![Complex::zero(); n / 2];
143    for (i, t) in t.into_iter().enumerate() {
144        if i & 1 == 0 {
145            f[i / 2].re = t;
146        } else {
147            f[i / 2].im = t;
148        }
149    }
150    fft(&mut f);
151    bit_reverse(&mut f);
152    f[0] = Complex::new(f[0].re + f[0].im, f[0].re - f[0].im);
153    f[n / 4] = f[n / 4].conjugate();
154    real_twiddles(n, false, |k, wk| {
155        let c = wk.conjugate().transpose() + 1.;
156        let d = c * (f[k] - f[n / 2 - k].conjugate()) * 0.5;
157        f[k] -= d;
158        f[n / 2 - k] += d.conjugate();
159    });
160    f
161}

163pub fn inverse_transform_real(mut f: Vec<Complex<f64>>, len: usize) -> Vec<f64> {
164    let n = len.max(4).next_power_of_two();
165    assert_eq!(f.len(), n / 2);
166    f[0] = Complex::new((f[0].re + f[0].im) * 0.5, (f[0].re - f[0].im) * 0.5);
167    f[n / 4] = f[n / 4].conjugate();
168    real_twiddles(n, true, |k, wk| {
169        let c = wk.transpose().conjugate() + 1.;
170        let d = c * (f[k] - f[n / 2 - k].conjugate()) * 0.5;
171        f[k] -= d;
172        f[n / 2 - k] += d.conjugate();
173    });
174    bit_reverse(&mut f);
175    ifft(&mut f);
176    let inv = 1. / (n / 2) as f64;
177    (0..len)
178        .map(|i| inv * if i & 1 == 0 { f[i / 2].re } else { f[i / 2].im })
179        .collect()
180}

183fn convolve_i64_naive(a: Vec<i64>, b: Vec<i64>, len: usize) -> Vec<i64> {
184    let (a, b) = if a.len() < b.len() { (b, a) } else { (a, b) };
185    if b.len() == 1 {
186        return a.into_iter().map(|a| a * b[0]).collect();
187    }
188    let mut c = vec![0; len];
189    for (i, a) in a.chunks(1024).enumerate() {
190        for (j, b) in b.iter().enumerate() {
191            let start = i * 1024 + j;
192            for (c, a) in c[start..start + a.len()].iter_mut().zip(a) {
193                *c += *a * *b;
194            }
195        }
196    }
197    c
198}

215    fn convolve(a: Self::T, b: Self::T) -> Self::T {
216        let len = (a.len() + b.len()).saturating_sub(1);
217        // Keep accumulation overflow-free and exact in the FFT's f64 representation.
218        if (a.len().min(b.len()) <= 32 || {
219            let size = len.next_power_of_two();
220            let log = size.ilog2();
221            let limit = crate::avx_helper!(@dispatch simd_backend, SimdBackend;
222                3 * log + 16, 2 * log + 16, 4 * log + 16
223            );
224            2 * a.len() as u128 * b.len() as u128 <= limit as u128 * size as u128
225        }) && a.iter().map(|x| x.unsigned_abs()).max().unwrap_or(0) as u128
226            * b.iter().map(|x| x.unsigned_abs()).max().unwrap_or(0) as u128
227            <= (1u128 << 53) / a.len().min(b.len()).max(1) as u128
228        {
229            return crate::avx_helper!(@dispatch simd_backend, SimdBackend;
230                unsafe {
231                    if a.len().max(b.len()) < 32 {
232                        simd::convolve_i64_avx2(a, b, len)
233                    } else {
234                        simd::convolve_i64_avx512(a, b, len)
235                    }
236                },
237                unsafe { simd::convolve_i64_avx2(a, b, len) },
238                convolve_i64_naive(a, b, len)
239            );
240        }
241        if !a.is_empty() && !b.is_empty() {
242            crate::avx_helper!(@dispatch_avx2_fma return unsafe {
243                simd::convolve_f64_avx2(
244                    a.into_iter().map(|x| x as f64),
245                    b.into_iter().map(|x| x as f64),
246                    0..len,
247                )
248                .into_iter()
249                .map(|x| x.round() as i64)
250                .collect()
251            }, ());
252        }
253        let mut a = Self::transform(a, len);
254        let b = Self::transform(b, len);
255        Self::multiply(&mut a, &b);
256        Self::inverse_transform(a, len)
257    }
