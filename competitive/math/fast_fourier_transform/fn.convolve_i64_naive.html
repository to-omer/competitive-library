336fn bit_reverse<T>(f: &mut [T]) {
337    let mut ip = vec![0u32];
338    let mut k = f.len();
339    let mut m = 1;
340    while 2 * m < k {
341        k /= 2;
342        for j in 0..m {
343            ip.push(ip[j] + k as u32);
344        }
345        m *= 2;
346    }
347    if m == k {
348        for i in 1..m {
349            for j in 0..i {
350                let ji = j + ip[i] as usize;
351                let ij = i + ip[j] as usize;
352                f.swap(ji, ij);
353            }
354        }
355    } else {
356        for i in 1..m {
357            for j in 0..i {
358                let ji = j + ip[i] as usize;
359                let ij = i + ip[j] as usize;
360                f.swap(ji, ij);
361                f.swap(ji + m, ij + m);
362            }
363        }
364    }
365}

367fn real_twiddles(n: usize, inverse: bool, mut f: impl FnMut(usize, Complex<f64>)) {
368    const BLOCK: usize = 256;
369    let sign = if inverse { 1.0 } else { -1.0 };
370    let step = Complex::primitive_nth_root_of_unity(sign * n as f64);
371    for start in (1..n / 4).step_by(BLOCK) {
372        let mut w = Complex::polar(1.0, sign * std::f64::consts::TAU * start as f64 / n as f64);
373        for k in start..(start + BLOCK).min(n / 4) {
374            f(k, w);
375            w *= step;
376        }
377    }
378}

380pub fn transform_real(t: impl IntoIterator<Item = f64>, len: usize) -> Vec<Complex<f64>> {
381    let n = len.max(4).next_power_of_two();
382    let mut f = vec![Complex::zero(); n / 2];
383    for (i, t) in t.into_iter().enumerate() {
384        if i & 1 == 0 {
385            f[i / 2].re = t;
386        } else {
387            f[i / 2].im = t;
388        }
389    }
390    fft(&mut f);
391    bit_reverse(&mut f);
392    f[0] = Complex::new(f[0].re + f[0].im, f[0].re - f[0].im);
393    f[n / 4] = f[n / 4].conjugate();
394    real_twiddles(n, false, |k, wk| {
395        let c = wk.conjugate().transpose() + 1.;
396        let d = c * (f[k] - f[n / 2 - k].conjugate()) * 0.5;
397        f[k] -= d;
398        f[n / 2 - k] += d.conjugate();
399    });
400    f
401}

403pub fn inverse_transform_real(mut f: Vec<Complex<f64>>, len: usize) -> Vec<f64> {
404    let n = len.max(4).next_power_of_two();
405    assert_eq!(f.len(), n / 2);
406    f[0] = Complex::new((f[0].re + f[0].im) * 0.5, (f[0].re - f[0].im) * 0.5);
407    f[n / 4] = f[n / 4].conjugate();
408    real_twiddles(n, true, |k, wk| {
409        let c = wk.transpose().conjugate() + 1.;
410        let d = c * (f[k] - f[n / 2 - k].conjugate()) * 0.5;
411        f[k] -= d;
412        f[n / 2 - k] += d.conjugate();
413    });
414    bit_reverse(&mut f);
415    ifft(&mut f);
416    let inv = 1. / (n / 2) as f64;
417    (0..len)
418        .map(|i| inv * if i & 1 == 0 { f[i / 2].re } else { f[i / 2].im })
419        .collect()
420}

423fn convolve_i64_naive(a: Vec<i64>, b: Vec<i64>, len: usize) -> Vec<i64> {
424    let (a, b) = if a.len() < b.len() { (b, a) } else { (a, b) };
425    if b.len() == 1 {
426        return a.into_iter().map(|a| a * b[0]).collect();
427    }
428    let mut c = vec![0; len];
429    for (i, a) in a.chunks(1024).enumerate() {
430        for (j, b) in b.iter().enumerate() {
431            let start = i * 1024 + j;
432            for (c, a) in c[start..start + a.len()].iter_mut().zip(a) {
433                *c += *a * *b;
434            }
435        }
436    }
437    c
438}

455    fn convolve(a: Self::T, b: Self::T) -> Self::T {
456        let len = (a.len() + b.len()).saturating_sub(1);
457        // Keep accumulation overflow-free and exact in the FFT's f64 representation.
458        if (a.len().min(b.len()) <= 32 || {
459            let size = len.next_power_of_two();
460            let log = size.ilog2();
461            let limit = crate::avx_helper!(@dispatch simd_backend, SimdBackend;
462                3 * log + 16, 2 * log + 16, 4 * log + 16
463            );
464            2 * a.len() as u128 * b.len() as u128 <= limit as u128 * size as u128
465        }) && a.iter().map(|x| x.unsigned_abs()).max().unwrap_or(0) as u128
466            * b.iter().map(|x| x.unsigned_abs()).max().unwrap_or(0) as u128
467            <= (1u128 << 53) / a.len().min(b.len()).max(1) as u128
468        {
469            return crate::avx_helper!(@dispatch simd_backend, SimdBackend;
470                unsafe {
471                    if a.len().max(b.len()) < 32 {
472                        simd::convolve_i64_avx2(a, b, len)
473                    } else {
474                        simd::convolve_i64_avx512(a, b, len)
475                    }
476                },
477                unsafe { simd::convolve_i64_avx2(a, b, len) },
478                convolve_i64_naive(a, b, len)
479            );
480        }
481        if !a.is_empty() && !b.is_empty() {
482            crate::avx_helper!(@dispatch_avx2_fma return unsafe {
483                simd::convolve_f64_avx2(
484                    a.into_iter().map(|x| x as f64),
485                    b.into_iter().map(|x| x as f64),
486                    0..len,
487                )
488                .into_iter()
489                .map(|x| x.round() as i64)
490                .collect()
491            }, ());
492        }
493        let mut a = Self::transform(a, len);
494        let b = Self::transform(b, len);
495        Self::multiply(&mut a, &b);
496        Self::inverse_transform(a, len)
497    }
