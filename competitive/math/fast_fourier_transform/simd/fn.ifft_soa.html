180    let len = a.len() + b.len() - 1;
181    let n = len.next_power_of_two() / 2;

183        3
184    } else {
185        5
186    };

190        3
191    } else {
192        5
193    };

195    drop(b);

198        fft_soa(&mut fb[part]);

207    drop(fb);

214        let imag: [[i64; 4]; 5] = std::array::from_fn(|part| round4(&fa[part][block].im));

217            if i < len {
