71fn power(mut a: u64, mut e: u64, mul: impl Fn(u64, u64) -> u64) -> u64 {
72    let mut x = 1;
73    while e > 0 {
74        if e & 1 == 1 {

77        a = mul(a, a);
78        e >>= 1;
79    }

83fn prime_field_log(a: u64, b: u64, p: u64) -> Option<(u64, u64)> {
84    if p <= u32::MAX.into() {
85        let br = BarrettReduction::<u64>::new(p);
86        pohlig_hellman::<4>(a, b, p, |x, y| br.rem(x * y))
87    } else {
88        let br = BarrettReduction::<u128>::new(p as _);
89        pohlig_hellman::<16>(a, b, p, |x, y| br.rem(x as u128 * y as u128) as u64)
90    }
91}

93fn pohlig_hellman<const LANES: usize>(
94    mut a: u64,
95    mut b: u64,
96    p: u64,
97    mul: impl Fn(u64, u64) -> u64 + Copy,
98) -> Option<(u64, u64)> {
99    let mut order = p - 1;
100    let mut x = 0;
101    let mut period = 1;
102    for (q, e) in prime_factors(order) {
103        for _ in 0..e {
104            order /= q;
105            let (g, h) = if order == 1 {
106                (a, b)
107            } else {
108                (power(a, order, mul), power(b, order, mul))
109            };
110            if g == 1 {
111                if h != 1 {
112                    return None;
113                }
114                continue;
115            }
116            let digit = if h == 1 {
117                0
118            } else {
119                prime_order_log::<LANES>(g, h, p, q, mul)
120            };

122            period *= q;
123            if order != 1 {

126                }

128            }
129        }
130    }
