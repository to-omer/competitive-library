pub fn with_roots<T>(n: usize, f: impl FnOnce(&[Complex<f64>]) -> T) -> T
Expand description

Borrows at least n cached roots, where n is a nonzero power of two. The callback must not reenter this cache.

96fn bit_reverse<T>(f: &mut [T]) {
97    let mut ip = vec![0u32];
98    let mut k = f.len();
99    let mut m = 1;
100    while 2 * m < k {
101        k /= 2;
102        for j in 0..m {
103            ip.push(ip[j] + k as u32);
104        }
105        m *= 2;
106    }
107    if m == k {
108        for i in 1..m {
109            for j in 0..i {
110                let ji = j + ip[i] as usize;
111                let ij = i + ip[j] as usize;
112                f.swap(ji, ij);
113            }
114        }
115    } else {
116        for i in 1..m {
117            for j in 0..i {
118                let ji = j + ip[i] as usize;
119                let ij = i + ip[j] as usize;
120                f.swap(ji, ij);
121                f.swap(ji + m, ij + m);
122            }
123        }
124    }
125}

127fn real_twiddles(n: usize, inverse: bool, mut f: impl FnMut(usize, Complex<f64>)) {
128    const BLOCK: usize = 256;
129    let sign = if inverse { 1.0 } else { -1.0 };
130    let step = Complex::primitive_nth_root_of_unity(sign * n as f64);
131    for start in (1..n / 4).step_by(BLOCK) {
132        let mut w = Complex::polar(1.0, sign * std::f64::consts::TAU * start as f64 / n as f64);
133        for k in start..(start + BLOCK).min(n / 4) {
134            f(k, w);
135            w *= step;
136        }
137    }
138}

140pub fn transform_real(t: impl IntoIterator<Item = f64>, len: usize) -> Vec<Complex<f64>> {
141    let n = len.max(4).next_power_of_two();
142    let mut f = vec![Complex::zero(); n / 2];
143    for (i, t) in t.into_iter().enumerate() {
144        if i & 1 == 0 {
145            f[i / 2].re = t;
146        } else {
147            f[i / 2].im = t;
148        }
149    }
150    fft(&mut f);
151    bit_reverse(&mut f);
152    f[0] = Complex::new(f[0].re + f[0].im, f[0].re - f[0].im);
153    f[n / 4] = f[n / 4].conjugate();
154    real_twiddles(n, false, |k, wk| {
155        let c = wk.conjugate().transpose() + 1.;
156        let d = c * (f[k] - f[n / 2 - k].conjugate()) * 0.5;
157        f[k] -= d;
158        f[n / 2 - k] += d.conjugate();
159    });
160    f
161}

163pub fn inverse_transform_real(mut f: Vec<Complex<f64>>, len: usize) -> Vec<f64> {
164    let n = len.max(4).next_power_of_two();
165    assert_eq!(f.len(), n / 2);
166    f[0] = Complex::new((f[0].re + f[0].im) * 0.5, (f[0].re - f[0].im) * 0.5);
167    f[n / 4] = f[n / 4].conjugate();
168    real_twiddles(n, true, |k, wk| {
169        let c = wk.transpose().conjugate() + 1.;
170        let d = c * (f[k] - f[n / 2 - k].conjugate()) * 0.5;
171        f[k] -= d;
172        f[n / 2 - k] += d.conjugate();
173    });
174    bit_reverse(&mut f);
175    ifft(&mut f);
176    let inv = 1. / (n / 2) as f64;
177    (0..len)
178        .map(|i| inv * if i & 1 == 0 { f[i / 2].re } else { f[i / 2].im })
179        .collect()
180}

183fn convolve_i64_naive(a: Vec<i64>, b: Vec<i64>, len: usize) -> Vec<i64> {
184    let (a, b) = if a.len() < b.len() { (b, a) } else { (a, b) };
185    if b.len() == 1 {
186        return a.into_iter().map(|a| a * b[0]).collect();
187    }
188    let mut c = vec![0; len];
189    for (i, a) in a.chunks(1024).enumerate() {
190        for (j, b) in b.iter().enumerate() {
191            let start = i * 1024 + j;
192            for (c, a) in c[start..start + a.len()].iter_mut().zip(a) {
193                *c += *a * *b;
194            }
195        }
196    }
197    c
198}

203    fn length(t: &Self::T) -> usize {

215    fn convolve(a: Self::T, b: Self::T) -> Self::T {
216        let len = (a.len() + b.len()).saturating_sub(1);
217        // Keep accumulation overflow-free and exact in the FFT's f64 representation.
218        if (a.len().min(b.len()) <= 32 || {
219            let size = len.next_power_of_two();
220            let log = size.ilog2();
221            let limit = crate::avx_helper!(@dispatch simd_backend, SimdBackend;
222                3 * log + 16, 2 * log + 16, 4 * log + 16
223            );
224            2 * a.len() as u128 * b.len() as u128 <= limit as u128 * size as u128
225        }) && a.iter().map(|x| x.unsigned_abs()).max().unwrap_or(0) as u128
226            * b.iter().map(|x| x.unsigned_abs()).max().unwrap_or(0) as u128
227            <= (1u128 << 53) / a.len().min(b.len()).max(1) as u128
228        {
229            return crate::avx_helper!(@dispatch simd_backend, SimdBackend;
230                unsafe {
231                    if a.len().max(b.len()) < 32 {
232                        simd::convolve_i64_avx2(a, b, len)
233                    } else {
234                        simd::convolve_i64_avx512(a, b, len)
235                    }
236                },
237                unsafe { simd::convolve_i64_avx2(a, b, len) },
238                convolve_i64_naive(a, b, len)
239            );
240        }
241        if !a.is_empty() && !b.is_empty() {
242            crate::avx_helper!(@dispatch_avx2_fma return unsafe {
243                simd::convolve_f64_avx2(
244                    a.into_iter().map(|x| x as f64),
245                    b.into_iter().map(|x| x as f64),
246                    0..len,
247                )
248                .into_iter()
249                .map(|x| x.round() as i64)
250                .collect()
251            }, ());
252        }
253        let mut a = Self::transform(a, len);
254        let b = Self::transform(b, len);
255        Self::multiply(&mut a, &b);
256        Self::inverse_transform(a, len)
257    }
258    fn multiply(f: &mut Self::F, g: &Self::F) {
259        assert_eq!(f.len(), g.len());
260        f[0].re *= g[0].re;
261        f[0].im *= g[0].im;
262        for (f, g) in f.iter_mut().zip(g.iter()).skip(1) {
263            *f *= *g;
264        }
265    }
266}

297macro_rules! fft_kernel {
298    ($a:expr, $cache:expr, $inverse:expr) => {{
299        let a = $a;
300        let cache = $cache;
301        let n = a.len();
302        if $inverse {
303            let mut v = 1;
304            if n.trailing_zeros() & 1 == 1 {
305                for (a, w) in a.as_chunks_mut::<2>().0.iter_mut().zip(cache) {
306                    let y = (a[0] - a[1]) * w.conjugate();
307                    a[0] += a[1];
308                    a[1] = y;
309                }
310                v = 2;
311            }
312            while v < n {
313                for (q, block) in a.chunks_exact_mut(v * 4).enumerate() {
314                    let (a, rest) = block.split_at_mut(v);
315                    let (b, rest) = rest.split_at_mut(v);
316                    let (c, d) = rest.split_at_mut(v);
317                    let w0 = cache[q].conjugate();
318                    let w1 = cache[q << 1].conjugate();
319                    let w3 = w0 * w1;
320                    for (((a, b), c), d) in a.iter_mut().zip(b).zip(c).zip(d) {
321                        let ac0 = *a + *b;
322                        let ac1 = *c + *d;
323                        let bd0 = *a - *b;
324                        let bd1 = *c - *d;
325                        let bd1 = Complex::new(-bd1.im, bd1.re);
326                        *a = ac0 + ac1;
327                        *b = (bd0 + bd1) * w1;
328                        *c = (ac0 - ac1) * w0;
329                        *d = (bd0 - bd1) * w3;
330                    }
331                }
332                v <<= 2;
333            }
334        } else {
335            let mut v = n / 2;
336            while v >= 2 {
337                let l = v / 2;
338                for (q, block) in a.chunks_exact_mut(l * 4).enumerate() {
339                    let (a, rest) = block.split_at_mut(l);
340                    let (b, rest) = rest.split_at_mut(l);
341                    let (c, d) = rest.split_at_mut(l);
342                    let w0 = cache[q];
343                    let w1 = cache[q << 1];
344                    let w3 = w0 * w1;
345                    for (((a, b), c), d) in a.iter_mut().zip(b).zip(c).zip(d) {
346                        let bv = *b * w1;
347                        let cv = *c * w0;
348                        let dv = *d * w3;
349                        let ac0 = *a + cv;
350                        let ac1 = *a - cv;
351                        let bd0 = bv + dv;
352                        let bd1 = bv - dv;
353                        let bd1 = Complex::new(bd1.im, -bd1.re);
354                        *a = ac0 + bd0;
355                        *b = ac0 - bd0;
356                        *c = ac1 + bd1;
357                        *d = ac1 - bd1;
358                    }
359                }
360                v >>= 2;
361            }
362            if v == 1 {
363                for (a, w) in a.as_chunks_mut::<2>().0.iter_mut().zip(cache) {
364                    let y = a[1] * *w;
365                    a[1] = a[0] - y;
366                    a[0] += y;
367                }
368            }
369        }
370    }};
371}

123unsafe fn split_u64_coefficients(values: &[u64], n: usize) -> [Vec<Complex4>; 5] {
124    let mut result: [Vec<Complex4>; 5] = std::array::from_fn(|_| {
125        let mut part = Vec::with_capacity(n / 4);
126        advise_huge_pages(&mut part);
127        part
128    });
129    for (i, chunk) in values.chunks(4).enumerate() {
130        let mut parts = [Complex4::default(); 5];
131        for (lane, mut value) in chunk.iter().copied().enumerate() {
132            for part in &mut parts {
133                let digit = ((value << 51) as i64) >> 51;
134                value = (value >> 13).wrapping_add(u64::from(digit < 0));
135                part.re[lane] = digit as f64;
136            }
137        }
138        for (result, part) in result.iter_mut().zip(parts) {
139            if i < n / 4 {
140                result.push(part);
141            } else {
142                result[i - n / 4].im = part.re;
143            }
144        }
145    }
146    for part in &mut result {
147        part.resize(n / 4, Complex4::default());
148    }
149    result
150}

153unsafe fn dot_u64_soa<const PARTS: usize>(a: &mut [Vec<Complex4>; 5], b: &[Vec<Complex4>; 5]) {
154    let n = a[0].len() * 4;
155    with_roots(n / 2, |cache| {
156        for block in 0..a[0].len() {
157            let mut br = [_mm256_setzero_pd(); PARTS];
158            let mut bi = br;
159            let mut rr = [_mm256_setzero_pd(); 5];
160            let mut ri = rr;
161            for part in 0..PARTS {
162                (br[part], bi[part]) = load(&b[part][block]);
163            }
164            let w = eval_twiddle(cache, 1, a[0].len(), block);
165            let wr = _mm256_setr_pd(w.re, 1.0, 1.0, 1.0);
166            let wi = _mm256_setr_pd(w.im, 0.0, 0.0, 0.0);
167            // Keep lane indices constant to avoid loop-carried SIMD spills.
168            macro_rules! accumulate_lane {
169                ($lane:expr) => {{
170                    // Descending right preserves ascending left for each output.
171                    for right in (0..PARTS).rev() {
172                        for left in 0..PARTS.min(5 - right) {
173                            let ar = _mm256_set1_pd(a[left][block].re[$lane]);
174                            let ai = _mm256_set1_pd(a[left][block].im[$lane]);
175                            multiply_accumulate(
176                                &mut rr[left + right],
177                                &mut ri[left + right],
178                                ar,
179                                ai,
180                                br[right],
181                                bi[right],
182                            );
183                        }
184                        if $lane != 3 {
185                            br[right] = _mm256_permute4x64_pd::<0x93>(br[right]);
186                            bi[right] = _mm256_permute4x64_pd::<0x93>(bi[right]);
187                            (br[right], bi[right]) = complex_mul(br[right], bi[right], wr, wi);
188                        }
189                    }
190                }};
191            }
192            accumulate_lane!(0);
193            accumulate_lane!(1);
194            accumulate_lane!(2);
195            accumulate_lane!(3);
