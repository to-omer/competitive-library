204pub unsafe fn convolve_u64_avx2(a: Vec<u64>, b: Vec<u64>) -> Vec<u64> {
205    let len = a.len() + b.len() - 1;
206    let n = len.next_power_of_two() / 2;
207    let parts = |values: &[u64]| {
208        let maximum = values
209            .iter()
210            .try_fold(0, |maximum, &value| {
211                (value <= 4095 * (1 + (1u64 << 13) + (1u64 << 26) + (1u64 << 39)))
212                    .then_some(maximum.max(value))
213            })
214            .unwrap_or(u64::MAX);
215        let mut limit = (1u64 << 12) - 1;
216        let mut parts = 1;
217        while parts < 5 && maximum > limit {
218            limit = (limit << 13) + ((1 << 12) - 1);
219            parts += 1;
220        }
221        parts
222    };
223    let a_parts = parts(&a);
224    let mut fa = split_u64_coefficients(&a, n);
225    drop(a);
226    let b_parts = parts(&b);
227    let mut fb = split_u64_coefficients(&b, n);
228    drop(b);
229    let shared = a_parts.min(b_parts).min(3);
230    for part in 0..shared {
231        fft_soa(&mut fa[part]);
232        fft_soa(&mut fb[part]);
233    }
234    for part in &mut fa[shared..a_parts] {
235        fft_soa(part);
236    }
237    for part in &mut fb[shared..b_parts] {
238        fft_soa(part);
239    }
240    match a_parts.max(b_parts) {
241        1 => dot_u64_soa::<1>(&mut fa, &fb),
242        2 => dot_u64_soa::<2>(&mut fa, &fb),
243        3 => dot_u64_soa::<3>(&mut fa, &fb),
244        4 => dot_u64_soa::<4>(&mut fa, &fb),
245        _ => dot_u64_soa::<5>(&mut fa, &fb),
246    }
247    drop(fb);
248    for part in &mut fa[..5.min(a_parts + b_parts - 1)] {
249        ifft_soa(part);
250    }
251    let mut result = vec![0; len];
252    for (block, _) in fa[0].iter().enumerate() {
253        let real: [[i64; 4]; 5] = std::array::from_fn(|part| round4(&fa[part][block].re));
254        let imag: [[i64; 4]; 5] = std::array::from_fn(|part| round4(&fa[part][block].im));
255        for lane in 0..4 {
256            let i = block * 4 + lane;
257            if i < len {
258                result[i] = (real[0][lane] as u64)
259                    .wrapping_add((real[1][lane] as u64) << 13)
260                    .wrapping_add((real[2][lane] as u64) << 26)
261                    .wrapping_add((real[3][lane] as u64) << 39)
262                    .wrapping_add((real[4][lane] as u64) << 52);
263            }
264            if i + n < len {
265                result[i + n] = (imag[0][lane] as u64)
266                    .wrapping_add((imag[1][lane] as u64) << 13)
267                    .wrapping_add((imag[2][lane] as u64) << 26)
268                    .wrapping_add((imag[3][lane] as u64) << 39)
269                    .wrapping_add((imag[4][lane] as u64) << 52);
270            }
271        }
272    }
273    result
274}

277pub unsafe fn convolve_mint_avx2<M>(a: Vec<MInt<M>>, b: Vec<MInt<M>>) -> Vec<MInt<M>>
278where
279    M: MIntConvert + MIntConvert<u32>,
280{
281    let len = a.len() + b.len() - 1;
282    let n = len.next_power_of_two() / 2;
283    let modulus = <M as MIntConvert<u32>>::mod_into() as i64;
284    let split = (modulus as f64).sqrt() as i64 + 1;
285    let (mut a0, mut a1) = split_coefficients(a, n, modulus, split);
286    let (mut b0, mut b1) = split_coefficients(b, n, modulus, split);
287    fft_soa(&mut a0);
288    fft_soa(&mut a1);
289    fft_soa(&mut b0);
290    fft_soa(&mut b1);
291    dot_soa(&mut a0, &mut a1, &mut b0, &b1);
292    drop(b1);
293    ifft_soa(&mut a0);
294    ifft_soa(&mut a1);
295    ifft_soa(&mut b0);
296    let split2 = (split * split % modulus) as f64;
297    let split = _mm256_set1_pd(split as f64);
298    let split2 = _mm256_set1_pd(split2);
299    let inverse = _mm256_set1_pd(1.0 / modulus as f64);
300    let modulus = _mm256_set1_pd(modulus as f64);
301    let magic = _mm256_set1_pd((3i64 << 51) as f64);
302    let mut result = vec![MInt::<M>::from(0u32); len];
303    for (block, ((a0, a1), b0)) in a0.iter().zip(&a1).zip(&b0).enumerate() {
304        for (part, (a0, a1, b0)) in [(&a0.re, &a1.re, &b0.re), (&a0.im, &a1.im, &b0.im)]
305            .into_iter()
306            .enumerate()
307        {
308            let a0 = _mm256_round_pd::<{ _MM_FROUND_TO_NEAREST_INT | _MM_FROUND_NO_EXC }>(
309                _mm256_load_pd(a0.as_ptr()),
310            );
311            let a1 = _mm256_round_pd::<{ _MM_FROUND_TO_NEAREST_INT | _MM_FROUND_NO_EXC }>(
312                _mm256_load_pd(a1.as_ptr()),
313            );
314            let b0 = _mm256_round_pd::<{ _MM_FROUND_TO_NEAREST_INT | _MM_FROUND_NO_EXC }>(
315                _mm256_load_pd(b0.as_ptr()),
316            );
317            let a0 = reduce_mod4(a0, modulus, inverse);
318            let a1 = reduce_mod4(a1, modulus, inverse);
319            let b0 = reduce_mod4(b0, modulus, inverse);
320            let value = _mm256_fmadd_pd(b0, split2, _mm256_fmadd_pd(a1, split, a0));
321            let value = reduce_mod4(value, modulus, inverse);
322            let value = _mm256_add_pd(
323                value,
324                _mm256_and_pd(
325                    _mm256_cmp_pd::<_CMP_LT_OQ>(value, _mm256_setzero_pd()),
326                    modulus,
327                ),
328            );
329            let value = _mm256_sub_epi64(
330                _mm256_castpd_si256(_mm256_add_pd(value, magic)),
331                _mm256_castpd_si256(magic),
332            );
333            let mut lanes = [0i64; 4];
334            _mm256_storeu_si256(lanes.as_mut_ptr().cast(), value);
335            for (lane, value) in lanes.into_iter().enumerate() {
336                let i = block * 4 + lane + part * n;
337                if i < len {
338                    let value = value as u32;
339                    let modulus = <M as MIntConvert<u32>>::mod_into();
340                    // Expose the reduced range to the conversion's remainder operation.
341                    result[i] = MInt::<M>::from(if value < modulus {
342                        value
343                    } else {
344                        value % modulus
345                    });
346                }
347            }
348        }
349    }
350    result
