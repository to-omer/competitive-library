298    fn next_column(&self, row: usize, mut start: usize, cols: usize) -> usize {
299        while start < cols {
300            let word = start / 64;
301            let bits = self.data[row..]
302                .iter()
303                .fold(0, |x, row| x | row.words()[word])
304                & (u64::MAX << (start & 63));
305            if bits != 0 {
306                return (word * 64 + bits.trailing_zeros() as usize).min(cols);
307            }
308            start = (word + 1) * 64;
309        }
310        cols

314    fn eliminate_sparse(&mut self, cols: usize, full: bool, require_full_rank: bool) -> Vec<usize> {
315        let n = self.shape.0;
316        let mut basis = vec![n; cols];
317        let mut pivots = Vec::new();
318        for i in 0..n {
319            loop {
320                let Some(c) = self[i].iter_ones().next().filter(|&c| c < cols) else {
321                    if require_full_rank {
322                        return pivots;
323                    }
324                    break;
325                };
326                if basis[c] == n {
327                    basis[c] = i;
328                    pivots.push(c);
329                    break;
330                }
331                let (upper, lower) = self.data.split_at_mut(i);
332                xor(
333                    &mut lower[0].words_mut()[c / 64..],
334                    &upper[basis[c]].words()[c / 64..],
335                );
336            }
337        }
338        pivots.sort_unstable();
339        self.data
340            .sort_by_cached_key(|row| row.iter_ones().next().filter(|&c| c < cols).unwrap_or(cols));
341        if full {
342            for (i, &c) in pivots.iter().enumerate() {
343                basis[c] = i;
344            }
345            for i in (0..pivots.len()).rev() {
346                let next = self[i].iter_ones().take_while(|&c| c < cols).nth(1);
347                if let Some(c) = next
348                    && basis[c] != n
349                {
350                    let (upper, lower) = self.data.split_at_mut(basis[c]);
351                    xor(
352                        &mut upper[i].words_mut()[c / 64..],
353                        &lower[0].words()[c / 64..],
354                    );
355                }
356            }
357        }
358        pivots

362    fn mul_impl(&self, rhs: &Self) -> Self {
363        let mut result = Self::zeros((self.shape.0, rhs.shape.1));
364        let ones = self.data.iter().map(BitSet::count_ones).sum::<u64>();
365        let size = self.shape.0 as u64 * self.shape.1 as u64;
366        if self.shape.0 < 256 || self.shape.1 < 32 || ones <= size / 8 {
367            for (a, c) in self.data.iter().zip(&mut result.data) {
368                for j in a.iter_ones() {
369                    xor(c.words_mut(), rhs[j].words());
370                }
371            }
372            return result;
373        }
374        if size - ones <= size / 8 {
375            let mut sum = BitSet::new(rhs.shape.1);
376            for row in &rhs.data {
377                sum ^= row;
378            }
379            for (a, c) in self.data.iter().zip(&mut result.data) {
380                c.words_mut().copy_from_slice(sum.words());
381                for j in (!a.clone()).iter_ones() {
382                    xor(c.words_mut(), rhs[j].words());
383                }
384            }
385            return result;
386        }
387        let width = rhs.shape.1.div_ceil(64);
388        if width == 0 {
389            return result;
390        }
391
392        // Separate the table groups by a cache line to avoid mapping them to the same sets.
393        let group = 256 * width + 8;
394        let mut storage = BitSet::new(8 * group * 64);
395        let table = storage.words_mut();
396        for start in (0..self.shape.1).step_by(64) {
397            for (t, table) in table.chunks_exact_mut(group).enumerate() {
398                let col = start + t * 8;
399                for bit in 0..self.shape.1.saturating_sub(col).min(8) {
400                    let row = rhs[col + bit].words();
401                    let half = (1 << bit) * width;
402                    let (lower, upper) = table.split_at_mut(half);
403                    for (source, dest) in
404                        lower.chunks_exact(width).zip(upper.chunks_exact_mut(width))
405                    {
406                        for ((x, y), z) in dest.iter_mut().zip(source).zip(row) {
407                            *x = y ^ z;
408                        }
409                    }
410                }
411            }
412            for (a, c) in self.data.iter().zip(&mut result.data) {
413                let key = a.words()[start / 64];
414                let offset = (key & 255) as usize * width;
415                let p0 = &table[offset..offset + width];
416                let offset = group + (key >> 8 & 255) as usize * width;
417                let p1 = &table[offset..offset + width];
418                let offset = 2 * group + (key >> 16 & 255) as usize * width;
419                let p2 = &table[offset..offset + width];
420                let offset = 3 * group + (key >> 24 & 255) as usize * width;
421                let p3 = &table[offset..offset + width];
422                let offset = 4 * group + (key >> 32 & 255) as usize * width;
423                let p4 = &table[offset..offset + width];
424                let offset = 5 * group + (key >> 40 & 255) as usize * width;
425                let p5 = &table[offset..offset + width];
426                let offset = 6 * group + (key >> 48 & 255) as usize * width;
427                let p6 = &table[offset..offset + width];
428                let offset = 7 * group + (key >> 56 & 255) as usize * width;
429                let p7 = &table[offset..offset + width];
430                for ((((((((x, p0), p1), p2), p3), p4), p5), p6), p7) in c
431                    .words_mut()
432                    .iter_mut()
433                    .zip(p0)
434                    .zip(p1)
435                    .zip(p2)
436                    .zip(p3)
437                    .zip(p4)
438                    .zip(p5)
439                    .zip(p6)
440                    .zip(p7)
441                {
442                    *x ^= p0 ^ p1 ^ p2 ^ p3 ^ p4 ^ p5 ^ p6 ^ p7;
443                }
444            }
445        }
