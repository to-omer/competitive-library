80    pub fn inverse(&self) -> Option<Self> {
81        let (n, m) = self.shape;
82        assert_eq!(n, m);
83        let mut a = Self::zeros((n, 2 * n));
84        for i in 0..n {
85            a[i].words_mut()[..self[i].words().len()].copy_from_slice(self[i].words());
86            a[i].set(n + i, true);
87        }
88        if a.eliminate(n, true, true).len() != n {
89            return None;
90        }
91        for row in &mut a.data {
92            *row >>= n;
93        }
94        let mut inverse = Self::zeros((n, n));
95        for (row, source) in inverse.data.iter_mut().zip(&a.data) {
96            let len = row.words().len();
97            row.words_mut().copy_from_slice(&source.words()[..len]);
98        }
99        Some(inverse)
100    }

104    pub fn solve_system_of_linear_equations(&self, b: &BitSet) -> Option<BitMatrixSolution> {
105        let (n, m) = self.shape;
106        assert_eq!(b.len(), n);
107        let mut a = Self::zeros((n, m + 1));
108        for i in 0..n {
109            a[i].words_mut()[..self[i].words().len()].copy_from_slice(self[i].words());
110            a[i].set(m, b.get(i));
111        }
112        let pivots = a.eliminate(m, true, false);
113        if a.data[pivots.len()..].iter().any(|row| row.get(m)) {
114            return None;
115        }
116        let mut particular = BitSet::new(m);
117        let mut free = BitSet::ones(m);
118        for (i, &c) in pivots.iter().enumerate() {
119            particular.set(c, a[i].get(m));
120            free.set(c, false);
121        }
122        let basis = free
123            .iter_ones()
124            .map(|c| {
125                let mut row = BitSet::new(m);
126                row.set(c, true);
127                for (i, &p) in pivots.iter().enumerate() {
128                    row.set(p, a[i].get(c));
129                }
130                row
131            })
132            .collect();
133        Some(BitMatrixSolution { particular, basis })
134    }

170    fn eliminate_impl(&mut self, cols: usize, full: bool, require_full_rank: bool) -> Vec<usize> {
171        let n = self.shape.0;
172        let mut pivots = Vec::with_capacity(n.min(cols));
173        if n < 32 {
174            let mut c = 0;
175            while c < cols {
176                let r = pivots.len();
177                if r == n {
178                    break;
179                }
180                let Some(p) = (r..n).find(|&i| self[i].get(c)) else {
181                    if require_full_rank {
182                        return pivots;
183                    }
184                    c = self.next_column(r, c + 1, cols);
185                    continue;
186                };
187                self.data.swap(r, p);
188                let (upper, lower) = self.data.split_at_mut(r);
189                let (pivot, lower) = lower.split_first_mut().unwrap();
190                for row in lower
191                    .iter_mut()
192                    .chain(upper.iter_mut().take(if full { r } else { 0 }))
193                {
194                    if row.get(c) {
195                        xor(&mut row.words_mut()[c / 64..], &pivot.words()[c / 64..]);
196                    }
197                }
198                pivots.push(c);
199                c += 1;
200            }
201            return pivots;
202        }
203        // Eliminating a shared leading coefficient preserves the two-coefficient bound.
204        if self
205            .data
206            .iter()
207            .all(|row| row.iter_ones().take_while(|&c| c < cols).take(3).count() <= 2)
208        {
209            return self.eliminate_sparse(cols, full, require_full_rank);
210        }
211        let block = if n < 512 { 4 } else { 8 };
212        let mut table = vec![BitSet::new(self.shape.1); 1 << block];
213        let mut reduced = vec![0; n];
214        let mut start = 0;
215        while start < cols {
216            let first = pivots.len();
217            let word = start / 64;
218            reduced.fill(first);
219            for c in start..cols.min(start + block) {
220                let r = pivots.len();
221                let mut pivot = None;
222                for (i, reduced) in reduced.iter_mut().enumerate().skip(r) {
223                    let (upper, lower) = self.data.split_at_mut(i);
224                    let row = &mut lower[0];
225                    for p in *reduced..r {
226                        if row.get(pivots[p]) {
227                            xor(&mut row.words_mut()[word..], &upper[p].words()[word..]);
228                        }
229                    }
230                    *reduced = r;
231                    if row.get(c) {
232                        pivot = Some(i);
233                        break;
234                    }
235                }
236                if let Some(p) = pivot {
237                    self.data.swap(r, p);
238                    reduced.swap(r, p);
239                    pivots.push(c);
240                } else if require_full_rank {
241                    return pivots;
242                }
243            }
244            let rank = pivots.len();
245            if first == rank {
246                let next = self.next_column(rank, start + block, cols);
247                if next == cols {
248                    break;
249                }
250                start = next / block * block;
251                continue;
252            }
253            // Make the panel's pivot columns an identity matrix before indexing its table.
254            for r in (first..rank).rev() {
255                let (upper, lower) = self.data.split_at_mut(r);
256                for row in &mut upper[first..] {
257                    if row.get(pivots[r]) {
258                        xor(&mut row.words_mut()[word..], &lower[0].words()[word..]);
259                    }
260                }
261            }
262            let mut masks = [0usize; 256];
263            for (bit, &c) in pivots[first..].iter().enumerate() {
264                let half = 1 << bit;
265                for index in 0..half {
266                    masks[index + half] = masks[index] | (1 << (c - start));
267                    let (lower, upper) = table.split_at_mut(index + half);
268                    let target = &mut upper[0].words_mut()[word..];
269                    let source = &lower[index].words()[word..];
270                    let pivot = &self[first + bit].words()[word..];
271                    for ((x, y), z) in target.iter_mut().zip(source).zip(pivot) {
272                        *x = y ^ z;
273                    }
274                }
275            }
276            let mut indices = [0usize; 256];
277            let mask = masks[(1 << (rank - first)) - 1];
278            for (i, &m) in masks[..1 << (rank - first)].iter().enumerate() {
279                indices[m] = i;
280            }
281            for i in (rank..n).chain(0..if full { first } else { 0 }) {
282                let index = indices[(self[i].words()[word] >> (start & 63)) as usize & mask];
283                if index != 0 {
284                    xor(
285                        &mut self[i].words_mut()[word..],
286                        &table[index].words()[word..],
287                    );
288                }
289            }
290            if rank == n {
291                break;
292            }
293            start += block;
294        }
295        pivots
296    }

298    fn next_column(&self, row: usize, mut start: usize, cols: usize) -> usize {
299        while start < cols {
300            let word = start / 64;
301            let bits = self.data[row..]
302                .iter()
303                .fold(0, |x, row| x | row.words()[word])
304                & (u64::MAX << (start & 63));
305            if bits != 0 {
306                return (word * 64 + bits.trailing_zeros() as usize).min(cols);
307            }
308            start = (word + 1) * 64;
309        }
310        cols
311    }

314    fn eliminate_sparse(&mut self, cols: usize, full: bool, require_full_rank: bool) -> Vec<usize> {
315        let n = self.shape.0;
316        let mut basis = vec![n; cols];
317        let mut pivots = Vec::new();
318        for i in 0..n {
319            loop {
320                let Some(c) = self[i].iter_ones().next().filter(|&c| c < cols) else {
321                    if require_full_rank {
322                        return pivots;
323                    }
324                    break;
325                };
326                if basis[c] == n {
327                    basis[c] = i;
328                    pivots.push(c);
329                    break;
330                }
331                let (upper, lower) = self.data.split_at_mut(i);
332                xor(
333                    &mut lower[0].words_mut()[c / 64..],
334                    &upper[basis[c]].words()[c / 64..],
335                );
336            }
337        }
338        pivots.sort_unstable();
339        self.data
340            .sort_by_cached_key(|row| row.iter_ones().next().filter(|&c| c < cols).unwrap_or(cols));
341        if full {
342            for (i, &c) in pivots.iter().enumerate() {
343                basis[c] = i;
344            }
345            for i in (0..pivots.len()).rev() {
346                let next = self[i].iter_ones().take_while(|&c| c < cols).nth(1);
347                if let Some(c) = next
348                    && basis[c] != n
349                {
350                    let (upper, lower) = self.data.split_at_mut(basis[c]);
351                    xor(
352                        &mut upper[i].words_mut()[c / 64..],
353                        &lower[0].words()[c / 64..],
354                    );
355                }
356            }
357        }
358        pivots
359    }

362    fn mul_impl(&self, rhs: &Self) -> Self {
363        let mut result = Self::zeros((self.shape.0, rhs.shape.1));
364        if self.shape.0 < 256
365            || self.shape.1 < 32
366            || self.data.iter().map(BitSet::count_ones).sum::<u64>()
367                <= self.shape.0 as u64 * self.shape.1 as u64 / 8
368        {
369            for (a, c) in self.data.iter().zip(&mut result.data) {
370                for j in a.iter_ones() {
371                    xor(c.words_mut(), rhs[j].words());
372                }
373            }
374            return result;
375        }
376        let mut table = vec![BitSet::new(rhs.shape.1); 8 * 256];
377        for start in (0..self.shape.1).step_by(64) {
378            for t in 0..8 {
379                let col = start + t * 8;
380                let table = &mut table[t * 256..(t + 1) * 256];
381                for bit in 0..self.shape.1.saturating_sub(col).min(8) {
382                    let row = rhs[col + bit].words();
383                    let half = 1 << bit;
384                    for index in 0..half {
385                        let (lower, upper) = table.split_at_mut(index + half);
386                        let source = lower[index].words();
387                        for ((x, y), z) in upper[0].words_mut().iter_mut().zip(source).zip(row) {
388                            *x = y ^ z;
389                        }
390                    }
391                }
392            }
393            for (a, c) in self.data.iter().zip(&mut result.data) {
394                let bits = a.words()[start / 64];
395                let p0 = table[(bits & 255) as usize].words();
396                let p1 = table[256 + (bits >> 8 & 255) as usize].words();
397                let p2 = table[512 + (bits >> 16 & 255) as usize].words();
398                let p3 = table[768 + (bits >> 24 & 255) as usize].words();
399                let p4 = table[1024 + (bits >> 32 & 255) as usize].words();
400                let p5 = table[1280 + (bits >> 40 & 255) as usize].words();
401                let p6 = table[1536 + (bits >> 48 & 255) as usize].words();
402                let p7 = table[1792 + (bits >> 56 & 255) as usize].words();
403                for ((((((((x, p0), p1), p2), p3), p4), p5), p6), p7) in c
404                    .words_mut()
405                    .iter_mut()
406                    .zip(p0)
407                    .zip(p1)
408                    .zip(p2)
409                    .zip(p3)
410                    .zip(p4)
411                    .zip(p5)
412                    .zip(p6)
413                    .zip(p7)
414                {
415                    *x ^= p0 ^ p1 ^ p2 ^ p3 ^ p4 ^ p5 ^ p6 ^ p7;
416                }
417            }
418        }
419        result
420    }
