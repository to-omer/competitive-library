116fn sieved_leaves_minus_semiprimes(
117    n: u64,
118    y: u32,
119    table: &PrimeCounts,
120    factors: &[i32],
121    split: usize,
122) -> i128 {
123    let z = n / u64::from(y);
124    // Every full segment starts at the same phase of the presieve.
125    const PAGE_SIZE: u64 = 240 * 4 * PATTERN.len() as u64;
126    let mut partial_counts = vec![0u64; split];
127    let mut prime_count = (SMALL_PRIMES.len() - 1) as u64;
128    let a = table.count(y);
129    let mut upper = table.primes.len();
130    let mut sum = 0;
131    for low in (0..=z).step_by(PAGE_SIZE as usize) {
132        let high = z.min(low + PAGE_SIZE - 1);
133        let mut segment = Segment::new(low, high);
134        let sieve_end = table.count(high.isqrt() as u32);
135        let partial_end = split.min(table.count((n / low.max(1)).isqrt() as u32));
136        for (i, base) in partial_counts[..partial_end]
137            .iter_mut()
138            .enumerate()
139            .skip(SMALL_PRIMES.len())
140        {
141            let prime = table.primes[i];
142            let p = u64::from(prime);
143            let t = n / p;
144            let count = segment.build_prefix(high.min(t / p));
145            // Sieving from p^2 retains the earlier primes as well as 1.
146            let offset = *base as i64 + SMALL_PRIMES.len() as i64 - i as i64;
147            sum += table.special_leaves(&segment, t, prime, y, factors, offset);
148            *base += u64::from(count);
149            segment.cross_off(p);
150        }
151        for &p in table.primes[..sieve_end]
152            .iter()
153            .skip(partial_end.max(SMALL_PRIMES.len()))
154        {
155            segment.cross_off(u64::from(p));
156        }
157        let count = segment.build_prefix(high);
158        while upper > a {
159            let x = n / u64::from(table.primes[upper - 1]);
160            if x > high {
161                break;
162            }
163            upper -= 1;
164            let pi = prime_count + u64::from(segment.count(x));
165            sum -= i128::from(pi) - upper as i128;
166        }
167        prime_count += u64::from(count);
168    }
169    sum
170}

177impl PrimeCounts {
178    fn new(limit: u32) -> Self {
179        let mut sieve = Segment::new(0, u64::from(limit));
180        for p in PrimeList::new(limit.isqrt())
181            .primes()
182            .skip(SMALL_PRIMES.len())
183        {
184            sieve.cross_off(u64::from(p));
185        }
186        sieve.words[0][0] &= !1;
187        let last = sieve.words.last_mut().unwrap();
188        *last = (u64::from_le_bytes(*last) & MASK[limit as usize % 240]).to_le_bytes();
189        sieve.build_prefix(u64::from(limit));
190
191        let mut primes = SMALL_PRIMES.to_vec();
192        for (i, &word) in sieve.words.iter().enumerate() {
193            let mut bits = u64::from_le_bytes(word);
194            while bits != 0 {
195                let bit = bits.trailing_zeros() as usize;
196                let prime = 240 * i + 30 * (bit / 8) + usize::from(RESIDUES[bit % 8]);
197                primes.push(prime as u32);
198                bits &= bits - 1;
199            }
200        }
201        Self { primes, sieve }
202    }

204    fn count(&self, n: u32) -> usize {
205        let small = if n >= 13 {
206            SMALL_PRIMES.len()
207        } else {
208            SMALL_PRIMES.partition_point(|&p| p <= n)
209        };
210        small + self.sieve.count(u64::from(n)) as usize
211    }

213    fn special_leaves(
214        &self,
215        segment: &Segment,
216        quotient: u64,
217        prime: u32,
218        y: u32,
219        factors: &[i32],
220        offset: i64,
221    ) -> i128 {
222        let min_m = (quotient / (segment.high + 1)).min(u64::from(y)) as u32;
223        let max_m = (quotient / segment.low.max(1)).min(u64::from(y)) as u32;
224        if u64::from(prime).pow(2) <= u64::from(y) {
225            let mut sum = 0;
226            let start = (min_m.max(y / prime) as usize).div_ceil(2);
227            let end = (max_m as usize).div_ceil(2);
228            for (i, &f) in factors[start..end].iter().enumerate() {
229                if f.unsigned_abs() > prime {
230                    let m = 2 * (start + i) + 1;
231                    let x = quotient / m as u64;
232                    let phi = offset + i64::from(segment.count(x));
233                    sum -= i128::from(f.signum()) * i128::from(phi);
234                }
235            }
236            sum
237        } else {
238            let start = self.count(min_m.max(prime));
239            let end = self.count(max_m);
240            let mut count = 0u64;
241            for &q in &self.primes[start..end] {
242                let x = quotient / u64::from(q);
243                count += u64::from(segment.count(x));
244            }
245            i128::from(count) + i128::from(offset) * (end - start) as i128
246        }
247    }
