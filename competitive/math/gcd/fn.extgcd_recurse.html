178pub fn extgcd(mut a: i64, mut b: i64) -> (i64, i64, i64) {
179    #[cfg(verus_keep_ghost)]
180    proof_decl! { let ghost original_a = a; let ghost original_b = b; }
181    let (mut u, mut v, mut x, mut y) = (1, 0, 0, 1);
182    #[cfg(verus_keep_ghost)]
183    proof! { assert(abs(0) == 0 && abs(1) == 1) by(compute_only); }
184    #[cfg_attr(verus_keep_ghost, verus_spec(
185        invariant
186            i64::MIN < a <= i64::MAX, i64::MIN < b <= i64::MAX,
187            i64::MIN < original_a, i64::MIN < original_b,
188            original_a >= 0 && original_b >= 0 ==> a >= 0 && b >= 0,
189            (a == 0 && b == 0) == (original_a == 0 && original_b == 0),
190            (original_a as int) * u + (original_b as int) * v == a,
191            (original_a as int) * x + (original_b as int) * y == b,
192            (a as int) * b * u * x <= 0, (a as int) * b * v * y <= 0,
193            (abs(a as int) as int) * abs(x as int) + (abs(b as int) as int) * abs(u as int) == abs(original_b as int),
194            (abs(a as int) as int) * abs(y as int) + (abs(b as int) as int) * abs(v as int) == abs(original_a as int),
195            abs(u as int) <= vstd::math::max(abs(original_b as int) as int, 1),
196            abs(x as int) <= vstd::math::max(abs(original_b as int) as int, 1),
197            abs(v as int) <= vstd::math::max(abs(original_a as int) as int, 1),
198            abs(y as int) <= vstd::math::max(abs(original_a as int) as int, 1),
199            forall|d: int| #![trigger original_a as int % d] d > 0 ==>
200                ((original_a as int % d == 0 && original_b as int % d == 0)
201                    == (a as int % d == 0 && b as int % d == 0)),
202        decreases abs(a as int),
203    ))]
204    while a != 0 {
205        #[cfg(verus_keep_ghost)]
206        proof! { signed_division(b as int, a as int); }
207        let k = b / a;
208        #[cfg(verus_keep_ghost)]
209        proof! {
210            signed_coefficient_step(a as int, b as int, u as int, x as int, abs(original_b as int) as int);
211            signed_coefficient_step(a as int, b as int, v as int, y as int, abs(original_a as int) as int);
212            let aa = abs(a as int) as int;
213            let bb = abs(b as int) as int;
214            let r = rust_rem(b as int, a as int);
215            lemma_fundamental_div_mod(bb, aa);
216            assert((bb / aa) * aa <= bb) by(nonlinear_arith)
217                requires bb == aa * (bb / aa) + bb % aa, bb % aa >= 0;
218            assert(-bb <= (k as int) * a <= bb) by(nonlinear_arith)
219                requires abs(k as int) == bb / aa, (bb / aa) * aa <= bb, aa == abs(a as int), aa > 0;
220            assert((original_a as int) * (x as int - (k as int) * u)
221                + (original_b as int) * (y as int - (k as int) * v) == b as int - (k as int) * a)
222                by(nonlinear_arith) requires
223                    (original_a as int) * u + (original_b as int) * v == a,
224                    (original_a as int) * x + (original_b as int) * y == b;
225            assert forall|d: int| #![trigger original_a as int % d] d > 0 implies
226                ((original_a as int % d == 0 && original_b as int % d == 0)
227                    == (a as int % d == 0 && r % d == 0)) by {
228                assert((original_a as int % d == 0 && original_b as int % d == 0)
229                    == (a as int % d == 0 && b as int % d == 0));
230                if a as int % d == 0 {
231                    vstd::arithmetic::mul::lemma_mul_basics(k as int);
232                    vstd::arithmetic::mul::lemma_mul_is_commutative(a as int, k as int);
233                    lemma_small_mod(0, d as nat);
234                    lemma_mul_mod_noop_left(a as int, k as int, d);
235                    lemma_add_mod_noop((a as int) * k, r, d);
236                    lemma_mod_twice(r, d);
237                }
238            }
239        }
240        x -= k * u;
241        y -= k * v;
242        b -= k * a;
243        std::mem::swap(&mut x, &mut u);
244        std::mem::swap(&mut y, &mut v);
245        std::mem::swap(&mut b, &mut a);
246    }
247    #[cfg(verus_keep_ghost)]
248    proof! {
249        if b != 0 {
250            let d = abs(b as int) as int;
251            lemma_small_mod(0, d as nat);
252            lemma_mod_self_0(d);
253            if b < 0 {
254                lemma_mul_mod_noop_right(-1, d, d);
255                vstd::arithmetic::mul::lemma_mul_basics(d);
256                vstd::arithmetic::mul::lemma_mul_basics(-1);
257            }
258            lemma_mod_bound(original_a as int, d);
259            lemma_mod_bound(original_b as int, d);
260            assert((original_a as int) % d == 0 && (original_b as int) % d == 0);
261        }
262        #[verifier::proof_note("For every i64 input pair excluding i64::MIN, iterative extended GCD terminates without integer overflow and returns a signed gcd with exact Bezout coefficients; its magnitude divides both inputs and the result is zero exactly for (0, 0)")]
263        assert((b == 0) == (original_a == 0 && original_b == 0)
264            && (original_a as int) * x + (original_b as int) * y == b
265            && (b != 0 ==> (original_a as int) % (abs(b as int) as int) == 0
266                && (original_b as int) % (abs(b as int) as int) == 0));
267    }
268    (b, x, y)
269}

271/// Requires nonnegative inputs.
272pub fn extgcd_binary(a: i64, b: i64) -> (i64, i64, i64) {
273    if b == 0 {
274        return (a, 1, 0);
275    }
276    if a == 0 {
277        return (b, 0, 1);
278    }
279    let k = (a | b).trailing_zeros();
280    let (mut c, mut d) = ((a >> k) as u64, (b >> k) as u64);
281    let swapped = d & 1 == 0;
282    if swapped {
283        std::mem::swap(&mut c, &mut d);
284    }
285    let (mut a, mut b) = (c, d);
286    // c*u == a and c*s == b modulo the odd d, with 0 <= u,s <= d.
287    let (mut u, mut s) = (1u64, 0u64);
288    let shift = a.trailing_zeros();
289    a >>= shift;
290    for _ in 0..shift {
291        if u & 1 == 1 {
292            u += d;
293        }
294        u /= 2;
295    }
296    while a != b {
297        if b < a {
298            std::mem::swap(&mut a, &mut b);
299            std::mem::swap(&mut u, &mut s);
300        }
301        b -= a;
302        if s < u {
303            s += d;
304        }
305        s -= u;
306        let shift = b.trailing_zeros();
307        b >>= shift;
308        for _ in 0..shift {
309            if s & 1 == 1 {
310                s += d;
311            }
312            s /= 2;
313        }
314    }
315    let x = s as i64;
316    let y = ((a as i128 - c as i128 * s as i128) / d as i128) as i64;
317    let g = (a << k) as i64;
318    if swapped { (g, y, x) } else { (g, x, y) }
319}
