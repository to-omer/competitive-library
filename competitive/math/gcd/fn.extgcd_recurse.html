219pub fn extgcd_recurse(a: i64, b: i64) -> (i64, i64, i64) {

648        assert((-difference) / (d as int) == -quotient);
649    }
650    let x = s as i64;
651    let y = ((a as i128 - c as i128 * s as i128) / d as i128) as i64;
652    let g = (a << k) as i64;
653    #[cfg(verus_keep_ghost)]
654    proof! {
655        assert((c as int) * x + (d as int) * y == a) by(nonlinear_arith)
656            requires (a as int - (c as int) * s) == (d as int) * ((a as int - (c as int) * s) / (d as int)),
657                x == s, y == (a as int - (c as int) * s) / (d as int);
658        assert((c as int) * scale * x + (d as int) * scale * y == g) by(nonlinear_arith)
659            requires (c as int) * x + (d as int) * y == a, g == (a as int) * scale;
660        lemma_mod_self_0(g as int);
661        assert((b as int) * scale == g);
662        assert(((original_a as int) % (g as int) == 0 && (original_b as int) % (g as int) == 0)
663            == ((a as int) * scale % (g as int) == 0 && (b as int) * scale % (g as int) == 0));
664        #[verifier::proof_note("For all nonnegative i64 inputs, binary extended GCD terminates without overflow and returns a nonnegative common divisor with exact Bezout coefficients; the result is zero exactly for (0, 0)")]
665        assert(g > 0 && original_a % g == 0 && original_b % g == 0
666            && (if swapped { (original_a as int) * y + (original_b as int) * x == g }
667                else { (original_a as int) * x + (original_b as int) * y == g }));
668    }
669    if swapped { (g, y, x) } else { (g, x, y) }
670}
