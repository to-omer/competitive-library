388    fn prime_band<G, const REMOVE: bool>(
389        &mut self,
390        primes: &[(u64, T)],
391        least: &[u16],
392        mul: &impl Fn(&T, &T) -> T,
393        next: &mut impl FnMut(u64, u32, &T, &T) -> T,
394    ) where
395        T: Clone,
396        G: AbelianGroup<T = T>,
397    {
398        let first = primes[0].0;
399        let rough = if REMOVE {
400            first
401        } else {
402            primes.last().unwrap().0 + 1
403        };
404        let max_k = self.cofactor_limit(first);
405        let keys = (rough as usize..=max_k)
406            .filter(|&k| least[k] == u16::MAX || u64::from(least[k]) >= rough)
407            .map(|k| {
408                (
409                    k as u64,

417            rough,
418            max_k,
419            keys,
420            offset,
421            difference: vec![G::unit(); self.data.len() - offset],
422            high: Vec::new(),
423        };

425        band.products::<G, REMOVE>(primes, 1, self.n, None, mul, next);
426        let PrimeBand {
427            difference, high, ..
428        } = band;

431            G::operate_assign(&mut sum, &change);
432            G::operate_assign(x, &sum);
433        }
434        for (x, change) in self.data.iter_mut().rev().zip(high) {
435            G::operate_assign(x, &change);
436        }

447    fn large_primes<G, const REMOVE: bool>(
448        &mut self,
449        primes: &[(u64, T)],
450        mul: &impl Fn(&T, &T) -> T,
451        next_weight: &mut impl FnMut(u64, u32, &T, &T) -> T,
452    ) where
453        T: Clone,
454        G: AbelianGroup<T = T>,
455    {
456        let n = self.n;
457        let root = self.root;
458        let len = self.data.len();
459        let large = len - root;
460        let lo = &self.data[..root];
461        let first = primes[0].0;
462        // Four primes above the fourth root exceed n, and a triple cannot contain a prime above the square root.
463        let mut pairs = vec![G::unit(); large];
464        let mut squares = Vec::with_capacity(primes.len());
465        let mut explicit_end = primes.len();
466        for (i, (p, weight)) in primes.iter().enumerate() {
467            let p = *p;
468            let m = n / p;
469            let square = next_weight(p, 2, weight, weight);
470            G::operate_assign(&mut pairs[(m / p) as usize - 1], &square);
471            squares.push(square);
472            while explicit_end > 0 && (explicit_end as u64) * primes[explicit_end - 1].0 > m {
473                explicit_end -= 1;
474            }
475            if let Some(&(bound, _)) = primes.get(explicit_end.max(i + 1)) {
476                let split = (m / root as u64) as usize;
477                let end = (m / bound) as usize;
478                G::operate_assign(&mut pairs[split - 1], &mul(weight, &lo[root - 1]));
479                G::rinv_operate_assign(&mut pairs[end - 1], &mul(weight, &lo[bound as usize - 2]));
480            }
481            for (q, other_weight) in &primes[i + 1..explicit_end.max(i + 1)] {
482                G::operate_assign(&mut pairs[(m / q) as usize - 1], &mul(weight, other_weight));
483            }
484        }
485        for i in (1..pairs.len()).rev() {
486            pairs[i - 1] = G::operate(&pairs[i - 1], &pairs[i]);
487        }
488        explicit_end = primes.len();
489        for (i, (p, weight)) in primes.iter().enumerate() {
490            let m = n / p;
491            while explicit_end > 0 && (explicit_end as u64) * primes[explicit_end - 1].0 > m {
492                explicit_end -= 1;
493            }
494            let Some(&(bound, _)) = primes.get(explicit_end.max(i + 1)) else {
495                continue;
496            };
497            let split = (m / root as u64) as usize;
498            for d in split + 1..=(m / bound) as usize {
499                let q = (m / d as u64) as usize;
500                let x = &lo[q - 1];
501                G::operate_assign(&mut pairs[d - 1], &mul(weight, x));
502            }
503        }
504        let mut triples = vec![G::unit(); (n / first / first / first) as usize];
505        for (i, (p, weight)) in primes.iter().enumerate() {
506            if *p > n / p / p {
507                break;
508            }
509            let cube = next_weight(*p, 3, weight, &squares[i]);
510            let m = n / p;
511            let mut end = primes.partition_point(|x| x.0 <= m.isqrt());
512            for d in 1..=(m / p / p) as usize {
513                let v = m / d as u64;
514                while primes[end - 1].0 * primes[end - 1].0 > v {
515                    end -= 1;
516                }
517                let prefix = i + 1 < end - (i + 1);
518                let partners = if prefix { 0..i + 1 } else { i + 1..end };
519                let mut sum = G::unit();
520                for j in partners {
521                    let (q, other_weight) = &primes[j];
522                    let other = G::rinv_operate(&lo[(v / q) as usize - 1], &lo[*q as usize - 1]);
523                    G::operate_assign(
524                        &mut sum,
525                        &G::operate(&mul(other_weight, &other), &squares[j]),
526                    );
527                }
528                if prefix {
529                    sum = G::rinv_operate(&pairs[d * *p as usize - 1], &sum);
530                }
531                let higher = G::rinv_operate(&lo[(v / p) as usize - 1], &lo[*p as usize - 1]);
532                let x = G::operate(&mul(weight, &sum), &mul(&squares[i], &higher));
533                G::operate_assign(&mut triples[d - 1], &G::operate(&x, &cube));
534            }
535        }
536        for (x, t) in pairs.iter_mut().zip(triples) {
537            G::operate_assign(x, &t);
538        }
539        let before = self.data[root - 1].clone();
540        for i in 0..large {
541            let d = if REMOVE { large - i } else { i + 1 };
542            let mut total = pairs[d - 1].clone();
543            for (p, weight) in primes {
544                if *p > (large / d) as u64 {
545                    break;
546                }
547                let other = G::rinv_operate(&self.data[len - d * *p as usize], &before);
548                G::operate_assign(&mut total, &mul(weight, &other));
549            }
550            update::<G, REMOVE>(&mut self.data[len - d], &total);
551        }
552    }

568    fn dirichlet_prefixes<R>(
569        &self,
570        operand: DirichletOperand<'_, T>,
571        f: Vec<T>,
572        g: Vec<T>,
573        mut result: Vec<T>,
574    ) -> Self
575    where
576        T: Clone,
577        R: Ring<T = T, Additive: Invertible>,
578    {
579        let root = self.root;
580        let len = self.data.len();
581        let large = len - root;
582        let other = match &operand {
583            DirichletOperand::Factor(rhs) => &rhs.data[..root],
584            DirichletOperand::Inverse(_) => &result,
585        };
586        let prefixes: Vec<_> = self.data[..root]
587            .iter()
588            .cloned()
589            .zip(other.iter().cloned())
590            .collect();
591        let weights: Vec<_> = f.into_iter().zip(g).collect();
592        let (f1, g1) = &weights[0];
593        let (pf, pg) = &prefixes[root - 1];
594        let boundary = R::add(&R::mul(f1, pg), &R::mul(pf, g1));
595        let mut running = match operand {
596            DirichletOperand::Factor(_) => R::sub(&result[root - 1], &boundary),
597            DirichletOperand::Inverse(_) => R::neg(&boundary),
598        };
599        R::add_assign(&mut running, &R::mul(f1, g1));
600        result.resize(len, R::zero());
601        let mut changes: Vec<_> = (0..large).map(|_| R::product_sum()).collect();
602        let reciprocals: Vec<_> = (2..=root).map(|i| u64::MAX / i as u64 + 1).collect();
603        for i in 2..=root {
604            let m = self.n / i as u64;
605            let split = m.isqrt() as usize;
606            if i > split {
607                break;
608            }
609            let (fi, gi) = &weights[i - 1];
610            if i > root / i {
611                changes[(m / i as u64) as usize - 1].add_products([(fi, gi)]);
612            }
613            let start = i.max(root / i) + 1;
614            let mut scatter = |j: usize, d: usize| {
615                let (fj, gj) = &weights[j - 1];
616                changes[d - 1].add_products([(fi, gj), (fj, gi)]);
617            };
618            let fast = quotient_reciprocals(m, start, split, &reciprocals);
619            for (j, &r) in (start..).zip(fast) {
620                scatter(j, quotient(m, r));
621            }
622            for j in start + fast.len()..=split {
623                scatter(j, (m / j as u64) as usize);
624            }
625            let row = (m / (split as u64 + 1)) as usize;
626            let (pf, pg) = &prefixes[split - 1];
627            // Rows d <= row include these scattered pairs in their hyperbola sum.
628            let counted = R::add(
629                &R::mul(fi, &R::sub(pg, &prefixes[i - 2].1)),
630                &R::mul(&R::sub(pf, &prefixes[i - 1].0), gi),
631            );
632            R::sub_assign(&mut result[len - row], &counted);
633        }
634        for (k, change) in changes.into_iter().enumerate().rev() {
635            let d = k + 1;
636            let q = self.n / d as u64;
637            let end = q.isqrt().min(q / (d as u64 + 1)) as usize;
638            let upper = (large / d).min(end);
639            R::add_assign(&mut running, &R::add(&change.finish(), &result[len - d]));
640            let other = match &operand {
641                DirichletOperand::Factor(rhs) => &rhs.data,
642                DirichletOperand::Inverse(_) => &result,
643            };
644            let mut sum = R::product_sum();
645            for i in 2..=upper {
646                let (fi, gi) = &weights[i - 1];
647                let j = len - d * i;
648                sum.add_products([(fi, &other[j]), (&self.data[j], gi)]);
649            }
650            let terms = |i: usize, j: usize| {
651                let (fi, gi) = &weights[i - 1];
652                let (pf, pg) = &prefixes[j - 1];
653                [(fi, pg), (pf, gi)]
654            };
655            let mut i = upper.max(1) + 1;
656            let (pairs, _) = quotient_reciprocals(q, i, end, &reciprocals).as_chunks::<2>();
657            for &[r, s] in pairs {
658                let [a, b] = terms(i, quotient(q, r));
659                let [c, e] = terms(i + 1, quotient(q, s));
660                sum.add_products([a, b, c, e]);
661                i += 2;
662            }
663            for i in i..=end {
664                sum.add_products(terms(i, (q / i as u64) as usize));
665            }
666            let (pf, pg) = &prefixes[end - 1];
667            let mut value = R::sub(&running, &R::mul(pf, pg));
668            R::add_assign(&mut value, &sum.finish());
669            R::add_assign(&mut value, &R::mul(&self.data[len - d], g1));
670            result[len - d] = match &operand {
671                DirichletOperand::Factor(rhs) => R::add(&value, &R::mul(f1, &rhs.data[len - d])),
672                DirichletOperand::Inverse(inv) => R::mul(inv, &R::neg(&value)),
673            };
674        }
675        Self {
676            n: self.n,
677            root,
678            data: result,
679        }
680    }

683struct PrimeBand<'a, T> {
684    input: &'a DirichletPrefixSums<T>,
685    rough: u64,
686    max_k: usize,
687    keys: Vec<(u64, T)>,
688    offset: usize,
689    difference: Vec<T>,

693impl<T> PrimeBand<'_, T> {
694    fn products<G, const REMOVE: bool>(
695        &mut self,
696        primes: &[(u64, T)],
697        product: u64,
698        limit: u64,
699        parent: Option<(&T, bool)>,
700        mul: &impl Fn(&T, &T) -> T,
701        next: &mut impl FnMut(u64, u32, &T, &T) -> T,
702    ) where
703        T: Clone,
704        G: AbelianGroup<T = T>,
705    {
706        for (j, (p, w)) in primes.iter().enumerate() {
707            if *p > limit {
708                break;
709            }
710            let mut t = product;
711            let mut m = limit;
712            let mut weight = w.clone();
713            let mut e = 1;
714            loop {
715                t *= p;
716                m /= p;
717                let (val, subtract) = match parent {
718                    Some((x, negative)) => (mul(x, &weight), REMOVE && !negative),
719                    None => (weight.clone(), REMOVE),
720                };
721                let prime = parent.is_none() && e == 1;
722                if subtract {
723                    self.term::<G, true>(t, m, &val, prime, mul);
724                } else {
725                    self.term::<G, false>(t, m, &val, prime, mul);
726                }
727                if *p > m {
728                    break;
729                }
730                self.products::<G, REMOVE>(
731                    &primes[j + 1..],
732                    t,
733                    m,
734                    Some((&val, subtract)),
735                    mul,
736                    next,
737                );
738                if REMOVE {
739                    break;
740                }
741                e += 1;
742                weight = next(*p, e, w, &weight);
743            }
744        }
745    }

747    fn term<G, const SUB: bool>(
748        &mut self,
749        product: u64,
750        limit: u64,
751        weight: &T,
752        prime: bool,
753        mul: &impl Fn(&T, &T) -> T,
754    ) where
755        T: Clone,
756        G: AbelianGroup<T = T>,
757    {
758        let input = self.input;
759        let root = input.root as u64;
760        let len = input.data.len();
761        if !prime {
762            let pos = if product <= root {
763                product as usize - 1
764            } else {
765                len - limit as usize
766            };
767            update::<G, SUB>(&mut self.difference[pos - self.offset], weight);
768        }
769        if limit < self.rough {
770            return;
771        }
772        // Enumerate small rough cofactors once; for the tail, read their prefix sum at each remaining quotient.
773        let small = root / product;
774        let split = if small < self.rough {
775            0
776        } else {
777            self.keys.partition_point(|x| x.0 <= small)
778        };
779        let mut left = split;
780        let max_count = usize::try_from(limit / self.rough).unwrap_or(usize::MAX);
781        let mut right = self.keys.len().min(max_count);
782        while left < right {
783            let mid = left + (right - left) / 2;
784            if (mid as u64 + 1) * self.keys[mid].0 <= limit {
785                left = mid + 1;
786            } else {
787                right = mid;
788            }
789        }
790        for (k, v) in &self.keys[..split] {
791            update::<G, SUB>(
792                &mut self.difference[(product * k) as usize - 1 - self.offset],
793                &mul(weight, v),
794            );
795        }
796        for (k, v) in &self.keys[split..left] {
797            let d = limit / k;
798            update::<G, SUB>(
799                &mut self.difference[len - d as usize - self.offset],
800                &mul(weight, v),
801            );
802        }
803        let cut = if left == self.keys.len() {
804            self.max_k as u64
805        } else if left == 0 {
806            self.rough - 1
807        } else {
808            self.keys[left - 1].0
809        };
810        let cut = cut.max(self.rough - 1).max(small);
811        let end = (limit / (cut + 1)) as usize;
812        if end == 0 {
813            return;
814        }
815        let previous = mul(weight, &input.data[cut as usize - 1]);
816        if SUB {
817            G::operate_assign(&mut self.difference[len - end - self.offset], &previous);
818        } else {
819            G::rinv_operate_assign(&mut self.difference[len - end - self.offset], &previous);
820        }
821        if self.high.len() < end {
822            self.high.resize(end, G::unit());
823        }
824        let big = (((len - input.root) as u64 / product).min(end as u64)) as usize;
825        for (d, x) in self.high[..big].iter_mut().enumerate() {
826            update::<G, SUB>(
827                x,
828                &mul(weight, &input.data[len - (d + 1) * product as usize]),
829            );
830        }
831        for d in big + 1..=end {
832            update::<G, SUB>(
833                &mut self.high[d - 1],
834                &mul(weight, &input.data[(limit / d as u64) as usize - 1]),
835            );
836        }
837    }
