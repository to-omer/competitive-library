116    fn smooth_sum<R>(
117        &self,
118        powers: &[Vec<(u64, T)>],
119        d: u64,
120        limit: u64,
121        cutoff: usize,
122        before: &T,
123    ) -> T
124    where
125        R: Ring<T = T, Additive: Invertible>,
126    {
127        let mut sum = R::one();
128        if limit > cutoff as u64 {
129            let i = if limit <= self.root as u64 {
130                limit as usize - 1
131            } else {
132                self.data.len() - d as usize
133            };
134            R::add_assign(&mut sum, &R::sub(&self.data[i], before));
135        }
136        for (j, powers_of_p) in powers.iter().enumerate() {
137            if powers_of_p[0].0 > limit {
138                break;
139            }
140            for (power, weight) in powers_of_p {
141                if *power > limit {
142                    break;
143                }
144                let rest = self.smooth_sum::<R>(
145                    &powers[j + 1..],
146                    d * power,
147                    limit / power,
148                    cutoff,
149                    before,
150                );
151                R::add_assign(&mut sum, &R::mul(weight, &rest));
152            }
153        }
154        sum
155    }

157    fn sieve<G, const REMOVE: bool>(
158        &mut self,
159        cutoff: usize,
160        mul: impl Fn(&T, &T) -> T,
161        mut next: impl FnMut(u64, u32, &T, &T) -> T,
162    ) where
163        T: Clone,
164        G: AbelianGroup<T = T>,
165    {
166        let primes: Vec<_> = PrimeList::new(self.root as u32)
167            .primes()
168            .map(|p| {
169                let i = p as usize - 1;
170                (
171                    u64::from(p),
172                    G::rinv_operate(&self.data[i], &self.data[i - 1]),
173                )
174            })
175            .collect();
176        let fourth = self.root.isqrt();
177        let small_problem = self.root <= 128;
178        let dense = if small_problem {
179            self.root
180        } else {
181            (fourth.isqrt() / 2).max(2)
182        };
183        let end = if small_problem {
184            primes.len()
185        } else {
186            primes.partition_point(|x| x.0 <= fourth as u64)
187        };
188        let (small, large) = primes.split_at(end);
189        if !REMOVE && !large.is_empty() {
190            self.large_primes::<G, REMOVE>(large, &mul, &mut next);
191        }
192        let max_k = self.cofactor_limit(dense as u64 + 1);
193        let mut least = Vec::new();
194        let large_len = self.data.len() - self.root;
195        let quotients: Vec<_> = (1..=large_len).map(|d| self.n / d as u64).collect();
196        let mut powers = Vec::new();
197        let bands = small.last().map_or(0, |x| x.0.ilog2());
198        for i in 1..=bands {
199            let shift = if REMOVE { i } else { bands + 1 - i };
200            let first = 1u64 << shift;
201            let start = small.partition_point(|x| x.0 < first || x.0 <= cutoff as u64);
202            let end = small.partition_point(|x| x.0 < first * 2);
203            if start >= end {
204                continue;
205            }
206            let band = &small[start..end];
207            if first <= dense as u64 {
208                for i in 0..band.len() {
209                    let (p, weight) = &band[if REMOVE { i } else { band.len() - 1 - i }];
210                    powers.clear();
211                    powers.push((*p, weight.clone()));
212                    if !REMOVE {
213                        while let Some((power, old)) = powers.last() {
214                            if *power > self.n / p {
215                                break;
216                            }
217                            powers
218                                .push((power * p, next(*p, powers.len() as u32 + 1, weight, old)));
219                        }
220                    }
221                    self.dense_prime::<G, REMOVE>(&powers, &quotients, &mul);
222                }
223            } else {
224                if least.is_empty() {
225                    // A composite cofactor up to sqrt(n) has a least prime factor smaller than 2^16.
226                    least.resize(max_k + 1, u16::MAX);
227                    for &(p, _) in small.iter().rev() {
228                        if p * p <= max_k as u64 {
229                            for k in (p as usize * p as usize..=max_k).step_by(p as usize) {
230                                least[k] = p as u16;
231                            }
232                        }
233                    }
234                }
235                let mut groups = band.chunks(32);
236                while let Some(batch) = if REMOVE {
237                    groups.next()
238                } else {
239                    groups.next_back()
240                } {
241                    self.prime_band::<G, REMOVE>(batch, &least, &mul, &mut next);
242                }
243            }
244        }
245        if REMOVE && !large.is_empty() {
246            self.large_primes::<G, REMOVE>(large, &mul, &mut next);
247        }
248    }

236                while let Some(batch) = if REMOVE {
237                    groups.next()
238                } else {
239                    groups.next_back()
240                } {
241                    self.prime_band::<G, REMOVE>(batch, &least, &mul, &mut next);
242                }
243            }

245        if REMOVE && !large.is_empty() {

250    fn dense_prime<G, const REMOVE: bool>(
251        &mut self,
252        powers: &[(u64, T)],
253        quotients: &[u64],
254        mul: &impl Fn(&T, &T) -> T,
255    ) where
256        T: Clone,
257        G: AbelianGroup<T = T>,
258    {
259        let (p, weight) = &powers[0];
260        let p = *p;
261        let before = self.data[p as usize - if REMOVE { 2 } else { 1 }].clone();
262        let len = self.data.len();
263        let large = quotients.len();
264        let delta = |v: &T, e: usize| {
265            let mut x = mul(&powers[e].1, &G::rinv_operate(v, &before));
266            if !REMOVE {
267                G::operate_assign(&mut x, &powers[e + 1].1);
268            }
269            x
270        };
271        let last = (self.n / (p * p)).min(large as u64) as usize;
272        let split = last.min(large / p as usize);
273        let repeated = if REMOVE {
274            0
275        } else {
276            (self.n / (p * p) / p).min(last as u64) as usize
277        };
278        for d in 1..=repeated {
279            let q = quotients[d - 1];
280            let mut x = G::unit();
281            for (e, pair) in powers.windows(2).enumerate() {
282                if pair[1].0 > q {
283                    break;
284                }
285                let power = pair[0].0;
286                let index = if power <= (large / d) as u64 {
287                    len - d * power as usize
288                } else {
289                    (q / power) as usize - 1
290                };
291                G::operate_assign(&mut x, &delta(&self.data[index], e));
292            }
293            G::operate_assign(&mut self.data[len - d], &x);
294        }
295        for d in repeated + 1..=split {
296            let x = delta(&self.data[len - d * p as usize], 0);
297            update::<G, REMOVE>(&mut self.data[len - d], &x);
298        }
299        for d in repeated.max(split) + 1..=last {
300            let q = quotients[d - 1];
301            let v = q / p;
302            let x = delta(&self.data[v as usize - 1], 0);
303            update::<G, REMOVE>(&mut self.data[len - d], &x);
304        }
305        for v in (p..=self.root as u64 / p).rev() {
306            let mut x = mul(
307                weight,
308                &G::rinv_operate(&self.data[v as usize - 1], &before),
309            );
310            if !REMOVE {
311                G::operate_assign(&mut x, &powers[1].1);
312                for e in 1..powers.len() - 1 {
313                    if powers[e].0 > v {
314                        break;
315                    }
316                    let index = (v / powers[e - 1].0) as usize - 1;
317                    G::operate_assign(&mut x, &delta(&self.data[index], e));
318                }
319            }
320            let from = (p * v) as usize - 1;
321            let end = from.saturating_add(p as usize).min(self.root);
322            for cell in &mut self.data[from..end] {
323                update::<G, REMOVE>(cell, &x);
324            }
325        }
326    }

328    fn cofactor_limit(&self, prime: u64) -> usize {
329        ((self.n / prime).isqrt() * 2).min(self.root as u64) as usize
330    }

332    fn prime_band<G, const REMOVE: bool>(
333        &mut self,
334        primes: &[(u64, T)],
335        least: &[u16],
336        mul: &impl Fn(&T, &T) -> T,
337        next: &mut impl FnMut(u64, u32, &T, &T) -> T,
338    ) where
339        T: Clone,
340        G: AbelianGroup<T = T>,
341    {
342        let first = primes[0].0;
343        let rough = if REMOVE {
344            first
345        } else {
346            primes.last().unwrap().0 + 1
347        };
348        let max_k = self.cofactor_limit(first);
349        let keys = (rough as usize..=max_k)
350            .filter(|&k| least[k] == u16::MAX || u64::from(least[k]) >= rough)
351            .map(|k| {
352                (
353                    k as u64,
354                    G::rinv_operate(&self.data[k - 1], &self.data[k - 2]),
355                )
356            })
357            .collect();
358        let offset = self.ceiling_index(first * first);
359        let mut band = PrimeBand {
360            input: self,
361            rough,
362            max_k,
363            keys,
364            offset,
365            difference: vec![G::unit(); self.data.len() - offset],
366            high: Vec::new(),
367        };
368        // Removing a block uses its squarefree Möbius weights. Inserting it uses all its prime powers.
369        band.products::<G, REMOVE>(primes, 1, self.n, None, mul, next);
370        let PrimeBand {
371            difference, high, ..
372        } = band;
373        let mut sum = G::unit();
374        for (x, change) in self.data[offset..].iter_mut().zip(difference) {
375            G::operate_assign(&mut sum, &change);
376            G::operate_assign(x, &sum);
377        }
378        for (x, change) in self.data.iter_mut().rev().zip(high) {
379            G::operate_assign(x, &change);
380        }
381    }
