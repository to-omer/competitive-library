250    fn dense_prime<G, const REMOVE: bool>(
251        &mut self,
252        powers: &[(u64, T)],
253        quotients: &[u64],
254        mul: &impl Fn(&T, &T) -> T,
255    ) where
256        T: Clone,
257        G: AbelianGroup<T = T>,
258    {
259        let (p, weight) = &powers[0];
260        let p = *p;
261        let before = self.data[p as usize - if REMOVE { 2 } else { 1 }].clone();
262        let len = self.data.len();
263        let large = quotients.len();
264        let delta = |v: &T, e: usize| {
265            let mut x = mul(&powers[e].1, &G::rinv_operate(v, &before));
266            if !REMOVE {
267                G::operate_assign(&mut x, &powers[e + 1].1);
268            }
269            x
270        };
271        let last = (self.n / (p * p)).min(large as u64) as usize;
272        let split = last.min(large / p as usize);
273        let repeated = if REMOVE {
274            0
275        } else {
276            (self.n / (p * p) / p).min(last as u64) as usize
277        };
278        for d in 1..=repeated {
279            let q = quotients[d - 1];
280            let mut x = G::unit();
281            for (e, pair) in powers.windows(2).enumerate() {
282                if pair[1].0 > q {
283                    break;
284                }
285                let power = pair[0].0;
286                let index = if power <= (large / d) as u64 {
287                    len - d * power as usize
288                } else {
289                    (q / power) as usize - 1
290                };
291                G::operate_assign(&mut x, &delta(&self.data[index], e));
292            }
293            G::operate_assign(&mut self.data[len - d], &x);
294        }
295        for d in repeated + 1..=split {
296            let x = delta(&self.data[len - d * p as usize], 0);
297            update::<G, REMOVE>(&mut self.data[len - d], &x);
298        }
299        for d in repeated.max(split) + 1..=last {
300            let q = quotients[d - 1];
301            let v = q / p;
302            let x = delta(&self.data[v as usize - 1], 0);
303            update::<G, REMOVE>(&mut self.data[len - d], &x);
304        }
305        for v in (p..=self.root as u64 / p).rev() {
306            let mut x = mul(
307                weight,
308                &G::rinv_operate(&self.data[v as usize - 1], &before),
309            );
310            if !REMOVE {
311                G::operate_assign(&mut x, &powers[1].1);
312                for e in 1..powers.len() - 1 {
313                    if powers[e].0 > v {
314                        break;
315                    }
316                    let index = (v / powers[e - 1].0) as usize - 1;
317                    G::operate_assign(&mut x, &delta(&self.data[index], e));
318                }
319            }
320            let from = (p * v) as usize - 1;
321            let end = from.saturating_add(p as usize).min(self.root);
322            for cell in &mut self.data[from..end] {
323                update::<G, REMOVE>(cell, &x);
324            }
325        }
326    }

332    fn prime_band<G, const REMOVE: bool>(
333        &mut self,
334        primes: &[(u64, T)],
335        least: &[u16],
336        mul: &impl Fn(&T, &T) -> T,
337        next: &mut impl FnMut(u64, u32, &T, &T) -> T,
338    ) where
339        T: Clone,
340        G: AbelianGroup<T = T>,
341    {
342        let first = primes[0].0;
343        let rough = if REMOVE {
344            first
345        } else {
346            primes.last().unwrap().0 + 1
347        };
348        let max_k = self.cofactor_limit(first);
349        let keys = (rough as usize..=max_k)
350            .filter(|&k| least[k] == u16::MAX || u64::from(least[k]) >= rough)
351            .map(|k| {
352                (
353                    k as u64,
354                    G::rinv_operate(&self.data[k - 1], &self.data[k - 2]),
355                )
356            })
357            .collect();
358        let offset = self.ceiling_index(first * first);
359        let mut band = PrimeBand {
360            input: self,
361            rough,
362            max_k,
363            keys,
364            offset,
365            difference: vec![G::unit(); self.data.len() - offset],
366            high: Vec::new(),
367        };
368        // Removing a block uses its squarefree Möbius weights. Inserting it uses all its prime powers.
369        band.products::<G, REMOVE>(primes, 1, self.n, None, mul, next);
370        let PrimeBand {
371            difference, high, ..
372        } = band;
373        let mut sum = G::unit();
374        for (x, change) in self.data[offset..].iter_mut().zip(difference) {
375            G::operate_assign(&mut sum, &change);
376            G::operate_assign(x, &sum);
377        }
378        for (x, change) in self.data.iter_mut().rev().zip(high) {
379            G::operate_assign(x, &change);
380        }
381    }

391    fn large_primes<G, const REMOVE: bool>(
392        &mut self,
393        primes: &[(u64, T)],
394        mul: &impl Fn(&T, &T) -> T,
395        next_weight: &mut impl FnMut(u64, u32, &T, &T) -> T,
396    ) where
397        T: Clone,
398        G: AbelianGroup<T = T>,
399    {
400        let n = self.n;
401        let root = self.root;
402        let len = self.data.len();
403        let large = len - root;
404        let lo = &self.data[..root];
405        let first = primes[0].0;
406        // Four primes above the fourth root exceed n, and a triple cannot contain a prime above the square root.
407        let mut pairs = vec![G::unit(); large];
408        let mut squares = Vec::with_capacity(primes.len());
409        let mut explicit_end = primes.len();
410        for (i, (p, weight)) in primes.iter().enumerate() {
411            let p = *p;
412            let m = n / p;
413            let square = next_weight(p, 2, weight, weight);
414            G::operate_assign(&mut pairs[(m / p) as usize - 1], &square);
415            squares.push(square);
416            while explicit_end > 0 && (explicit_end as u64) * primes[explicit_end - 1].0 > m {
417                explicit_end -= 1;
418            }
419            if let Some(&(bound, _)) = primes.get(explicit_end.max(i + 1)) {
420                let split = (m / root as u64) as usize;
421                let end = (m / bound) as usize;
422                G::operate_assign(&mut pairs[split - 1], &mul(weight, &lo[root - 1]));
423                G::rinv_operate_assign(&mut pairs[end - 1], &mul(weight, &lo[bound as usize - 2]));
424            }
425            for (q, other_weight) in &primes[i + 1..explicit_end.max(i + 1)] {
426                G::operate_assign(&mut pairs[(m / q) as usize - 1], &mul(weight, other_weight));
427            }
428        }
429        for i in (1..pairs.len()).rev() {
430            pairs[i - 1] = G::operate(&pairs[i - 1], &pairs[i]);
431        }
432        explicit_end = primes.len();
433        for (i, (p, weight)) in primes.iter().enumerate() {
434            let m = n / p;
435            while explicit_end > 0 && (explicit_end as u64) * primes[explicit_end - 1].0 > m {
436                explicit_end -= 1;
437            }
438            let Some(&(bound, _)) = primes.get(explicit_end.max(i + 1)) else {
439                continue;
440            };
441            let split = (m / root as u64) as usize;
442            for d in split + 1..=(m / bound) as usize {
443                let q = (m / d as u64) as usize;
444                let x = &lo[q - 1];
445                G::operate_assign(&mut pairs[d - 1], &mul(weight, x));
446            }
447        }
448        let mut triples = vec![G::unit(); (n / first / first / first) as usize];
449        for (i, (p, weight)) in primes.iter().enumerate() {
450            if *p > n / p / p {
451                break;
452            }
453            let cube = next_weight(*p, 3, weight, &squares[i]);
454            let m = n / p;
455            let mut end = primes.partition_point(|x| x.0 <= m.isqrt());
456            for d in 1..=(m / p / p) as usize {
457                let v = m / d as u64;
458                while primes[end - 1].0 * primes[end - 1].0 > v {
459                    end -= 1;
460                }
461                let prefix = i + 1 < end - (i + 1);
462                let partners = if prefix { 0..i + 1 } else { i + 1..end };
463                let mut sum = G::unit();
464                for j in partners {
465                    let (q, other_weight) = &primes[j];
466                    let other = G::rinv_operate(&lo[(v / q) as usize - 1], &lo[*q as usize - 1]);
467                    G::operate_assign(
468                        &mut sum,
469                        &G::operate(&mul(other_weight, &other), &squares[j]),
470                    );
471                }
472                if prefix {
473                    sum = G::rinv_operate(&pairs[d * *p as usize - 1], &sum);
474                }
475                let higher = G::rinv_operate(&lo[(v / p) as usize - 1], &lo[*p as usize - 1]);
476                let x = G::operate(&mul(weight, &sum), &mul(&squares[i], &higher));
477                G::operate_assign(&mut triples[d - 1], &G::operate(&x, &cube));
478            }
479        }
480        for (x, t) in pairs.iter_mut().zip(triples) {
481            G::operate_assign(x, &t);
482        }
483        let before = self.data[root - 1].clone();
484        for i in 0..large {
485            let d = if REMOVE { large - i } else { i + 1 };
486            let mut total = pairs[d - 1].clone();
487            for (p, weight) in primes {
488                if *p > (large / d) as u64 {
489                    break;
490                }
491                let other = G::rinv_operate(&self.data[len - d * *p as usize], &before);
492                G::operate_assign(&mut total, &mul(weight, &other));
493            }
494            update::<G, REMOVE>(&mut self.data[len - d], &total);
495        }
496    }

510    fn products<G, const REMOVE: bool>(
511        &mut self,
512        primes: &[(u64, T)],
513        product: u64,
514        limit: u64,
515        parent: Option<(&T, bool)>,
516        mul: &impl Fn(&T, &T) -> T,
517        next: &mut impl FnMut(u64, u32, &T, &T) -> T,
518    ) where
519        T: Clone,
520        G: AbelianGroup<T = T>,
521    {
522        for (j, (p, w)) in primes.iter().enumerate() {
523            if *p > limit {
524                break;
525            }
526            let mut t = product;
527            let mut m = limit;
528            let mut weight = w.clone();
529            let mut e = 1;
530            loop {
531                t *= p;
532                m /= p;
533                let (val, subtract) = match parent {
534                    Some((x, negative)) => (mul(x, &weight), REMOVE && !negative),
535                    None => (weight.clone(), REMOVE),
536                };
537                let prime = parent.is_none() && e == 1;
538                if subtract {
539                    self.term::<G, true>(t, m, &val, prime, mul);
540                } else {
541                    self.term::<G, false>(t, m, &val, prime, mul);
542                }
543                if *p > m {
544                    break;
545                }
546                self.products::<G, REMOVE>(
547                    &primes[j + 1..],
548                    t,
549                    m,
550                    Some((&val, subtract)),
551                    mul,
552                    next,
553                );
554                if REMOVE {
555                    break;
556                }
557                e += 1;
558                weight = next(*p, e, w, &weight);
559            }
560        }
561    }

563    fn term<G, const SUB: bool>(
564        &mut self,
565        product: u64,
566        limit: u64,
567        weight: &T,
568        prime: bool,
569        mul: &impl Fn(&T, &T) -> T,
570    ) where
571        T: Clone,
572        G: AbelianGroup<T = T>,
573    {
574        let input = self.input;
575        let root = input.root as u64;
576        let len = input.data.len();
577        if !prime {
578            let pos = if product <= root {
579                product as usize - 1
580            } else {
581                len - limit as usize
582            };
583            update::<G, SUB>(&mut self.difference[pos - self.offset], weight);
584        }
585        if limit < self.rough {
586            return;
587        }
588        // Enumerate small rough cofactors once; for the tail, read their prefix sum at each remaining quotient.
589        let small = root / product;
590        let split = if small < self.rough {
591            0
592        } else {
593            self.keys.partition_point(|x| x.0 <= small)
594        };
595        let mut left = split;
596        let max_count = usize::try_from(limit / self.rough).unwrap_or(usize::MAX);
597        let mut right = self.keys.len().min(max_count);
598        while left < right {
599            let mid = left + (right - left) / 2;
600            if (mid as u64 + 1) * self.keys[mid].0 <= limit {
601                left = mid + 1;
602            } else {
603                right = mid;
604            }
605        }
606        for (k, v) in &self.keys[..split] {
607            update::<G, SUB>(
608                &mut self.difference[(product * k) as usize - 1 - self.offset],
609                &mul(weight, v),
610            );
611        }
612        for (k, v) in &self.keys[split..left] {
613            let d = limit / k;
614            update::<G, SUB>(
615                &mut self.difference[len - d as usize - self.offset],
616                &mul(weight, v),
617            );
618        }
619        let cut = if left == self.keys.len() {
620            self.max_k as u64
621        } else if left == 0 {
622            self.rough - 1
623        } else {
624            self.keys[left - 1].0
625        };
626        let cut = cut.max(self.rough - 1).max(small);
627        let end = (limit / (cut + 1)) as usize;
628        if end == 0 {
629            return;
630        }
631        let previous = mul(weight, &input.data[cut as usize - 1]);
632        if SUB {
633            G::operate_assign(&mut self.difference[len - end - self.offset], &previous);
634        } else {
635            G::rinv_operate_assign(&mut self.difference[len - end - self.offset], &previous);
636        }
637        if self.high.len() < end {
638            self.high.resize(end, G::unit());
639        }
640        let big = (((len - input.root) as u64 / product).min(end as u64)) as usize;
641        for (d, x) in self.high[..big].iter_mut().enumerate() {
642            update::<G, SUB>(
643                x,
644                &mul(weight, &input.data[len - (d + 1) * product as usize]),
645            );
646        }
647        for d in big + 1..=end {
648            update::<G, SUB>(
649                &mut self.high[d - 1],
650                &mul(weight, &input.data[(limit / d as u64) as usize - 1]),
651            );
652        }
653    }
