827    pub fn mul_strassen(&self, rhs: &Matrix<R>) -> Matrix<R> {
828        assert_eq!(self.shape.1, rhs.shape.0);
829        if let Some(data) = R::try_matrix_product(&self.data, &rhs.data) {
830            return Matrix::from_vec(data);
831        }
832        let (n, m) = self.shape;
833        let p = rhs.shape.1;
834        if n == 0 || m == 0 || p == 0 {
835            return Matrix::zeros((n, p));
836        }
837        let split = n.min(m).min(p).div_ceil(128).next_power_of_two();
838        if split <= 2 {
839            return self * rhs;
840        }
841        let rows = n.div_ceil(split) * split;
842        let inner = m.div_ceil(split) * split;
843        let cols = p.div_ceil(split) * split;
844        let mut a = vec![R::zero(); rows * inner];
845        for (a, data) in a.chunks_exact_mut(inner).zip(&self.data) {
846            a[..m].clone_from_slice(data);
847        }
848        let mut b = vec![R::zero(); inner * cols];
849        for (b, data) in b.chunks_exact_mut(cols).zip(&rhs.data) {
850            b[..p].clone_from_slice(data);
851        }
852        let mut c = vec![R::zero(); rows * cols];
853        strassen_rec::<R>(&a, &b, &mut c, (rows, inner, cols), inner, cols);
854        let mut res = Matrix::zeros((n, p));
855        for (data, c) in res.data.iter_mut().zip(c.chunks_exact(cols)) {
856            data.clone_from_slice(&c[..p]);
857        }
858        res
859    }
