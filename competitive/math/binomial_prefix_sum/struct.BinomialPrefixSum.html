227    where
228        F: FnMut(usize, MInt<M>),

230        let mut binom = BinomialPrefixSum::<M>::with_capacity(self.query.len() * K);
231        let mut derived = Vec::with_capacity(self.query.len() * K);
232        let mut stirling = [[MInt::<M>::zero(); K]; K];
233        if K > 0 {
234            stirling[0][0] = MInt::one();
235        }
236        for n in 1..K {
237            for r in 1..=n {
238                stirling[n][r] = stirling[n - 1][r - 1] + MInt::from(r) * stirling[n - 1][r];
239            }
240        }
241        let mut coef_cache = HashMap::with_capacity(self.query.len());
242        for (i, (n, m, coef)) in self.query.iter().enumerate() {
243            let (n, m) = (*n, *m);
244            let coef = *coef_cache.entry(*coef).or_insert_with(|| {
245                let mut falling = [MInt::<M>::zero(); K];
246                for (k, &c) in coef.iter().enumerate() {
247                    if c.is_zero() {
248                        continue;
249                    }
250                    for r in 0..=k {
251                        falling[r] += c * stirling[k][r];
252                    }
253                }
254                falling
255            });
256            let mut falling = MInt::one();
257            for (r, &coef) in coef.iter().enumerate() {
258                if r > n || r > m {
259                    break;
260                }
261                if !coef.is_zero() {
262                    binom.push(n - r, m - r);
263                    derived.push((i, coef * falling));
264                }
265                if r + 1 < K {
266                    falling *= MInt::from(n - r);
267                }
268            }
269        }
270        binom.for_each(|i, x| {
271            let (q, coef) = derived[i];
272            f(q, coef * x);
273        });
274    }

119    fn divide<F>(
120        points: &mut [(usize, usize, usize)],
121        (m, n): (usize, usize),
122        mut sum: MInt<M>,
123        fact: &MemorizedFactorial<M>,
124        power2: &[MInt<M>],
125        f: &mut F,
126    ) where
127        F: FnMut(usize, MInt<M>),
128    {
129        let (mut min_m, mut max_m, mut min_n, mut max_n) = (usize::MAX, 0, usize::MAX, 0);
130        for &(m, n, _) in points.iter() {
131            min_m = min_m.min(m);
132            max_m = max_m.max(m);
133            min_n = min_n.min(n);
134            max_n = max_n.max(n);
135        }
136        // S(n + 1, m) = 2 S(n, m) - C(n, m), with 1 / m! factored out.
137        let mut tail = MInt::zero();
138        for i in n..min_n {
139            tail = tail + tail + fact.fact[i] * fact.inv_fact[i - m];
140        }
141        sum = sum * power2[min_n - n] - tail * fact.inv_fact[m];
142        // S(n, m + 1) = S(n, m) + C(n, m + 1), with n! factored out.
143        let mut tail = MInt::zero();
144        for i in m + 1..=min_m {
145            tail += fact.inv_fact[i] * fact.inv_fact[min_n - i];
146        }
147        sum += tail * fact.fact[min_n];
148        if min_m == max_m && min_n == max_n {
149            for &(_, _, i) in points.iter() {
150                f(i, sum);
151            }
152            return;
153        }
154        let split_n = max_m - min_m < max_n - min_n;
155        let middle = if split_n {
156            min_n + (max_n - min_n) / 2
157        } else {
158            min_m + (max_m - min_m) / 2
159        };
160        let mut left = 0;
161        let mut right = points.len();
162        while left < right {
163            let (m, n, _) = points[left];
164            if (if split_n { n } else { m }) <= middle {
165                left += 1;
166            } else {
167                right -= 1;
168                points.swap(left, right);
169            }
170        }
171        let (left, right) = points.split_at_mut(left);
172        Self::divide(left, (min_m, min_n), sum, fact, power2, f);
173        Self::divide(right, (min_m, min_n), sum, fact, power2, f);
174    }

60    pub fn for_each<F>(self, mut f: F)
61    where
62        F: FnMut(usize, MInt<M>),
63    {
64        let query = self.query;
65        if query.is_empty() {
66            return;
67        }
68        let max_n = query.iter().map(|&(_, n)| n).max().unwrap_or(0);
69        let modulus = M::mod_into();
70        debug_assert!(modulus > 2 && modulus % 2 == 1);
71        debug_assert!(max_n < modulus);
72        debug_assert!(query.iter().all(|&(m, n)| m <= n));
73
74        let mut power2 = vec![MInt::<M>::one(); max_n + 1];
75        for n in 1..=max_n {
76            power2[n] = power2[n - 1] + power2[n - 1];
77        }
78        let mut points = Vec::with_capacity(query.len());
79        for (i, &(m, n)) in query.iter().enumerate() {
80            if m == n {
81                f(i, power2[n]);
82                continue;
83            }
84            let k = m.min(n - m - 1);
85            let value = if k == 0 {
86                MInt::one()
87            } else if k * 2 + 1 == n {
88                power2[n - 1]
89            } else {
90                points.push((k, n, i));
91                continue;
92            };
93            f(i, if m == k { value } else { power2[n] - value });
94        }
95        if points.is_empty() {
96            return;
97        }
98        let max_n = points.iter().map(|&(_, n, _)| n).max().unwrap();
99        let fact = MemorizedFactorial::<M>::new(max_n);
100        Self::divide(
101            &mut points,
102            (0, 0),
103            MInt::one(),
104            &fact,
105            &power2,
106            &mut |i, value| {
107                let (m, n) = query[i];
108                f(i, if m < n - m { value } else { power2[n] - value });
109            },
110        );
111    }
