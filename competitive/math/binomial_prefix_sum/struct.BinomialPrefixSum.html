288        M: MIntDotProduct,

291        let mut binom = BinomialPrefixSum::<M>::with_capacity(self.query.len() * K);
292        let mut derived = Vec::with_capacity(self.query.len() * K);
293        let mut stirling = [[MInt::<M>::zero(); K]; K];
294        if K > 0 {
295            stirling[0][0] = MInt::one();
296        }
297        for n in 1..K {
298            for r in 1..=n {

300            }
301        }
302        let mut coef_cache = HashMap::with_capacity(self.query.len());
303        for (i, (n, m, coef)) in self.query.iter().enumerate() {
304            let (n, m) = (*n, *m);
305            let coef = *coef_cache.entry(*coef).or_insert_with(|| {
306                let mut falling = [MInt::<M>::zero(); K];
307                for (k, &c) in coef.iter().enumerate() {
308                    if c.is_zero() {
309                        continue;
310                    }
311                    for r in 0..=k {
312                        falling[r] += c * stirling[k][r];
313                    }
314                }
315                falling
316            });
317            let mut falling = MInt::one();
318            for (r, &coef) in coef.iter().enumerate() {
319                if r > n || r > m {
320                    break;
321                }
322                if !coef.is_zero() {
323                    binom.push(n - r, m - r);
324                    derived.push((i, coef * falling));
325                }
326                if r + 1 < K {
327                    falling *= MInt::from(n - r);
328                }
329            }
330        }
331        binom.for_each(|i, x| {
332            let (q, coef) = derived[i];

334        });

130    fn evaluate<F>(
131        points: &[(usize, usize, usize)],
132        weighted_fact: &[MInt<M>],
133        inv_fact: &[MInt<M>],
134        power2: &[MInt<M>],
135        f: &mut F,
136    ) where
137        M: MIntDotProduct,
138        F: FnMut(usize, MInt<M>),
139    {
140        let (_, n, _) = points[0];
141        if points.iter().all(|&(_, l, _)| l == n) {
142            let min_m = points.iter().map(|&(m, _, _)| m).min().unwrap();
143            let max_m = points.iter().map(|&(m, _, _)| m).max().unwrap();
144            let factor = (weighted_fact[n] + weighted_fact[n]) * power2[n];
145            let mut sum = M::dot_product_rev(&inv_fact[..=min_m], &inv_fact[n - min_m..=n]);
146            let mut prefix = Vec::with_capacity(max_m - min_m + 1);
147            prefix.push(sum * factor);
148            for m in min_m + 1..=max_m {
149                sum += inv_fact[m] * inv_fact[n - m];
150                prefix.push(sum * factor);
151            }
152            for &(m, _, i) in points.iter() {
153                f(i, prefix[m - min_m]);
154            }
155            return;
156        }
157        let max_n = weighted_fact.len() - 1;
158        let block = ((max_n as f64 / (points.len() as f64).sqrt()).round() as usize).max(1);
159        let rows = max_n / block;
160        // Store S(row * block, column * block) / 2^(row * block).
161        let mut grid = vec![MInt::<M>::zero(); (rows + 1) * (rows + 2) / 2];
162        for row in 0..=rows {
163            let n = row * block;
164            let factor = weighted_fact[n] + weighted_fact[n];
165            let base = row * (row + 1) / 2;
166            let mut sum = MInt::<M>::zero();
167            for column in 0..=row / 2 {
168                let k = column * block;
169                if column > 0 {
170                    sum += M::dot_product_rev(
171                        &inv_fact[k - block..k],
172                        &inv_fact[n - k + 1..=n - k + block],
173                    );
174                }
175                let value = sum * factor;
176                grid[base + row - column] = MInt::one() - value;
177                grid[base + column] = value + inv_fact[k] * inv_fact[n - k] * factor;
178            }
179        }
180        for &(m, n, i) in points.iter() {
181            let row = ((n + block / 2) / block).min(rows);
182            let column = (m + block / 2) / block;
183            let start = row * block;
184            let anchor = column * block;
185            let distance = n.abs_diff(start) + m.abs_diff(anchor);
186            let center = (n - 1) / 2;
187            if m.min(center - m) <= distance {
188                let factor = weighted_fact[n] + weighted_fact[n];
189                let sum = if m <= center - m {
190                    factor
191                        * (inv_fact[n] + M::dot_product_rev(&inv_fact[1..=m], &inv_fact[n - m..n]))
192                } else {
193                    let half = weighted_fact[0];
194                    let mid = n / 2;
195                    let center_value = if n % 2 == 0 {
196                        half - weighted_fact[n] * inv_fact[mid] * inv_fact[mid]
197                    } else {
198                        half
199                    };
200                    center_value
201                        - factor
202                            * M::dot_product_rev(
203                                &inv_fact[m + 1..=center],
204                                &inv_fact[n - center..n - m],
205                            )
206                };
207                f(i, sum * power2[n]);
208                continue;
209            }
210            let mut sum = grid[row * (row + 1) / 2 + column];
211            if start < n {
212                sum -= inv_fact[anchor]
213                    * M::dot_product(
214                        &weighted_fact[start..n],
215                        &inv_fact[start - anchor..n - anchor],
216                    );
217            } else if n < start {
218                sum += inv_fact[anchor]
219                    * M::dot_product(
220                        &weighted_fact[n..start],
221                        &inv_fact[n - anchor..start - anchor],
222                    );
223            }
224            let factor = weighted_fact[n] + weighted_fact[n];
225            if anchor < m {
226                sum += factor
227                    * M::dot_product_rev(&inv_fact[anchor + 1..=m], &inv_fact[n - m..n - anchor]);
228            } else if m < anchor {
229                sum -= factor
230                    * M::dot_product_rev(&inv_fact[m + 1..=anchor], &inv_fact[n - anchor..n - m]);
231            }
232            f(i, sum * power2[n]);
233        }
234    }

60    pub fn for_each<F>(self, mut f: F)
61    where
62        M: MIntDotProduct,
63        F: FnMut(usize, MInt<M>),
64    {
65        let query = self.query;
66        if query.is_empty() {
67            return;
68        }
69        let max_n = query.iter().map(|&(_, n)| n).max().unwrap_or(0);
70        let modulus = M::mod_into();
71        debug_assert!(modulus > 2 && modulus % 2 == 1);
72        debug_assert!(max_n < modulus);
73        debug_assert!(query.iter().all(|&(m, n)| m <= n));
74
75        let mut power2 = vec![MInt::<M>::one(); max_n + 1];
76        for n in 1..=max_n {
77            power2[n] = power2[n - 1] + power2[n - 1];
78        }
79        let mut points = Vec::with_capacity(query.len());
80        for (i, &(m, n)) in query.iter().enumerate() {
81            if m == n {
82                f(i, power2[n]);
83                continue;
84            }
85            let k = m.min(n - m - 1);
86            let value = if k == 0 {
87                MInt::one()
88            } else if k * 2 + 1 == n {
89                power2[n - 1]
90            } else {
91                points.push((k, n, i));
92                continue;
93            };
94            f(i, if m == k { value } else { power2[n] - value });
95        }
96        if points.is_empty() {
97            return;
98        }
99        let max_n = points.iter().map(|&(_, n, _)| n).max().unwrap();
100        let MemorizedFactorial {
101            fact: mut weighted_fact,
102            inv_fact,
103        } = MemorizedFactorial::<M>::new(max_n);
104        let mut scale = (power2[max_n] + power2[max_n]).inv();
105        for value in weighted_fact.iter_mut().rev() {
106            *value *= scale;
107            scale += scale;
108        }
109        Self::evaluate(
110            &points,
111            &weighted_fact,
112            &inv_fact,
113            &power2,
114            &mut |i, value| {
115                let (m, n) = query[i];
116                f(i, if m < n - m { value } else { power2[n] - value });
117            },
118        );
119    }
