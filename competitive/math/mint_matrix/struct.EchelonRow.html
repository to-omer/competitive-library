148fn generate_frobenius_block<M>(
149    a: &Matrix<AddMulOperation<MInt<M>>>,
150    mut v: Vec<MInt<M>>,
151    rows: &mut Vec<EchelonRow<M>>,
152    t: &mut Vec<Vec<MInt<M>>>,
153) -> Polynomial<M>
154where
155    M: MIntBase,
156{
157    let n = a.shape.0;
158    loop {
159        let mut row = vec![MInt::zero(); 2 * n + 1];
160        let (x, c) = row.split_at_mut(n);
161        x.copy_from_slice(&v);
162        c[rows.len()] = MInt::one();
163        for r in rows.iter() {
164            r.reduce(&mut row);
165        }
166        if let Some(pivot) = row[..n].iter().position(|x| !x.is_zero()) {
167            t.push(v);
168            let u = t.last().unwrap();
169            v = a.data.iter().map(|row| MInt::dot_product(u, row)).collect();
170            rows.push(EchelonRow {
171                pivot,
172                inv: row[pivot].inv(),
173                row,
174            });
175        } else {
176            let mut p = row.split_off(n);
177            while p.last().is_some_and(|x| x.is_zero()) {
178                p.pop();
179            }
180            return Polynomial(p);
181        }
182    }
183}

189    fn exact_div(mut self, rhs: &Self) -> Option<Self> {
190        let mut q = vec![MInt::zero(); self.0.len() - rhs.0.len() + 1];
191        let inv = rhs.0.last().unwrap().inv();
192        for i in (0..q.len()).rev() {
193            q[i] = self.0[i + rhs.0.len() - 1] * inv;
194            for (x, &y) in self.0[i..].iter_mut().zip(&rhs.0) {
195                *x -= q[i] * y;
196            }
197        }
198        self.0.iter().all(|x| x.is_zero()).then_some(Self(q))
199    }

201    fn mul_mod(&self, rhs: &Self, p: &Self) -> Self {
202        let d = p.0.len() - 1;
203        let mut c = vec![MInt::zero(); 2 * d - 1];
204        for (i, &x) in self.0.iter().enumerate() {
205            for (z, &y) in c[i..].iter_mut().zip(&rhs.0) {
206                *z += x * y;
207            }
208        }
209        for i in (d..c.len()).rev() {
210            let x = c[i];
211            for (y, &m) in c[i - d..].iter_mut().zip(&p.0) {
212                *y -= x * m;
213            }
214        }
215        c.truncate(d);
216        Self(c)
217    }

219    fn x_pow_mod(&self, mut k: usize) -> Self {
220        let d = self.0.len() - 1;
221        let mut r = Self(vec![MInt::zero(); d]);
222        r.0[0] = MInt::one();
223        let mut x = Self(vec![MInt::zero(); d]);
224        if d == 1 {

226        } else {

231                r = r.mul_mod(&x, self);
232            }
233            k >>= 1;
234            if k > 0 {

236            }
237        }

242fn frobenius_decomposition<M>(
243    a: &Matrix<AddMulOperation<MInt<M>>>,
244    rng: &mut Xorshift,
245) -> Option<FrobeniusDecomposition<M>>
246where
247    M: MIntBase + MIntConvert<u64>,
248{
249    let n = a.shape.0;
250    let mut rows = Vec::with_capacity(n);
251    let mut t = Vec::with_capacity(n);
252    let mut blocks = Vec::new();
253    while rows.len() < n {
254        let s = rows.len();
255        let v = (0..n).map(|_| MInt::from(rng.rand64())).collect();
256        let mut c = generate_frobenius_block(a, v, &mut rows, &mut t);
257        if rows.len() == s {
258            continue;
259        }
260        if c.0[..s].iter().any(|x| !x.is_zero()) {
261            let p = Polynomial(c.0[s..].to_vec());
262            let q = c.exact_div(&p)?;
263            let mut v = t[s].clone();
264            for (&x, u) in q.0.iter().zip(&t[..s]) {
265                for (v, &u) in v.iter_mut().zip(u) {
266                    *v += x * u;
267                }
268            }
269            rows.truncate(s);
270            t.truncate(s);
271            c = generate_frobenius_block(a, v, &mut rows, &mut t);
272        }
273        blocks.push(Polynomial(c.0[s..].to_vec()));
274    }
275
276    for i in 0..n {
277        let (left, right) = rows.split_at_mut(i + 1);
278        for row in right {
279            row.reduce(&mut left[i].row);
280        }
281    }
282    let mut t_inv = vec![vec![MInt::zero(); n]; n];
283    for row in rows {
284        let (_, c) = row.row.split_at(n);
285        for (x, &y) in t_inv[row.pivot].iter_mut().zip(c) {
286            *x = row.inv * y;
287        }
288    }
289    Some(FrobeniusDecomposition {
290        t: Matrix::from_vec(t),
291        t_inv: Matrix::from_vec(t_inv),
292        blocks,
293    })
