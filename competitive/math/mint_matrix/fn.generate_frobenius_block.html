242fn frobenius_decomposition<M>(
243    a: &Matrix<AddMulOperation<MInt<M>>>,
244    rng: &mut Xorshift,
245) -> Option<FrobeniusDecomposition<M>>
246where
247    M: MIntBase + MIntConvert<u64>,
248{
249    let n = a.shape.0;
250    let mut rows = Vec::with_capacity(n);
251    let mut t = Vec::with_capacity(n);
252    let mut blocks = Vec::new();
253    while rows.len() < n {
254        let s = rows.len();
255        let v = (0..n).map(|_| MInt::from(rng.rand64())).collect();
256        let mut c = generate_frobenius_block(a, v, &mut rows, &mut t);
257        if rows.len() == s {
258            continue;
259        }
260        if c.0[..s].iter().any(|x| !x.is_zero()) {
261            let p = Polynomial(c.0[s..].to_vec());
262            let q = c.exact_div(&p)?;
263            let mut v = t[s].clone();
264            for (&x, u) in q.0.iter().zip(&t[..s]) {
265                for (v, &u) in v.iter_mut().zip(u) {
266                    *v += x * u;
267                }
268            }
269            rows.truncate(s);
270            t.truncate(s);
271            c = generate_frobenius_block(a, v, &mut rows, &mut t);
272        }
273        blocks.push(Polynomial(c.0[s..].to_vec()));
274    }
275
276    for i in 0..n {
277        let (left, right) = rows.split_at_mut(i + 1);
278        for row in right {
279            row.reduce(&mut left[i].row);
280        }
281    }
282    let mut t_inv = vec![vec![MInt::zero(); n]; n];
283    for row in rows {
284        let (_, c) = row.row.split_at(n);
285        for (x, &y) in t_inv[row.pivot].iter_mut().zip(c) {
286            *x = row.inv * y;
287        }
288    }
289    Some(FrobeniusDecomposition {
290        t: Matrix::from_vec(t),
291        t_inv: Matrix::from_vec(t_inv),
292        blocks,
293    })
