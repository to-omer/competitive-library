604fn reconstruct_mint_crt<M, N1, N2, N3>(f: (MVec<N1>, MVec<N2>, MVec<N3>)) -> MVec<M>
605where
606    M: MIntConvert + MIntConvert<u32>,
607    N1: Montgomery32NttModulus,
608    N2: Montgomery32NttModulus,
609    N3: Montgomery32NttModulus,
610{
611    let t1 = MInt::<N2>::new(N1::get_mod()).inv();
612    let m1_3 = MInt::<N3>::new(N1::get_mod());
613    let t2 = (m1_3 * MInt::<N3>::new(N2::get_mod())).inv();
614    let modulus = <M as MIntConvert<u32>>::mod_into() as u64;
615    let m1 = N1::get_mod() as u64;
616    let m2 = m1 * N2::get_mod() as u64 % modulus;
617    let fits_u64 = (N1::get_mod() - 1) as u128
618        + (N2::get_mod() - 1) as u128 * m1 as u128
619        + (N3::get_mod() - 1) as u128 * m2 as u128
620        <= u64::MAX as u128;
621    f.0.into_iter()
622        .zip(f.1)
623        .zip(f.2)
624        .map(|((c1, c2), c3)| {
625            let d1 = c1.inner();
626            let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
627            let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
628            let d3 = ((c3 - x) * t2).inner();
629            let value = if fits_u64 {
630                (d1 as u64 + d2 as u64 * m1 + d3 as u64 * m2) % modulus
631            } else {
632                ((d1 as u128 + d2 as u128 * m1 as u128 + d3 as u128 * m2 as u128) % modulus as u128)
633                    as u64
634            };
635            MInt::<M>::from(value as u32)
636        })
637        .collect()
638}

881        let qr_fft = Self::even_mul_normal_neg(&q_fft, &q_fft);
882        (
883            Self::inverse_transform_ntt(pr_fft, half),
884            Self::inverse_transform_ntt(qr_fft, half),

571    /// solve: $X(QF)'=\alpha P'(QF)+\beta P(Q'F)$ in $O(deg * max(nz(P), nz(Q), nz(X)))$
572    pub fn solve_sparse_differential2(
573        p: &Self,
574        q: &Self,
575        x: &Self,
576        alpha: T,
577        beta: T,
578        deg: usize,
579    ) -> Self {
580        if deg == 0 {
581            return Self::zero();
582        }
583        let collect_sparse = |p: &Self| -> Vec<(usize, T)> {
584            p.iter()
585                .enumerate()
586                .filter(|&(_, x)| !x.is_zero())
587                .map(|(i, x)| (i, x.clone()))
588                .collect()
589        };
590        assert!(q.coeff(0).is_one());
591        assert!(x.coeff(0).is_one());
592        let p = collect_sparse(p);
593        let q = collect_sparse(q);
594        let x = collect_sparse(x);
595        let diff = |p: &[(usize, T)]| -> Vec<(usize, T)> {
596            p.iter()
597                .filter(|&&(i, _)| i > 0)
598                .map(|&(i, ref x)| (i - 1, x.clone() * T::from(i)))
599                .collect()
600        };
601        let dp = diff(&p);
602        let dq = diff(&q);
603
604        let mf = T::memorized_factorial(deg);
605        let mut f = Self::zeros(deg);
606        let mut qf = Self::zeros(deg);
607        let mut dq_f = Self::zeros(deg);
608        let mut d_qf = Self::zeros(deg);
609        f[0] = T::one();
610        for i in 0..deg - 1 {
611            qf[i] = f.sparse_fold(q.iter().cloned(), i);
612            dq_f[i] = f.sparse_fold(dq.iter().cloned(), i);
613            let dp_qf_i = qf.sparse_fold(dp.iter().cloned(), i);
614            let p_dq_f_i = dq_f.sparse_fold(p.iter().cloned(), i);
615            let x_d_qf_i = d_qf.sparse_fold(
616                x.iter()
617                    .map(|&(i, ref x)| (i, x.clone() - T::from((i == 0) as usize))),
618                i,
619            );
620            d_qf[i] = alpha.clone() * dp_qf_i + beta.clone() * p_dq_f_i - x_d_qf_i;
621
622            let mut f_ip1 = d_qf[i].clone();
623            for &(j, ref q) in q.iter().take_while(|&&(j, _)| j <= i) {
624                if j > 0 {
625                    f_ip1 -= q.clone() * &f[i - (j - 1)] * T::from(i - (j - 1));
626                }
627            }
628            f[i + 1] = f_ip1 * T::memorized_inv(&mf, i + 1);
629        }
630        f
631    }

669        let (sq, cq, mut q) = normalize(q, exp_q);

673            .saturating_add(exp_q.saturating_mul(sq as _)) as usize;
674        if shift >= deg {

681            &p,
682            &q,
683            &p,
684            T::from(exp_p),
685            T::from(exp_q),
686            deg - shift,
687        );
688        f *= cp.signed_pow(exp_p) * cq.signed_pow(exp_q);
689        if shift > 0 {
690            f <<= shift;
691        }
692        f.prefix(deg)
693    }
