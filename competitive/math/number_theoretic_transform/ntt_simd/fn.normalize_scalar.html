111pub unsafe fn ntt_avx2<M>(a: &mut [MInt<M>])
112where
113    M: Montgomery32NttModulus,
114{
115    let n = a.len();
116    if n <= 1 {
117        return;
118    }
119    let ptr = a.as_mut_ptr() as *mut u32;
120    let a = std::slice::from_raw_parts_mut(ptr, n);
121    let mod_vec = _mm256_set1_epi32(M::MOD as i32);
122    let mod2_vec = _mm256_set1_epi32(M::MOD.wrapping_add(M::MOD) as i32);
123    let r_vec = _mm256_set1_epi32(M::R as i32);
124    let imag = M::INFO.root[2];
125    let imag_vec = _mm256_set1_epi32(imag as i32);
126
127    let mut v = n / 2;
128    if n.trailing_zeros() & 1 == 1 {
129        let mut i = 0;
130        while i + 8 <= v {
131            let x0 = _mm256_loadu_si256(a.as_ptr().add(i) as *const __m256i);
132            let x1 = _mm256_loadu_si256(a.as_ptr().add(v + i) as *const __m256i);
133            let y0 = add_vec_avx2::<M>(x0, x1, mod_vec, mod2_vec);
134            let y1 = sub_vec_avx2::<M>(x0, x1, mod_vec, mod2_vec);
135            _mm256_storeu_si256(a.as_mut_ptr().add(i) as *mut __m256i, y0);
136            _mm256_storeu_si256(a.as_mut_ptr().add(v + i) as *mut __m256i, y1);
137            i += 8;
138        }
139        while i < v {
140            let x0 = a[i];
141            let x1 = a[v + i];
142            a[i] = M::mod_add(x0, x1);
143            a[v + i] = M::mod_sub(x0, x1);
144            i += 1;
145        }
146        v >>= 1;
147    }
148    while v > 1 {
149        let half = v >> 1;
150        let mut w1 = M::N1;
151        for (s, block) in a.chunks_exact_mut(v << 1).enumerate() {
152            let base = block.as_mut_ptr();
153            let ll = base;
154            let lr = base.add(half);
155            let rl = base.add(v);
156            let rr = base.add(v + half);
157
158            let w2 = M::mod_mul(w1, w1);
159            let w3 = M::mod_mul(w2, w1);
160            let w1v = _mm256_set1_epi32(w1 as i32);
161            let w2v = _mm256_set1_epi32(w2 as i32);
162            let w3v = _mm256_set1_epi32(w3 as i32);
163
164            let mut i = 0;
165            while i + 8 <= half {
166                let x0 = _mm256_loadu_si256(ll.add(i) as *const __m256i);
167                let x1 = _mm256_loadu_si256(lr.add(i) as *const __m256i);
168                let x2 = _mm256_loadu_si256(rl.add(i) as *const __m256i);
169                let x3 = _mm256_loadu_si256(rr.add(i) as *const __m256i);
170
171                let (a1, a2, a3) = if s == 0 {
172                    (x1, x2, x3)
173                } else {
174                    (
175                        mul_vec_avx2::<M>(x1, w1v, r_vec, mod_vec),
176                        mul_vec_avx2::<M>(x2, w2v, r_vec, mod_vec),
177                        mul_vec_avx2::<M>(x3, w3v, r_vec, mod_vec),
178                    )
179                };
180
181                let a0pa2 = add_vec_avx2::<M>(x0, a2, mod_vec, mod2_vec);
182                let a0na2 = sub_vec_avx2::<M>(x0, a2, mod_vec, mod2_vec);
183                let a1pa3 = add_vec_avx2::<M>(a1, a3, mod_vec, mod2_vec);
184                let a1na3 = sub_vec_avx2::<M>(a1, a3, mod_vec, mod2_vec);
185                let a1na3imag = mul_vec_avx2::<M>(a1na3, imag_vec, r_vec, mod_vec);
186
187                let y0 = add_vec_avx2::<M>(a0pa2, a1pa3, mod_vec, mod2_vec);
188                let y1 = sub_vec_avx2::<M>(a0pa2, a1pa3, mod_vec, mod2_vec);
189                let y2 = add_vec_avx2::<M>(a0na2, a1na3imag, mod_vec, mod2_vec);
190                let y3 = sub_vec_avx2::<M>(a0na2, a1na3imag, mod_vec, mod2_vec);
191
192                _mm256_storeu_si256(ll.add(i) as *mut __m256i, y0);
193                _mm256_storeu_si256(lr.add(i) as *mut __m256i, y1);
194                _mm256_storeu_si256(rl.add(i) as *mut __m256i, y2);
195                _mm256_storeu_si256(rr.add(i) as *mut __m256i, y3);
196                i += 8;
197            }
198            while i < half {
199                let a0 = *ll.add(i);
200                let a1 = M::mod_mul(*lr.add(i), w1);
201                let a2 = M::mod_mul(*rl.add(i), w2);
202                let a3 = M::mod_mul(*rr.add(i), w3);
203                let a0pa2 = M::mod_add(a0, a2);
204                let a0na2 = M::mod_sub(a0, a2);
205                let a1pa3 = M::mod_add(a1, a3);
206                let a1na3 = M::mod_sub(a1, a3);
207                let a1na3imag = M::mod_mul(a1na3, imag);
208                *ll.add(i) = M::mod_add(a0pa2, a1pa3);
209                *lr.add(i) = M::mod_sub(a0pa2, a1pa3);
210                *rl.add(i) = M::mod_add(a0na2, a1na3imag);
211                *rr.add(i) = M::mod_sub(a0na2, a1na3imag);
212                i += 1;
213            }
214            w1 = M::mod_mul(w1, M::INFO.rate3[s.trailing_ones() as usize]);
215        }
216        v >>= 2;
217    }
218    normalize_avx2::<M>(a);
219}

304                let a3 = M::mod_mul(normalize_scalar::<M>(*rr.add(i)), w3);
305                let a0pa2 = M::mod_add(a0, a2);
306                let a0na2 = M::mod_sub(a0, a2);
307                let a1pa3 = M::mod_add(a1, a3);
308                let a1na3 = M::mod_sub(a1, a3);
309                let a1na3imag = M::mod_mul(a1na3, imag);
310                *ll.add(i) = M::mod_add(a0pa2, a1pa3);
311                *lr.add(i) = M::mod_sub(a0pa2, a1pa3);
312                *rl.add(i) = M::mod_add(a0na2, a1na3imag);
313                *rr.add(i) = M::mod_sub(a0na2, a1na3imag);
314                i += 1;
315            }
316            w1 = M::mod_mul(w1, M::INFO.rate3[s.trailing_ones() as usize]);
317        }
318        v >>= 2;
319    }
320    normalize_avx512::<M>(a);
321}

323#[target_feature(enable = "avx512f,avx512dq,avx512cd,avx512bw,avx512vl")]
324pub unsafe fn intt_avx512<M>(a: &mut [MInt<M>])
325where
326    M: Montgomery32NttModulus,
327{
328    let n = a.len();
329    if n <= 1 {
330        return;
331    }
332    let ptr = a.as_mut_ptr() as *mut u32;
333    let a = std::slice::from_raw_parts_mut(ptr, n);
334    let mod_vec = _mm512_set1_epi32(M::MOD as i32);
335    let mod2_vec = _mm512_set1_epi32(M::MOD.wrapping_add(M::MOD) as i32);
336    let r_vec = _mm512_set1_epi32(M::R as i32);
337    let iimag = M::INFO.inv_root[2];
338    let iimag_vec = _mm512_set1_epi32(iimag as i32);
339
340    let mut v = 1;
341    let limit = if n.trailing_zeros() & 1 == 1 {
342        n / 2
343    } else {
344        n
345    };
346    while v < limit {
347        let mut w1 = M::N1;
348        for (s, block) in a.chunks_exact_mut(v << 2).enumerate() {
349            let base = block.as_mut_ptr();
350            let ll = base;
351            let lr = base.add(v);
352            let rl = base.add(v << 1);
353            let rr = base.add(v * 3);
354            let w2 = M::mod_mul(w1, w1);
355            let w3 = M::mod_mul(w2, w1);
356            let w1v = _mm512_set1_epi32(w1 as i32);
357            let w2v = _mm512_set1_epi32(w2 as i32);
358            let w3v = _mm512_set1_epi32(w3 as i32);
359
360            let mut i = 0;
361            while i + 16 <= v {
362                let x0 = _mm512_loadu_si512(ll.add(i) as *const __m512i);
363                let x1 = _mm512_loadu_si512(lr.add(i) as *const __m512i);
364                let x2 = _mm512_loadu_si512(rl.add(i) as *const __m512i);
365                let x3 = _mm512_loadu_si512(rr.add(i) as *const __m512i);
366
367                let a0pa1 = add_vec_avx512::<M>(x0, x1, mod_vec, mod2_vec);
368                let a0na1 = sub_vec_avx512::<M>(x0, x1, mod_vec, mod2_vec);
369                let a2pa3 = add_vec_avx512::<M>(x2, x3, mod_vec, mod2_vec);
370                let a2na3 = sub_vec_avx512::<M>(x2, x3, mod_vec, mod2_vec);
371                let a2na3iimag = mul_vec_avx512::<M>(a2na3, iimag_vec, r_vec, mod_vec);
372
373                let y0 = add_vec_avx512::<M>(a0pa1, a2pa3, mod_vec, mod2_vec);
374                let y1 = add_vec_avx512::<M>(a0na1, a2na3iimag, mod_vec, mod2_vec);
375                let y2 = sub_vec_avx512::<M>(a0pa1, a2pa3, mod_vec, mod2_vec);
376                let y3 = sub_vec_avx512::<M>(a0na1, a2na3iimag, mod_vec, mod2_vec);
377
378                let (y1, y2, y3) = if s == 0 {
379                    (y1, y2, y3)
380                } else {
381                    (
382                        mul_vec_avx512::<M>(y1, w1v, r_vec, mod_vec),
383                        mul_vec_avx512::<M>(y2, w2v, r_vec, mod_vec),
384                        mul_vec_avx512::<M>(y3, w3v, r_vec, mod_vec),
385                    )
386                };
387
388                _mm512_storeu_si512(ll.add(i) as *mut __m512i, y0);
389                _mm512_storeu_si512(lr.add(i) as *mut __m512i, y1);
390                _mm512_storeu_si512(rl.add(i) as *mut __m512i, y2);
391                _mm512_storeu_si512(rr.add(i) as *mut __m512i, y3);
392                i += 16;
393            }
394            while i < v {
395                let a0 = *ll.add(i);
396                let a1 = *lr.add(i);
397                let a2 = *rl.add(i);
398                let a3 = *rr.add(i);
399                let a0pa1 = M::mod_add(a0, a1);
400                let a0na1 = M::mod_sub(a0, a1);
401                let a2pa3 = M::mod_add(a2, a3);
402                let a2na3iimag = M::mod_mul(M::mod_sub(a2, a3), iimag);
403                *ll.add(i) = M::mod_add(a0pa1, a2pa3);
404                *lr.add(i) = M::mod_mul(M::mod_add(a0na1, a2na3iimag), w1);
405                *rl.add(i) = M::mod_mul(M::mod_sub(a0pa1, a2pa3), w2);
406                *rr.add(i) = M::mod_mul(M::mod_sub(a0na1, a2na3iimag), w3);
407                i += 1;
408            }
409            w1 = M::mod_mul(w1, M::INFO.inv_rate3[s.trailing_ones() as usize]);
410        }
411        v <<= 2;
412    }
413    if n.trailing_zeros() & 1 == 1 {
414        let half = n / 2;
415        let mut i = 0;
416        while i + 16 <= half {
417            let x0 = _mm512_loadu_si512(a.as_ptr().add(i) as *const __m512i);
418            let x1 = _mm512_loadu_si512(a.as_ptr().add(half + i) as *const __m512i);
419            let y0 = add_vec_avx512::<M>(x0, x1, mod_vec, mod2_vec);
420            let y1 = sub_vec_avx512::<M>(x0, x1, mod_vec, mod2_vec);
421            _mm512_storeu_si512(a.as_mut_ptr().add(i) as *mut __m512i, y0);
422            _mm512_storeu_si512(a.as_mut_ptr().add(half + i) as *mut __m512i, y1);
423            i += 16;
424        }
425        while i < half {
426            let x0 = a[i];
427            let x1 = a[half + i];
428            a[i] = M::mod_add(x0, x1);
429            a[half + i] = M::mod_sub(x0, x1);
430            i += 1;
431        }
432    }
433    let inv = M::mod_inv(<M as MIntConvert<u32>>::from(n as u32));
434    let inv_vec = _mm512_set1_epi32(inv as i32);
435    let mut i = 0;
436    while i + 16 <= n {
437        let x = _mm512_loadu_si512(a.as_ptr().add(i) as *const __m512i);
438        let y = simd32::montgomery_mul_512_canon(x, inv_vec, r_vec, mod_vec);
439        _mm512_storeu_si512(a.as_mut_ptr().add(i) as *mut __m512i, y);
440        i += 16;
441    }
442    while i < n {
443        a[i] = M::mod_mul(a[i], inv);
444        i += 1;
445    }
446}
