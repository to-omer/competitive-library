530                _mm256_storeu_si256(rr.add(i) as *mut __m256i, y3);
531                i += 8;
532            }
533            while i < quarter {
534                let a0 = normalize_scalar::<M>(*ll.add(i));
535                let a1 = normalize_scalar::<M>(*lr.add(i));
536                let a2 = normalize_scalar::<M>(*rl.add(i));
537                let a3 = normalize_scalar::<M>(*rr.add(i));
538                let a0pa1 = M::mod_add(a0, a1);
539                let a0na1 = M::mod_sub(a0, a1);
540                let a2pa3 = M::mod_add(a2, a3);
541                let a2na3iimag = M::mod_mul(M::mod_sub(a2, a3), iimag);
542                *ll.add(i) = M::mod_add(a0pa1, a2pa3);
543                *lr.add(i) = M::mod_mul(M::mod_add(a0na1, a2na3iimag), w1);
544                *rl.add(i) = M::mod_mul(M::mod_sub(a0pa1, a2pa3), w2);
545                *rr.add(i) = M::mod_mul(M::mod_sub(a0na1, a2na3iimag), w3);
546                i += 1;
547            }
548            w1 = M::mod_mul(w1, M::INFO.inv_rate3[s.trailing_ones() as usize]);
549        }
550        v <<= 2;
551    }
552    if n.trailing_zeros() & 1 == 1 {
553        let half = (n >> 1) * width;
554        let mut i = 0;
555        while i + 8 <= half {
556            let x0 = _mm256_loadu_si256(a.as_ptr().add(i) as *const __m256i);
557            let x1 = _mm256_loadu_si256(a.as_ptr().add(half + i) as *const __m256i);
558            let y0 = add_vec_avx2::<M>(x0, x1, mod_vec, mod2_vec);
559            let y1 = sub_vec_avx2::<M>(x0, x1, mod_vec, mod2_vec);
560            _mm256_storeu_si256(a.as_mut_ptr().add(i) as *mut __m256i, y0);
561            _mm256_storeu_si256(a.as_mut_ptr().add(half + i) as *mut __m256i, y1);
562            i += 8;
563        }
564        while i < half {
565            let x0 = normalize_scalar::<M>(a[i]);
566            let x1 = normalize_scalar::<M>(a[half + i]);
567            a[i] = M::mod_add(x0, x1);
568            a[half + i] = M::mod_sub(x0, x1);
569            i += 1;
570        }
571    }
572    let inv = M::mod_inv(<M as MIntConvert<u32>>::from(n as u32));
573    let inv_vec = _mm256_set1_epi32(inv as i32);
574    let mut i = 0;
575    while i + 8 <= a.len() {
576        let x = _mm256_loadu_si256(a.as_ptr().add(i) as *const __m256i);
577        let y = simd32::montgomery_mul_256_canon(x, inv_vec, r_vec, mod_vec);
578        _mm256_storeu_si256(a.as_mut_ptr().add(i) as *mut __m256i, y);
579        i += 8;
580    }
581    while i < a.len() {
582        a[i] = M::mod_mul(a[i], inv);
583        i += 1;
584    }
585}
