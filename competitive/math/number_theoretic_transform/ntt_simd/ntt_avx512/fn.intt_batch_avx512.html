336fn intt_batch_scalar<M>(a: &mut [MInt<M>], width: usize)
337where
338    M: Montgomery32NttModulus,
339{
340    let n = a.len() / width;
341    if n <= 1 {
342        return;
343    }
344    // MInt is transparent over u32; lazy residues stay below 2 * MOD and are
345    // normalized before the typed slice is used again.
346    let a = unsafe { std::slice::from_raw_parts_mut(a.as_mut_ptr().cast::<u32>(), a.len()) };
347    let mut v = 1;
348    let limit = if n.trailing_zeros() & 1 == 1 {
349        n / 2
350    } else {
351        n
352    };
353    let iimag = M::INFO.inv_root[2];
354    while v < limit {
355        let mut w1 = M::N1;
356        let mut w2 = w1;
357        let mut w3 = w1;
358        for (s, a) in a.chunks_exact_mut((v << 2) * width).enumerate() {
359            let (l, r) = a.split_at_mut((v << 1) * width);
360            let (ll, lr) = l.split_at_mut(v * width);
361            let (rl, rr) = r.split_at_mut(v * width);
362            for (((x0, x1), x2), x3) in ll.iter_mut().zip(lr).zip(rl).zip(rr) {
363                let a0 = *x0;
364                let a1 = *x1;
365                let a2 = *x2;
366                let a3 = *x3;
367                let a0pa1 = add_scalar::<M>(a0, a1);
368                let a0na1 = sub_scalar::<M>(a0, a1);
369                let a2pa3 = add_scalar::<M>(a2, a3);
370                let a2na3iimag = mul_scalar::<M>(sub_scalar::<M>(a2, a3), iimag);
371                *x0 = add_scalar::<M>(a0pa1, a2pa3);
372                *x1 = mul_scalar::<M>(add_scalar::<M>(a0na1, a2na3iimag), w1);
373                *x2 = mul_scalar::<M>(sub_scalar::<M>(a0pa1, a2pa3), w2);
374                *x3 = mul_scalar::<M>(sub_scalar::<M>(a0na1, a2na3iimag), w3);
375            }
376            let rate = &M::INFO.inv_rate3_packed[s.trailing_ones() as usize];
377            w1 = M::mod_mul(w1, rate[1]);
378            w2 = M::mod_mul(w2, rate[3]);
379            w3 = M::mod_mul(w3, rate[5]);
380        }
381        v <<= 2;
382    }
383    if n.trailing_zeros() & 1 == 1 {
384        let (l, r) = a.split_at_mut(n / 2 * width);
385        for (x0, x1) in l.iter_mut().zip(r) {
386            let a0 = *x0;
387            let a1 = *x1;
388            *x0 = add_scalar::<M>(a0, a1);
389            *x1 = sub_scalar::<M>(a0, a1);
390        }
391    }
392    let inv = M::mod_inv(<M as MIntConvert<u32>>::from(n as u32));
393    for a in a {
394        *a = M::mod_mul(*a, inv);
395    }
396}

398fn ntt<M>(a: &mut [MInt<M>])
399where
400    M: Montgomery32NttModulus,
