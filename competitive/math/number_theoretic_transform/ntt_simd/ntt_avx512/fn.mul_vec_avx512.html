515                _mm512_storeu_si512(lr.add(i) as *mut __m512i, y1);
516                _mm512_storeu_si512(rl.add(i) as *mut __m512i, y2);
517                _mm512_storeu_si512(rr.add(i) as *mut __m512i, y3);
518                i += 16;
519            }
520            while i < quarter {
521                let a0 = normalize_scalar::<M>(*ll.add(i));
522                let a1 = normalize_scalar::<M>(*lr.add(i));
523                let a2 = normalize_scalar::<M>(*rl.add(i));
524                let a3 = normalize_scalar::<M>(*rr.add(i));
525                let a0pa1 = M::mod_add(a0, a1);
526                let a0na1 = M::mod_sub(a0, a1);
527                let a2pa3 = M::mod_add(a2, a3);
528                let a2na3iimag = M::mod_mul(M::mod_sub(a2, a3), iimag);
529                *ll.add(i) = M::mod_add(a0pa1, a2pa3);
530                *lr.add(i) = M::mod_mul(M::mod_add(a0na1, a2na3iimag), w1);
531                *rl.add(i) = M::mod_mul(M::mod_sub(a0pa1, a2pa3), w2);
532                *rr.add(i) = M::mod_mul(M::mod_sub(a0na1, a2na3iimag), w3);
533                i += 1;
534            }
535            w1 = M::mod_mul(w1, M::INFO.inv_rate3[s.trailing_ones() as usize]);
536        }
537        v <<= 2;
538    }
539    if n.trailing_zeros() & 1 == 1 {
540        let half = n / 2 * width;
541        let mut i = 0;
542        while i + 16 <= half {
543            let x0 = _mm512_loadu_si512(a.as_ptr().add(i) as *const __m512i);
544            let x1 = _mm512_loadu_si512(a.as_ptr().add(half + i) as *const __m512i);
545            let y0 = add_vec_avx512::<M>(x0, x1, mod_vec, mod2_vec);
546            let y1 = sub_vec_avx512::<M>(x0, x1, mod_vec, mod2_vec);
547            _mm512_storeu_si512(a.as_mut_ptr().add(i) as *mut __m512i, y0);
548            _mm512_storeu_si512(a.as_mut_ptr().add(half + i) as *mut __m512i, y1);
549            i += 16;
550        }
551        while i < half {
552            let x0 = normalize_scalar::<M>(a[i]);
553            let x1 = normalize_scalar::<M>(a[half + i]);
554            a[i] = M::mod_add(x0, x1);
555            a[half + i] = M::mod_sub(x0, x1);
556            i += 1;
557        }
558    }
559    let inv = M::mod_inv(<M as MIntConvert<u32>>::from(n as u32));
560    let inv_vec = _mm512_set1_epi32(inv as i32);
561    let mut i = 0;
562    while i + 16 <= a.len() {
563        let x = _mm512_loadu_si512(a.as_ptr().add(i) as *const __m512i);
564        let y = simd32::montgomery_mul_512_canon(x, inv_vec, r_vec, mod_vec);
565        _mm512_storeu_si512(a.as_mut_ptr().add(i) as *mut __m512i, y);
566        i += 16;
567    }
568    while i < a.len() {
569        a[i] = M::mod_mul(a[i], inv);
570        i += 1;
571    }
