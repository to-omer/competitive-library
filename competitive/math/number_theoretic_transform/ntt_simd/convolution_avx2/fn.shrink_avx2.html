35unsafe fn update_root_avx2(root: __m256i, rate: __m256i, modulus: __m256i) -> __m256i {
36    let correction = montgomery_simd::mul_epu32_256(root, rate);

83            let r1_r = ($broadcast)(r1_r);

85            let lazy_sub = |x, y, m| $add(x, $sub(m, y));

106                    let h1 = $sub_mod(g1, ng3, modulus2);
107                    let h0 = $add_mod(g0, g2, modulus2);
108                    let h2 = $sub_mod(g0, g2, modulus2);
109                    (h0, h1, h2, h3)
110                };
111                // Only y0 feeds the next first group; keep its inputs below 2*MOD.
112                let mut y0 = if FIRST {
113                    $add_mod(h0, h1, modulus2)
114                } else {
115                    $add(h0, h1)
116                };
117                let mut y1 = lazy_sub(h0, h1, modulus2);
118                let mut y2 = $add(h2, h3);
119                let mut y3 = lazy_sub(h2, h3, modulus2);
120                if FIRST && quarter == 1 {

122                    y1 = shrink(shrink(y1, modulus2), modulus);

124                    y3 = shrink(shrink(y3, modulus2), modulus);
125                }
126                $store(base.add(p0).cast(), y0);
127                $store(base.add(p0 + (quarter << 3)).cast(), y1);
128                $store(base.add(p0 + (quarter << 4)).cast(), y2);
129                $store(base.add(p0 + quarter * 24).cast(), y3);

149            let indices = _mm256_setr_epi32(0, 2, 0, 4, 0, 2, 0, 4);

196                    )
197                };

200                    y1 = shrink(y1, modulus);
201                    y2 = shrink(y2, modulus);
202                    y3 = shrink(y3, modulus);
203                }
204                $store(base.add(p0).cast(), y0);
205                $store(base.add(p0 + (quarter << 3)).cast(), y1);
206                $store(base.add(p0 + (quarter << 4)).cast(), y2);
207                $store(base.add(p0 + quarter * 24).cast(), y3);
208                j += $step;
209            }

243#[target_feature(enable = "avx2")]
244unsafe fn ntt_blocks_avx2<M>(a: *mut u32, n: usize, input_len: usize)
245where
246    M: Montgomery32NttModulus,
247{
248    let modulus = _mm256_set1_epi32(M::MOD as i32);
249    let modulus2 = _mm256_set1_epi32(M::MOD.wrapping_mul(2) as i32);
250    let wide = matches!(simd_backend(), SimdBackend::Avx512);
251    let imag = M::INFO.root[2];
252    let imag_r = _mm256_set1_epi32(imag.wrapping_mul(M::R) as i32);
253    let imag = _mm256_set1_epi32(imag as i32);
254    let initial_root = _mm256_setr_epi32(
255        M::INFO.root[3] as i32,
256        0,
257        M::INFO.root[2] as i32,
258        0,
259        (M::MOD - M::mod_mul(M::INFO.root[2], M::INFO.root[3])) as i32,
260        0,
261        0,
262        0,
263    );
264    let log_n = n.trailing_zeros() as usize;
265    let mut roots = [initial_root; 16];
266    let nn = n >> (log_n & 1);
267    let tile_len = n.min(64);
268    let mut stage_log = log_n & !1;
269    let sparse = nn != n && input_len <= nn << 3;
270    if nn != n && !sparse {
271        let mut i = 0;
272        while i < nn {
273            let x0 = load_block_avx2(a, i);
274            let x1 = load_block_avx2(a, nn + i);
275            store_block_avx2(a, i, montgomery_simd::add_mod_256(x0, x1, modulus2));
276            store_block_avx2(a, nn + i, lazy_sub_avx2(x0, x1, modulus2));
277            i += 1;
278        }
279    }
280    if sparse {
281        let quarter = nn >> 2;
282        let w1 = M::INFO.root[3];
283        let w3 = M::MOD - M::mod_mul(w1, M::INFO.root[2]);
284        let r1 = _mm256_set1_epi32(w1 as i32);
285        let r1_r = _mm256_set1_epi32(w1.wrapping_mul(M::R) as i32);
286        let nr3 = _mm256_set1_epi32(w3 as i32);
287        let nr3_r = _mm256_set1_epi32(w3.wrapping_mul(M::R) as i32);
288        let store = |offset, values: [__m256i; 4]| {
289            for (k, mut y) in values.into_iter().enumerate() {
290                if quarter == 1 {
291                    y = normalize_avx2(y, modulus, modulus2);
292                }
293                store_block_avx2(a, offset + quarter * k, y);
294            }
295        };
296        for j in 0..quarter {
297            let x0 = load_block_avx2(a, j);
298            let x1 = load_block_avx2(a, j + quarter);
299            let x2 = load_block_avx2(a, j + quarter * 2);
300            let x3 = load_block_avx2(a, j + quarter * 3);
301            let h0 = _mm256_add_epi32(x0, x2);
302            let h1 = _mm256_add_epi32(x1, x3);
303            let h2 = lazy_sub_avx2(x0, x2, modulus);
304            let h3 = montgomery_simd::montgomery_mul_256_fixed(
305                lazy_sub_avx2(x1, x3, modulus),
306                imag,
307                imag_r,
308                modulus,
309            );
310            store(
311                j,
312                [
313                    montgomery_simd::add_mod_256(h0, h1, modulus2),
314                    lazy_sub_avx2(h0, h1, modulus2),
315                    _mm256_add_epi32(h2, h3),
316                    lazy_sub_avx2(h2, h3, modulus2),
317                ],
318            );
319            let g1 = montgomery_simd::montgomery_mul_256_fixed(x1, r1, r1_r, modulus);
320            let ng3 = montgomery_simd::montgomery_mul_256_fixed(x3, nr3, nr3_r, modulus);
321            let g2 = montgomery_simd::montgomery_mul_256_fixed(x2, imag, imag_r, modulus);
322            let h3 = montgomery_simd::montgomery_mul_256_fixed(
323                _mm256_add_epi32(g1, ng3),
324                imag,
325                imag_r,
326                modulus,
327            );
328            let h1 = montgomery_simd::sub_mod_256(g1, ng3, modulus2);
329            let h0 = montgomery_simd::add_mod_256(x0, g2, modulus2);
330            let h2 = montgomery_simd::sub_mod_256(x0, g2, modulus2);
331            store(
332                nn + j,
333                [
334                    _mm256_add_epi32(h0, h1),
335                    lazy_sub_avx2(h0, h1, modulus2),
336                    _mm256_add_epi32(h2, h3),
337                    lazy_sub_avx2(h2, h3, modulus2),
338                ],
339            );
340        }
341        stage_log -= 2;
342    }
343    if nn == n && input_len <= n << 2 {
344        let quarter = n >> 2;
345        let mut i = 0;
346        while i < quarter {
347            let x0 = load_block_avx2(a, i);
348            let x1 = load_block_avx2(a, quarter + i);
349            let x3 = montgomery_simd::montgomery_mul_256_fixed(x1, imag, imag_r, modulus);
350            store_block_avx2(a, i, _mm256_add_epi32(x0, x1));
351            store_block_avx2(a, quarter + i, lazy_sub_avx2(x0, x1, modulus2));
352            store_block_avx2(a, 2 * quarter + i, _mm256_add_epi32(x0, x3));
353            store_block_avx2(a, 3 * quarter + i, lazy_sub_avx2(x0, x3, modulus2));
354            i += 1;
355        }
356        stage_log -= 2;
357    }
358    if stage_log == 0 {
359        return;
360    }
361    let max_stage_log = stage_log;
362    while stage_log >= 2 {
363        let quarter = 1 << (stage_log - 2);
364        if wide && quarter > 1 {
365            ntt_butterfly_avx512::<M, true>(a, quarter, initial_root);
366        } else {
367            ntt_butterfly_avx2::<M, true>(a, quarter, initial_root);
368        }
369        stage_log -= 2;
370    }
371    stage_log = max_stage_log.min(6);
372    let mut tile = 0;
373    while tile < n {
374        let base = a.add(tile << 3);
375        let mut group_len = 1usize << stage_log;
376        let mut quarter = group_len >> 2;
377        while quarter > 1 {
378            let root_slot = (stage_log - 2) >> 1;
379            let mut root = roots[root_slot];
380            let mut i = if tile == 0 { group_len } else { 0 };
381            let mut group = (tile + i) >> stage_log;
382            while i < tile_len {
383                let current = root;
384                root = update_root_avx2(
385                    root,
386                    packed_rate_avx2::<M>((!group).trailing_zeros() as usize, false),
387                    modulus,
388                );
389                let base = base.add(i << 3);
390                if wide {
391                    ntt_butterfly_avx512::<M, false>(base, quarter, current);
392                } else {
393                    ntt_butterfly_avx2::<M, false>(base, quarter, current);
394                }
395                i += group_len;
396                group += 1;
397            }
398            roots[root_slot] = root;
399            group_len = quarter;
400            quarter >>= 2;
401            stage_log -= 2;
402        }
403        // Final-stage roots are used once; skip precomputing root*R.
404        let r = _mm256_set1_epi32(M::R as i32);
405        let store = |i, y| store_block_avx2(a, i, normalize_avx2(y, modulus, modulus2));
406        let mut root = roots[0];
407        let mut i = tile + if tile == 0 { 4 } else { 0 };
408        while i < tile + tile_len {
409            let r1 = _mm256_permutevar8x32_epi32(root, _mm256_setr_epi32(0, 2, 0, 4, 0, 2, 0, 4));
410            root = update_root_avx2(
411                root,
412                packed_rate_avx2::<M>((!(i >> 2)).trailing_zeros() as usize, false),
413                modulus,
414            );
415            let r2 = _mm256_shuffle_epi32::<0x55>(r1);
416            let nr3 = _mm256_shuffle_epi32::<0xff>(r1);
417            let x0 = load_block_avx2(a, i);
418            let x1 = load_block_avx2(a, i + 1);
419            let x2 = load_block_avx2(a, i + 2);
420            let x3 = load_block_avx2(a, i + 3);
421            let g1 = montgomery_simd::montgomery_mul_256(
422                x1,
423                _mm256_shuffle_epi32::<0xa0>(r1),
424                r,
425                modulus,
426            );
427            let ng3 = montgomery_simd::montgomery_mul_256(x3, nr3, r, modulus);
428            let g2 = montgomery_simd::montgomery_mul_256(x2, r2, r, modulus);
429            let g0 = shrink_avx2(x0, modulus2);
430            let h3 = montgomery_simd::montgomery_mul_256_fixed(
431                _mm256_add_epi32(g1, ng3),
432                imag,
433                imag_r,
434                modulus,
435            );
436            let h1 = montgomery_simd::sub_mod_256(g1, ng3, modulus2);
437            let h0 = montgomery_simd::add_mod_256(g0, g2, modulus2);
438            let h2 = montgomery_simd::sub_mod_256(g0, g2, modulus2);
439            store(i, _mm256_add_epi32(h0, h1));
440            store(i + 1, lazy_sub_avx2(h0, h1, modulus2));
441            store(i + 2, _mm256_add_epi32(h2, h3));
442            store(i + 3, lazy_sub_avx2(h2, h3, modulus2));
443            i += 4;
444        }
445        roots[0] = root;
446
447        tile += tile_len;
448        if tile < n {
449            stage_log = (tile.trailing_zeros() as usize & !1).min(max_stage_log);
450        }
451    }
452}

454#[target_feature(enable = "avx2")]
455unsafe fn intt_blocks_avx2<M>(a: *mut u32, n: usize)
456where
457    M: Montgomery32NttModulus,
458{
459    let inv = M::mod_inv(<M as MIntConvert<u32>>::from(n as u32));
460    let modulus = _mm256_set1_epi32(M::MOD as i32);
461    let modulus2 = _mm256_set1_epi32(M::MOD.wrapping_mul(2) as i32);
462    let wide = matches!(simd_backend(), SimdBackend::Avx512);
463
464    let initial_root = _mm256_setr_epi32(M::N1 as i32, 0, M::N1 as i32, 0, M::N1 as i32, 0, 0, 0);
465    let log_n = n.trailing_zeros() as usize;
466    let mut roots = [initial_root; 16];
467    let nn = n >> (log_n & 1);
468    let tile_len = n.min(64);
469    let inv_vec = _mm256_set1_epi32(inv as i32);
470
471    roots[0] = inv_vec;
472    let mut tile = 0;
473    while tile < n {
474        let max_stage_log = (tile + tile_len).trailing_zeros() as usize;
475        let mut stage_log = 2usize;
476        let mut group_len = 4usize;
477        let mut quarter = 1usize;
478        while stage_log <= max_stage_log {
479            let offset = tile + tile_len - group_len.max(tile_len);
480            let base = a.add(offset << 3);
481            let mut i = 0;
482            let root_slot = (stage_log - 2) >> 1;
483            let mut root = roots[root_slot];
484            let mut group = (tile + i) >> stage_log;
485            while i < tile_len {
486                let current = root;
487                root = update_root_avx2(
488                    root,
489                    packed_rate_avx2::<M>((!group).trailing_zeros() as usize, true),
490                    modulus,
491                );
492                let base = base.add(i << 3);
493                let last = group_len == n;
494                if wide && quarter > 1 {
495                    intt_butterfly_avx512::<M>(base, quarter, current, inv, group, last);
496                } else {
497                    intt_butterfly_avx2::<M>(base, quarter, current, inv, group, last);
498                }
499                i += group_len;
500                group += 1;
501            }
502            roots[root_slot] = root;
503            quarter = group_len;
504            group_len <<= 2;
505            stage_log += 2;
506        }
507        tile += tile_len;
508    }
509    if nn != n {
510        let mut i = 0;
511        while i < nn {
512            let x0 = load_block_avx2(a, i);
513            let x1 = load_block_avx2(a, nn + i);
514            store_block_avx2(
515                a,
516                i,
517                shrink_avx2(montgomery_simd::add_mod_256(x0, x1, modulus2), modulus),
518            );
519            store_block_avx2(
520                a,
521                nn + i,
522                shrink_avx2(montgomery_simd::sub_mod_256(x0, x1, modulus2), modulus),
523            );
524            i += 1;
525        }
526    }
