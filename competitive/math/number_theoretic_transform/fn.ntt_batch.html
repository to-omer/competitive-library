175fn ntt_batch<M>(a: &mut [MInt<M>], width: usize)
176where
177    M: Montgomery32NttModulus,
178{
179    let n = a.len() / width;
180    if n <= 1 {
181        return;
182    }
183    let mut v = n / 2;
184    if n.trailing_zeros() & 1 == 1 {
185        let (l, r) = a.split_at_mut(v * width);
186        for (x0, x1) in l.iter_mut().zip(r) {
187            let a0 = *x0;
188            let a1 = *x1;
189            *x0 = a0 + a1;
190            *x1 = a0 - a1;
191        }
192        v >>= 1;
193    }
194    let imag = MInt::<M>::new_unchecked(M::INFO.root[2]);
195    while v > 1 {
196        let mut w1 = MInt::<M>::one();
197        for (s, a) in a.chunks_exact_mut((v << 1) * width).enumerate() {
198            let (l, r) = a.split_at_mut(v * width);
199            let (ll, lr) = l.split_at_mut((v >> 1) * width);
200            let (rl, rr) = r.split_at_mut((v >> 1) * width);
201            let w2 = w1 * w1;
202            let w3 = w1 * w2;
203            for (((x0, x1), x2), x3) in ll.iter_mut().zip(lr).zip(rl).zip(rr) {
204                let a0 = *x0;
205                let a1 = *x1 * w1;
206                let a2 = *x2 * w2;
207                let a3 = *x3 * w3;
208                let a0pa2 = a0 + a2;
209                let a0na2 = a0 - a2;
210                let a1pa3 = a1 + a3;
211                let a1na3imag = (a1 - a3) * imag;
212                *x0 = a0pa2 + a1pa3;
213                *x1 = a0pa2 - a1pa3;
214                *x2 = a0na2 + a1na3imag;
215                *x3 = a0na2 - a1na3imag;
216            }
217            w1 *= MInt::<M>::new_unchecked(M::INFO.rate3[s.trailing_ones() as usize]);
218        }
219        v >>= 2;
220    }
221}

230fn intt_batch<M>(a: &mut [MInt<M>], width: usize)
231where
232    M: Montgomery32NttModulus,
233{
234    let n = a.len() / width;
235    if n <= 1 {
236        return;
237    }
238    let mut v = 1;
239    let limit = if n.trailing_zeros() & 1 == 1 {
240        n / 2
241    } else {
242        n
243    };
244    let iimag = MInt::<M>::new_unchecked(M::INFO.inv_root[2]);
245    while v < limit {
246        let mut w1 = MInt::<M>::one();
247        for (s, a) in a.chunks_exact_mut((v << 2) * width).enumerate() {
248            let (l, r) = a.split_at_mut((v << 1) * width);
249            let (ll, lr) = l.split_at_mut(v * width);
250            let (rl, rr) = r.split_at_mut(v * width);
251            let w2 = w1 * w1;
252            let w3 = w1 * w2;
253            for (((x0, x1), x2), x3) in ll.iter_mut().zip(lr).zip(rl).zip(rr) {
254                let a0 = *x0;
255                let a1 = *x1;
256                let a2 = *x2;
257                let a3 = *x3;
258                let a0pa1 = a0 + a1;
259                let a0na1 = a0 - a1;
260                let a2pa3 = a2 + a3;
261                let a2na3iimag = (a2 - a3) * iimag;
262                *x0 = a0pa1 + a2pa3;
263                *x1 = (a0na1 + a2na3iimag) * w1;
264                *x2 = (a0pa1 - a2pa3) * w2;
265                *x3 = (a0na1 - a2na3iimag) * w3;
266            }
267            w1 *= MInt::<M>::new_unchecked(M::INFO.inv_rate3[s.trailing_ones() as usize]);
268        }
269        v <<= 2;
270    }
271    if n.trailing_zeros() & 1 == 1 {
272        let (l, r) = a.split_at_mut(n / 2 * width);
273        for (x0, x1) in l.iter_mut().zip(r) {
274            let a0 = *x0;
275            let a1 = *x1;
276            *x0 = a0 + a1;
277            *x1 = a0 - a1;
278        }
279    }
280    let inv = MInt::<M>::from(n as u32).inv();
281    for a in a {
282        *a *= inv;
283    }
284}

377fn convolve_naive<T>(a: &[T], b: &[T]) -> Vec<T>
378where
379    T: Copy + Zero + AddAssign<T> + Mul<Output = T>,
380{
381    if a.is_empty() && b.is_empty() {
382        return Vec::new();
383    }
384    let len = a.len() + b.len() - 1;
385    let mut c = vec![T::zero(); len];
386    if a.len() < b.len() {
387        for (i, &b) in b.iter().enumerate() {
388            for (a, c) in a.iter().zip(&mut c[i..]) {
389                *c += *a * b;
390            }
391        }
392    } else {
393        for (i, &a) in a.iter().enumerate() {
394            for (b, c) in b.iter().zip(&mut c[i..]) {
395                *c += *b * a;
396            }
397        }
398    }
399    c
400}

402fn convolve_karatsuba<T>(a: &[T], b: &[T]) -> Vec<T>
403where
404    T: Copy + Zero + AddAssign<T> + SubAssign<T> + Mul<Output = T>,
405{
406    if a.len().min(b.len()) <= 30 {
407        return convolve_naive(a, b);
408    }
409    let m = a.len().max(b.len()).div_ceil(2);
410    let (a0, a1) = if a.len() <= m {
411        (a, &[][..])
412    } else {
413        a.split_at(m)
414    };
415    let (b0, b1) = if b.len() <= m {
416        (b, &[][..])
417    } else {
418        b.split_at(m)
419    };
420    let f00 = convolve_karatsuba(a0, b0);
421    let f11 = convolve_karatsuba(a1, b1);
422    let mut a0a1 = a0.to_vec();
423    for (a0a1, &a1) in a0a1.iter_mut().zip(a1) {
424        *a0a1 += a1;
425    }
426    let mut b0b1 = b0.to_vec();
427    for (b0b1, &b1) in b0b1.iter_mut().zip(b1) {
428        *b0b1 += b1;
429    }
430    let mut f01 = convolve_karatsuba(&a0a1, &b0b1);
431    for (f01, &f00) in f01.iter_mut().zip(&f00) {
432        *f01 -= f00;
433    }
434    for (f01, &f11) in f01.iter_mut().zip(&f11) {
435        *f01 -= f11;
436    }
437    let mut c = vec![T::zero(); a.len() + b.len() - 1];
438    for (c, &f00) in c.iter_mut().zip(&f00) {
439        *c += f00;
440    }
441    for (c, &f01) in c[m..].iter_mut().zip(&f01) {
442        *c += f01;
443    }
444    for (c, &f11) in c[m << 1..].iter_mut().zip(&f11) {
445        *c += f11;
446    }
447    c
448}

451fn convolve_large_ntt<M>(a: Vec<MInt<M>>, b: Vec<MInt<M>>) -> Vec<MInt<M>>
452where
453    M: Montgomery32NttModulus,
454{
455    let len = a.len() + b.len() - 1;
456    let ntt_len = 1usize << M::RANK;
457    let block_len = ntt_len / 2;
458    let same = a == b;
459    let fa: Vec<_> = a
460        .chunks(block_len)
461        .map(|a| Convolve::<M>::transform_ntt(a.to_vec(), ntt_len))
462        .collect();
463    let fb: Option<Vec<_>> = if same {
464        None
465    } else {
466        Some(
467            b.chunks(block_len)
468                .map(|b| Convolve::<M>::transform_ntt(b.to_vec(), ntt_len))
469                .collect(),
470        )
471    };
472    let b_blocks = fb.as_ref().map_or(fa.len(), Vec::len);
473    let mut result = vec![MInt::<M>::zero(); len];
474    for diagonal in 0..fa.len() + b_blocks - 1 {
475        let mut spectrum = vec![MInt::<M>::zero(); ntt_len];
476        let start = diagonal.saturating_sub(b_blocks - 1);
477        for i in start..=diagonal.min(fa.len() - 1) {
478            let j = diagonal - i;
479            let g = if let Some(fb) = &fb { &fb[j] } else { &fa[j] };
480            pointwise_multiply_add(&mut spectrum, &fa[i], g);
481        }
482        spectrum = Convolve::<M>::inverse_transform_ntt(spectrum, ntt_len);
483        let offset = diagonal * block_len;
484        for (result, value) in result[offset..].iter_mut().zip(spectrum) {
485            *result += value;
486        }
487    }
488    result
489}

491impl<M> ConvolveSteps for Convolve<M>
492where
493    M: Montgomery32NttModulus,
494{
495    type T = Vec<MInt<M>>;
496    type F = Vec<MInt<M>>;
497    fn length(t: &Self::T) -> usize {
498        t.len()
499    }
500    fn transform(mut t: Self::T, len: usize) -> Self::F {
501        t.resize_with(len.max(1).next_power_of_two(), Zero::zero);
502        #[cfg(target_arch = "x86_64")]
503        if use_block_ntt::<M>(t.len()) {
504            unsafe { ntt_simd::transform_blocks_avx2(&mut t) };
505            return t;
506        }
507        ntt(&mut t);
508        t
509    }
510    fn inverse_transform(mut f: Self::F, len: usize) -> Self::T {
511        #[cfg(target_arch = "x86_64")]
512        if use_block_ntt::<M>(f.len()) {
513            unsafe { ntt_simd::inverse_transform_blocks_avx2(&mut f) };
514            f.truncate(len);
515            return f;
516        }
517        intt(&mut f);
518        f.truncate(len);
519        f
520    }
521    fn multiply(f: &mut Self::F, g: &Self::F) {
522        assert_eq!(f.len(), g.len());
523        #[cfg(target_arch = "x86_64")]
524        if use_block_ntt::<M>(f.len()) {
525            unsafe { ntt_simd::multiply_blocks_avx2(f, g) };
526            return;
527        }
528        pointwise_multiply(f, g);
529    }
530    fn convolve(mut a: Self::T, mut b: Self::T) -> Self::T {
531        if Self::length(&a).max(Self::length(&b)) <= 100 {
532            return convolve_karatsuba(&a, &b);
533        }
534        if Self::length(&a).min(Self::length(&b)) <= 60 {
535            return convolve_naive(&a, &b);
536        }
537        let len = (Self::length(&a) + Self::length(&b)).saturating_sub(1);
538        let size = len.max(1).next_power_of_two();
539        if size > 1usize << M::RANK {
540            return convolve_large_ntt(a, b);
541        }
542        if len <= size / 2 + 2 {
543            let xa = a.pop().unwrap();
544            let xb = b.pop().unwrap();
545            let mut c = vec![MInt::<M>::zero(); len];
546            *c.last_mut().unwrap() = xa * xb;
547            for (a, c) in a.iter().zip(&mut c[b.len()..]) {
548                *c += *a * xb;
549            }
550            for (b, c) in b.iter().zip(&mut c[a.len()..]) {
551                *c += *b * xa;
552            }
553            let d = Self::convolve(a, b);
554            for (d, c) in d.into_iter().zip(&mut c) {
555                *c += d;
556            }
557            return c;
558        }
559        let same = a == b;
560        #[cfg(target_arch = "x86_64")]
561        if use_block_ntt::<M>(size) {
562            a.resize_with(size, Zero::zero);
563            b.resize_with(size, Zero::zero);
564            unsafe { ntt_simd::convolve_blocks_avx2(&mut a, &mut b, same) };
565            a.truncate(len);
566            return a;
567        }
568        let mut a = Self::transform(a, len);
569        if same {
570            for a in a.iter_mut() {
571                *a *= *a;
572            }
573        } else {
574            let b = Self::transform(b, len);
575            Self::multiply(&mut a, &b);
576        }
577        Self::inverse_transform(a, len)
578    }
579}

583fn convert_crt_input<M, N1, N2, N3>(t: MVec<M>, capacity: usize) -> (MVec<N1>, MVec<N2>, MVec<N3>)
584where
585    M: MIntConvert<u32>,

591        MVec::<N1>::with_capacity(capacity),
592        MVec::<N2>::with_capacity(capacity),
593        MVec::<N3>::with_capacity(capacity),

595    for t in t {

598        f.1.push(t.into());

604fn reconstruct_mint_crt<M, N1, N2, N3>(f: (MVec<N1>, MVec<N2>, MVec<N3>)) -> MVec<M>
605where
606    M: MIntConvert + MIntConvert<u32>,
607    N1: Montgomery32NttModulus,
608    N2: Montgomery32NttModulus,
609    N3: Montgomery32NttModulus,
610{
611    let t1 = MInt::<N2>::new(N1::get_mod()).inv();
612    let m1_3 = MInt::<N3>::new(N1::get_mod());
613    let t2 = (m1_3 * MInt::<N3>::new(N2::get_mod())).inv();
614    let modulus = <M as MIntConvert<u32>>::mod_into() as u64;
615    let m1 = N1::get_mod() as u64;
616    let m2 = m1 * N2::get_mod() as u64 % modulus;
617    let fits_u64 = (N1::get_mod() - 1) as u128
618        + (N2::get_mod() - 1) as u128 * m1 as u128
619        + (N3::get_mod() - 1) as u128 * m2 as u128
620        <= u64::MAX as u128;
621    f.0.into_iter()
622        .zip(f.1)
623        .zip(f.2)
624        .map(|((c1, c2), c3)| {
625            let d1 = c1.inner();
626            let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
627            let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
628            let d3 = ((c3 - x) * t2).inner();
629            let value = if fits_u64 {
630                (d1 as u64 + d2 as u64 * m1 + d3 as u64 * m2) % modulus
631            } else {
632                ((d1 as u128 + d2 as u128 * m1 as u128 + d3 as u128 * m2 as u128) % modulus as u128)
633                    as u64
634            };
635            MInt::<M>::from(value as u32)
636        })
637        .collect()
638}

640impl<M, N1, N2, N3> ConvolveSteps for Convolve<(M, (N1, N2, N3))>
641where
642    M: MIntConvert + MIntConvert<u32>,

651    }

667    }

670        Convolve::<N2>::multiply(&mut f.1, &g.1);
671        Convolve::<N3>::multiply(&mut f.2, &g.2);

676        }

678            return convolve_naive(&a, &b);
679        }
680        if (a.len() + b.len() - 1).next_power_of_two() <= 1 << 20 {
681            crate::avx_helper!(@dispatch_avx2_fma return unsafe {
682                super::mint_fft_convolve::convolve_mint_avx2(a, b)
683            }, ());
684        }
685        let a_len = a.len();
686        let b_len = b.len();
687        let a = convert_crt_input(a, a_len);
688        let b = convert_crt_input(b, b_len);
689        reconstruct_mint_crt((
690            Convolve::<N1>::convolve(a.0, b.0),
691            Convolve::<N2>::convolve(a.1, b.1),
692            Convolve::<N3>::convolve(a.2, b.2),
693        ))
694    }

733        let t2 = (m1_3 * MInt::<N3>::new(N2::get_mod())).inv();
734        let m2 = m1 * N2::get_mod() as u64;
735        Convolve::<N1>::inverse_transform(f.0, len)
736            .into_iter()
737            .zip(Convolve::<N2>::inverse_transform(f.1, len))
738            .zip(Convolve::<N3>::inverse_transform(f.2, len))
739            .map(|((c1, c2), c3)| {
740                let d1 = c1.inner();
741                let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
742                let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
743                let d3 = ((c3 - x) * t2).inner();
744                (d1 as u64)
745                    .wrapping_add((d2 as u64).wrapping_mul(m1))
746                    .wrapping_add((d3 as u64).wrapping_mul(m2))
747            })
748            .collect()
749    }
750
751    fn multiply(f: &mut Self::F, g: &Self::F) {
752        Convolve::<N1>::multiply(&mut f.0, &g.0);
753        Convolve::<N2>::multiply(&mut f.1, &g.1);
754        Convolve::<N3>::multiply(&mut f.2, &g.2);
755    }
756
757    fn convolve(a: Self::T, b: Self::T) -> Self::T {
758        let max_len = Self::length(&a).max(Self::length(&b));
759        if max_len <= 300 || Self::length(&a).min(Self::length(&b)) <= 60 {
760            let a_wrapping: &[Wrapping<u64>] =
761                unsafe { std::slice::from_raw_parts(a.as_ptr().cast(), a.len()) };
762            let b_wrapping: &[Wrapping<u64>] =
763                unsafe { std::slice::from_raw_parts(b.as_ptr().cast(), b.len()) };
764            let mut c = std::mem::ManuallyDrop::new(if max_len <= 300 {
765                convolve_karatsuba(a_wrapping, b_wrapping)
766            } else {
767                convolve_naive(a_wrapping, b_wrapping)
768            });
769            return unsafe { Vec::from_raw_parts(c.as_mut_ptr().cast(), c.len(), c.capacity()) };
770        }
771        let len = (Self::length(&a) + Self::length(&b)).saturating_sub(1);
772        if len.next_power_of_two() <= 1 << 21 {
773            let factor = Xorshift::new().rand64() | 1;
774            let mut inverse = factor;
775            for _ in 0..5 {
776                inverse = inverse.wrapping_mul(2u64.wrapping_sub(factor.wrapping_mul(inverse)));
777            }
778            crate::avx_helper!(@dispatch_avx2_fma return unsafe {
779                super::mint_fft_convolve::convolve_u64_avx2(a, b, factor, inverse)
780            }, ());
781            return convolve_u64_fft_scalar(a, b, factor, inverse);
782        }
783        let mut a = Self::transform(a, len);
784        let b = Self::transform(b, len);
785        Self::multiply(&mut a, &b);
786        Self::inverse_transform(a, len)
787    }
788}
789
790fn convolve_u64_fft_scalar(a: Vec<u64>, b: Vec<u64>, factor: u64, inverse: u64) -> Vec<u64> {
791    fn split(values: &[u64], factor: u64) -> [Vec<i64>; 4] {
792        let mut result = std::array::from_fn(|_| Vec::with_capacity(values.len()));
793        let mut multiplier = 1u64;
794        for &value in values {
795            let mut value = value.wrapping_mul(multiplier);
796            for part in &mut result {
797                let digit = value as i16;
798                part.push(digit as i64);
799                value = (value >> 16).wrapping_add(u64::from(digit < 0));
800            }
801            multiplier = multiplier.wrapping_mul(factor);
802        }
803        result
804    }
805
806    let len = a.len() + b.len() - 1;
807    let fa = split(&a, factor).map(|a| ConvolveRealFft::transform(a, len));
808    drop(a);
809    let fb = split(&b, factor).map(|b| ConvolveRealFft::transform(b, len));
810    drop(b);
811    let values: [Vec<i64>; 4] = std::array::from_fn(|part| {
812        let mut sum = fa[0].clone();
813        ConvolveRealFft::multiply(&mut sum, &fb[part]);
814        for left in 1..=part {
815            let mut product = fa[left].clone();
816            ConvolveRealFft::multiply(&mut product, &fb[part - left]);
817            for (sum, product) in sum.iter_mut().zip(product) {
818                *sum += product;
819            }
820        }
821        ConvolveRealFft::inverse_transform(sum, len)
822    });
823    let mut multiplier = 1u64;
824    (0..len)
825        .map(|i| {
826            let value = (values[0][i] as u64)
827                .wrapping_add((values[1][i] as u64) << 16)
828                .wrapping_add((values[2][i] as u64) << 32)
829                .wrapping_add((values[3][i] as u64) << 48)
830                .wrapping_mul(multiplier);
831            multiplier = multiplier.wrapping_mul(inverse);
832            value
833        })
834        .collect()
835}

837pub trait NttReuse: ConvolveSteps {
838    const MULTIPLE: bool = true;
839
840    /// Transforms coefficients into the usual NTT frequency order.
841    fn transform_ntt(t: Self::T, len: usize) -> Self::F {
842        Self::transform(t, len)
843    }
844
845    /// Inverts a value produced by `transform_ntt`.
846    fn inverse_transform_ntt(f: Self::F, len: usize) -> Self::T {
847        Self::inverse_transform(f, len)
848    }
849
850    /// Extends a value produced by `transform_ntt` to twice its length.
851    fn ntt_doubling(f: Self::F) -> Self::F;
852
853    /// Extracts the even coefficients of `a(x) * b(-x)` in the usual NTT frequency order.
854    fn even_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F;
855
856    /// Extracts the odd coefficients of `a(x) * b(-x)` in the usual NTT frequency order.
857    fn odd_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F;
858
859    /// Multiplies a usual NTT transform by the corresponding prefix of another one.
860    fn multiply_prefix(f: &mut Self::F, g: &Self::F);
861
862    /// Adds the pointwise product of two usual NTT transforms to `sum`.
863    fn multiply_add(sum: &mut Self::F, f: &Self::F, g: &Self::F);
864
865    fn power_projection_step(
866        p_flat: Self::T,
867        q_flat: Self::T,
868        n: usize,
869        py: usize,
870        qy: usize,
871    ) -> (Self::T, Self::T) {
872        let base = n * 2;
873        let len_p = base * py;
874        let len_q = base * qy;
875        let len = (len_p + len_q - 1).max(len_q + len_q - 1);
876        let half = len.max(1).next_power_of_two() / 2;
877
878        let p_fft = Self::transform_ntt(p_flat, len);
879        let q_fft = Self::transform_ntt(q_flat, len);
880        let pr_fft = Self::odd_mul_normal_neg(&p_fft, &q_fft);
881        let qr_fft = Self::even_mul_normal_neg(&q_fft, &q_fft);
882        (
883            Self::inverse_transform_ntt(pr_fft, half),
884            Self::inverse_transform_ntt(qr_fft, half),
885        )
886    }
887}

889thread_local!(
890    static BIT_REVERSE: UnsafeCell<Vec<Vec<usize>>> = const { UnsafeCell::new(vec![]) };
891);
892
893impl<M> NttReuse for Convolve<M>
894where
895    M: Montgomery32NttModulus,
896{
897    const MULTIPLE: bool = false;
898
899    fn transform_ntt(mut t: Self::T, len: usize) -> Self::F {
900        t.resize_with(len.max(1).next_power_of_two(), Zero::zero);
901        ntt(&mut t);
902        t
903    }
904
905    fn inverse_transform_ntt(mut f: Self::F, len: usize) -> Self::T {
906        intt(&mut f);
907        f.truncate(len);
908        f
909    }
910
911    fn ntt_doubling(mut f: Self::F) -> Self::F {
912        let n = f.len();
913        let k = n.trailing_zeros() as usize;
914        let mut a = Self::inverse_transform_ntt(f.clone(), n);
915        let mut rot = MInt::<M>::one();
916        let zeta = MInt::<M>::new_unchecked(M::INFO.root[k + 1]);
917        for a in a.iter_mut() {
918            *a *= rot;
919            rot *= zeta;
920        }
921        f.extend(Self::transform_ntt(a, n));
922        f
923    }
924
925    fn even_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F {
926        assert_eq!(f.len(), g.len());
927        assert!(f.len().is_power_of_two());
928        assert!(f.len() >= 2);
929        let inv2 = MInt::<M>::from(2).inv();
930        let n = f.len() / 2;
931        (0..n)
932            .map(|i| (f[i << 1] * g[i << 1 | 1] + f[i << 1 | 1] * g[i << 1]) * inv2)
933            .collect()
934    }
935
936    fn odd_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F {
937        assert_eq!(f.len(), g.len());
938        assert!(f.len().is_power_of_two());
939        assert!(f.len() >= 2);
940        let mut inv2 = MInt::<M>::from(2).inv();
941        let n = f.len() / 2;
942        let k = f.len().trailing_zeros() as usize;
943        let mut h = vec![MInt::<M>::zero(); n];
944        let w = MInt::<M>::new_unchecked(M::INFO.inv_root[k]);
945        BIT_REVERSE.with(|br| {
946            let br = unsafe { &mut *br.get() };
947            if br.len() < k {
948                br.resize_with(k, Default::default);
949            }
950            let k = k - 1;
951            if br[k].is_empty() {
952                let mut v = vec![0; 1 << k];
953                for i in 0..1 << k {
954                    v[i] = (v[i >> 1] >> 1) | ((i & 1) << k.saturating_sub(1));
955                }
956                br[k] = v;
957            }
958            for &i in &br[k] {
959                h[i] = (f[i << 1] * g[i << 1 | 1] - f[i << 1 | 1] * g[i << 1]) * inv2;
960                inv2 *= w;
961            }
962        });
963        h
964    }
965
966    fn multiply_prefix(f: &mut Self::F, g: &Self::F) {
967        pointwise_multiply(f, g);
968    }
969
970    fn multiply_add(sum: &mut Self::F, f: &Self::F, g: &Self::F) {
971        assert!(sum.len() == f.len() && sum.len() == g.len());
972        pointwise_multiply_add(sum, f, g);
973    }
974
975    fn power_projection_step(
976        p_flat: Vec<MInt<M>>,
977        q_flat: Vec<MInt<M>>,
978        n: usize,
979        py: usize,
980        qy: usize,
981    ) -> (Vec<MInt<M>>, Vec<MInt<M>>) {
982        let high_degree = (qy - 1) * 2;
983        let rows = (py + qy - 1).max(high_degree).next_power_of_two();
984        let cols = n * 2;
985        let size = rows * cols;
986        let mut p = p_flat;
987        p.resize_with(size, MInt::<M>::zero);
988        ntt_rows(&mut p, cols);
989        ntt_batch(&mut p, cols);
990
991        let mut q = q_flat;
992        q.resize_with(size, MInt::<M>::zero);
993        ntt_rows(&mut q, cols);
994        let q_high = (rows == high_degree).then(|| q[(qy - 1) * cols..qy * cols].to_vec());
995        ntt_batch(&mut q, cols);
996
997        let half = cols / 2;
998        let mut odd_factor = vec![MInt::<M>::zero(); half];
999        let mut factor = MInt::<M>::from(2).inv();
1000        let k = cols.trailing_zeros() as usize;
1001        let w = MInt::<M>::new_unchecked(M::INFO.inv_root[k]);
1002        BIT_REVERSE.with(|br| {
1003            let br = unsafe { &mut *br.get() };
1004            if br.len() < k {
1005                br.resize_with(k, Default::default);
1006            }
1007            let k = k - 1;
1008            if br[k].is_empty() {
1009                let mut v = vec![0; 1 << k];
1010                for i in 0..1 << k {
1011                    v[i] = (v[i >> 1] >> 1) | ((i & 1) << k.saturating_sub(1));
1012                }
1013                br[k] = v;
1014            }
1015            for &i in &br[k] {
1016                odd_factor[i] = factor;
1017                factor *= w;
1018            }
1019        });
1020
1021        let mut pr = vec![MInt::<M>::zero(); rows * half];
1022        let mut qr = vec![MInt::<M>::zero(); rows * half];
1023        for i in 0..pr.len() {
1024            pr[i] = (p[i << 1] * q[i << 1 | 1] - p[i << 1 | 1] * q[i << 1])
1025                * odd_factor[i & (half - 1)];
1026            qr[i] = q[i << 1] * q[i << 1 | 1];
1027        }
1028        intt_batch(&mut pr, half);
1029        intt_rows(&mut pr, half);
1030        intt_batch(&mut qr, half);
1031        intt_rows(&mut qr, half);
1032
1033        if let Some(q_high) = q_high {
1034            let mut q_high_even = vec![MInt::<M>::zero(); half];
1035            for i in 0..half {
1036                q_high_even[i] = q_high[i << 1] * q_high[i << 1 | 1];
1037            }
1038            intt(&mut q_high_even);
1039            for (value, high) in qr.iter_mut().zip(&q_high_even) {
1040                *value -= *high;
1041            }
1042            qr.extend_from_slice(&q_high_even);
1043        }
1044        (pr, qr)
1045    }
