773    fn transform(t: Self::T, len: usize) -> Self::F {
774        let npot = len.max(1).next_power_of_two();
775        let mut f = (
776            MVec::<N1>::with_capacity(npot),

792    fn inverse_transform(f: Self::F, len: usize) -> Self::T {
793        let t1 = MInt::<N2>::new(N1::get_mod()).inv();
794        let m1 = N1::get_mod() as u64;
795        let m1_3 = MInt::<N3>::new(N1::get_mod());
796        let t2 = (m1_3 * MInt::<N3>::new(N2::get_mod())).inv();
797        let m2 = m1 * N2::get_mod() as u64;
798        Convolve::<N1>::inverse_transform(f.0, len)
799            .into_iter()
800            .zip(Convolve::<N2>::inverse_transform(f.1, len))
801            .zip(Convolve::<N3>::inverse_transform(f.2, len))
802            .map(|((c1, c2), c3)| {
803                let d1 = c1.inner();
804                let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
805                let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
806                let d3 = ((c3 - x) * t2).inner();
807                (d1 as u64)
808                    .wrapping_add((d2 as u64).wrapping_mul(m1))
809                    .wrapping_add((d3 as u64).wrapping_mul(m2))
810            })
811            .collect()
812    }

820    fn convolve(a: Self::T, b: Self::T) -> Self::T {
821        let max_len = Self::length(&a).max(Self::length(&b));
822        if max_len <= 300 || Self::length(&a).min(Self::length(&b)) <= 60 {
823            let a_wrapping: &[Wrapping<u64>] =
824                unsafe { std::slice::from_raw_parts(a.as_ptr().cast(), a.len()) };
825            let b_wrapping: &[Wrapping<u64>] =
826                unsafe { std::slice::from_raw_parts(b.as_ptr().cast(), b.len()) };
827            let mut c = std::mem::ManuallyDrop::new(if max_len <= 300 {
828                convolve_karatsuba(a_wrapping, b_wrapping)
829            } else {
830                convolve_naive(a_wrapping, b_wrapping)
831            });
832            return unsafe { Vec::from_raw_parts(c.as_mut_ptr().cast(), c.len(), c.capacity()) };
833        }
834        let len = (Self::length(&a) + Self::length(&b)).saturating_sub(1);
835        if len.next_power_of_two() <= 1 << 21 {
836            let factor = Xorshift::new().rand64() | 1;
837            let mut inverse = factor;
838            for _ in 0..5 {
839                inverse = inverse.wrapping_mul(2u64.wrapping_sub(factor.wrapping_mul(inverse)));
840            }
841            crate::avx_helper!(@dispatch_avx2_fma return unsafe {
842                super::mint_fft_convolve::convolve_u64_avx2(a, b, factor, inverse)
843            }, ());
844            return convolve_u64_fft_scalar(a, b, factor, inverse);
845        }
846        let mut a = Self::transform(a, len);
847        let b = Self::transform(b, len);
848        Self::multiply(&mut a, &b);
849        Self::inverse_transform(a, len)
850    }

853fn convolve_u64_fft_scalar(a: Vec<u64>, b: Vec<u64>, factor: u64, inverse: u64) -> Vec<u64> {
854    fn split(values: &[u64], factor: u64) -> [Vec<i64>; 4] {
855        let mut result = std::array::from_fn(|_| Vec::with_capacity(values.len()));
856        let mut multiplier = 1u64;
857        for &value in values {
858            let mut value = value.wrapping_mul(multiplier);
859            for part in &mut result {
860                let digit = value as i16;
861                part.push(digit as i64);
862                value = (value >> 16).wrapping_add(u64::from(digit < 0));
863            }
864            multiplier = multiplier.wrapping_mul(factor);
865        }
866        result
867    }
868
869    let len = a.len() + b.len() - 1;
870    let fa = split(&a, factor).map(|a| ConvolveRealFft::transform(a, len));
871    drop(a);
872    let fb = split(&b, factor).map(|b| ConvolveRealFft::transform(b, len));
873    drop(b);
874    let values: [Vec<i64>; 4] = std::array::from_fn(|part| {
875        let mut sum = fa[0].clone();
876        ConvolveRealFft::multiply(&mut sum, &fb[part]);
877        for left in 1..=part {
878            let mut product = fa[left].clone();
879            ConvolveRealFft::multiply(&mut product, &fb[part - left]);
880            for (sum, product) in sum.iter_mut().zip(product) {
881                *sum += product;
882            }
883        }
884        ConvolveRealFft::inverse_transform(sum, len)
885    });
886    let mut multiplier = 1u64;
887    (0..len)
888        .map(|i| {
889            let value = (values[0][i] as u64)
890                .wrapping_add((values[1][i] as u64) << 16)
891                .wrapping_add((values[2][i] as u64) << 32)
892                .wrapping_add((values[3][i] as u64) << 48)
893                .wrapping_mul(multiplier);
894            multiplier = multiplier.wrapping_mul(inverse);
895            value
896        })
897        .collect()
898}

903    /// Transforms coefficients into the usual NTT frequency order.
904    fn transform_ntt(t: Self::T, len: usize) -> Self::F {
905        Self::transform(t, len)
906    }
907
908    /// Inverts a value produced by `transform_ntt`.
909    fn inverse_transform_ntt(f: Self::F, len: usize) -> Self::T {

913    /// Extends a value produced by `transform_ntt` to twice its length.
914    fn ntt_doubling(f: Self::F) -> Self::F;

916    /// Extracts the even coefficients of `a(x) * b(-x)` in the usual NTT frequency order.
917    fn even_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F;
918
919    /// Extracts the odd coefficients of `a(x) * b(-x)` in the usual NTT frequency order.
920    fn odd_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F;
921
922    /// Multiplies a usual NTT transform by the corresponding prefix of another one.
923    fn multiply_prefix(f: &mut Self::F, g: &Self::F);
924
925    /// Adds the pointwise product of two usual NTT transforms to `sum`.
926    fn multiply_add(sum: &mut Self::F, f: &Self::F, g: &Self::F);
927
928    fn power_projection_step(
929        p_flat: Self::T,
930        q_flat: Self::T,
931        n: usize,
932        py: usize,
933        qy: usize,
934    ) -> (Self::T, Self::T) {
935        let base = n * 2;
936        let len_p = base * py;
937        let len_q = base * qy;
938        let len = (len_p + len_q - 1).max(len_q + len_q - 1);
939        let half = len.max(1).next_power_of_two() / 2;
940
941        let p_fft = Self::transform_ntt(p_flat, len);
942        let q_fft = Self::transform_ntt(q_flat, len);
943        let pr_fft = Self::odd_mul_normal_neg(&p_fft, &q_fft);
944        let qr_fft = Self::even_mul_normal_neg(&q_fft, &q_fft);
945        (
946            Self::inverse_transform_ntt(pr_fft, half),
947            Self::inverse_transform_ntt(qr_fft, half),
948        )
949    }

974    fn ntt_doubling(mut f: Self::F) -> Self::F {
975        let n = f.len();
976        let k = n.trailing_zeros() as usize;
977        let mut a = Self::inverse_transform_ntt(f.clone(), n);
978        let mut rot = MInt::<M>::one();
979        let zeta = MInt::<M>::new_unchecked(M::INFO.root[k + 1]);
980        for a in a.iter_mut() {
981            *a *= rot;
982            rot *= zeta;
983        }

988    fn even_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F {
989        assert_eq!(f.len(), g.len());
990        assert!(f.len().is_power_of_two());
991        assert!(f.len() >= 2);
992        let inv2 = MInt::<M>::from(2).inv();
993        let n = f.len() / 2;
994        (0..n)
995            .map(|i| (f[i << 1] * g[i << 1 | 1] + f[i << 1 | 1] * g[i << 1]) * inv2)
996            .collect()
997    }

999    fn odd_mul_normal_neg(f: &Self::F, g: &Self::F) -> Self::F {
1000        assert_eq!(f.len(), g.len());
1001        assert!(f.len().is_power_of_two());
1002        assert!(f.len() >= 2);
1003        let mut inv2 = MInt::<M>::from(2).inv();
1004        let n = f.len() / 2;
1005        let k = f.len().trailing_zeros() as usize;
1006        let mut h = vec![MInt::<M>::zero(); n];
1007        let w = MInt::<M>::new_unchecked(M::INFO.inv_root[k]);
1008        BIT_REVERSE.with(|br| {
1009            let br = unsafe { &mut *br.get() };
1010            if br.len() < k {
1011                br.resize_with(k, Default::default);
1012            }
1013            let k = k - 1;
1014            if br[k].is_empty() {
1015                let mut v = vec![0; 1 << k];
1016                for i in 0..1 << k {
1017                    v[i] = (v[i >> 1] >> 1) | ((i & 1) << k.saturating_sub(1));
1018                }
1019                br[k] = v;
1020            }
1021            for &i in &br[k] {
1022                h[i] = (f[i << 1] * g[i << 1 | 1] - f[i << 1 | 1] * g[i << 1]) * inv2;
1023                inv2 *= w;
1024            }
1025        });
1026        h
1027    }

1038    fn power_projection_step(
1039        p_flat: Vec<MInt<M>>,
1040        q_flat: Vec<MInt<M>>,
1041        n: usize,
1042        py: usize,
1043        qy: usize,
1044    ) -> (Vec<MInt<M>>, Vec<MInt<M>>) {
1045        let high_degree = (qy - 1) * 2;
1046        let rows = (py + qy - 1).max(high_degree).next_power_of_two();
1047        let cols = n * 2;
1048        let size = rows * cols;
1049        let mut p = p_flat;
1050        p.resize_with(size, MInt::<M>::zero);
1051        ntt_rows(&mut p, cols);
1052        ntt_batch(&mut p, cols);
1053
1054        let mut q = q_flat;
1055        q.resize_with(size, MInt::<M>::zero);
1056        ntt_rows(&mut q, cols);
1057        let q_high = (rows == high_degree).then(|| q[(qy - 1) * cols..qy * cols].to_vec());
1058        ntt_batch(&mut q, cols);
1059
1060        let half = cols / 2;
1061        let mut odd_factor = vec![MInt::<M>::zero(); half];
1062        let mut factor = MInt::<M>::from(2).inv();
1063        let k = cols.trailing_zeros() as usize;
1064        let w = MInt::<M>::new_unchecked(M::INFO.inv_root[k]);
1065        BIT_REVERSE.with(|br| {
1066            let br = unsafe { &mut *br.get() };
1067            if br.len() < k {
1068                br.resize_with(k, Default::default);
1069            }
1070            let k = k - 1;
1071            if br[k].is_empty() {
1072                let mut v = vec![0; 1 << k];
1073                for i in 0..1 << k {
1074                    v[i] = (v[i >> 1] >> 1) | ((i & 1) << k.saturating_sub(1));
1075                }
1076                br[k] = v;
1077            }
1078            for &i in &br[k] {
1079                odd_factor[i] = factor;
1080                factor *= w;
1081            }
1082        });
1083
1084        let mut pr = vec![MInt::<M>::zero(); rows * half];
1085        let mut qr = vec![MInt::<M>::zero(); rows * half];
1086        for i in 0..pr.len() {
1087            pr[i] = (p[i << 1] * q[i << 1 | 1] - p[i << 1 | 1] * q[i << 1])
1088                * odd_factor[i & (half - 1)];
1089            qr[i] = q[i << 1] * q[i << 1 | 1];
1090        }
1091        intt_batch(&mut pr, half);
1092        intt_rows(&mut pr, half);
1093        intt_batch(&mut qr, half);
1094        intt_rows(&mut qr, half);
1095
1096        if let Some(q_high) = q_high {
1097            let mut q_high_even = vec![MInt::<M>::zero(); half];
1098            for i in 0..half {
1099                q_high_even[i] = q_high[i << 1] * q_high[i << 1 | 1];
1100            }
1101            intt(&mut q_high_even);
1102            for (value, high) in qr.iter_mut().zip(&q_high_even) {
1103                *value -= *high;
1104            }
1105            qr.extend_from_slice(&q_high_even);
1106        }
1107        (pr, qr)
1108    }
1109}
