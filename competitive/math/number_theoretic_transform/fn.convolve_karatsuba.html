559fn convolve_large_ntt<M>(a: Vec<MInt<M>>, b: Vec<MInt<M>>) -> Vec<MInt<M>>
560where
561    M: Montgomery32NttModulus,
562{
563    let len = a.len() + b.len() - 1;
564    let ntt_len = 1usize << M::RANK;
565    let block_len = ntt_len / 2;
566    let same = a == b;
567    let fa: Vec<_> = a
568        .chunks(block_len)
569        .map(|a| Convolve::<M>::transform_ntt(a.to_vec(), ntt_len))
570        .collect();
571    let fb: Option<Vec<_>> = if same {
572        None
573    } else {
574        Some(
575            b.chunks(block_len)
576                .map(|b| Convolve::<M>::transform_ntt(b.to_vec(), ntt_len))
577                .collect(),
578        )
579    };
580    let b_blocks = fb.as_ref().map_or(fa.len(), Vec::len);
581    let mut result = vec![MInt::<M>::zero(); len];
582    for diagonal in 0..fa.len() + b_blocks - 1 {
583        let mut spectrum = vec![MInt::<M>::zero(); ntt_len];
584        let start = diagonal.saturating_sub(b_blocks - 1);
585        for i in start..=diagonal.min(fa.len() - 1) {
586            let j = diagonal - i;
587            let g = if let Some(fb) = &fb { &fb[j] } else { &fa[j] };
588            pointwise_multiply_add(&mut spectrum, &fa[i], g);
589        }
590        spectrum = Convolve::<M>::inverse_transform_ntt(spectrum, ntt_len);
591        let offset = diagonal * block_len;
592        for (result, value) in result[offset..].iter_mut().zip(spectrum) {
593            *result += value;
594        }
595    }
596    result
597}

607    fn length(t: &Self::T) -> usize {

609    }
610    fn transform(mut t: Self::T, len: usize) -> Self::F {
611        t.resize_with(len.max(1).next_power_of_two(), Zero::zero);

620    fn inverse_transform(mut f: Self::F, len: usize) -> Self::T {
621        #[cfg(target_arch = "x86_64")]
622        if use_block_ntt::<M>(f.len()) {
623            unsafe { ntt_simd::inverse_transform_blocks_avx2(&mut f) };
624            f.truncate(len);
625            return f;
626        }
627        intt(&mut f);
628        f.truncate(len);
629        f

631    fn multiply(f: &mut Self::F, g: &Self::F) {
632        assert_eq!(f.len(), g.len());
633        #[cfg(target_arch = "x86_64")]
634        if use_block_ntt::<M>(f.len()) {
635            unsafe { ntt_simd::multiply_blocks_avx2(f, g) };
636            return;
637        }
638        pointwise_multiply(f, g);
639    }
640    fn square(t: Self::T, len: usize) -> Self::T {
641        let mut f = Self::transform(t, len);
642        let g = f.clone();
643        Self::multiply(&mut f, &g);
644        Self::inverse_transform(f, len)
645    }
646    fn convolve(mut a: Self::T, mut b: Self::T) -> Self::T {
647        let (threshold, naive_threshold) = (100, 60);
648        #[cfg(target_arch = "x86_64")]
649        let (threshold, naive_threshold) = if use_block_ntt::<M>(64) {
650            (
651                60,
652                if M::RANK >= 13 && a.len().max(b.len()) <= 4096 {
653                    18
654                } else if M::RANK >= 19 && a.len().max(b.len()) <= 262144 {
655                    32
656                } else {
657                    34
658                },
659            )
660        } else {
661            (threshold, naive_threshold)
662        };
663        if Self::length(&a).max(Self::length(&b)) <= threshold {
664            return convolve_karatsuba(&a, &b);
665        }
666        if Self::length(&a).min(Self::length(&b)) <= naive_threshold {
667            return convolve_naive(&a, &b);
668        }
669        let len = (Self::length(&a) + Self::length(&b)).saturating_sub(1);
670        let size = len.max(1).next_power_of_two();
671        if size > 1usize << M::RANK {
672            return convolve_large_ntt(a, b);
673        }
674        if len <= size / 2 + 2 {
675            let xa = a.pop().unwrap();
676            let xb = b.pop().unwrap();
677            let mut c = vec![MInt::<M>::zero(); len];
678            *c.last_mut().unwrap() = xa * xb;
679            for (a, c) in a.iter().zip(&mut c[b.len()..]) {
680                *c += *a * xb;
681            }
682            for (b, c) in b.iter().zip(&mut c[a.len()..]) {
683                *c += *b * xa;
684            }
685            let d = Self::convolve(a, b);
686            for (d, c) in d.into_iter().zip(&mut c) {
687                *c += d;
688            }
689            return c;
690        }
691        let same = a == b;
692        #[cfg(target_arch = "x86_64")]
693        if use_block_ntt::<M>(size) {
694            a.resize_with(size, Zero::zero);
695            b.resize_with(size, Zero::zero);
696            unsafe { ntt_simd::convolve_blocks_avx2(&mut a, &mut b, same) };
697            a.truncate(len);
698            return a;
699        }
700        let mut a = Self::transform(a, len);
701        if same {
702            for a in a.iter_mut() {
703                *a *= *a;
704            }
705        } else {
706            let b = Self::transform(b, len);
707            Self::multiply(&mut a, &b);
708        }
709        Self::inverse_transform(a, len)
710    }
711}

736fn reconstruct_mint_crt<M, N1, N2, N3>(f: (MVec<N1>, MVec<N2>, MVec<N3>)) -> MVec<M>
737where
738    M: MIntConvert + MIntConvert<u32>,
739    N1: Montgomery32NttModulus,
740    N2: Montgomery32NttModulus,
741    N3: Montgomery32NttModulus,
742{
743    let t1 = MInt::<N2>::new(N1::get_mod()).inv();
744    let m1_3 = MInt::<N3>::new(N1::get_mod());
745    let t2 = (m1_3 * MInt::<N3>::new(N2::get_mod())).inv();
746    let modulus = <M as MIntConvert<u32>>::mod_into() as u64;
747    let m1 = N1::get_mod() as u64;
748    let m2 = m1 * N2::get_mod() as u64 % modulus;
749    let fits_u64 = (N1::get_mod() - 1) as u128
750        + (N2::get_mod() - 1) as u128 * m1 as u128
751        + (N3::get_mod() - 1) as u128 * m2 as u128
752        <= u64::MAX as u128;
753    f.0.into_iter()
754        .zip(f.1)
755        .zip(f.2)
756        .map(|((c1, c2), c3)| {
757            let d1 = c1.inner();
758            let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
759            let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
760            let d3 = ((c3 - x) * t2).inner();
761            let value = if fits_u64 {
762                (d1 as u64 + d2 as u64 * m1 + d3 as u64 * m2) % modulus
763            } else {
764                ((d1 as u128 + d2 as u128 * m1 as u128 + d3 as u128 * m2 as u128) % modulus as u128)
765                    as u64
766            };
767            MInt::<M>::from(value as u32)
768        })
769        .collect()
770}

846        let mut f = (
847            MVec::<N1>::with_capacity(npot),
848            MVec::<N2>::with_capacity(npot),
849            MVec::<N3>::with_capacity(npot),

857            Convolve::<N1>::transform(f.0, npot),

860        )

863    fn inverse_transform(f: Self::F, len: usize) -> Self::T {
864        let t1 = MInt::<N2>::new(N1::get_mod()).inv();
865        let m1 = N1::get_mod() as u64;
866        let m1_3 = MInt::<N3>::new(N1::get_mod());
867        let t2 = (m1_3 * MInt::<N3>::new(N2::get_mod())).inv();
868        let m2 = m1 * N2::get_mod() as u64;
869        Convolve::<N1>::inverse_transform(f.0, len)
870            .into_iter()
871            .zip(Convolve::<N2>::inverse_transform(f.1, len))
872            .zip(Convolve::<N3>::inverse_transform(f.2, len))
873            .map(|((c1, c2), c3)| {
874                let d1 = c1.inner();
875                let d2 = ((c2 - MInt::<N2>::from(d1)) * t1).inner();
876                let x = MInt::<N3>::new(d1) + MInt::<N3>::new(d2) * m1_3;
877                let d3 = ((c3 - x) * t2).inner();
878                (d1 as u64)
879                    .wrapping_add((d2 as u64).wrapping_mul(m1))
880                    .wrapping_add((d3 as u64).wrapping_mul(m2))
881            })
882            .collect()
883    }

887        Convolve::<N2>::multiply(&mut f.1, &g.1);
888        Convolve::<N3>::multiply(&mut f.2, &g.2);
889    }

892        let max_len = Self::length(&a).max(Self::length(&b));

921    }
