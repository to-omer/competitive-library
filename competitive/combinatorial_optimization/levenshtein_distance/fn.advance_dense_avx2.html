289unsafe fn levenshtein_distance_simd<I, F>(x: I, y: I, avx512: bool, mut row: F) -> usize
290where
291    I: ExactSizeIterator,
292    I::Item: Eq + Hash,
293    F: FnMut(usize),
294{
295    let lanes = if avx512 { 8 } else { 4 };
296    let m = x.len();
297    let n = y.len();
298    let mut ids = HashMap::new();
299    let mut pattern: Vec<_> = x
300        .map(|value| {
301            let id = ids.len();
302            *ids.entry(value).or_insert(id)
303        })
304        .collect();
305    let absent = ids.len();
306    // Padding covers the staggered lanes when short mask tiles use gather.
307    let mut text = vec![absent; n + 14];
308    for (id, value) in text[7..n + 7].iter_mut().zip(y) {
309        *id = ids.get(&value).copied().unwrap_or(absent);
310    }
311    drop(ids);
312    let mut offsets = vec![usize::MAX; absent + 1];
313    for &id in &text[7..n + 7] {
314        offsets[id] = 0;
315    }
316    let shared = offsets[..absent]
317        .iter()
318        .filter(|&&offset| offset == 0)
319        .count();
320    if shared == 0 {
321        for len in 1..=n {
322            row(m.max(len));
323        }
324        return m.max(n);
325    }
326    // Bound dense mask storage by O(m + n), even for a large alphabet.
327    let stride = (m.saturating_add(n) / (shared + 4)).max(lanes) / lanes * lanes;
328    let stride = stride.min(m.div_ceil(64).next_multiple_of(lanes));
329    // Short tiles favor staggered lanes over scanning carries between words.
330    let dense = stride >= 32;
331    let stride = if dense { stride } else { lanes };
332    let mut mask_len = 0;
333    for offset in &mut offsets[..absent] {
334        if *offset == 0 {
335            *offset = mask_len;
336            mask_len += stride;
337        }
338    }
339    offsets[absent] = mask_len;
340    // Pattern-only values must not overwrite the zero mask for text-only values.
341    for offset in &mut offsets[..absent] {
342        if *offset == usize::MAX {
343            *offset = mask_len + stride;
344        }
345    }
346    for id in pattern.iter_mut().chain(&mut text) {
347        *id = offsets[*id];
348    }
349    drop(offsets);
350    let mut horizontal = vec![0; n + 14];
351    horizontal[7..n + 7].fill(1);
352    if dense {
353        let storage_len = mask_len + 4 * stride;
354        let mut storage = vec![0u64; storage_len + 7];
355        let alignment = storage.as_ptr().align_offset(64);
356        let storage = &mut storage[alignment..alignment + storage_len];
357        let (masks, state) = storage.split_at_mut(mask_len + 2 * stride);
358        let (positive, negative) = state.split_at_mut(stride);
359        // SAFETY: the caller checked the requested ISA. Storage starts on a
360        // 64-byte boundary; every mask/state row has a whole number of vectors.
361        unsafe {
362            if avx512 {
363                advance_dense_avx512(
364                    &pattern,
365                    &text[7..n + 7],
366                    &mut horizontal[7..n + 7],
367                    masks,
368                    positive,
369                    negative,
370                );
371            } else {
372                advance_dense_avx2(
373                    &pattern,
374                    &text[7..n + 7],
375                    &mut horizontal[7..n + 7],
376                    masks,
377                    positive,
378                    negative,
379                );
380            }
381        }
382    } else {
383        let mut masks = vec![0; mask_len + 2 * lanes];
384        let mut start = 0;
385        if avx512 {
386            start = pattern.len() / 512 * 512;
387            // SAFETY: the caller checked AVX-512F. Mask offsets identify complete
388            // eight-word rows, and text/horizontal have seven padding entries on each side.
389            unsafe {
390                advance_avx512(&pattern[..start], &text, &mut horizontal, &mut masks);
391            }
392        }
393        let end = pattern.len() / 256 * 256;
394        // SAFETY: the caller checked AVX2. The same padding and offset bounds apply;
395        // each mask row has at least four words, and pattern chunks are complete.
396        unsafe {
397            advance_avx2(&pattern[start..end], &text, &mut horizontal, &mut masks);
398        }
399        for block in pattern[end..].chunks(64) {
400            for (i, &id) in block.iter().enumerate() {
401                masks[id] |= 1 << i;
402            }
403            let shift = block.len() - 1;
404            let (mut pv, mut mv) = (u64::MAX, 0);
405            for (&id, h) in text[7..n + 7].iter().zip(&mut horizontal[7..n + 7]) {
406                let neg = u64::from(*h < 0);
407                let pos = u64::from(*h > 0);
408                let eq = masks[id];
409                let xv = eq | mv;
410                let eq = eq | neg;
411                let xh = ((eq & pv).wrapping_add(pv) ^ pv) | eq;
412                let ph = mv | !(xh | pv);
413                let mh = pv & xh;
414                *h = ((ph >> shift) & 1) as i8 - ((mh >> shift) & 1) as i8;
415                let ph = (ph << 1) | pos;
416                let mh = (mh << 1) | neg;
417                pv = mh | !(xv | ph);
418                mv = ph & xv;
419            }
420            for &id in block {
421                masks[id] = 0;
422            }
423        }
424    }
425    let mut delta = 0isize;
426    for &h in &horizontal[7..n + 7] {
427        delta += h as isize;
428        row((m as isize + delta) as usize);
429    }
430    (m as isize + delta) as usize
