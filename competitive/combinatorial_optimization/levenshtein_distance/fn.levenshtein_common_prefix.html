7pub fn levenshtein_distance<T>(x: &[T], y: &[T]) -> usize
8where
9    T: Eq + Hash,
10{
11    let (x, y) = if x.len() <= y.len() { (x, y) } else { (y, x) };
12    if x.is_empty() {
13        return y.len();
14    }
15    let prefix = levenshtein_common_prefix(x, y);
16    let (x, y) = (&x[prefix..], &y[prefix..]);
17    if x.is_empty() {
18        return y.len();
19    }
20    let suffix = if x.last() == y.last() {
21        let blocks = x
22            .rchunks_exact(32)
23            .zip(y.rchunks_exact(32))
24            .take_while(|(a, b)| a == b)
25            .count();
26        let suffix = blocks * 32;
27        suffix
28            + x[..x.len() - suffix]
29                .iter()
30                .rev()
31                .zip(y[..y.len() - suffix].iter().rev())
32                .take_while(|(a, b)| a == b)
33                .count()
34    } else {
35        0
36    };
37    let (x, y) = (&x[..x.len() - suffix], &y[..y.len() - suffix]);
38    if x.is_empty() {
39        return y.len();
40    }
41    let work = y.len().saturating_mul(x.len().div_ceil(64));
42    if work >= 65_536
43        && let Some(distance) = levenshtein_wavefront(x, y, (work / 64).clamp(2048, 1 << 22))
44    {
45        return distance;
46    }
47    levenshtein_distance_impl(x.iter(), y.iter(), |_| {})
48}

70fn levenshtein_common_prefix<T>(x: &[T], y: &[T]) -> usize
71where
72    T: Eq,
73{
74    if x.first() != y.first() {
75        return 0;
76    }
77    let blocks = x
78        .as_chunks::<32>()
79        .0
80        .iter()
81        .zip(y.as_chunks().0)
82        .take_while(|(a, b)| a == b)
83        .count();
84    let prefix = blocks * 32;
85    prefix
86        + x[prefix..]
87            .iter()
88            .zip(&y[prefix..])
89            .take_while(|(a, b)| a == b)
90            .count()
91}

93fn levenshtein_wavefront<T>(x: &[T], y: &[T], mut budget: usize) -> Option<usize>
94where
95    T: Eq,
96{
97    let limit = budget.isqrt().min(x.len().max(y.len()));
98    if x.len().abs_diff(y.len()) > limit {
99        return None;
100    }
101    let first = levenshtein_common_prefix(x, y);
102    if first == x.len() && first == y.len() {
103        return Some(0);
104    }
105    budget = budget.checked_sub(1 + first / 32)?;
106    let base = limit + 1;
107    let mut previous = vec![-1isize; 2 * limit + 3];
108    let mut current = previous.clone();
109    previous[base] = first as isize;
110    for distance in 1..=limit {
111        for diagonal in -(distance.min(y.len()) as isize)..=distance.min(x.len()) as isize {
112            budget = budget.checked_sub(1)?;
113            let index = (base as isize + diagonal) as usize;
114            let delete = previous[index - 1];
115            let substitute = previous[index];
116            let insert = previous[index + 1];
117            let mut position = -1;
118            if delete >= 0 && delete < x.len() as isize {
119                position = delete + 1;
120            }
121            if substitute >= 0
122                && substitute < x.len() as isize
123                && substitute - diagonal < y.len() as isize
124            {
125                position = position.max(substitute + 1);
126            }
127            if insert >= 0 && insert - (diagonal + 1) < y.len() as isize {
128                position = position.max(insert);
129            }
130            if position >= 0 {
131                let matched = levenshtein_common_prefix(
132                    &x[position as usize..],
133                    &y[(position - diagonal) as usize..],
134                );
135                position += matched as isize;
136                if position == x.len() as isize && position - diagonal == y.len() as isize {
137                    return Some(distance);
138                }
139                // One match run can exceed the budget, adding at most linear work.
140                budget = budget.checked_sub(1 + matched / 32)?;
141            }
142            current[index] = position;
143        }
144        std::mem::swap(&mut previous, &mut current);
145    }
146    None
147}
