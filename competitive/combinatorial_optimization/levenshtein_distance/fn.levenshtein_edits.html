38fn levenshtein_edits<T>(x: &[T], y: &[T], flipped: bool, edits: &mut Vec<LevenshteinEdit>) -> usize
39where
40    T: Eq + Hash,
41{
42    if x.len() > y.len() {
43        return levenshtein_edits(y, x, !flipped, edits);
44    }
45    if x.len() <= 1 {
46        let insert = if flipped {
47            LevenshteinEdit::Delete
48        } else {
49            LevenshteinEdit::Insert
50        };
51        if x.is_empty() {
52            edits.extend(std::iter::repeat_n(insert, y.len()));
53            return y.len();
54        }
55        let position = y.iter().position(|value| value == &x[0]);
56        let index = position.unwrap_or(0);
57        edits.extend(std::iter::repeat_n(insert, index));
58        edits.push(if position.is_some() {
59            LevenshteinEdit::Match
60        } else {
61            LevenshteinEdit::Substitute
62        });
63        edits.extend(std::iter::repeat_n(insert, y.len() - index - 1));
64        return y.len() - usize::from(position.is_some());
65    }
66    let middle = x.len() / 2;
67    let mut forward = Vec::with_capacity(y.len() + 1);
68    forward.push(middle);
69    levenshtein_distance_impl(x[..middle].iter(), y.iter(), |score| forward.push(score));
70    let mut split = y.len();
71    let mut distance = forward[split] + x.len() - middle;
72    let mut index = y.len();
73    // Reversed prefix scores are suffix scores in the original coordinates.
74    levenshtein_distance_impl(x[middle..].iter().rev(), y.iter().rev(), |score| {
75        index -= 1;
76        let candidate = forward[index] + score;
77        if candidate < distance {
78            distance = candidate;
79            split = index;
80        }
81    });
82    drop(forward);
83    levenshtein_edits(&x[..middle], &y[..split], flipped, edits);
84    levenshtein_edits(&x[middle..], &y[split..], flipped, edits);
85    distance
