128unsafe fn normalize_avx2<M>(a: &mut [u32])
129where
130    M: Montgomery32NttModulus,
131{
132    let mod_vec = _mm256_set1_epi32(M::MOD as i32);
133    let mut i = 0;
134    while i + 8 <= a.len() {
135        let x = _mm256_loadu_si256(a.as_ptr().add(i) as *const __m256i);
136        let y = _mm256_min_epu32(x, _mm256_sub_epi32(x, mod_vec));
137        _mm256_storeu_si256(a.as_mut_ptr().add(i) as *mut __m256i, y);
138        i += 8;
139    }
140    while i < a.len() {
141        a[i] = normalize_scalar::<M>(a[i]);
142        i += 1;
143    }
