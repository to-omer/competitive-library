51fn dot_product<M, const REVERSE: bool>(x: &[MInt<M>], y: &[MInt<M>]) -> MInt<M>
52where
53    M: MontgomeryReduction32,
54{
55    assert_eq!(x.len(), y.len());
56    let reduce = |sum: u128| {
57        // Montgomery reduction accepts values below modulus * 2^32.
58        let value = if sum < (M::MOD as u128) << 32 {
59            sum as u64
60        } else if sum <= u64::MAX as u128 {
61            sum as u64 % M::MOD as u64
62        } else {
63            (sum % M::MOD as u128) as u64
64        };
65        MInt::new_unchecked(M::reduce(value))
66    };
67    // SAFETY: MInt is transparent over u32; retain the Montgomery residues.
68    let (x, y) = unsafe {
69        (
70            std::slice::from_raw_parts(x.as_ptr().cast::<u32>(), x.len()),
71            std::slice::from_raw_parts(y.as_ptr().cast::<u32>(), y.len()),
72        )
73    };
74    #[cfg(target_arch = "x86_64")]
75    if x.len() >= 64 && M::MOD < 1 << 31 {
76        if avx512_enabled() && avx512_supported() {
77            let sum = unsafe { simd::dot_product_avx512::<REVERSE>(x, y) };
78            return reduce(sum);
79        }
80        if is_x86_feature_detected!("avx2") {
81            let sum = unsafe { simd::dot_product_avx2::<REVERSE>(x, y) };
82            return reduce(sum);
83        }
84    }
85    let sum = dot_product_scalar::<M, REVERSE>(x, y);
86    reduce(sum)
