58fn dot_product<M, const REVERSE: bool>(x: &[MInt<M>], y: &[MInt<M>]) -> MInt<M>
59where
60    M: MontgomeryReduction32,
61{
62    assert_eq!(x.len(), y.len());
63    // SAFETY: MInt is transparent over u32; retain the Montgomery residues.
64    let (x, y) = unsafe {
65        (
66            std::slice::from_raw_parts(x.as_ptr().cast::<u32>(), x.len()),
67            std::slice::from_raw_parts(y.as_ptr().cast::<u32>(), y.len()),
68        )
69    };
70    #[cfg(target_arch = "x86_64")]
71    if x.len() >= 64 && M::MOD < 1 << 31 {
72        if avx512_enabled() && avx512_supported() {
73            let sum = unsafe { simd::dot_product_avx512::<REVERSE>(x, y) };
74            return reduce_sum(sum);
75        }
76        if is_x86_feature_detected!("avx2") {
77            let sum = unsafe { simd::dot_product_avx2::<REVERSE>(x, y) };
78            return reduce_sum(sum);
79        }
80    }
81    let sum = dot_product_scalar::<M, REVERSE>(x, y);
82    reduce_sum(sum)
83}

85fn dot_product_scalar<M, const REVERSE: bool>(x: &[u32], y: &[u32]) -> u128
86where
87    M: MontgomeryReduction32,
88{
89    let max_value = (M::MOD - 1) as u64;
90    let block = (u64::MAX / (max_value * max_value)).min(16) as usize;
91    let mut sum = 0u128;
92    for (i, x) in x.chunks(block).enumerate() {
93        let start = i * block;
94        let part: u64 = if REVERSE {
95            x.iter()
96                .zip(y[y.len() - start - x.len()..y.len() - start].iter().rev())
97                .map(|(&x, &y)| x as u64 * y as u64)
98                .sum()
99        } else {
100            x.iter()
101                .zip(&y[start..start + x.len()])
102                .map(|(&x, &y)| x as u64 * y as u64)
103                .sum()
104        };
105        sum += part as u128;
106    }
107    sum
108}

116    fn add_products<const N: usize>(&mut self, terms: [(&MInt<M>, &MInt<M>); N]) {
117        let product =
118            |(x, y): (&MInt<M>, &MInt<M>)| u64::from(x.into_raw()) * u64::from(y.into_raw());
119        let max_product = u128::from(M::MOD - 1).pow(2);
120        let add = if N as u128 * max_product <= u128::from(u64::MAX) {
121            u128::from(terms.into_iter().map(product).sum::<u64>())
122        } else {
123            terms.into_iter().map(|p| u128::from(product(p))).sum()
124        };
125        self.0 = self.0.checked_add(add).unwrap_or_else(|| {
126            let modulus = u128::from(M::MOD);
127            self.0 % modulus + add % modulus
128        });
