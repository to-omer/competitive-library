141) {
142    unsafe {
143        let (n, m, p) = shape;
144        let (modulus, inverse) = (kernel.modulus, kernel.inverse);
145        if n.min(m).min(p) <= 64 || n % 16 != 0 || m % 16 != 0 || p % 16 != 0 {
146            if kernel.avx512 {
147                leaf_avx512(a, b, c, shape, modulus, inverse);
148            } else {
149                leaf_avx2(a, b, c, shape, modulus, inverse);
150            }
151            return;
152        }
153        let (n, m, p) = (n / 2, m / 2, p / 2);
154        let (s, t, q) = (work, work.add(n * m), work.add(n * m + m * p));
155        let work = q.add(n * p);
156        let (a00, a01, a10, a11) = (a, a.add(n * m), a.add(2 * n * m), a.add(3 * n * m));
157        let (b00, b01, b10, b11) = (b, b.add(m * p), b.add(2 * m * p), b.add(3 * m * p));
158        let (c00, c01, c10, c11) = (c, c.add(n * p), c.add(2 * n * p), c.add(3 * n * p));
159        multiply(a00, b00, c11, (n, m, p), work, kernel);

162        combine::<false>(a10, a11, s, n * m, modulus);

184fn blocks(rows: usize, cols: usize, depth: usize) -> impl Iterator<Item = (usize, usize, usize)> {
185    (0..1usize << (2 * depth)).map(move |block| {
186        let (mut row, mut col) = (0, 0);
187        for bit in 0..depth {
188            row |= (block >> (2 * bit + 1) & 1) << bit;
189            col |= (block >> (2 * bit) & 1) << bit;
190        }
191        (
192            block * (rows >> depth) * (cols >> depth),
193            row * (rows >> depth),
194            col * (cols >> depth),
195        )
196    })
197}

199impl<M> MInt<M>
200where
201    M: MIntBase<Inner = u32>,
202{
203    /// # Safety
204    /// AVX2 must be available. The modulus must be odd, greater than one and below 2^30.
205    /// `scale` must be below the modulus; entries must have canonical raw representations.
206    #[target_feature(enable = "avx2")]
207    pub unsafe fn matrix_product_avx2(
208        a: &[Vec<Self>],
209        b: &[Vec<Self>],
210        scale: u32,
211    ) -> Vec<Vec<Self>> {
212        let (n, m, p) = (a.len(), b.len(), b.first().map_or(0, Vec::len));
213        assert!(a.iter().all(|row| row.len() == m));
214        assert!(b.iter().all(|row| row.len() == p));
215        let modulus = M::get_mod();
216        let alignment = if n.min(m).min(p) <= 64 { 8 } else { 32 };
217        let (nn, mm, pp) = (
218            n.div_ceil(alignment) * alignment,
219            m.div_ceil(alignment) * alignment,
220            p.div_ceil(alignment) * alignment,
221        );
222        let mut depth = 0;
223        let (mut x, mut y, mut z) = (nn, mm, pp);
224        while x.min(y).min(z) > 64 && x % 16 == 0 && y % 16 == 0 && z % 16 == 0 {
225            depth += 1;
226            x /= 2;
227            y /= 2;
228            z /= 2;
229        }
230        let entries = nn * mm + mm * pp + nn * pp;
231        // A recursive level uses one quarter of its parent's storage; siblings reuse it.
232        let mut data = if entries + entries / 3 >= 1 << 20 {
233            let mut data = Vec::with_capacity(entries + entries / 3);
234            advise_huge_pages(&mut data);
235            data.resize(entries + entries / 3, 0u32);
236            data
237        } else {
238            vec![0u32; entries + entries / 3]
239        };
240        let quotient = (((scale as u64) << 32) / modulus as u64) as u32;
241        let mut inverse = 1u32;
242        for _ in 0..5 {
243            inverse = inverse.wrapping_mul(2u32.wrapping_sub(modulus.wrapping_mul(inverse)));
244        }
245        let inverse = inverse.wrapping_neg();
246        for (offset, row, col) in blocks(nn, mm, depth) {
247            let (nr, nc) = (nn >> depth, mm >> depth);
248            for i in row..(row + nr).min(n) {
249                // SAFETY: MInt is transparent over u32. Reading raw words preserves Montgomery encoding.
250                let values: &[u32] = unsafe { std::slice::from_raw_parts(a[i].as_ptr().cast(), m) };
251                for j in col..(col + nc).min(m) {
252                    let x = values[j];
253                    let q = ((x as u64 * quotient as u64) >> 32) as u32;
254                    let x = x.wrapping_mul(scale).wrapping_sub(q.wrapping_mul(modulus));
255                    data[offset + (i - row) * nc + j - col] = x.min(x.wrapping_sub(modulus));
256                }
257            }
258        }
259        for (offset, row, col) in blocks(mm, pp, depth) {
260            let (nr, nc) = (mm >> depth, pp >> depth);
261            for i in row..(row + nr).min(m) {
262                // SAFETY: MInt is transparent over u32 and row lengths were checked above.
263                let values: &[u32] = unsafe { std::slice::from_raw_parts(b[i].as_ptr().cast(), p) };
264                for j in col..(col + nc).min(p) {
265                    data[nn * mm + offset + (i - row) * nc + j - col] = values[j];
266                }
267            }
268        }
269        let kernel = Kernel {
270            modulus,
271            inverse,
272            avx512: avx512_enabled() && is_x86_feature_detected!("avx512f"),
273        };
274        // SAFETY: padding keeps every leaf dimension divisible by eight. The three matrices
275        // and the geometric scratch space are disjoint parts of the allocated buffer.
276        unsafe {
277            let ptr = data.as_mut_ptr();
278            multiply(
279                ptr,
280                ptr.add(nn * mm),
281                ptr.add(nn * mm + mm * pp),
282                (nn, mm, pp),
283                ptr.add(entries),
284                &kernel,
285            );
286        }
287        let mut result = vec![vec![MInt::new_unchecked(M::mod_zero()); p]; n];
288        for (offset, row, col) in blocks(nn, pp, depth) {
289            let (nr, nc) = (nn >> depth, pp >> depth);
290            for i in row..(row + nr).min(n) {
291                for j in col..(col + nc).min(p) {
292                    result[i][j] = MInt::new_unchecked(
293                        data[nn * mm + mm * pp + offset + (i - row) * nc + j - col],
294                    );
295                }
296            }
297        }
298        result
299    }
