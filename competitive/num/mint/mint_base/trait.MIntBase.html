1021    impl AssignmentPowerStore {
1022        fn new(len: usize) -> Self {
1023            let width = len.max(1).ilog2() as usize + 1;
1024            let mut slots = 0;
1025            let mut count = len.div_ceil(B);
1026            while count > 1 {
1027                count = count.div_ceil(B);
1028                slots += count * B;
1029            }
1030            // A sweep leaves more than slots free records, even when every tag is live.
1031            let capacity = 2 * slots + 2;
1032            assert!(u32::try_from(capacity).is_ok());
1033            Self {
1034                powers: vec![(ONE, 0); capacity * width],
1035                free: (1..capacity as u32).rev().collect(),
1036                live: vec![false; capacity],
1037                width,
1038                lasting: false,
1039            }
1040        }
1041
1042        fn insert(&mut self, value: Affine, len: usize, tags: impl Iterator<Item = u32>) -> u32 {
1043            if self.free.is_empty() {
1044                self.live.fill(false);
1045                for tag in tags {
1046                    self.live[tag as usize] = true;
1047                }
1048                self.free.extend(
1049                    (1..self.live.len())
1050                        .rev()
1051                        .filter(|&i| !self.live[i])
1052                        .map(|i| i as u32),
1053                );
1054            }
1055            let tag = self.free.pop().expect("live tag capacity");
1056            self.lasting = false;
1057            let start = tag as usize * self.width;
1058            let record = &mut self.powers[start..][..=len.ilog2() as usize];
1059            let mut power = value;
1060            record[0] = power;
1061            for next in &mut record[1..] {
1062                let (a, b) = power;
1063                power = (mod_mul(a, a), mod_mul(mod_add(a, ONE), b));
1064                *next = power;
1065            }
1066            tag
1067        }
1068
1069        fn retain(&mut self) {
1070            self.lasting = true;
1071        }
1072
1073        fn finish(&mut self, tag: u32) {
1074            if !self.lasting {
1075                self.free.push(tag);
1076            }
1077        }

1084        fn power(&self, tag: u32, mut count: usize) -> Affine {
1085            if count == 0 {
1086                return (ONE, 0);
1087            }
1088            let start = tag as usize * self.width;
1089            let mut value = self.powers[start + count.trailing_zeros() as usize];
1090            count &= count - 1;
1091            while count != 0 {
1092                value = compose(value, self.powers[start + count.trailing_zeros() as usize]);
1093                count &= count - 1;
1094            }

1096        }
