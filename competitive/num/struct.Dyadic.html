169    pub fn options(&self, player: Player) -> impl Iterator<Item = Self> + '_ {
170        let mut offset = Dyadic::from(0);
171        let (single, nimbers, general) = match &self.0 {
172            GameValueNode::NumberUpNimber(n, 0, 0) => {
173                let option = if n.exponent() == 0 {
174                    match player {
175                        Player::Left if n.numerator() > 0 => Some(Dyadic::from(n.numerator() - 1)),
176                        Player::Right if n.numerator() < 0 => Some(Dyadic::from(n.numerator() + 1)),
177                        _ => None,
178                    }
179                } else {
180                    let numerator = (n.numerator() >> 1) + i64::from(player == Player::Right);
181                    Some(Dyadic::new(numerator, n.exponent() - 1))
182                };
183                (option.map(Self::from), None, &[][..])
184            }
185            GameValueNode::NumberUpNimber(q, 0, n) => (None, Some((*q, *n)), &[][..]),
186            GameValueNode::NumberUpNimber(q, k, 1)
187                if matches!((*k, player), (1, Player::Left) | (-1, Player::Right)) =>
188            {
189                (None, Some((*q, 2)), &[][..])
190            }
191            GameValueNode::NumberUpNimber(q, k, n) => {
192                let sign = if *k > 0 { 1 } else { -1 };
193                let moves_to_number = (*k > 0) == (player == Player::Left);
194                let option = if moves_to_number {
195                    Self::from(*q)
196                } else {
197                    Self::number_up_nimber(*q, k - sign, n ^ 1)
198                };
199                (Some(option), None, &[][..])
200            }
201            GameValueNode::Switches(value) => {
202                let temperatures = &value.temperatures[value.start..];
203                let t = if player == Player::Left {
204                    temperatures[0]
205                } else {
206                    -temperatures[0]
207                };
208                let option = if temperatures.len() == 1 {
209                    Self::number_up_nimber(value.number + t, value.up, value.nimber)
210                } else {
211                    Self(GameValueNode::Switches(Rc::new(GameValueSwitches {
212                        number: value.number + t,
213                        start: value.start + 1,
214                        fingerprint: value.fingerprint ^ Self::temperature_hash(temperatures[0]),
215                        ..(**value).clone()
216                    })))
217                };
218                (Some(option), None, &[][..])
219            }
220            GameValueNode::Options(q, options) => {
221                offset = *q;
222                let (left, right) = &**options;
223                let moves = if player == Player::Left { left } else { right };
224                (None, None, moves.as_slice())
225            }
226        };
227        single
228            .into_iter()
229            .chain(
230                nimbers
231                    .into_iter()
232                    .flat_map(|(q, n)| (0..n).map(move |m| Self::number_up_nimber(q, 0, m))),
233            )
234            .chain(general.iter().map(move |g| g.translated(offset)))
235    }

237    fn from_options_memoized(
238        mut left: Vec<Self>,
239        mut right: Vec<Self>,
240        cache: &mut HashMap<GameValueNode, HashMap<GameValueNode, bool>>,
241    ) -> Self {
242        loop {
243            if left.iter().chain(&right).all(|g| g.as_number().is_some()) {
244                let l = left.iter().filter_map(Self::as_number).max();
245                let r = right.iter().filter_map(Self::as_number).min();
246                if let Some(n) = Dyadic::simplest_between(l, r) {
247                    return n.into();
248                }
249                let (l, r) = (l.unwrap(), r.unwrap());
250                if l == r {
251                    return Self::number_up_nimber(l, 0, 1);
252                }
253                let (sum, difference) = (l + r, l - r);
254                let temperature = Dyadic::new(difference.numerator(), difference.exponent() + 1);
255                return Self::from_switches(
256                    Dyadic::new(sum.numerator(), sum.exponent() + 1),
257                    0,
258                    0,
259                    vec![temperature],
260                    Self::temperature_hash(temperature),
261                );
262            }
263            if !right.is_empty()
264                && let Some((number, _)) = left.first().and_then(Self::as_number_nimber)
265            {
266                let nimber = |g: &Self| {
267                    g.as_number_nimber()
268                        .filter(|&(q, _)| q == number)
269                        .map(|(_, n)| n)
270                };
271                let l: Option<Vec<_>> = left.iter().map(nimber).collect();
272                let r: Option<Vec<_>> = right.iter().map(nimber).collect();
273                if let (Some(l), Some(r)) = (l, r) {
274                    let value = game_mex(l);
275                    if value == game_mex(r) {
276                        return Self::number_up_nimber(number, 0, value);
277                    }
278                }
279            }
280            for (moves, replies, sign) in [(&left, &right, 1), (&right, &left, -1)] {
281                if replies.len() != 1 {
282                    continue;
283                }
284                if let Some((q, k, n)) = replies[0].as_number_up_nimber() {
285                    if moves.len() == 1
286                        && moves[0].as_number() == Some(q)
287                        && (k == 0 || k.signum() == sign)
288                    {
289                        return Self::number_up_nimber(q, k + sign, n ^ 1);
290                    }
291                    if k == 0 && n == 0 && moves.len() == 2 {
292                        let a = moves[0].as_number_nimber();
293                        let b = moves[1].as_number_nimber();
294                        if (a == Some((q, 0)) && b == Some((q, 1)))
295                            || (a == Some((q, 1)) && b == Some((q, 0)))
296                        {
297                            return Self::number_up_nimber(q, sign, 1);
298                        }
299                    }
300                }
301            }
302            if left.len() == 1
303                && right.len() == 1
304                && let (Some((a, k, n, ts)), Some((b, l, m, us))) =
305                    (left[0].as_switch_sum(), right[0].as_switch_sum())
306                && k == l
307                && n == m
308                && ts == us
309                && a > b
310            {
311                let (sum, difference) = (a + b, a - b);
312                let temperature = Dyadic::new(difference.numerator(), difference.exponent() + 1);
313                if ts.first().is_none_or(|&t| temperature > t) {
314                    return Self::from_switches(
315                        Dyadic::new(sum.numerator(), sum.exponent() + 1),
316                        k,
317                        n,
318                        std::iter::once(temperature)
319                            .chain(ts.iter().copied())
320                            .collect(),
321                        Self::temperature_hash(temperature) ^ left[0].switch_hash(),
322                    );
323                }
324            }
325            let count = left.len() + right.len();
326            Self::remove_dominated(&mut left, Player::Left, cache);
327            Self::remove_dominated(&mut right, Player::Right, cache);
328            if count != left.len() + right.len() {
329                continue;
330            }
331            let options = Rc::new((left, right));
332            let game = Self(GameValueNode::Options(0.into(), options.clone()));
333            // Bypasses preserve the game, so all checks use the same candidate.
334            let mut reversible = Vec::new();
335            for player in [Player::Left, Player::Right] {
336                let moves = match player {
337                    Player::Left => &options.0,
338                    Player::Right => &options.1,
339                };
340                for (i, g) in moves.iter().enumerate() {
341                    if let Some((q, n)) = g.as_number_nimber()
342                        && n > 1
343                    {
344                        let number = Self::from(q);
345                        let (lower, upper) = match player {
346                            Player::Left => (&game, &number),
347                            Player::Right => (&number, &game),
348                        };
349                        // A strict inequality against q excludes every reply q + *m.
350                        if lower.leq_memoized(upper, cache) && !upper.leq_memoized(lower, cache) {
351                            continue;
352                        }
353                    }
354                    if let Some(reply) = g.options(player.opponent()).find(|r| match player {
355                        Player::Left => r.leq_memoized(&game, cache),
356                        Player::Right => game.leq_memoized(r, cache),
357                    }) {
358                        reversible.push((player, i, reply));
359                    }
360                }
361            }
362            if reversible.is_empty() {
363                return game;
364            }
365            drop(game);
366            (left, right) = Rc::unwrap_or_clone(options);
367            for (player, i, reply) in reversible.into_iter().rev() {
368                let moves = if player == Player::Left {
369                    &mut left
370                } else {
371                    &mut right
372                };
373                moves.remove(i);
374                moves.extend(reply.options(player));
375            }
376        }
377    }

383    fn leq_memoized(
384        &self,
385        rhs: &Self,
386        cache: &mut HashMap<GameValueNode, HashMap<GameValueNode, bool>>,
387    ) -> bool {
388        if let (Some((a, k, n, ts)), Some((b, l, m, us))) =
389            (self.as_switch_sum(), rhs.as_switch_sum())
390        {
391            // Both players take the hottest remaining switch. At zero, the tail
392            // decides the winner with the turn reversed once per switch.
393            let mut number = a;
394            let mut odd = false;
395            for t in Self::switch_difference(ts, us) {
396                // The unplayed alternating tail has the sign of the next term.
397                if !odd && number >= b || odd && number <= b {
398                    return odd;
399                }
400                number = if odd { number - t } else { number + t };
401                odd = !odd;
402            }
403            return match number.cmp(&b) {
404                Ordering::Less => true,
405                Ordering::Greater => false,
406                Ordering::Equal => {
407                    let up = if odd { l - k } else { k - l };
408                    let nimber = n ^ m;
409                    let wins = up > 0 || up == 0 && nimber != 0 || up == -1 && nimber == 1;
410                    wins == odd
411                }
412            };
413        }
414        if let (GameValueNode::Options(a, g), GameValueNode::Options(b, h)) = (&self.0, &rhs.0)
415            && Rc::ptr_eq(g, h)
416        {
417            return a <= b;
418        }
419        // Normalize only memo keys: from_options also compares unreduced candidates.
420        let offset = -self.number_offset();
421        let left = self.translated(offset).0;
422        let right = rhs.translated(offset).0;
423        if let Some(&value) = cache.get(&left).and_then(|row| row.get(&right)) {
424            return value;
425        }
426        // Try explicit options before expanding a compact value.
427        let mut sides = [(self, rhs, Player::Left), (rhs, self, Player::Right)];
428        if self.as_switch_sum().is_some() {
429            sides.swap(0, 1);
430        }
431        let value = sides.into_iter().all(|(game, other, player)| {
432            let mut preferred = |a: &Self, b: &Self| match player {
433                Player::Left => b.leq_memoized(a, cache),
434                Player::Right => a.leq_memoized(b, cache),
435            };
436            if let GameValueNode::NumberUpNimber(q, 0, n) = game.0
437                && n != 0
438            {
439                let number = Self::from(q);
440                !preferred(&number, other)
441                    && (preferred(other, &number)
442                        || !game.options(player).skip(1).any(|g| preferred(&g, other)))
443            } else {
444                !game.options(player).any(|g| preferred(&g, other))
445            }
446        });
447        cache.entry(left).or_default().insert(right, value);
448        value
449    }

480    fn sum(
481        &self,
482        rhs: &Self,
483        cache: &mut HashMap<(GameValueNode, GameValueNode), Self>,
484        comparisons: &mut HashMap<GameValueNode, HashMap<GameValueNode, bool>>,
485    ) -> Self {
486        if let (GameValueNode::NumberUpNimber(a, k, n), GameValueNode::NumberUpNimber(b, l, m)) =
487            (&self.0, &rhs.0)
488        {
489            return Self::number_up_nimber(*a + *b, k + l, n ^ m);
490        }
491        if let Some(q) = self.as_number() {
492            return rhs.translated(q);
493        }
494        if let Some(q) = rhs.as_number() {
495            return self.translated(q);
496        }
497        if let (Some((a, k, n, ts)), Some((b, l, m, us))) =
498            (self.as_switch_sum(), rhs.as_switch_sum())
499        {
500            return Self::from_switches(
501                a + b,
502                k + l,
503                n ^ m,
504                Self::switch_difference(ts, us).collect(),
505                self.switch_hash() ^ rhs.switch_hash(),
506            );
507        }
508        // Cache the sum of the untranslated parts, sharing it across numeric offsets.
509        let (a, b) = (self.number_offset(), rhs.number_offset());
510        if a.numerator() != 0 || b.numerator() != 0 {
511            return self
512                .translated(-a)
513                .sum(&rhs.translated(-b), cache, comparisons)
514                .translated(a + b);
515        }
516        let mut key = (self.0.clone(), rhs.0.clone());
517        let swap = match (&key.0, &key.1) {
518            (GameValueNode::Options(_, a), GameValueNode::Options(_, b)) => {
519                Rc::as_ptr(a) > Rc::as_ptr(b)
520            }
521            (GameValueNode::Options(..), _) => true,
522            _ => false,
523        };
524        if swap {
525            std::mem::swap(&mut key.0, &mut key.1);
526        }
527        if let Some(value) = cache.get(&key) {
528            return value.clone();
529        }
530        let mut options = |player| {
531            let mut moves = Vec::new();
532            for g in self.options(player) {
533                moves.push(g.sum(rhs, cache, comparisons));
534            }
535            for g in rhs.options(player) {
536                moves.push(self.sum(&g, cache, comparisons));
537            }
538            moves
539        };
540        let value =
541            Self::from_options_memoized(options(Player::Left), options(Player::Right), comparisons);
542        cache.insert(key, value.clone());
543        value
544    }

606    fn switch_difference<'a>(
607        mut left: &'a [Dyadic<i64>],
608        mut right: &'a [Dyadic<i64>],
609    ) -> impl Iterator<Item = Dyadic<i64>> + 'a {
610        std::iter::from_fn(move || {
611            while let (Some(a), Some(b)) = (left.first(), right.first()) {
612                if a != b {
613                    break;
614                }
615                left = &left[1..];
616                right = &right[1..];
617            }
618            if left.first() < right.first() {
619                std::mem::swap(&mut left, &mut right);
620            }
621            let (&value, tail) = left.split_first()?;
622            left = tail;
623            Some(value)
624        })
625    }

627    fn remove_dominated(
628        options: &mut Vec<Self>,
629        player: Player,
630        cache: &mut HashMap<GameValueNode, HashMap<GameValueNode, bool>>,
631    ) {
632        let mut retained = vec![true; options.len()];
633        for i in 0..options.len() {
634            let dominated = (0..options.len()).any(|j| {
635                i != j
636                    && retained[j]
637                    && match player {
638                        Player::Left => options[i].leq_memoized(&options[j], cache),
639                        Player::Right => options[j].leq_memoized(&options[i], cache),
640                    }
641            });
642            retained[i] = !dominated;
643        }
644        let mut retained = retained.into_iter();
645        options.retain(|_| retained.next().unwrap());
646    }
