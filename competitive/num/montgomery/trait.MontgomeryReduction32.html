40        let p = Self::get_mod() as i32;
41        let (mut a, mut b) = (x as i32, p);
42        let (mut u, mut x) = (1, 0);
43        while a != 0 {

45            x -= k * u;
46            b -= k * a;

26    fn dot_product(x: &[MInt<Self>], y: &[MInt<Self>]) -> MInt<Self> {
27        assert_eq!(x.len(), y.len());
28        // reduce() needs sum < modulus * 2^32 to return a canonical residue.
29        let modulus = <Self as MontgomeryReduction32>::MOD as u64;
30        let block = (((modulus << 32) - 1) / ((modulus - 1) * (modulus - 1))).min(16) as usize;
31        let mut result = 0;
32        for (x, y) in x.chunks(block).zip(y.chunks(block)) {
33            // SAFETY: MInt is transparent over u32 and both slices have equal lengths.
34            let sum = unsafe {
35                let a = std::slice::from_raw_parts(x.as_ptr().cast::<u32>(), x.len());
36                let b = std::slice::from_raw_parts(y.as_ptr().cast::<u32>(), y.len());
37                a.iter().zip(b).map(|(&a, &b)| a as u64 * b as u64).sum()
38            };
39            result = Self::mod_add(result, Self::reduce(sum));
40        }
41        MInt::new_unchecked(result)
42    }
