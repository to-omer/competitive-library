246    M: Montgomery32NttModulus,

248    let modulus = _mm256_set1_epi32(M::MOD as i32);
249    let modulus2 = _mm256_set1_epi32(M::MOD.wrapping_mul(2) as i32);

276            store_block_avx2(a, nn + i, lazy_sub_avx2(x0, x1, modulus2));

333                [

339            );
340        }

342    }

376        let mut quarter = group_len >> 2;
377        while quarter > 1 {

386                    packed_rate_avx2::<M>((!group).trailing_zeros() as usize, false),

388                );
389                let base = base.add(i << 3);
390                if wide {
391                    ntt_butterfly_avx512::<M, false>(base, quarter, current);
392                } else {
393                    ntt_butterfly_avx2::<M, false>(base, quarter, current);
394                }
395                i += group_len;
396                group += 1;
397            }
398            roots[root_slot] = root;
399            group_len = quarter;
400            quarter >>= 2;
401            stage_log -= 2;
402        }
403        // Final-stage roots are used once; skip precomputing root*R.
404        let r = _mm256_set1_epi32(M::R as i32);
405        let store = |i, y| store_block_avx2(a, i, normalize_avx2(y, modulus, modulus2));

407        let mut i = tile + if tile == 0 { 4 } else { 0 };

410            root = update_root_avx2(
411                root,
412                packed_rate_avx2::<M>((!(i >> 2)).trailing_zeros() as usize, false),
413                modulus,
414            );
415            let r2 = _mm256_shuffle_epi32::<0x55>(r1);
416            let nr3 = _mm256_shuffle_epi32::<0xff>(r1);
417            let x0 = load_block_avx2(a, i);
418            let x1 = load_block_avx2(a, i + 1);
419            let x2 = load_block_avx2(a, i + 2);
420            let x3 = load_block_avx2(a, i + 3);
421            let g1 = montgomery_simd::montgomery_mul_256(
422                x1,
423                _mm256_shuffle_epi32::<0xa0>(r1),
424                r,
425                modulus,
426            );
427            let ng3 = montgomery_simd::montgomery_mul_256(x3, nr3, r, modulus);
428            let g2 = montgomery_simd::montgomery_mul_256(x2, r2, r, modulus);
429            let g0 = shrink_avx2(x0, modulus2);
430            let h3 = montgomery_simd::montgomery_mul_256_fixed(
431                _mm256_add_epi32(g1, ng3),
432                imag,
433                imag_r,
434                modulus,
435            );
436            let h1 = montgomery_simd::sub_mod_256(g1, ng3, modulus2);
437            let h0 = montgomery_simd::add_mod_256(g0, g2, modulus2);
438            let h2 = montgomery_simd::sub_mod_256(g0, g2, modulus2);
439            store(i, _mm256_add_epi32(h0, h1));
440            store(i + 1, lazy_sub_avx2(h0, h1, modulus2));
441            store(i + 2, _mm256_add_epi32(h2, h3));
442            store(i + 3, lazy_sub_avx2(h2, h3, modulus2));
443            i += 4;
444        }
445        roots[0] = root;
446
447        tile += tile_len;
448        if tile < n {
449            stage_log = (tile.trailing_zeros() as usize & !1).min(max_stage_log);
450        }
451    }

455unsafe fn intt_blocks_avx2<M>(a: *mut u32, n: usize)
456where
457    M: Montgomery32NttModulus,
458{
459    let inv = M::mod_inv(<M as MIntConvert<u32>>::from(n as u32));
460    let modulus = _mm256_set1_epi32(M::MOD as i32);
461    let modulus2 = _mm256_set1_epi32(M::MOD.wrapping_mul(2) as i32);
462    let wide = matches!(simd_backend(), SimdBackend::Avx512);
463
464    let initial_root = _mm256_setr_epi32(M::N1 as i32, 0, M::N1 as i32, 0, M::N1 as i32, 0, 0, 0);
465    let log_n = n.trailing_zeros() as usize;
466    let mut roots = [initial_root; 16];
467    let nn = n >> (log_n & 1);
468    let tile_len = n.min(64);
469    let inv_vec = _mm256_set1_epi32(inv as i32);
470
471    roots[0] = inv_vec;
472    let mut tile = 0;
473    while tile < n {
474        let max_stage_log = (tile + tile_len).trailing_zeros() as usize;
475        let mut stage_log = 2usize;
476        let mut group_len = 4usize;
477        let mut quarter = 1usize;
478        while stage_log <= max_stage_log {
479            let offset = tile + tile_len - group_len.max(tile_len);
480            let base = a.add(offset << 3);
481            let mut i = 0;
482            let root_slot = (stage_log - 2) >> 1;
483            let mut root = roots[root_slot];
484            let mut group = (tile + i) >> stage_log;
485            while i < tile_len {
486                let current = root;
487                root = update_root_avx2(
488                    root,
489                    packed_rate_avx2::<M>((!group).trailing_zeros() as usize, true),
490                    modulus,
491                );
492                let base = base.add(i << 3);
493                let last = group_len == n;
494                if wide && quarter > 1 {
495                    intt_butterfly_avx512::<M>(base, quarter, current, inv, group, last);
496                } else {
497                    intt_butterfly_avx2::<M>(base, quarter, current, inv, group, last);
498                }
499                i += group_len;
500                group += 1;
501            }
502            roots[root_slot] = root;
503            quarter = group_len;
504            group_len <<= 2;
505            stage_log += 2;
506        }
507        tile += tile_len;
508    }
509    if nn != n {
510        let mut i = 0;
511        while i < nn {
512            let x0 = load_block_avx2(a, i);
513            let x1 = load_block_avx2(a, nn + i);
514            store_block_avx2(
515                a,
516                i,
517                shrink_avx2(montgomery_simd::add_mod_256(x0, x1, modulus2), modulus),
518            );
519            store_block_avx2(
520                a,
521                nn + i,
522                shrink_avx2(montgomery_simd::sub_mod_256(x0, x1, modulus2), modulus),
523            );
524            i += 1;
525        }
526    }
527}

531unsafe fn reduce_sum_avx2(
532    even: __m256i,
533    odd: __m256i,
534    r_vec: __m256i,
535    mod_vec: __m256i,
536) -> __m256i {
537    let even_m = _mm256_mul_epu32(even, r_vec);
538    let odd_m = _mm256_mul_epu32(odd, r_vec);
539    let even = _mm256_add_epi64(even, _mm256_mul_epu32(even_m, mod_vec));
540    let odd = _mm256_add_epi64(odd, _mm256_mul_epu32(odd_m, mod_vec));
541    _mm256_or_si256(_mm256_bsrli_epi128::<4>(even), odd)
542}

544#[target_feature(enable = "avx2")]
545unsafe fn convolve_8_avx2<M>(f: *mut u32, g: *const u32, n: usize)
546where
547    M: Montgomery32NttModulus,
548{
549    let mod_vec = _mm256_set1_epi32(M::MOD as i32);
550    let mod2_vec = _mm256_set1_epi32(M::MOD.wrapping_add(M::MOD) as i32);
551    let r_vec = _mm256_set1_epi32(M::R as i32);
552    let mut rr = M::N1;
553    let mut i = 0;
554    while i < n {
555        let rr_i = M::mod_mul(rr, M::INFO.root[2]);
556        let mut work = std::mem::MaybeUninit::<[__m256i; 8]>::uninit();
557        let work = work.as_mut_ptr().cast::<u32>();
558        for (j, ww) in [
559            rr,
560            M::MOD.wrapping_mul(2) - rr,
561            rr_i,
562            M::MOD.wrapping_mul(2) - rr_i,
563        ]
564        .into_iter()
565        .enumerate()
566        {
567            let k = i + j;
568            let ff = load_block_avx2(f, k);
569            // ff, g < MOD and fw < 3 * MOD / 2, so the sum reduces below 4 * MOD.
570            let fw = montgomery_simd::montgomery_mul_256_fixed(
571                ff,
572                _mm256_set1_epi32(ww as i32),
573                _mm256_set1_epi32(ww.wrapping_mul(M::R) as i32),
574                mod_vec,
575            );
576            _mm256_store_si256(work.add(j << 4).cast(), fw);
577            _mm256_store_si256(work.add((j << 4) + 8).cast(), ff);
578        }
579        let mut even = [_mm256_setzero_si256(); 4];
580        let mut odd = [_mm256_setzero_si256(); 4];
581        let mut l = 0;
582        while l < 8 {
583            let mut j = 0;
584            while j < 4 {
585                let x = _mm256_loadu_si256(work.add((j << 4) + 8 - l).cast());
586                let y = _mm256_set1_epi32(*g.add(((i + j) << 3) + l) as i32);
587                even[j] = _mm256_add_epi64(even[j], _mm256_mul_epu32(x, y));
588                odd[j] = _mm256_add_epi64(odd[j], _mm256_mul_epu32(_mm256_bsrli_epi128::<4>(x), y));
589                j += 1;
590            }
591            l += 1;
592        }
593        let mut j = 0;
594        while j < 4 {
595            let x = reduce_sum_avx2(even[j], odd[j], r_vec, mod_vec);
596            // Repeated products must preserve the canonical input range.
597            store_block_avx2(f, i + j, normalize_avx2(x, mod_vec, mod2_vec));
598            j += 1;
599        }
600        i += 4;
601        let rate = M::INFO.rate3_packed[(i >> 2).trailing_zeros() as usize][1];
602        rr = M::mod_mul(rr, rate);
603    }
