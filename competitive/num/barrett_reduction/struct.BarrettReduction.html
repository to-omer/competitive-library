95    mut b: u64,
96    p: u64,

99    let mut order = p - 1;

101    let mut period = 1;
102    for (q, e) in prime_factors(order) {
103        for _ in 0..e {
104            order /= q;

107            } else {
108                (power(a, order, mul), power(b, order, mul))
109            };
110            if g == 1 {
111                if h != 1 {
112                    return None;
113                }
114                continue;
115            }
116            let digit = if h == 1 {
117                0
118            } else {
119                prime_order_log::<LANES>(g, h, p, q, mul)

122            period *= q;

127                a = power(a, q, mul);

129        }

134/// h is a nonidentity element of the subgroup of prime order generated by g.
135fn prime_order_log<const LANES: usize>(
136    g: u64,
137    h: u64,
138    p: u64,
139    order: u64,
140    mul: impl Fn(u64, u64) -> u64 + Copy,
141) -> u64 {
142    if order <= 128 {
143        let (mut value, mut i) = (g, 1);
144        while value != h {
145            value = mul(value, g);
146            i += 1;
147        }
148        return i;
149    }
150
151    let sqrt = p.isqrt();
152    // At most one prime divisor of p-1 exceeds √p. Field logarithms modulo
153    // that divisor distinguish all elements of the subgroup generated by g.
154    if order > sqrt
155        && let Some(digit) = LOGS.with_borrow_mut(|cache| {
156            let entry = cache.entry(p).or_insert_with(|| LogCache {
157                remaining_searches: ((sqrt << 12) / order).isqrt(),
158                table: None,
159            });
160            if entry.remaining_searches > 0 {
161                entry.remaining_searches -= 1;
162                return None;
163            }
164            let ic = entry
165                .table
166                .get_or_insert_with(|| Box::new(IndexCalculus::new(p, order)));
167            Some(ic.log(h).mod_mul(ic.log(g).mod_inv(order), order))
168        })
169    {
170        return digit;
171    }
172
173    let blocks = order.isqrt() as usize / LANES + 1;
174    let step = blocks * LANES;
175    let capacity = (4 * step).next_power_of_two();
176    let mask = capacity as u64 - 1;
177    let mut heads = vec![0u32; capacity];
178    let mut entries = Vec::with_capacity(step);
179    for block in power_blocks::<LANES>(1, g, mul).take(blocks) {
180        for value in block {
181            let head = &mut heads[(value & mask) as usize];
182            // The bucket supplies the low bits of the residue. Use those bits for the chain link.
183            entries.push((value & !mask) | u64::from(*head));
184            *head = entries.len() as u32;
185        }
186    }
187
188    let stride = power(g, step as u64, mul).mod_inv(p);
189    for (i, block) in power_blocks::<LANES>(h, stride, mul).enumerate() {
190        for (k, target) in block.into_iter().enumerate() {
191            let mut j = heads[(target & mask) as usize];
192            while j != 0 {
193                let entry = entries[j as usize - 1];
194                if (entry ^ target) & !mask == 0 {
195                    return (i * LANES + k) as u64 * step as u64 + u64::from(j) - 1;
196                }
197                j = (entry & mask) as u32;
198            }
199        }
200    }
201    unreachable!()
202}

204fn power_blocks<const LANES: usize>(
205    start: u64,
206    g: u64,
207    mul: impl Fn(u64, u64) -> u64,
208) -> impl Iterator<Item = [u64; LANES]> {
209    let mut values = [start; LANES];
210    for i in 1..LANES {
211        values[i] = mul(values[i - 1], g);
212    }
213    let jump = power(g, LANES as u64, &mul);
214    std::iter::from_fn(move || {
215        let block = values;
216        for v in &mut values {
217            *v = mul(*v, jump);
218        }
219        Some(block)
220    })
221}

223fn discrete_logarithm_prime_power(a: u64, b: u64, p: u64, e: u32) -> Option<(u64, u64)> {
224    let (mut x, mut period) = prime_field_log(a % p, b % p, p)?;
225    let mut m = p;
226    for _ in 1..e {
227        let next = m * p;
228        let mul = |x: u64, y| x.mod_mul(y, next);
229        let value = power(a, x, mul);
230        let step = power(a, period, mul);
231        // (1 + k*m)^d ≡ 1 + d*k*m (mod p*m).
232        let delta = (b % next).mod_sub(value, next) / m;
233        let slope = (step / m).mod_mul(value, p);
234        let (digit, radix) = solve_linear_congruence(slope, delta, p)?;
235        x += period * digit;
236        period *= radix;
237        m = next;
238    }
239    Some((x, period))
240}

266            .find(|&k| gcd(k, p - 1) == 1)

278                limit: u32::MAX / p,
279            })
280            .collect();

291        let mut pivots = vec![Vec::<u64>::new(); count];
292        let br = BarrettReduction::<u128>::new(q as _);

294        let (mut value, mut exponent) = (1, 0);
295        let mut rank = 0;
296        while rank < count {
297            let mut row = ic.relation(&mut value, &mut exponent);
298            for (i, pivot) in pivots.iter_mut().enumerate() {
299                let coefficient = row[i];
300                if coefficient == 0 {
301                    continue;
302                }
303                if pivot.is_empty() {
304                    let inverse = coefficient.mod_inv(q);
305                    for x in &mut row[i..] {
306                        *x = mul(*x, inverse);
307                    }
308                    *pivot = row;
309                    rank += 1;
310                    break;
311                }
312                for (x, &y) in row[i + 1..].iter_mut().zip(&pivot[i + 1..]) {
313                    *x = x.mod_sub(mul(coefficient, y), q);
314                }
315            }
316        }
317        ic.logs = vec![0; count];
318        for (i, row) in pivots.iter().enumerate().rev() {
319            ic.logs[i] = row[i + 1..count]
320                .iter()
321                .zip(&ic.logs[i + 1..])
322                .fold(row[count], |v, (&x, &y)| v.mod_sub(mul(x, y), q));
323        }
324        ic
325    }

71fn find_factor(n: u64) -> Option<u64> {
72    let br = BarrettReduction::<u128>::new(n as u128);
73    if miller_rabin_with_br(n, &br) {
74        return None;
75    }
76    let mr = MontgomeryReduction64::new(n);
77    let mut rng = Xorshift::default();
78    let (mut y0, mut c) = (0, n - 1);
79    loop {
80        let (mut x, mut y, mut ys, mut g, mut q, mut r, mut k) = (0, y0, 0, 1, 1, 1, 0);
81        while g == 1 && r <= 1 << 20 {
82            x = y;
83            while k < r && g == 1 {
84                ys = y;
85                for _ in 0..1024.min(r - k) {
86                    y = mr.sub(mr.mul(y, y), c);
87                    q = mr.mul(q, mr.sub(x, y));
88                }
89                g = gcd(q, n);
90                k += 1024;
91            }
92            k = r;
93            r <<= 1;
94        }
95        if g == n {
96            g = 1;
97            y = ys;
98            while g == 1 {
99                y = mr.sub(mr.mul(y, y), c);
100                g = gcd(mr.sub(x, y), n);
101            }
102        }
103        if g != 1 && g != n {
104            return Some(g);
105        }
106        y0 = ((rng.rand64() as u128 * (n - 2) as u128) >> 64) as u64 + 2;
107        c = ((rng.rand64() as u128 * (n - 1) as u128) >> 64) as u64 + 1;
108    }

337    fn relation(&self, value: &mut u64, exponent: &mut u64) -> Vec<u64> {
338        let p = self.modulus.get_mod() as u64;
339        let (mut num, mut den) = loop {
340            *value = self.modulus.rem(*value as u128 * self.generator as u128) as u64;
341            *exponent = exponent.mod_add(1, self.q);
342            let (mut x, mut y) = (p, *value);
343            let (mut u, mut v) = (0, 1);
344            // y ≡ ±value*v (mod p). At the first y ≤ √p, both y and v fit in u32.
345            // The sign vanishes in logarithms modulo the odd prime q.
346            while y > self.sqrt {
347                let quotient = x / y;
348                (x, y) = (y, x - quotient * y);
349                (u, v) = (v, u + quotient * v);
350            }
351            let (num, den) = (y as u32, v as u32);
352            if self.is_smooth(num) && self.is_smooth(den) {
353                break (num, den);
354            }
355        };
356        let mut row = vec![0; self.primes.len() + 1];
357        for (i, prime) in (0..self.primes.len()).rev().zip(&self.primes) {
358            let p = prime.p;
359            while num.is_multiple_of(p) {
360                num /= p;
361                row[i] += 1;
362            }
363            while den.is_multiple_of(p) {
364                den /= p;
365                row[i] = row[i].mod_sub(1, self.q);
366            }
367            if num == 1 && den == 1 {
368                break;
369            }
370        }
371        row[self.primes.len()] = *exponent;
372        row
373    }

135fn prime_order_log<const LANES: usize>(
136    g: u64,
137    h: u64,
138    p: u64,
139    order: u64,
140    mul: impl Fn(u64, u64) -> u64 + Copy,
141) -> u64 {
142    if order <= 128 {
143        let (mut value, mut i) = (g, 1);
144        while value != h {
145            value = mul(value, g);
146            i += 1;
147        }
148        return i;
149    }
150
151    let sqrt = p.isqrt();
152    // At most one prime divisor of p-1 exceeds √p. Field logarithms modulo
153    // that divisor distinguish all elements of the subgroup generated by g.
154    if order > sqrt
155        && let Some(digit) = LOGS.with_borrow_mut(|cache| {
156            let entry = cache.entry(p).or_insert_with(|| LogCache {
157                remaining_searches: ((sqrt << 12) / order).isqrt(),
158                table: None,

161                entry.remaining_searches -= 1;
162                return None;
163            }
164            let ic = entry
165                .table
166                .get_or_insert_with(|| Box::new(IndexCalculus::new(p, order)));
167            Some(ic.log(h).mod_mul(ic.log(g).mod_inv(order), order))
168        })
169    {
170        return digit;
171    }
172
173    let blocks = order.isqrt() as usize / LANES + 1;
174    let step = blocks * LANES;
175    let capacity = (4 * step).next_power_of_two();
176    let mask = capacity as u64 - 1;
177    let mut heads = vec![0u32; capacity];
178    let mut entries = Vec::with_capacity(step);
179    for block in power_blocks::<LANES>(1, g, mul).take(blocks) {
180        for value in block {
181            let head = &mut heads[(value & mask) as usize];
182            // The bucket supplies the low bits of the residue. Use those bits for the chain link.
183            entries.push((value & !mask) | u64::from(*head));
184            *head = entries.len() as u32;
185        }
186    }
187
188    let stride = power(g, step as u64, mul).mod_inv(p);
189    for (i, block) in power_blocks::<LANES>(h, stride, mul).enumerate() {
190        for (k, target) in block.into_iter().enumerate() {
191            let mut j = heads[(target & mask) as usize];
192            while j != 0 {
193                let entry = entries[j as usize - 1];
194                if (entry ^ target) & !mask == 0 {
195                    return (i * LANES + k) as u64 * step as u64 + u64::from(j) - 1;
196                }
197                j = (entry & mask) as u32;
198            }
199        }
200    }
201    unreachable!()
202}

205    start: u64,
206    g: u64,
207    mul: impl Fn(u64, u64) -> u64,
208) -> impl Iterator<Item = [u64; LANES]> {

215        let block = values;
216        for v in &mut values {

218        }

223fn discrete_logarithm_prime_power(a: u64, b: u64, p: u64, e: u32) -> Option<(u64, u64)> {
224    let (mut x, mut period) = prime_field_log(a % p, b % p, p)?;
225    let mut m = p;
226    for _ in 1..e {
227        let next = m * p;
228        let mul = |x: u64, y| x.mod_mul(y, next);
229        let value = power(a, x, mul);
230        let step = power(a, period, mul);
231        // (1 + k*m)^d ≡ 1 + d*k*m (mod p*m).
232        let delta = (b % next).mod_sub(value, next) / m;
233        let slope = (step / m).mod_mul(value, p);
234        let (digit, radix) = solve_linear_congruence(slope, delta, p)?;
235        x += period * digit;
236        period *= radix;
237        m = next;
238    }
