167fn two_sum(a: f64, b: f64) -> (f64, f64) {
168    let sum = a + b;
169    let v = sum - a;
170    (sum, (a - (sum - v)) + (b - v))
171}

173fn quick_two_sum(a: f64, b: f64) -> (f64, f64) {
174    let sum = a + b;
175    (sum, b - (sum - a))

178fn two_prod(a: f64, b: f64) -> (f64, f64) {
179    let product = a * b;
180    // Avoid a libm call when the target has no hardware FMA.
181    let error = if cfg!(any(target_arch = "aarch64", target_feature = "fma")) {
182        a.mul_add(b, -product)
183    } else {
184        let split = |value| {
185            let scaled = 134217729.0 * value;
186            let high = scaled - (scaled - value);
187            (high, value - high)
188        };
189        let (ah, al) = split(a);
190        let (bh, bl) = split(b);
191        ((ah * bh - product) + ah * bl + al * bh) + al * bl
192    };
193    (product, error)
194}

196fn renormalize<const N: usize, const M: usize>(mut values: [f64; M]) -> [f64; N] {
197    if N == 2 && M == 3 {
198        let (s, e) = two_sum(values[1], values[2]);
199        let (s, t) = two_sum(values[0], s);
200        let (high, low) = two_sum(s, t + e);
201        let mut result = scalar(high);
202        result[1] = low;
203        return result;
204    }
205    if N == 4 && M == 5 {
206        let (s, e3) = quick_two_sum(values[3], values[4]);
207        let (s, e2) = quick_two_sum(values[2], s);
208        let (s, e1) = quick_two_sum(values[1], s);
209        let (mut r0, mut r1) = quick_two_sum(values[0], s);
210        let (mut r2, mut r3) = (0.0, 0.0);
211        if r1 != 0.0 {
212            (r1, r2) = quick_two_sum(r1, e1);
213            if r2 != 0.0 {
214                (r2, r3) = quick_two_sum(r2, e2);
215            } else {
216                (r1, r2) = quick_two_sum(r1, e2);
217            }
218        } else {
219            (r0, r1) = quick_two_sum(r0, e1);
220            if r1 != 0.0 {
221                (r1, r2) = quick_two_sum(r1, e2);
222            } else {
223                (r0, r1) = quick_two_sum(r0, e2);
224            }
225        }
226        if r3 != 0.0 {
227            r3 += e3;
228        } else if r2 != 0.0 {
229            (r2, r3) = quick_two_sum(r2, e3);
230        } else if r1 != 0.0 {
231            (r1, r2) = quick_two_sum(r1, e3);
232        } else {
233            (r0, r1) = quick_two_sum(r0, e3);
234        }
235        let mut result = [0.0; N];
236        result[..4].copy_from_slice(&[r0, r1, r2, r3]);
237        return result;
238    }
239    let mut sum = values[M - 1];
240    for i in (0..M - 1).rev() {
241        (sum, values[i + 1]) = two_sum(values[i], sum);
242    }
243    let mut result = scalar(sum);
244    let mut index = 0;
245    for value in values.into_iter().skip(1) {
246        let (sum, error) = two_sum(result[index], value);
247        result[index] = sum;
248        if error != 0.0 && index + 1 < N {
249            index += 1;
250            result[index] = error;
251        }
252    }
253    result
254}

256// Retain cancellation terms until the final expansion is rounded.
257fn sum_terms<const N: usize, const M: usize>(values: [f64; M]) -> [f64; N] {
258    let mut expansion = [0.0; M];
259    let mut len = 0;
260    for mut value in values {
261        if value == 0.0 {
262            continue;
263        }
264        let mut end = 0;
265        for i in 0..len {
266            let error;
267            (value, error) = two_sum(value, expansion[i]);
268            if error != 0.0 {
269                expansion[end] = error;
270                end += 1;
271            }
272        }
273        expansion[end] = value;
274        len = end + 1;
275    }
276    expansion[..len].reverse();
277    expansion[len..].fill(0.0);
278    renormalize(expansion)
279}

281fn float_exponent(value: f64) -> i32 {
282    let bits = value.abs().to_bits();
283    let biased = (bits >> 52) as i32;
284    if biased == 0 {
285        63 - bits.leading_zeros() as i32 - 1074
286    } else {
287        biased - 1023
288    }
289}

291fn scale<const N: usize>(mut values: [f64; N], mut shift: i32) -> [f64; N] {
292    while shift != 0 {
293        let step = shift.clamp(-1022, 1023);
294        let factor = f64::from_bits(((step + 1023) as u64) << 52);
295        values = values.map(|x| x * factor);
296        if !values[0].is_finite() || values[0] == 0.0 {
297            return scalar(values[0]);
298        }
299        shift -= step;
300    }
301    values
