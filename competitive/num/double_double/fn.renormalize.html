257fn sum_terms<const N: usize, const M: usize>(values: [f64; M]) -> [f64; N] {
258    let mut expansion = [0.0; M];
259    let mut len = 0;
260    for mut value in values {
261        if value == 0.0 {
262            continue;
263        }
264        let mut end = 0;
265        for i in 0..len {
266            let error;
267            (value, error) = two_sum(value, expansion[i]);
268            if error != 0.0 {
269                expansion[end] = error;
270                end += 1;
271            }
272        }
273        expansion[end] = value;
274        len = end + 1;
275    }
276    expansion[..len].reverse();
277    expansion[len..].fill(0.0);
278    renormalize(expansion)
279}

281fn float_exponent(value: f64) -> i32 {
282    let bits = value.abs().to_bits();
283    let biased = (bits >> 52) as i32;
284    if biased == 0 {
285        63 - bits.leading_zeros() as i32 - 1074
286    } else {
287        biased - 1023
288    }
289}

291fn scale<const N: usize>(mut values: [f64; N], mut shift: i32) -> [f64; N] {
292    while shift != 0 {
293        let step = shift.clamp(-1022, 1023);
294        let factor = f64::from_bits(((step + 1023) as u64) << 52);
295        values = values.map(|x| x * factor);
296        if !values[0].is_finite() || values[0] == 0.0 {
297            return scalar(values[0]);
298        }
299        shift -= step;
300    }
301    values
302}

304fn parse<const N: usize>(text: &str) -> Result<[f64; N], ParseFloatError> {
305    let first: f64 = text.parse()?;
306    let mut result = [0.0; N];
307    result[0] = first;
308    if !first.is_finite() || first == 0.0 {
309        return Ok(result);
310    }
311    // The primitive parser validates the grammar before expanding the exponent.
312    let (mantissa, exponent) = text
313        .split_once(['e', 'E'])
314        .map_or((text, 0), |(m, e)| (m, e.parse::<i64>().unwrap()));
315    let negative = mantissa.starts_with('-');
316    let mantissa = mantissa.trim_start_matches(['+', '-']);
317    let point = mantissa.find('.').unwrap_or(mantissa.len()) as i64 + exponent;
318    let mut digits = mantissa.replace('.', "");
319    if point <= 0 {
320        digits = "0.".to_owned() + &"0".repeat((-point) as usize) + &digits;
321    } else if point as usize >= digits.len() {
322        digits.push_str(&"0".repeat(point as usize - digits.len()));
323    } else {
324        digits.insert(point as usize, '.');
325    }
326    if negative {
327        digits.insert(0, '-');
328    }
329    let mut remainder: Decimal = digits.parse().unwrap();
330    for component in &mut result {
331        *component = remainder.to_string().parse()?;
332        remainder -= Decimal::from(*component);
333    }
334    Ok(renormalize(result))
335}

343fn format(values: &[f64], f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
344    if !values[0].is_finite() || values[0] == 0.0 {
345        return std::fmt::Display::fmt(&values[0], f);
346    }
347    let value = decimal(values).to_string();
348    let mut digits = value.trim_start_matches('-').to_owned();
349    if let Some(precision) = f.precision() {
350        let point = digits.find('.').unwrap_or(digits.len());
351        if point < digits.len() {
352            digits.remove(point);
353        }
354        let end = point + precision;
355        if end < digits.len() {
356            let bytes = digits.as_bytes();
357            let increment = bytes[end] > b'5'
358                || bytes[end] == b'5'
359                    && (bytes[end + 1..].iter().any(|&x| x != b'0')
360                        || !(bytes[end - 1] - b'0').is_multiple_of(2));
361            digits.truncate(end);
362            if increment {
363                let mut bytes = digits.into_bytes();
364                let mut carry = true;
365                for digit in bytes.iter_mut().rev() {
366                    if *digit != b'9' {
367                        *digit += 1;
368                        carry = false;
369                        break;
370                    }
371                    *digit = b'0';
372                }
373                if carry {
374                    bytes.insert(0, b'1');
375                }
376                digits = String::from_utf8(bytes).unwrap();
377            }
378        } else {
379            digits.push_str(&"0".repeat(end - digits.len()));
380        }
381        if precision != 0 {
382            digits.insert(digits.len() - precision, '.');
383        }
384    }
385    f.pad_integral(!values[0].is_sign_negative(), "", &digits)
386}

844                let value = self.abs();

856                let value = self.abs();
857                let result = if value.0 > 1e150 {
858                    value.ln() + Self::LN_2
859                } else {
860                    let square = value * value;
861                    (value + square / ((square + 1.0).sqrt() + 1.0)).ln_1p()
862                };

864            }

867                if self < Self::one() { return Self::NAN; }
868                if self.0 > 1e150 { return self.ln() + Self::LN_2; }
869                let t = self - Self::one();
870                (t + (t * (self + 1.0)).sqrt()).ln_1p()
