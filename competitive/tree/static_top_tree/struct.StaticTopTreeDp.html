308    fn build_compress(&mut self, mut vertex: usize, heavy_child: &[usize], mask: &[u64]) -> Node {
309        let start = vertex;
310        let mut stack = Vec::new();
311        while vertex != usize::MAX {
312            stack.push(Node {
313                depth: bit_ceil(mask[vertex]).trailing_zeros() as usize,
314                slot: Slot::CompressLeaf(vertex),
315            });
316            loop {
317                let len = stack.len();
318                if len >= 3
319                    && (stack[len - 3].depth == stack[len - 2].depth
320                        || stack[len - 3].depth <= stack[len - 1].depth)
321                {
322                    let tail = stack.pop().unwrap();
323                    let right = stack.pop().unwrap();
324                    let left = stack.pop().unwrap();
325                    let node = self.merge_compress(left, right);
326                    stack.push(node);
327                    stack.push(tail);
328                } else if len >= 2 && stack[len - 2].depth <= stack[len - 1].depth {
329                    let right = stack.pop().unwrap();
330                    let left = stack.pop().unwrap();
331                    stack.push(self.merge_compress(left, right));
332                } else {
333                    break;
334                }
335            }
336            vertex = heavy_child[vertex];
337        }
338        while stack.len() > 1 {
339            let right = stack.pop().unwrap();
340            let left = stack.pop().unwrap();
341            stack.push(self.merge_compress(left, right));
342        }
343        let root = stack.pop().unwrap();
344        self.compress_roots[start] = Some(root.slot);
345        root
346    }

348    fn merge_compress(&mut self, left: Node, right: Node) -> Node {
349        let id = self.compressed.len();
350        self.set_parent(left.slot, id << 1);
351        self.set_parent(right.slot, id << 1 | 1);
352        self.compressed.push(InnerNode {
353            left: left.slot,
354            right: right.slot,

356        });
357        Node {
358            depth: left.depth.max(right.depth) + 1,
359            slot: Slot::CompressInner(id),
360        }

388        &self,
389        data: &mut StaticTopTreeDataBuilder<C>,
390        vertices: &[<C as Cluster>::Vertex],
391        edges: &[<C as Cluster>::Edge],
392        slot: Slot,
393    ) -> <C as Cluster>::Path
394    where

396    {
397        match slot {
398            Slot::CompressLeaf(vertex) => {
399                let point = self.init_point(data, vertices, edges, vertex);
400                C::add_vertex(
401                    &point,
402                    &vertices[vertex],
403                    self.parent_edge_ref(edges, vertex),
404                )
405            }
406            Slot::CompressInner(id) => {
407                let node = &self.compressed[id];

409                let right = self.init_compress(data, vertices, edges, node.right);
410                data.compressed[id].write(InnerValue {
411                    parent: node.parent,
412                    left: left.clone(),
413                    right: right.clone(),

417            Slot::RakeLeaf(_) | Slot::RakeInner(_) => unreachable!(),
418        }
