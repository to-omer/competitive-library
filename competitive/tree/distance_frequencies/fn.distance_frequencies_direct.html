7        let n = self.vertices_size();

10            return table;
11        }

14        let root = (0..n).max_by_key(|&u| self.neighbors(u).len()).unwrap();
15        if self.neighbors(root).len() <= 2 {
16            for (d, a) in table.iter_mut().enumerate().skip(1) {
17                *a = 2 * (n - d) as u64;
18            }
19            return table;
20        }
21        // BFS numbering makes each vertex's children a contiguous interval.
22        let mut order = Vec::with_capacity(n);
23        let mut children = Vec::with_capacity(n + 1);
24        order.push((root, n));
25        children.push(1);
26        for u in 0..n {
27            let (v, parent) = order[u];
28            order.extend(
29                self.neighbors(v)
30                    .filter(|a| a.to != parent)
31                    .map(|a| (a.to, v)),
32            );
33            children.push(order.len());
34        }

41                    height[u] = height[v] + 1;
42                    heavy[u] = v;
43                }
44            }
45        }

48            let next = children[end];

57        }

59        let mut work = 0usize;

63                let mut leaves = false;

68                    .skip(children[u])

75                        }
76                    }
77                }
