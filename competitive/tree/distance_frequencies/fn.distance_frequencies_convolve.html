7        let n = self.vertices_size();
8        let mut table = vec![0u64; n];

10            return table;
11        }
12        table[0] = n as u64;
13
14        let root = (0..n).max_by_key(|&u| self.neighbors(u).len()).unwrap();
15        if self.neighbors(root).len() <= 2 {
16            for (d, a) in table.iter_mut().enumerate().skip(1) {
17                *a = 2 * (n - d) as u64;
18            }
19            return table;
20        }
21        // BFS numbering makes each vertex's children a contiguous interval.
22        let mut order = Vec::with_capacity(n);
23        let mut children = Vec::with_capacity(n + 1);
24        order.push((root, n));
25        children.push(1);
26        for u in 0..n {
27            let (v, parent) = order[u];
28            order.extend(
29                self.neighbors(v)
30                    .filter(|a| a.to != parent)
31                    .map(|a| (a.to, v)),
32            );
33            children.push(order.len());
34        }
35        drop(order);
36        let mut height = vec![1usize; n];
37        let mut heavy = vec![n; n];
38        for u in (0..n).rev() {
39            for v in children[u]..children[u + 1] {
40                if height[u] < height[v] + 1 {
41                    height[u] = height[v] + 1;
42                    heavy[u] = v;
43                }
44            }
45        }
46        let mut end = 1;
47        for a in table.iter_mut().take(height[0]).skip(1) {
48            let next = children[end];
49            *a = (next - end) as u64;
50            end = next;
51        }
52        // Each vertex at depth >= d has exactly one ancestor at distance d.
53        let mut sum = 0;
54        for a in table[1..height[0]].iter_mut().rev() {
55            sum += *a;
56            *a = 2 * sum;
57        }
58        // Bound the direct algorithm's product count, aggregating leaf children first.
59        let mut work = 0usize;
60        if height[0] > 128 {
61            for u in 0..n {
62                let mut light = 0;
63                let mut leaves = false;
64                for (v, &h) in height
65                    .iter()
66                    .enumerate()
67                    .take(children[u + 1])
68                    .skip(children[u])
69                {
70                    if v != heavy[u] {
71                        if h == 1 {
72                            leaves = true;
73                        } else {
74                            light += h;
75                        }
76                    }
77                }
78                work = work
79                    .saturating_add((height[u] - 1).saturating_mul(light + usize::from(leaves)));
80                if work > n.saturating_mul(128) {
81                    break;
82                }
83            }
84        }

87        let mut heads = vec![0];
88        let mut size = height[0];
89        for u in 0..n {
90            for v in children[u]..children[u + 1] {
91                if v == heavy[u] {
92                    offset[v] = offset[u] + 1;
93                } else {
94                    offset[v] = size;
95                    size += height[v];
96                    if !direct && height[v] > 1 {

98                    }
99                }
100            }
101        }
102        if direct {

104            return table;
105        }
106

108        let mut counts = vec![1u64; n];
109        let mut branch_begin = vec![0; n + 1];
110        for u in 0..n {
111            let len = (children[u]..children[u + 1])
112                .filter(|&v| v != heavy[u])
113                .map(|v| height[v])
114                .max()
115                .unwrap_or(0);
116            branch_begin[u + 1] = branch_begin[u] + len;
117        }

121            if branch.is_empty() {

123            }

129                .iter()
130                .enumerate()
131                .take(children[u + 1])
132                .skip(children[u])
133            {

135                    continue;
136                }
137                if h == 1 {
138                    leaves += 1;
139                } else {
140                    light += 1;
141                    if first == n {
142                        first = v;
143                    } else {
144                        second = v;
145                    }
146                    for (a, &b) in branch.iter_mut().zip(&counts[offset[v]..offset[v] + h]) {
147                        *a += b;
148                    }
149                }
150            }
151            for (a, &b) in table[2..].iter_mut().zip(branch.iter()) {
152                *a += 2 * leaves * b;
153            }
154            table[2] += leaves * leaves.saturating_sub(1);
155            if light == 2 {
156                let [v, w] = [first, second];
157                let product = distance_frequencies_convolve(
158                    &counts[offset[v]..offset[v] + height[v]],
159                    &counts[offset[w]..offset[w] + height[w]],

161                for (a, b) in table[2..].iter_mut().zip(product) {
162                    *a += 2 * b;
163                }
164            } else if light > 2 {

168                {
169                    *a = a.wrapping_add(b);
170                }

173                        continue;
174                    }
175                    let child = &counts[offset[v]..offset[v] + height[v]];
176                    for (a, b) in table[2..]
177                        .iter_mut()
178                        .zip(distance_frequencies_convolve(child, child))
179                    {
180                        *a = a.wrapping_sub(b);
181                    }
182                }
183            }
184            branch[0] += leaves;
185            for (d, &a) in branch.iter().enumerate() {
186                counts[offset[u] + d + 1] += a;
187            }
188        }
189
190        drop(counts);
191        drop(offset);
192        drop(children);
193        let mut difference = vec![0u64; n + 1];
194        for head in heads {
195            let len = height[head];
196            let mut path = Vec::new();
197            let mut u = head;
198            for i in 0..len {
199                let branch = &branches[branch_begin[u]..branch_begin[u + 1]];
200                if !branch.is_empty() {
201                    // Ancestor pairs are already counted; add distances to later path vertices.
202                    for (d, &a) in branch.iter().enumerate() {
203                        difference[d + 2] = difference[d + 2].wrapping_add(2 * a);
204                        difference[d + len - i + 1] =
205                            difference[d + len - i + 1].wrapping_sub(2 * a);
206                    }
207                    path.push((i, branch));
208                }
209                u = heavy[u];
210            }
211            distance_frequencies_path(&path, &mut table);
212        }
213        let mut sum = 0u64;
214        for (a, d) in table.iter_mut().zip(difference) {
215            sum = sum.wrapping_add(d);
216            *a = a.wrapping_add(sum);
217        }
218        table
219    }

222fn distance_frequencies_direct(
223    children: &[usize],
224    height: &[usize],
225    heavy: &[usize],
226    offset: &[usize],
227    table: &mut [u64],
228) {
229    let n = table.len();
230    let mut counts = vec![1u64; n];
231    for u in (0..n).rev() {
232        let mut leaves = 0u64;
233        for v in children[u]..children[u + 1] {
234            if v == heavy[u] {
235                continue;
236            }
237            if height[v] == 1 {
238                leaves += 1;
239                continue;
240            }
241            for d in 0..height[v] {
242                let b = 2 * counts[offset[v] + d];
243                for (a, &c) in table[d + 2..]
244                    .iter_mut()
245                    .zip(&counts[offset[u] + 1..offset[u] + height[u]])
246                {
247                    *a += b * c;
248                }
249            }
250            for d in 0..height[v] {
251                counts[offset[u] + d + 1] += counts[offset[v] + d];
252            }
253        }
254        if leaves != 0 {
255            for (a, &b) in table[2..]
256                .iter_mut()
257                .zip(&counts[offset[u] + 1..offset[u] + height[u]])
258            {
259                *a += 2 * leaves * b;
260            }
261            table[2] += leaves * (leaves - 1);
262            counts[offset[u] + 1] += leaves;
263        }
264    }
265}

267fn distance_frequencies_path(path: &[(usize, &[u64])], table: &mut [u64]) {
268    if path.len() < 2 {
269        return;
270    }
271    if path.len() <= 4 || path.iter().map(|(_, a)| a.len()).sum::<usize>() <= 64 {
272        for (k, (i, a)) in path.iter().enumerate() {
273            for (j, b) in &path[k + 1..] {
274                for (value, product) in table[j - i + 2..]
275                    .iter_mut()
276                    .zip(distance_frequencies_convolve(a, b))
277                {
278                    *value += 2 * product;
279                }
280            }
281        }
282        return;
283    }
284    let first = path[0].0;
285    let len = path.last().unwrap().0 - first + 1;
286    let max_len = path.iter().map(|(_, a)| a.len()).max().unwrap();
287    let mut forward = vec![0u64; len + max_len - 1];
288    let mut backward = vec![0u64; len + max_len - 1];
289    for (i, a) in path {
290        for (d, &a) in a.iter().enumerate() {
291            forward[i - first + d] += a;
292            backward[len - 1 - (i - first) + d] += a;
293        }
294    }
295    while forward.last() == Some(&0) {
296        forward.pop();
297    }
298    while backward.last() == Some(&0) {
299        backward.pop();
300    }
301    // P+ P- contains the desired pairs, self products, and reverse-direction pairs.
302    let skip = usize::from(max_len == 1);
303    for (a, b) in table[2 + skip..].iter_mut().zip(
304        distance_frequencies_convolve(&forward, &backward)
305            .into_iter()
306            .skip(len - 1 + skip),
307    ) {
308        *a = a.wrapping_add(2 * b);
309    }
310    if max_len == 1 {
311        return;
312    }
313    drop(forward);
314    drop(backward);
315    for (_, a) in path {
316        if a.len() <= 9 {
317            for (d, &x) in a.iter().enumerate() {
318                for (b, &y) in table[d + 2..].iter_mut().zip(a.iter()) {
319                    *b = b.wrapping_sub(2 * x * y);
320                }
321            }
322        }
323    }
324    // Short profiles can only contribute reverse terms between nearby positions.
325    if max_len <= 9 {
326        for (k, (i, a)) in path.iter().enumerate() {
327            for (j, b) in path[k + 1..]
328                .iter()
329                .take_while(|(j, _)| j - i < 2 * max_len - 1)
330            {
331                for (d, &x) in a.iter().enumerate() {
332                    for (e, &y) in b.iter().enumerate().skip((j - i).saturating_sub(d)) {
333                        let value = &mut table[d + e + 2 - (j - i)];
334                        *value = value.wrapping_sub(2 * x * y);
335                    }
336                }
337            }
338        }
339        return;
340    }
341
342    let mut order: Vec<_> = (0..path.len()).collect();
343    order.sort_unstable_by_key(|&i| (path[i].1.len(), i));
344    let mut rank = vec![0; path.len()];
345    let mut begin = vec![0; path.len() + 1];
346    for (k, &i) in order.iter().enumerate() {
347        rank[i] = k;
348    }
349    for (k, (_, a)) in path.iter().enumerate() {
350        begin[k + 1] = begin[k] + a.len();
351    }
352    let mut diagonals = None;
353    for (k, &index) in order.iter().enumerate() {
354        let (i, a) = path[index];
355        let degree = a.len() - 1;
356        if degree == 0 {
357            continue;
358        }
359        let previous_degree = if k == 0 {
360            0
361        } else {
362            path[order[k - 1]].1.len() - 1
363        };
364        // Only exponents -degree .. previous_degree-1 can survive multiplication by a.
365        let mut correction = vec![0u64; degree + previous_degree];
366        if previous_degree <= 8 {
367            let radius = degree + previous_degree;
368            let left = path.partition_point(|(j, _)| *j < i.saturating_sub(radius));
369            for (j, &(position, b)) in path.iter().enumerate().skip(left) {
370                if position > i + radius {
371                    break;
372                }
373                if rank[j] >= k {
374                    continue;
375                }
376                let distance = position.abs_diff(i);
377                for (d, &value) in b.iter().enumerate().skip(distance.saturating_sub(degree)) {
378                    correction[degree + d - distance] += value;
379                }
380            }
381        } else {
382            let diagonals = diagonals.get_or_insert_with(|| {
383                [false, true]
384                    .map(|reverse| DistanceFrequencyDiagonal::new(path, &begin, &rank, k, reverse))
385            });
386            for (d, value) in correction.iter_mut().enumerate() {
387                let exponent = d as isize - degree as isize;
388                *value = diagonals[0].query(i - first, exponent, begin[index])
389                    + diagonals[1].query(len - 1 - (i - first), exponent, begin[index]);
390            }
391        }
392        let nonzero_begin = correction
393            .iter()
394            .position(|&a| a != 0)
395            .unwrap_or(correction.len());
396        let nonzero_end = correction
397            .iter()
398            .rposition(|&a| a != 0)
399            .map_or(nonzero_begin, |i| i + 1);
400        let combined_begin = nonzero_begin.min(degree);
401        // Fold a^2 into a * (correction + x^degree a) when the transform stays the same size.
402        let fuse = a.len() > 9
403            && nonzero_begin < nonzero_end
404            && a.len().saturating_mul(nonzero_end - nonzero_begin) > 4096
405            && (3 * degree + 1 - combined_begin).next_power_of_two()
406                == (degree + nonzero_end - nonzero_begin).next_power_of_two();
407        if a.len() > 9 && !fuse {
408            for (b, c) in table[2..]
409                .iter_mut()
410                .zip(distance_frequencies_convolve(a, a))
411            {
412                *b = b.wrapping_sub(2 * c);
413            }
414        }
415        if nonzero_begin < nonzero_end {
416            let (product, shift) = if fuse {
417                let mut combined = vec![0; 2 * degree + 1 - combined_begin];
418                combined[nonzero_begin - combined_begin..nonzero_end - combined_begin]
419                    .copy_from_slice(&correction[nonzero_begin..nonzero_end]);
420                for (c, &value) in combined[degree - combined_begin..].iter_mut().zip(a) {
421                    *c += value;
422                }
423                (distance_frequencies_convolve(a, &combined), combined_begin)
424            } else {
425                (
426                    distance_frequencies_convolve(a, &correction[nonzero_begin..nonzero_end]),
427                    nonzero_begin,
428                )
429            };
430            for (b, c) in table[2 + shift.saturating_sub(degree)..]
431                .iter_mut()
432                .zip(product.into_iter().skip(degree.saturating_sub(shift)))
433            {
434                *b = b.wrapping_sub(2 * c);
435            }
436        }
437        if k + 1 < order.len()
438            && let Some(diagonals) = &mut diagonals
439        {
440            diagonals[0].insert(i - first, begin[index], a);
441            diagonals[1].insert(len - 1 - (i - first), begin[index], a);
442        }
443    }
