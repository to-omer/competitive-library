357    pub fn build_fold<M: Monoid>(&self, values: &[M::T]) -> HeavyLightPathFold<'_, M> {
358        assert_eq!(values.len(), self.len());
359        let mut fold = HeavyLightPathFold {
360            tree: self,
361            nodes: self
362                .order
363                .iter()
364                .map(|&v| PathFoldNode {
365                    parent: u32::MAX,
366                    children: [u32::MAX; 2],
367                    priority: 0,
368                    value: values[v].clone(),
369                    aggregate: [values[v].clone(), values[v].clone()],
370                    prefix: [values[v].clone(), values[v].clone()],
371                })
372                .collect(),
373        };
374        let mut start = 0;
375        let mut priorities = Vec::new();
376        let mut stack = Vec::new();
377        while start < self.len() {
378            let mut end = start + 1;
379            while end < self.len() && self.nodes[self.order[end]].head as usize == start {
380                end += 1;
381            }
382            priorities.clear();
383            let mut sum = 0usize;
384            for i in start..end {
385                let weight = self.subtree_size(self.order[i])
386                    - if i + 1 < end {
387                        self.subtree_size(self.order[i + 1])
388                    } else {
389                        0
390                    };
391                let priority = (sum ^ (sum + weight)).ilog2();
392                sum += weight;
393                fold.nodes[i].priority = priority;
394                priorities.push(std::cmp::Reverse(priority));
395            }
396            let cartesian = CartesianTree::new(&priorities);
397            for i in start..end {
398                let parent = cartesian.parents[i - start];
399                fold.nodes[i].parent = if parent == usize::MAX {
400                    u32::MAX
401                } else {
402                    (parent + start) as u32
403                };
404                fold.nodes[i].children = cartesian.children[i - start].map(|v| {
405                    if v == usize::MAX {
406                        u32::MAX
407                    } else {
408                        (v + start) as u32
409                    }
410                });
411            }
412            stack.clear();
413            stack.push(cartesian.root + start);
414            let mut i = 0;
415            while i < stack.len() {
416                stack.extend(
417                    fold.nodes[stack[i]]
418                        .children
419                        .into_iter()
420                        .filter(|&v| v != u32::MAX)
421                        .map(|v| v as usize),
422                );
423                i += 1;
424            }
425            for &i in stack.iter().rev() {
426                fold.pull(i);
427            }
428            start = end;
429        }

536        let (mut left, mut right) = (M::unit(), M::unit());
537        while self.tree.nodes[u].head != self.tree.nodes[v].head {
538            if self.tree.index(u) > self.tree.index(v) {
539                left = M::operate(&left, &self.fold_prefix::<true>(self.tree.index(u)));
540                u = self.tree.nodes[u].parent as usize;

542                right = M::operate(&self.fold_prefix::<false>(self.tree.index(v)), &right);

545        }

547            self.fold_range::<true>(self.tree.index(v), self.tree.index(u))
