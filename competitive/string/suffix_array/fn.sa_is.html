199fn induced_sort(s: &[usize], upper: usize, is_s: &[bool], lms: &[usize]) -> Vec<usize> {
200    let n = s.len();
201    let mut sa = vec![!0usize; n];
202
203    let mut bucket_size = vec![0usize; upper + 1];
204    for &c in s {
205        bucket_size[c] += 1;
206    }
207
208    let mut bucket_tail = vec![0usize; upper + 1];
209    {
210        let mut sum = 0usize;
211        for i in 0..=upper {
212            sum += bucket_size[i];
213            bucket_tail[i] = sum;
214        }
215    }
216    for &pos in lms.iter().rev() {
217        let c = s[pos];
218        bucket_tail[c] -= 1;
219        sa[bucket_tail[c]] = pos;
220    }
221
222    let mut bucket_head = vec![0usize; upper + 1];
223    {
224        let mut sum = 0usize;
225        for i in 0..=upper {
226            bucket_head[i] = sum;
227            sum += bucket_size[i];
228        }
229    }
230    for i in 0..n {
231        let v = sa[i];
232        if v == !0 || v == 0 {
233            continue;
234        }
235        let p = v - 1;
236        if !is_s[p] {
237            let c = s[p];
238            sa[bucket_head[c]] = p;
239            bucket_head[c] += 1;
240        }
241    }
242
243    let mut bucket_tail = vec![0usize; upper + 1];
244    {
245        let mut sum = 0usize;
246        for i in 0..=upper {
247            sum += bucket_size[i];
248            bucket_tail[i] = sum;
249        }
250    }
251    for i in (0..n).rev() {
252        let v = sa[i];
253        if v == !0 || v == 0 {
254            continue;
255        }
256        let p = v - 1;
257        if is_s[p] {
258            let c = s[p];
259            bucket_tail[c] -= 1;
260            sa[bucket_tail[c]] = p;
261        }
262    }
263
264    sa
265}

267fn sa_is(s: &[usize], upper: usize) -> Vec<usize> {
268    let n = s.len();
269    match n {
270        0 => return vec![],
271        1 => return vec![0],
272        _ => {}
273    }
274
275    let mut is_s = vec![false; n];
276    is_s[n - 1] = true;
277    for i in (0..n - 1).rev() {
278        is_s[i] = if s[i] < s[i + 1] {
279            true
280        } else if s[i] > s[i + 1] {
281            false
282        } else {
283            is_s[i + 1]
284        };
285    }
286
287    let mut lms_map = vec![!0; n];
288    let mut lms = vec![];
289    for i in 1..n {
290        if is_s[i] && !is_s[i - 1] {
291            lms_map[i] = lms.len();
292            lms.push(i);
293        }
294    }
295
296    let mut sa = induced_sort(s, upper, &is_s, &lms);
297
298    if lms.len() > 1 {
299        let mut rec_s = vec![0usize; lms.len()];
300        let mut rec_upper = 0usize;
301        let mut prev = !0usize;
302        for &pos in &sa {
303            if pos != !0 && lms_map[pos] != !0 {
304                if prev != !0 {
305                    let mut offset = 0usize;
306                    loop {
307                        if s[prev + offset] != s[pos + offset]
308                            || is_s[prev + offset] != is_s[pos + offset]
309                        {
310                            rec_upper += 1;
311                            break;
312                        }
313                        if offset != 0 {
314                            let prev_is_lms = lms_map[prev + offset] != !0;
315                            let pos_is_lms = lms_map[pos + offset] != !0;
316                            if prev_is_lms || pos_is_lms {
317                                rec_upper += (prev_is_lms != pos_is_lms) as usize;
318                                break;
319                            }
320                        }
321                        offset += 1;
322                    }
323                }
324                rec_s[lms_map[pos]] = rec_upper;
325                prev = pos;
326            }
327        }
328
329        let rec_sa = if rec_upper + 1 == lms.len() {
330            let mut rec_sa = vec![0usize; lms.len()];
331            for i in 0..lms.len() {
332                rec_sa[rec_s[i]] = i;
333            }
334            rec_sa
335        } else {
336            sa_is(&rec_s, rec_upper)
337        };
338
339        let mut ordered_lms = Vec::with_capacity(lms.len());
340        for &i in &rec_sa {
341            ordered_lms.push(lms[i]);
342        }
343        sa = induced_sort(s, upper, &is_s, &ordered_lms);
344    }
345
346    sa
